export function sum(options: ReadonlyArray<Option.Option<number>>, trace: string[], n: number): Option.Option<number> {
  return Option.gen(function* () {
    trace.push("start:" + n)
    const a = yield* options[0]
    trace.push("a")
    const b = yield* options[1]
    return a + b
  })
}
export function loop(options: ReadonlyArray<Option.Option<number>>, trace: string[], n: number): Option.Option<number> {
  return Option.gen(function* () {
    let total = 0
    for (let i = 0; i < n; i++) {
      trace.push("loop:" + i)
      const next = yield* options[i]
      total += next
    }
    return total
  })
}
export function discard(options: ReadonlyArray<Option.Option<number>>, trace: string[], n: number): Option.Option<number> {
  return Option.gen(function* () {
    yield* options[0]
    trace.push("discard:" + n)
    return 1
  })
}
export function early(options: ReadonlyArray<Option.Option<number>>, trace: string[], n: number): Option.Option<number> {
  return Option.gen(function* () {
    if (n > 0) return yield* options[0]
    trace.push("early")
    return 0
  })
}
export function capture(options: ReadonlyArray<Option.Option<number>>, trace: string[], n: number): Option.Option<number> {
  const $option0 = n
  return Option.gen(function* () {
    const a = yield* options[0]
    trace.push("capture")
    return a + $option0
  })
}
