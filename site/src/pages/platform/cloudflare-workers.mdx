# Cloudflare Workers

A Cloudflare Worker exports a `fetch` handler rather than running `main()`. Write the application logic in `.vibe`, invoke it from a thin `.ts` `fetch` adapter with `Effect.runPromise`, bundle both with `esbuild` + `vibelang-unplugin`, and run locally in `workerd` via Wrangler.

## Install

```sh
npm install effect vibelang-runtime
npm install -D vibelang vibelang-unplugin esbuild wrangler
```

## Configure

Include `.vibe` sources in `tsconfig.json` (required by `vibelang-unplugin`; omitting it fails with `TS5058`):

```json
{
  "compilerOptions": {
    "target": "ES2022",
    "module": "ESNext",
    "moduleResolution": "Bundler",
    "lib": ["ES2022", "DOM"],
    "strict": true,
    "skipLibCheck": true,
    "noEmit": true,
    "rewriteRelativeImportExtensions": true
  },
  "include": ["src/**/*.ts", "src/**/*.vibe"]
}
```

Bundle to a single ES module using `workerd` export conditions in `build.mjs`:

```js
// build.mjs
import * as esbuild from "esbuild"
import vibelang from "vibelang-unplugin/esbuild"

await esbuild.build({
  entryPoints: ["src/worker.ts"],
  bundle: true,
  format: "esm",
  platform: "browser",
  conditions: ["workerd", "worker"],
  outfile: "dist/worker.js",
  plugins: [vibelang()]
})
```

Point `wrangler.toml` at the prebuilt bundle:

```toml
name = "greeter"
main = "dist/worker.js"
compatibility_date = "2026-10-01"
no_bundle = true

[build]
command = "node build.mjs"
```

## Write the Worker

```vibe
// src/greeter.vibe
export error EmptyName

service Greeter {
  greet(name: string): string!EmptyName
}

const GreeterLive = Layer.succeed(Greeter, {
  effect greet(name) {
    if (name.length === 0) throw new EmptyName()
    return `Hello, ${name}!`
  }
})

export effect function greeting(name: string) {
  provide GreeterLive {
    const greeter = try Greeter
    return greeter.greet(name) catch () => "Hello, stranger!"
  }
}
```

```ts
// src/worker.ts
import { Effect } from "effect"
import { greeting } from "./greeter.vibe"

export default {
  async fetch(request: Request): Promise<Response> {
    const name = new URL(request.url).searchParams.get("name") ?? "Worker"
    return new Response(await Effect.runPromise(greeting(name)))
  }
}
```

Provide all required layers inside the invoked Effect (`FetchHttpClient.layer` uses the Worker's native `fetch` for outbound HTTP).

## Run locally and deploy

```sh
npx wrangler dev
curl "http://localhost:8787/?name=Ada"
# Hello, Ada!
curl "http://localhost:8787/?name="
# Hello, stranger!
```

Deploy with `npx wrangler deploy`.
