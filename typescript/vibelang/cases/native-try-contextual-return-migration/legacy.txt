export error Bad
export type Outcome<T> = { _tag: "Ready"; value: T } | { _tag: "Failed"; error: Bad }
export effect function project<T>(value: T, fail: boolean): Outcome<T>!! {
  try {
    if (fail) throw new Bad()
    return { _tag: "Ready", value }
  } catch (error) {
    return { _tag: "Failed", error }
  }
}
export effect function pair(value: number): ["value", number]!! {
  try { return ["value", value] } finally {}
}
export effect function conditional(flag: boolean): ({ tag: "yes" } | { tag: "no" })!! {
  try { return flag ? { tag: "yes" } : { tag: "no" } } finally {}
}
