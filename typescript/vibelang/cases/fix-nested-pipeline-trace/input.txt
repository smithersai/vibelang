import * as Effect from "effect/Effect";
export const run = Effect.fn("run")(function* () {
  const value = yield* Effect.succeed(1).pipe(
    (program: Effect.Effect<number>) => program,
    Effect.flatMap((n: number) => Effect.succeed(n).pipe(Effect.map((value: number) => value + 1)))
  );
  return value;
});
