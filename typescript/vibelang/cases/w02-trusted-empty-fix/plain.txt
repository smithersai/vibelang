let captureReads = 0
let thrown: unknown
const parse = function (this: unknown, text: string): number {
  if (this !== undefined) throw new Error("receiver")
  const n = Number(text)
  if (Number.isNaN(n)) {
    thrown = new Error("bad " + text)
    throw thrown
  }
  return n
}
export const Dialect = { get parse() { captureReads++; return parse } }
export const reads = (): number => captureReads
export const lastThrown = (): unknown => thrown
