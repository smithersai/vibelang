package main

// Checked native-pin assertions over the Go frontend's whole-project graph.
//
// The source spelling is an ordinary imported call:
//
//	import { native } from "smithers:native"
//	native(checksum)
//
// Authority comes only from the resolved declaration in the compiler-owned
// virtual module below. The authored spelling of the callee is never tested,
// so aliases and namespace reads retain authority while local/lookalike
// functions named `native` remain ordinary runtime calls.

import (
	"sort"
	"strconv"
	"strings"

	"github.com/microsoft/TypeScript/tsc/internal/ast"
	"github.com/microsoft/TypeScript/tsc/internal/checker"
)

const (
	nativeLogicalPath     = "__smithers_native.d.ts"
	nativeVirtualName     = "/src/" + nativeLogicalPath
	nativeModuleAlias     = "./__smithers_native"
	nativeModuleSpecifier = "smithers:native"
)

// This declaration file has its own module identity. Mapping the native
// specifier to the shared runtime prelude would make the same exported symbol
// reachable through smithers:exceptions or smthrs/context as well. The
// lowerer rewrites the accepted import to the runtime prelude, whose `native`
// export is the identity operation.
const nativeDeclarationText = `
export declare function native<F extends (...args: never[]) => unknown>(pinned: F): F;
`

func nativeResolvedModule(typeChecker *checker.Checker, specifier *ast.Node) bool {
	if typeChecker == nil || specifier == nil {
		return false
	}
	symbol := typeChecker.ResolveExternalModuleName(specifier)
	if symbol == nil {
		return false
	}
	symbol = typeChecker.GetMergedSymbol(symbol)
	for _, declaration := range symbol.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && file.FileName() == nativeVirtualName {
			return true
		}
	}
	return false
}

// nativePinCall recognizes the exact function declaration exported by the
// compiler-owned native module. declarationName is consulted only after the
// declaration-file identity proof; no use-site name participates.
func (a *rowAnalysis) nativePinCall(node *ast.Node) bool {
	if node == nil || !ast.IsCallExpression(node) {
		return false
	}
	expression := ast.SkipParentheses(node.AsCallExpression().Expression)
	symbol := a.symbolAt(expression)
	if symbol == nil && ast.IsPropertyAccessExpression(expression) {
		symbol = a.symbolAt(expression.AsPropertyAccessExpression().Name())
	}
	for _, declaration := range symbolDeclarations(symbol) {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && file.FileName() == nativeVirtualName &&
			ast.IsFunctionDeclaration(declaration) && declarationName(declaration) == "native" {
			return true
		}
	}
	return false
}

func nativeFunctionPathName(fn *functionRows) string {
	file := "main.sm"
	if fn != nil && fn.file != nil {
		file = strings.TrimPrefix(fn.file.FileName(), "/src/")
	}
	name := "<anonymous>"
	if fn != nil && fn.name != "" {
		name = fn.name
	}
	return file + "#" + name
}

func addNativeRequirement(paths map[string][]string, requirement string, path []string) bool {
	if requirement == "" {
		return false
	}
	if _, exists := paths[requirement]; exists {
		return false
	}
	paths[requirement] = append([]string(nil), path...)
	return true
}

func nativePinBlocked(requirement string) bool {
	return requirement == "TypeScript" || strings.HasPrefix(requirement, "Module<") ||
		strings.HasPrefix(requirement, "Host<")
}

// nativeRequirementForModule classifies one checker-resolved module edge. The
// compiler-owned registry is exact because identity is asked first; there is
// deliberately no smithers:/smthrs/ prefix test here.
func (a *rowAnalysis) nativeRequirementForModule(specifier *ast.Node) string {
	if specifier == nil || !ast.IsStringLiteralLike(specifier) {
		return ""
	}
	if a.prelude.isCompilerModuleSpecifier(specifier) {
		return ""
	}
	moduleName := specifier.Text()
	if strings.HasPrefix(moduleName, "node:") {
		return "Module<" + strconv.Quote(moduleName) + ">"
	}
	resolved := a.checker.ResolveExternalModuleName(specifier)
	if resolved != nil {
		resolved = a.checker.GetMergedSymbol(resolved)
		for _, declaration := range resolved.Declarations {
			file := ast.GetSourceFileOfNode(declaration)
			if file != nil && strings.HasSuffix(strings.ToLower(file.FileName()), ".sm") {
				return ""
			}
		}
	}
	return "TypeScript"
}

type nativeDependency struct {
	requirement string
	hops        []string
}

// nativeFlowValue is the finite value domain used only by native-pin graph
// construction. A value is a project syntax node (callable, object literal, or
// class), a positional list, or a union of values that may reach the same use.
// Lists and unions are keyed structurally below: both are rebuilt at calls, so
// pointer identity would make the recursion guard grow without converging.
type nativeFlowValue struct {
	kind    nativeFlowKind
	node    *ast.Node
	values  []*nativeFlowValue
	options []*nativeFlowValue
}

type nativeFlowKind uint8

const (
	nativeFlowNode nativeFlowKind = iota + 1
	nativeFlowList
	nativeFlowUnion
)

type nativeFlowBindings map[*ast.Symbol]*nativeFlowValue

type nativeFlowSeen struct {
	symbols map[*ast.Symbol]bool
	nodes   map[*ast.Node]bool
}

func newNativeFlowSeen() *nativeFlowSeen {
	return &nativeFlowSeen{symbols: make(map[*ast.Symbol]bool), nodes: make(map[*ast.Node]bool)}
}

func (seen *nativeFlowSeen) clone() *nativeFlowSeen {
	cloned := newNativeFlowSeen()
	if seen == nil {
		return cloned
	}
	for symbol := range seen.symbols {
		cloned.symbols[symbol] = true
	}
	for node := range seen.nodes {
		cloned.nodes[node] = true
	}
	return cloned
}

func nativeFlowNodeValue(node *ast.Node) *nativeFlowValue {
	if node == nil {
		return nil
	}
	return &nativeFlowValue{kind: nativeFlowNode, node: node}
}

func nativeFlowListValue(values []*nativeFlowValue) *nativeFlowValue {
	return &nativeFlowValue{kind: nativeFlowList, values: values}
}

func nativeFlowOptions(value *nativeFlowValue) []*nativeFlowValue {
	if value == nil {
		return nil
	}
	if value.kind == nativeFlowUnion {
		return value.options
	}
	return []*nativeFlowValue{value}
}

func nativeFlowValueKey(value *nativeFlowValue) string {
	if value == nil {
		return "?"
	}
	switch value.kind {
	case nativeFlowList:
		parts := make([]string, len(value.values))
		for index, element := range value.values {
			parts[index] = nativeFlowValueKey(element)
		}
		return "[" + strings.Join(parts, ",") + "]"
	case nativeFlowUnion:
		parts := make([]string, len(value.options))
		for index, option := range value.options {
			parts[index] = nativeFlowValueKey(option)
		}
		sort.Strings(parts)
		return "{" + strings.Join(parts, "|") + "}"
	case nativeFlowNode:
		file := ast.GetSourceFileOfNode(value.node)
		name := "<unknown>"
		if file != nil {
			name = file.FileName()
		}
		return name + ":" + strconv.Itoa(value.node.Pos()) + ":" + strconv.Itoa(value.node.End())
	default:
		return "?"
	}
}

func nativeFlowUnionValue(values ...*nativeFlowValue) *nativeFlowValue {
	options := make([]*nativeFlowValue, 0, len(values))
	keys := make(map[string]bool)
	for _, value := range values {
		for _, option := range nativeFlowOptions(value) {
			key := nativeFlowValueKey(option)
			if keys[key] {
				continue
			}
			keys[key] = true
			options = append(options, option)
		}
	}
	if len(options) == 0 {
		return nil
	}
	if len(options) == 1 {
		return options[0]
	}
	return &nativeFlowValue{kind: nativeFlowUnion, options: options}
}

func nativeFlowNodeOf(value *nativeFlowValue) *ast.Node {
	if value == nil || value.kind != nativeFlowNode {
		return nil
	}
	return value.node
}

func nativeFlowListOf(value *nativeFlowValue) ([]*nativeFlowValue, bool) {
	if value == nil || value.kind != nativeFlowList {
		return nil, false
	}
	return value.values, true
}

func nativeProjectSourceNode(node *ast.Node) bool {
	file := ast.GetSourceFileOfNode(node)
	return file != nil && strings.HasSuffix(strings.ToLower(file.FileName()), ".sm")
}

func nativeProjectCallable(node *ast.Node) *ast.Node {
	if node == nil || !ast.IsFunctionLikeDeclaration(node) || node.Body() == nil || !nativeProjectSourceNode(node) {
		return nil
	}
	return node
}

func nativeProjectClass(node *ast.Node) *ast.Node {
	if node == nil || !ast.IsClassLike(node) || !nativeProjectSourceNode(node) {
		return nil
	}
	return node
}

func nativeUnwrapExpression(expression *ast.Node) *ast.Node {
	for expression != nil {
		switch expression.Kind {
		case ast.KindParenthesizedExpression, ast.KindNonNullExpression, ast.KindAsExpression,
			ast.KindTypeAssertionExpression, ast.KindSatisfiesExpression:
			expression = expression.Expression()
		default:
			return expression
		}
	}
	return nil
}

func nativeNamedMember(member *ast.Node, name string) bool {
	if member == nil || member.Name() == nil {
		return false
	}
	named := member.Name()
	return (ast.IsIdentifier(named) || ast.IsStringLiteralLike(named)) && named.Text() == name
}

func nativeCallableName(node *ast.Node) string {
	if node == nil {
		return ""
	}
	if name := node.Name(); name != nil && (ast.IsIdentifier(name) || ast.IsStringLiteralLike(name)) {
		return name.Text()
	}
	if parent := node.Parent; parent != nil &&
		(ast.IsVariableDeclaration(parent) || ast.IsPropertyAssignment(parent) || ast.IsPropertyDeclaration(parent)) &&
		parent.Name() != nil && ast.IsIdentifier(parent.Name()) {
		return parent.Name().Text()
	}
	return ""
}

func nativeCallableHop(node *ast.Node) string {
	name := nativeCallableName(node)
	if name == "" {
		return ""
	}
	return nativePortableFileName(ast.GetSourceFileOfNode(node)) + "#" + name
}

func nativeFlowBindingsKey(callee *ast.Node, bindings nativeFlowBindings) string {
	values := make([]string, 0, len(bindings))
	for _, value := range bindings {
		values = append(values, nativeFlowValueKey(value))
	}
	sort.Strings(values)
	return nativeFlowValueKey(nativeFlowNodeValue(callee)) + "|" + strings.Join(values, ",")
}

func nativeBindingSymbol(typeChecker *checker.Checker, name *ast.Node) *ast.Symbol {
	if typeChecker == nil || name == nil {
		return nil
	}
	return typeChecker.GetSymbolAtLocation(name)
}

func (a *rowAnalysis) nativeFlowValueOfSymbol(
	local *ast.Symbol,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if local == nil {
		return nil
	}
	if bound := bindings[local]; bound != nil {
		return bound
	}
	symbol := a.canonical(local)
	if symbol == nil || seen.symbols[symbol] {
		return nil
	}
	seen.symbols[symbol] = true
	for _, declaration := range symbol.Declarations {
		if callable := nativeProjectCallable(declaration); callable != nil {
			return nativeFlowNodeValue(callable)
		}
		if class := nativeProjectClass(declaration); class != nil {
			return nativeFlowNodeValue(class)
		}
		if ast.IsBindingElement(declaration) {
			if value := a.nativeBindingElementValue(declaration, bindings, seen); value != nil {
				return value
			}
		}
		if ast.IsVariableDeclaration(declaration) {
			if value := a.nativeIteratedValue(declaration, bindings, seen); value != nil {
				return value
			}
		}
		if value := a.nativeFlowValue(nativeLaunderedInitializer(declaration), bindings, seen); value != nil {
			return value
		}
	}
	return nil
}

func (a *rowAnalysis) nativeFlowValue(
	expression *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if expression == nil {
		return nil
	}
	node := nativeUnwrapExpression(expression)
	switch {
	case ast.IsObjectLiteralExpression(node):
		return nativeFlowNodeValue(node)
	case ast.IsFunctionExpression(node), ast.IsArrowFunction(node):
		return nativeFlowNodeValue(nativeProjectCallable(node))
	case ast.IsArrayLiteralExpression(node):
		values, _ := a.nativePositionalValues(node.Elements(), bindings, seen)
		return nativeFlowListValue(values)
	case ast.IsNewExpression(node):
		value := a.nativeFlowValue(node.AsNewExpression().Expression, bindings, seen)
		classes := make([]*nativeFlowValue, 0)
		for _, option := range nativeFlowOptions(value) {
			if class := nativeProjectClass(nativeFlowNodeOf(option)); class != nil {
				classes = append(classes, nativeFlowNodeValue(class))
			}
		}
		return nativeFlowUnionValue(classes...)
	case ast.IsCallExpression(node):
		return a.nativeFlowReturnValue(node, bindings, seen)
	case ast.IsConditionalExpression(node):
		conditional := node.AsConditionalExpression()
		return nativeFlowUnionValue(
			a.nativeFlowValue(conditional.WhenTrue, bindings, seen.clone()),
			a.nativeFlowValue(conditional.WhenFalse, bindings, seen.clone()),
		)
	case ast.IsPropertyAccessExpression(node):
		access := node.AsPropertyAccessExpression()
		return a.nativeFlowMember(access.Expression, access.Name().Text(), bindings, seen)
	case ast.IsElementAccessExpression(node):
		access := node.AsElementAccessExpression()
		argument := nativeUnwrapExpression(access.ArgumentExpression)
		if ast.IsStringLiteralLike(argument) {
			return a.nativeFlowMember(access.Expression, argument.Text(), bindings, seen)
		}
		if argument == nil || argument.Kind != ast.KindNumericLiteral {
			return nil
		}
		index, err := strconv.Atoi(argument.Text())
		if err != nil || index < 0 {
			return nil
		}
		values := make([]*nativeFlowValue, 0)
		for _, option := range nativeFlowOptions(a.nativeFlowValue(access.Expression, bindings, seen)) {
			if list, ok := nativeFlowListOf(option); ok && index < len(list) {
				values = append(values, list[index])
			}
		}
		return nativeFlowUnionValue(values...)
	case ast.IsIdentifier(node):
		return a.nativeFlowValueOfSymbol(a.checker.GetSymbolAtLocation(node), bindings, seen)
	}
	return nil
}

func (a *rowAnalysis) nativePositionalValues(
	elements []*ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) ([]*nativeFlowValue, bool) {
	values := make([]*nativeFlowValue, 0, len(elements))
	for _, element := range elements {
		if ast.IsOmittedExpression(element) {
			values = append(values, nil)
			continue
		}
		if ast.IsSpreadElement(element) {
			spread := a.nativeFlowValue(element.AsSpreadElement().Expression, bindings, seen.clone())
			list, ok := nativeFlowListOf(spread)
			if !ok {
				return values, true
			}
			values = append(values, list...)
			continue
		}
		values = append(values, a.nativeFlowValue(element, bindings, seen.clone()))
	}
	return values, false
}

func (a *rowAnalysis) nativeFlowMember(
	target *ast.Node,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	values := make([]*nativeFlowValue, 0)
	for _, option := range nativeFlowOptions(a.nativeFlowValue(target, bindings, seen)) {
		values = append(values, a.nativeMemberOfValue(option, name, bindings, seen.clone()))
	}
	return nativeFlowUnionValue(values...)
}

func (a *rowAnalysis) nativeMemberOfValue(
	value *nativeFlowValue,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	node := nativeFlowNodeOf(value)
	if node == nil {
		return nil
	}
	if ast.IsObjectLiteralExpression(node) {
		return a.nativeObjectMemberValue(node, name, bindings, seen)
	}
	if ast.IsClassLike(node) {
		return a.nativeClassMemberValue(node, name, bindings, seen)
	}
	return nil
}

func (a *rowAnalysis) nativeFlowReturnValue(
	call *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	values := make([]*nativeFlowValue, 0)
	for _, callee := range a.nativeInvokedCallables(call, bindings) {
		if seen.nodes[callee] {
			continue
		}
		nextSeen := seen.clone()
		nextSeen.nodes[callee] = true
		inner := a.nativeArgumentBindings(callee, call, bindings)
		for _, returned := range nativeReturnedExpressions(callee) {
			values = append(values, a.nativeFlowValue(returned, inner, nextSeen.clone()))
		}
	}
	return nativeFlowUnionValue(values...)
}

func nativeReturnedExpressions(callee *ast.Node) []*ast.Node {
	body := callee.Body()
	if body == nil {
		return nil
	}
	if !ast.IsBlock(body) {
		return []*ast.Node{body}
	}
	returned := make([]*ast.Node, 0)
	var visit func(*ast.Node) bool
	visit = func(node *ast.Node) bool {
		if ast.IsFunctionLikeDeclaration(node) || ast.IsClassLike(node) {
			return false
		}
		if ast.IsReturnStatement(node) {
			if expression := node.AsReturnStatement().Expression; expression != nil {
				returned = append(returned, expression)
			}
			return false
		}
		node.ForEachChild(visit)
		return false
	}
	body.ForEachChild(visit)
	return returned
}

func (a *rowAnalysis) nativeIteratedValue(
	declaration *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	list := declaration.Parent
	if list == nil || !ast.IsVariableDeclarationList(list) || list.Parent == nil || !ast.IsForOfStatement(list.Parent) {
		return nil
	}
	iterated := a.nativeFlowValue(list.Parent.AsForInOrOfStatement().Expression, bindings, seen)
	values := make([]*nativeFlowValue, 0)
	for _, option := range nativeFlowOptions(iterated) {
		if elements, ok := nativeFlowListOf(option); ok {
			values = append(values, elements...)
		}
	}
	return nativeFlowUnionValue(values...)
}

func (a *rowAnalysis) nativeFlowCallables(
	expression *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) []*ast.Node {
	callables := make([]*ast.Node, 0)
	known := make(map[*ast.Node]bool)
	for _, option := range nativeFlowOptions(a.nativeFlowValue(expression, bindings, seen)) {
		node := nativeProjectCallable(nativeFlowNodeOf(option))
		if node != nil && !known[node] {
			known[node] = true
			callables = append(callables, node)
		}
	}
	return callables
}

func (a *rowAnalysis) nativeObjectMemberValue(
	literal *ast.Node,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	var found *nativeFlowValue
	for _, property := range literal.Properties() {
		if ast.IsSpreadAssignment(property) {
			source := a.nativeFlowValue(property.AsSpreadAssignment().Expression, bindings, seen.clone())
			if member := a.nativeSpreadMemberValue(source, name, bindings, seen.clone()); member != nil {
				found = member
			}
			continue
		}
		if !nativeNamedMember(property, name) {
			continue
		}
		switch {
		case ast.IsMethodDeclaration(property):
			found = nativeFlowNodeValue(nativeProjectCallable(property))
		case ast.IsPropertyAssignment(property):
			found = a.nativeFlowValue(property.AsPropertyAssignment().Initializer, bindings, seen)
		case ast.IsShorthandPropertyAssignment(property):
			found = a.nativeFlowValueOfSymbol(a.checker.GetShorthandAssignmentValueSymbol(property), bindings, seen)
		default:
			found = nil
		}
	}
	return found
}

func (a *rowAnalysis) nativeSpreadMemberValue(
	value *nativeFlowValue,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	node := nativeFlowNodeOf(value)
	if node == nil {
		return nil
	}
	if ast.IsObjectLiteralExpression(node) {
		return a.nativeObjectMemberValue(node, name, bindings, seen)
	}
	if ast.IsClassLike(node) {
		return a.nativeClassPropertyValue(node, name, bindings, seen)
	}
	return nil
}

func (a *rowAnalysis) nativeClassPropertyValue(
	class *ast.Node,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if seen.nodes[class] {
		return nil
	}
	seen.nodes[class] = true
	for _, member := range class.Members() {
		if !nativeNamedMember(member, name) {
			continue
		}
		if ast.IsPropertyDeclaration(member) {
			return a.nativeFlowValue(member.AsPropertyDeclaration().Initializer, bindings, seen)
		}
		return nil
	}
	base := a.nativeBaseClass(class, bindings, seen)
	if base == nil {
		return nil
	}
	return a.nativeClassPropertyValue(base, name, bindings, seen)
}

func (a *rowAnalysis) nativeClassMemberValue(
	class *ast.Node,
	name string,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if seen.nodes[class] {
		return nil
	}
	seen.nodes[class] = true
	for _, member := range class.Members() {
		if !nativeNamedMember(member, name) {
			continue
		}
		if ast.IsGetAccessorDeclaration(member) || ast.IsSetAccessorDeclaration(member) {
			return nil
		}
		if ast.IsMethodDeclaration(member) {
			if callable := nativeProjectCallable(member); callable != nil {
				return nativeFlowNodeValue(callable)
			}
			continue
		}
		if ast.IsPropertyDeclaration(member) {
			if value := a.nativeFlowValue(member.AsPropertyDeclaration().Initializer, bindings, seen); value != nil {
				return value
			}
		}
	}
	base := a.nativeBaseClass(class, bindings, seen)
	if base == nil {
		return nil
	}
	return a.nativeClassMemberValue(base, name, bindings, seen)
}

func (a *rowAnalysis) nativeBaseClass(
	class *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *ast.Node {
	data := class.ClassLikeData()
	if data == nil || data.HeritageClauses == nil {
		return nil
	}
	for _, clauseNode := range data.HeritageClauses.Nodes {
		clause := clauseNode.AsHeritageClause()
		if clause.Token != ast.KindExtendsKeyword || clause.Types == nil || len(clause.Types.Nodes) == 0 {
			continue
		}
		expression := clause.Types.Nodes[0].AsExpressionWithTypeArguments().Expression
		for _, option := range nativeFlowOptions(a.nativeFlowValue(expression, bindings, seen)) {
			if base := nativeProjectClass(nativeFlowNodeOf(option)); base != nil {
				return base
			}
		}
	}
	return nil
}

func (a *rowAnalysis) nativeBindingElementValue(
	element *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	pattern := element.Parent
	source := a.nativePatternSource(pattern, bindings, seen)
	if source != nil {
		if member := a.nativePatternMember(element, pattern, source, bindings, seen); member != nil {
			return member
		}
	}
	return a.nativeFlowValue(element.AsBindingElement().Initializer, bindings, seen)
}

func (a *rowAnalysis) nativePatternSource(
	pattern *ast.Node,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if pattern == nil || pattern.Parent == nil {
		return nil
	}
	parent := pattern.Parent
	if ast.IsBindingElement(parent) {
		return a.nativeBindingElementValue(parent, bindings, seen)
	}
	if ast.IsVariableDeclaration(parent) {
		return a.nativeFlowValue(parent.AsVariableDeclaration().Initializer, bindings, seen)
	}
	if ast.IsParameterDeclaration(parent) {
		return a.nativeFlowValue(parent.AsParameterDeclaration().Initializer, bindings, seen)
	}
	return nil
}

func (a *rowAnalysis) nativePatternMember(
	element *ast.Node,
	pattern *ast.Node,
	source *nativeFlowValue,
	bindings nativeFlowBindings,
	seen *nativeFlowSeen,
) *nativeFlowValue {
	if element.AsBindingElement().DotDotDotToken != nil {
		return nil
	}
	if ast.IsArrayBindingPattern(pattern) {
		index := -1
		for candidate, current := range pattern.Elements() {
			if current == element {
				index = candidate
				break
			}
		}
		if index < 0 {
			return nil
		}
		values := make([]*nativeFlowValue, 0)
		for _, option := range nativeFlowOptions(source) {
			if list, ok := nativeFlowListOf(option); ok && index < len(list) {
				values = append(values, list[index])
			}
		}
		return nativeFlowUnionValue(values...)
	}
	named := element.AsBindingElement().PropertyName
	if named == nil {
		named = element.Name()
	}
	if named == nil || !(ast.IsIdentifier(named) || ast.IsStringLiteralLike(named)) {
		return nil
	}
	values := make([]*nativeFlowValue, 0)
	for _, option := range nativeFlowOptions(source) {
		values = append(values, a.nativeMemberOfValue(option, named.Text(), bindings, seen.clone()))
	}
	return nativeFlowUnionValue(values...)
}

func (a *rowAnalysis) nativeBindPattern(
	name *ast.Node,
	value *nativeFlowValue,
	next nativeFlowBindings,
	bindings nativeFlowBindings,
) {
	if ast.IsIdentifier(name) {
		if symbol := nativeBindingSymbol(a.checker, name); symbol != nil {
			next[symbol] = value
		}
		return
	}
	if !ast.IsArrayBindingPattern(name) && !ast.IsObjectBindingPattern(name) {
		return
	}
	for _, element := range name.Elements() {
		if ast.IsOmittedExpression(element) {
			continue
		}
		member := a.nativePatternMember(element, name, value, bindings, newNativeFlowSeen())
		if member != nil {
			a.nativeBindPattern(element.Name(), member, next, bindings)
		}
	}
}

func nativeTemplateSubstitutions(call *ast.Node) []*ast.Node {
	if call == nil || !ast.IsTaggedTemplateExpression(call) {
		return nil
	}
	template := call.AsTaggedTemplateExpression().Template
	if template == nil || !ast.IsTemplateExpression(template) {
		return nil
	}
	spans := template.AsTemplateExpression().TemplateSpans
	if spans == nil {
		return nil
	}
	expressions := make([]*ast.Node, 0, len(spans.Nodes))
	for _, span := range spans.Nodes {
		expressions = append(expressions, span.AsTemplateSpan().Expression)
	}
	return expressions
}

func (a *rowAnalysis) nativeArgumentBindings(
	callee *ast.Node,
	call *ast.Node,
	bindings nativeFlowBindings,
) nativeFlowBindings {
	next := make(nativeFlowBindings)
	tagged := ast.IsTaggedTemplateExpression(call)
	var written []*ast.Node
	if tagged {
		written = nativeTemplateSubstitutions(call)
	} else {
		written = call.Arguments()
	}
	arguments, _ := a.nativePositionalValues(written, bindings, newNativeFlowSeen())
	offset := 0
	if tagged {
		offset = 1
	}
	parameters := callee.Parameters()
	for index := 0; index+offset < len(parameters); index++ {
		parameter := parameters[index+offset]
		if parameter.AsParameterDeclaration().DotDotDotToken != nil {
			if !ast.IsIdentifier(parameter.Name()) || index >= len(arguments) {
				break
			}
			if symbol := nativeBindingSymbol(a.checker, parameter.Name()); symbol != nil {
				next[symbol] = nativeFlowListValue(append([]*nativeFlowValue(nil), arguments[index:]...))
			}
			break
		}
		if index >= len(arguments) {
			break
		}
		if arguments[index] != nil {
			a.nativeBindPattern(parameter.Name(), arguments[index], next, bindings)
		}
	}
	return next
}

func (a *rowAnalysis) nativeCalleeTarget(call *ast.Node) *ast.Node {
	if call == nil {
		return nil
	}
	var target *ast.Node
	if ast.IsTaggedTemplateExpression(call) {
		target = call.AsTaggedTemplateExpression().Tag
	} else if ast.IsCallExpression(call) {
		target = call.AsCallExpression().Expression
	} else if ast.IsNewExpression(call) {
		target = call.AsNewExpression().Expression
	}
	inner := nativeUnwrapExpression(target)
	if ast.IsCallExpression(call) && ast.IsPropertyAccessExpression(inner) {
		access := inner.AsPropertyAccessExpression()
		method := access.Name().Text()
		if method == "call" || method == "apply" {
			return access.Expression
		}
	}
	return target
}

func (a *rowAnalysis) nativeSelectedCallable(call *ast.Node) *ast.Node {
	if call == nil {
		return nil
	}
	if signature := a.checker.GetResolvedSignature(call); signature != nil {
		if callable := nativeProjectCallable(signature.Declaration()); callable != nil {
			return callable
		}
	}
	return nil
}

func (a *rowAnalysis) nativeInvokedCallables(call *ast.Node, bindings nativeFlowBindings) []*ast.Node {
	if selected := a.nativeSelectedCallable(call); selected != nil {
		return []*ast.Node{selected}
	}
	return a.nativeFlowCallables(a.nativeCalleeTarget(call), bindings, newNativeFlowSeen())
}

func (a *rowAnalysis) nativeAccessorOfValue(
	value *nativeFlowValue,
	name string,
	seen *nativeFlowSeen,
) *ast.Node {
	node := nativeFlowNodeOf(value)
	if node == nil {
		return nil
	}
	if ast.IsObjectLiteralExpression(node) {
		for _, property := range node.Properties() {
			if !nativeNamedMember(property, name) {
				continue
			}
			if ast.IsGetAccessorDeclaration(property) {
				return nativeProjectCallable(property)
			}
			return nil
		}
		return nil
	}
	if !ast.IsClassLike(node) {
		return nil
	}
	return a.nativeClassAccessor(node, name, seen)
}

func (a *rowAnalysis) nativeClassAccessor(class *ast.Node, name string, seen *nativeFlowSeen) *ast.Node {
	if seen.nodes[class] {
		return nil
	}
	seen.nodes[class] = true
	for _, member := range class.Members() {
		if !nativeNamedMember(member, name) {
			continue
		}
		if ast.IsGetAccessorDeclaration(member) {
			return nativeProjectCallable(member)
		}
		return nil
	}
	base := a.nativeBaseClass(class, nil, seen)
	if base == nil {
		return nil
	}
	return a.nativeClassAccessor(base, name, seen)
}

func (a *rowAnalysis) nativeInvokedAccessors(access *ast.Node, bindings nativeFlowBindings) []*ast.Node {
	if access == nil || !ast.IsPropertyAccessExpression(access) {
		return nil
	}
	property := access.AsPropertyAccessExpression()
	for _, declaration := range symbolDeclarations(a.checker.GetSymbolAtLocation(property.Name())) {
		if ast.IsGetAccessorDeclaration(declaration) {
			if callable := nativeProjectCallable(declaration); callable != nil {
				return []*ast.Node{callable}
			}
		}
	}
	value := a.nativeFlowValue(property.Expression, bindings, newNativeFlowSeen())
	found := make([]*ast.Node, 0)
	known := make(map[*ast.Node]bool)
	for _, option := range nativeFlowOptions(value) {
		accessor := a.nativeAccessorOfValue(option, property.Name().Text(), newNativeFlowSeen())
		if accessor != nil && !known[accessor] {
			known[accessor] = true
			found = append(found, accessor)
		}
	}
	return found
}

type nativeFlowCallEdge struct {
	callee *functionRows
	hops   []string
}

func nativeSameCallable(values []*ast.Node, target *ast.Node) bool {
	for _, value := range values {
		if value == target {
			return true
		}
	}
	return false
}

func nativeTopLevelAnalyzedCallable(node *ast.Node) bool {
	return node != nil && ast.IsFunctionDeclaration(node) && node.Parent != nil && ast.IsSourceFile(node.Parent)
}

func (a *rowAnalysis) collectNativeFlowEdges(fn *functionRows) []nativeFlowCallEdge {
	if fn == nil || fn.node == nil || fn.node.Body() == nil {
		return nil
	}
	edges := make([]nativeFlowCallEdge, 0)
	edgeKeys := make(map[string]bool)
	followed := make(map[string]bool)
	addEdge := func(callable *ast.Node, hops []string) {
		if callable == nil || nativeInvokedWhereDefined(callable) {
			return
		}
		callee := a.byNode[callable]
		if callee == nil || callee == fn {
			return
		}
		if nativeTopLevelAnalyzedCallable(callable) {
			hops = nil
		}
		key := nativeFlowValueKey(nativeFlowNodeValue(callable)) + "|" + strings.Join(hops, "->")
		if edgeKeys[key] {
			return
		}
		edgeKeys[key] = true
		edges = append(edges, nativeFlowCallEdge{callee: callee, hops: append([]string(nil), hops...)})
	}
	appendCallableHop := func(hops []string, callable *ast.Node) []string {
		if nativeInvokedWhereDefined(callable) {
			return append([]string(nil), hops...)
		}
		return nativeAppendHop(hops, nativeCallableHop(callable))
	}
	var follow func(*ast.Node, nativeFlowBindings, []string)
	follow = func(callee *ast.Node, bindings nativeFlowBindings, hops []string) {
		if len(bindings) == 0 {
			return
		}
		key := nativeFlowBindingsKey(callee, bindings)
		if followed[key] {
			return
		}
		followed[key] = true
		var step func(*ast.Node) bool
		step = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node) {
				return false
			}
			if ast.IsCallExpression(node) || ast.IsNewExpression(node) || ast.IsTaggedTemplateExpression(node) {
				ordinary := []*ast.Node(nil)
				if selected := a.nativeSelectedCallable(node); selected != nil {
					ordinary = []*ast.Node{selected}
				}
				flowed := a.nativeInvokedCallables(node, bindings)
				for _, target := range flowed {
					if !nativeSameCallable(ordinary, target) {
						addEdge(target, hops)
					}
				}
				targets := flowed
				if len(targets) == 0 {
					targets = ordinary
				}
				for _, target := range targets {
					follow(target, a.nativeArgumentBindings(target, node, bindings), appendCallableHop(hops, target))
				}
			}
			if ast.IsPropertyAccessExpression(node) {
				ordinary := a.nativeInvokedAccessors(node, nil)
				for _, accessor := range a.nativeInvokedAccessors(node, bindings) {
					if !nativeSameCallable(ordinary, accessor) {
						addEdge(accessor, hops)
					}
				}
			}
			node.ForEachChild(step)
			return false
		}
		callee.ForEachChild(step)
	}
	var visit func(*ast.Node) bool
	visit = func(node *ast.Node) bool {
		if node != fn.node && ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node) {
			return false
		}
		if ast.IsCallExpression(node) || ast.IsNewExpression(node) || ast.IsTaggedTemplateExpression(node) {
			ordinary := []*ast.Node(nil)
			if selected := a.nativeSelectedCallable(node); selected != nil {
				ordinary = []*ast.Node{selected}
			}
			flowed := a.nativeInvokedCallables(node, nil)
			for _, target := range flowed {
				hasExisting := false
				for _, edge := range fn.calls {
					if edge.node == node && edge.callee == a.byNode[target] {
						hasExisting = true
						break
					}
				}
				if !nativeSameCallable(ordinary, target) || !hasExisting {
					addEdge(target, nil)
				}
				follow(target, a.nativeArgumentBindings(target, node, nil), appendCallableHop(nil, target))
			}
		}
		if ast.IsPropertyAccessExpression(node) {
			for _, accessor := range a.nativeInvokedAccessors(node, nil) {
				addEdge(accessor, nil)
			}
		}
		node.ForEachChild(visit)
		return false
	}
	visit(fn.node)
	return edges
}

// nativeLaunderWalk is branch-local provenance state. symbols and starLookups
// are stacks rather than memo tables: whether a namespace symbol is clean
// depends on the member being read, so retaining a completed branch's answer
// would let an earlier clean member hide a later foreign one.
type nativeLaunderWalk struct {
	analysis    *rowAnalysis
	symbols     map[*ast.Symbol]bool
	starLookups map[string]bool
	ambient     map[string][]string
	origin      *ast.SourceFile
	loaded      map[string][]string
	entered     map[*ast.Node]bool
	followed    map[string]bool
	asset       bool
}

func nativePortableFileName(file *ast.SourceFile) string {
	if file == nil {
		return "main.sm"
	}
	return strings.TrimPrefix(file.FileName(), "/src/")
}

func nativeAppendHop(hops []string, hop string) []string {
	if hop == "" || len(hops) != 0 && hops[len(hops)-1] == hop {
		return append([]string(nil), hops...)
	}
	return append(append([]string(nil), hops...), hop)
}

func nativeAliasOwner(declaration *ast.Node) *ast.Node {
	if declaration == nil || !(ast.IsImportClause(declaration) || ast.IsNamespaceImport(declaration) ||
		ast.IsImportSpecifier(declaration) || ast.IsExportSpecifier(declaration) ||
		ast.IsNamespaceExport(declaration)) {
		return nil
	}
	for current := declaration; current != nil; current = current.Parent {
		if ast.IsImportDeclaration(current) || ast.IsExportDeclaration(current) {
			return current
		}
		if ast.IsSourceFile(current) {
			return nil
		}
	}
	return nil
}

func nativeExportIsTypeOnly(declaration *ast.ExportDeclaration) bool {
	if declaration == nil || declaration.IsTypeOnly {
		return true
	}
	clause := declaration.ExportClause
	if clause == nil || !ast.IsNamedExports(clause) {
		return false
	}
	elements := clause.AsNamedExports().Elements
	if elements == nil || len(elements.Nodes) == 0 {
		return false
	}
	for _, element := range elements.Nodes {
		if !element.AsExportSpecifier().IsTypeOnly {
			return false
		}
	}
	return true
}

func nativeBindingIsTypeOnly(declaration *ast.Node, owner *ast.Node) bool {
	if owner == nil {
		return false
	}
	if ast.IsImportDeclaration(owner) {
		if importIsTypeOnly(owner.AsImportDeclaration()) {
			return true
		}
		return ast.IsImportSpecifier(declaration) && declaration.AsImportSpecifier().IsTypeOnly
	}
	export := owner.AsExportDeclaration()
	return export.IsTypeOnly || ast.IsExportSpecifier(declaration) && declaration.AsExportSpecifier().IsTypeOnly
}

func nativeBoundExportName(declaration *ast.Node) string {
	switch {
	case ast.IsImportSpecifier(declaration):
		specifier := declaration.AsImportSpecifier()
		if specifier.PropertyName != nil {
			return specifier.PropertyName.Text()
		}
		return specifier.Name().Text()
	case ast.IsExportSpecifier(declaration):
		specifier := declaration.AsExportSpecifier()
		if specifier.PropertyName != nil {
			return specifier.PropertyName.Text()
		}
		return specifier.Name().Text()
	case ast.IsImportClause(declaration):
		return "default"
	}
	return ""
}

func (a *rowAnalysis) nativeProjectModuleFile(specifier *ast.Node) *ast.SourceFile {
	if specifier == nil {
		return nil
	}
	module := a.checker.ResolveExternalModuleName(specifier)
	if module == nil {
		return nil
	}
	module = a.checker.GetMergedSymbol(module)
	for _, declaration := range module.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && strings.HasSuffix(strings.ToLower(file.FileName()), ".sm") {
			return file
		}
	}
	return nil
}

// collectNativeLoadGraph records what evaluating file unconditionally loads.
// Only binding-free imports belong here: bound imports remain name-directed and
// are charged when their values are read. onStack is path-scoped rather than a
// completed-module memo so an edge after a cycle is still visited by the outer
// frame.
func (a *rowAnalysis) collectNativeLoadGraph(
	file *ast.SourceFile,
	route []string,
	onStack map[string]bool,
	into map[string][]string,
) {
	if file == nil || onStack[file.FileName()] {
		return
	}
	onStack[file.FileName()] = true
	defer delete(onStack, file.FileName())
	for _, statement := range file.Statements.Nodes {
		if !ast.IsImportDeclaration(statement) {
			continue
		}
		declaration := statement.AsImportDeclaration()
		if declaration.ImportClause != nil || declaration.ModuleSpecifier == nil ||
			!ast.IsStringLiteralLike(declaration.ModuleSpecifier) {
			continue
		}
		if target := a.nativeProjectModuleFile(declaration.ModuleSpecifier); target != nil {
			a.collectNativeLoadGraph(
				target,
				nativeAppendHop(route, nativePortableFileName(target)),
				onStack,
				into,
			)
			continue
		}
		requirement := a.nativeRequirementForModule(declaration.ModuleSpecifier)
		if requirement == "" {
			continue
		}
		if _, exists := into[requirement]; exists {
			continue
		}
		if len(route) == 0 {
			into[requirement] = nil
		} else {
			into[requirement] = nativeAppendHop(route, declaration.ModuleSpecifier.Text())
		}
	}
}

func nativeMemberRead(node *ast.Node) string {
	if node == nil || node.Parent == nil {
		return ""
	}
	parent := node.Parent
	if ast.IsPropertyAccessExpression(parent) && parent.AsPropertyAccessExpression().Expression == node {
		return parent.AsPropertyAccessExpression().Name().Text()
	}
	if ast.IsElementAccessExpression(parent) && parent.AsElementAccessExpression().Expression == node {
		argument := parent.AsElementAccessExpression().ArgumentExpression
		if argument != nil && ast.IsStringLiteralLike(argument) {
			return argument.Text()
		}
	}
	return ""
}

func nativeImmediateAlias(typeChecker *checker.Checker, symbol *ast.Symbol) *ast.Symbol {
	if symbol == nil || symbol.Flags&ast.SymbolFlagsAlias == 0 {
		return nil
	}
	return typeChecker.GetImmediateAliasedSymbol(symbol)
}

func nativeExtendRoute(route []string, aliased *ast.Symbol) []string {
	for _, declaration := range symbolDeclarations(aliased) {
		file := ast.GetSourceFileOfNode(declaration)
		if file == nil || !strings.HasSuffix(strings.ToLower(file.FileName()), ".sm") {
			continue
		}
		return nativeAppendHop(route, nativePortableFileName(file))
	}
	return route
}

// recordModuleLoad charges the unconditional load graph of every project
// module a name-directed binding walk crosses. The reference's own module is
// already charged by collectDirectNativeRequirements and is omitted here so
// its established direct routes remain first.
func (walk *nativeLaunderWalk) recordModuleLoad(file *ast.SourceFile, hops []string) {
	if file == nil || file == walk.origin || !strings.HasSuffix(strings.ToLower(file.FileName()), ".sm") {
		return
	}
	route := nativeAppendHop(hops, nativePortableFileName(file))
	walk.analysis.collectNativeLoadGraph(file, route, make(map[string]bool), walk.loaded)
}

func (walk *nativeLaunderWalk) dependencyForSymbol(symbol *ast.Symbol, member string, hops []string) *nativeDependency {
	if symbol == nil || walk.symbols[symbol] {
		return nil
	}
	walk.symbols[symbol] = true
	defer delete(walk.symbols, symbol)
	for _, declaration := range symbol.Declarations {
		walk.recordModuleLoad(ast.GetSourceFileOfNode(declaration), hops)
		if edge := walk.dependencyForDeclaration(declaration, symbol, member, hops); edge != nil {
			return edge
		}
	}
	return nil
}

func (walk *nativeLaunderWalk) dependencyForDeclaration(declaration *ast.Node, symbol *ast.Symbol, member string, hops []string) *nativeDependency {
	a := walk.analysis
	if owner := nativeAliasOwner(declaration); owner != nil {
		if nativeBindingIsTypeOnly(declaration, owner) {
			return nil
		}
		var specifier *ast.Node
		if ast.IsImportDeclaration(owner) {
			specifier = owner.AsImportDeclaration().ModuleSpecifier
		} else {
			specifier = owner.AsExportDeclaration().ModuleSpecifier
		}
		if specifier == nil {
			return walk.dependencyForSymbol(nativeImmediateAlias(a.checker, symbol), member, hops)
		}
		if target := a.nativeProjectModuleFile(specifier); target != nil {
			route := nativeAppendHop(hops, nativePortableFileName(target))
			if aliased := nativeImmediateAlias(a.checker, symbol); aliased != nil {
				if edge := walk.dependencyForSymbol(aliased, member, nativeExtendRoute(route, aliased)); edge != nil {
					return edge
				}
			}
			name := nativeBoundExportName(declaration)
			if name == "" {
				name = member
			}
			return walk.dependencyThroughStarExports(target, name, route)
		}
		if _, asset := a.assetImports[owner]; asset {
			walk.asset = true
			return nil
		}
		requirement := a.nativeRequirementForModule(specifier)
		if requirement == "" {
			return nil
		}
		route := append([]string(nil), hops...)
		if len(route) != 0 {
			route = nativeAppendHop(route, specifier.Text())
		}
		return &nativeDependency{requirement: requirement, hops: route}
	}

	if ast.IsSourceFile(declaration) {
		file := declaration.AsSourceFile()
		if !strings.HasSuffix(strings.ToLower(file.FileName()), ".sm") {
			return nil
		}
		for _, exported := range a.checker.GetExportsOfModule(symbol) {
			if member != "" && exported.Name != member {
				continue
			}
			if edge := walk.dependencyForSymbol(exported, "", hops); edge != nil {
				return edge
			}
		}
		return walk.dependencyThroughStarExports(file, member, hops)
	}

	if initializer := nativeLaunderedInitializer(declaration); initializer != nil {
		return walk.dependencyInExpression(initializer, hops, nativeBindingRoute(declaration, hops))
	}
	return nil
}

func (walk *nativeLaunderWalk) dependencyThroughStarExports(file *ast.SourceFile, member string, hops []string) *nativeDependency {
	if file == nil {
		return nil
	}
	key := file.FileName() + "#" + member
	if walk.starLookups[key] {
		return nil
	}
	walk.starLookups[key] = true
	defer delete(walk.starLookups, key)
	for _, statement := range file.Statements.Nodes {
		if !ast.IsExportDeclaration(statement) {
			continue
		}
		declaration := statement.AsExportDeclaration()
		if declaration.ExportClause != nil || declaration.IsTypeOnly || declaration.ModuleSpecifier == nil {
			continue
		}
		if target := walk.analysis.nativeProjectModuleFile(declaration.ModuleSpecifier); target != nil {
			if edge := walk.dependencyThroughStarExports(target, member,
				nativeAppendHop(hops, nativePortableFileName(target))); edge != nil {
				return edge
			}
			continue
		}
		// An asset `export *` is rejected by asset admission and is therefore
		// intentionally not exempted as a compile-time edge.
		requirement := walk.analysis.nativeRequirementForModule(declaration.ModuleSpecifier)
		if requirement != "" {
			return &nativeDependency{requirement: requirement,
				hops: nativeAppendHop(hops, declaration.ModuleSpecifier.Text())}
		}
	}
	return nil
}

func nativeLaunderedInitializer(declaration *ast.Node) *ast.Node {
	if declaration == nil {
		return nil
	}
	switch {
	case ast.IsVariableDeclaration(declaration):
		return declaration.AsVariableDeclaration().Initializer
	case ast.IsParameterDeclaration(declaration):
		return declaration.AsParameterDeclaration().Initializer
	case ast.IsExportAssignment(declaration):
		return declaration.AsExportAssignment().Expression
	case ast.IsBindingElement(declaration):
		for current := declaration; current != nil; current = current.Parent {
			if ast.IsVariableDeclaration(current) {
				return current.AsVariableDeclaration().Initializer
			}
			if ast.IsParameterDeclaration(current) {
				return current.AsParameterDeclaration().Initializer
			}
			if ast.IsSourceFile(current) {
				return nil
			}
		}
	}
	return nil
}

func nativeBindingRoute(declaration *ast.Node, hops []string) []string {
	file := nativePortableFileName(ast.GetSourceFileOfNode(declaration))
	name := declarationName(declaration)
	if ast.IsExportAssignment(declaration) {
		name = "default"
	}
	route := append([]string(nil), hops...)
	if len(route) != 0 && route[len(route)-1] == file {
		route = route[:len(route)-1]
	}
	if name != "" {
		file += "#" + name
	}
	return nativeAppendHop(route, file)
}

func nativeAmbientSourceName(node *ast.Node) string {
	if member := nativeMemberRead(node); member != "" {
		return node.Text() + "." + member
	}
	return node.Text()
}

func nativeInvokedWhereDefined(callable *ast.Node) bool {
	if callable == nil {
		return false
	}
	current := callable
	parent := current.Parent
	for parent != nil && (ast.IsParenthesizedExpression(parent) || ast.IsNonNullExpression(parent) ||
		ast.IsAsExpression(parent) || ast.IsTypeAssertion(parent) || parent.Kind == ast.KindSatisfiesExpression) {
		current = parent
		parent = current.Parent
	}
	if parent == nil {
		return false
	}
	if ast.IsCallExpression(parent) && parent.AsCallExpression().Expression == current {
		return true
	}
	if ast.IsNewExpression(parent) && parent.AsNewExpression().Expression == current {
		return true
	}
	if ast.IsTaggedTemplateExpression(parent) && parent.AsTaggedTemplateExpression().Tag == current {
		return true
	}
	if !ast.IsPropertyAccessExpression(parent) || parent.AsPropertyAccessExpression().Expression != current {
		return false
	}
	applied := parent.Parent
	if applied == nil || !ast.IsCallExpression(applied) || applied.AsCallExpression().Expression != parent {
		return false
	}
	method := parent.AsPropertyAccessExpression().Name().Text()
	if method == "call" || method == "apply" {
		return true
	}
	return method == "bind" && nativeInvokedWhereDefined(applied)
}

func (walk *nativeLaunderWalk) recordAmbient(node *ast.Node, route []string) {
	if !nativeValueReference(node) {
		return
	}
	requirements := make([]string, 0, 2)
	if _, forbidden := forbiddenHostGlobals[node.Text()]; forbidden && walk.analysis.isAmbientGlobalReference(node) {
		requirements = append(requirements, "Host<"+strconv.Quote(node.Text())+">")
	}
	for _, use := range walk.analysis.ambientAuthorityUses(node) {
		requirement := use.requirement
		if requirement == "Host" {
			requirement = "Host<" + strconv.Quote(node.Text()) + ">"
		}
		requirements = append(requirements, requirement)
	}
	if len(requirements) == 0 {
		return
	}
	reached := nativeAppendHop(route, nativeAmbientSourceName(node))
	for _, requirement := range requirements {
		if _, exists := walk.ambient[requirement]; !exists {
			walk.ambient[requirement] = reached
		}
	}
}

func (walk *nativeLaunderWalk) dependencyInExpression(expression *ast.Node, hops []string, ambientRoute []string) *nativeDependency {
	var found *nativeDependency
	calleeHops := func(callable *ast.Node, route []string) []string {
		return nativeAppendHop(route, nativePortableFileName(ast.GetSourceFileOfNode(callable)))
	}
	calleeRoute := func(callable *ast.Node, route []string) []string {
		name := nativeCallableName(callable)
		if name == "" {
			return append([]string(nil), route...)
		}
		return nativeAppendHop(route,
			nativePortableFileName(ast.GetSourceFileOfNode(callable))+"#"+name)
	}
	var visit func(*ast.Node, bool, []string, []string)
	var enter func(*ast.Node, []string, []string)
	var follow func(*ast.Node, nativeFlowBindings, []string, []string)
	enter = func(callable *ast.Node, route []string, authorityRoute []string) {
		if callable == nil || walk.entered[callable] || nativeInvokedWhereDefined(callable) {
			return
		}
		walk.entered[callable] = true
		callable.ForEachChild(func(child *ast.Node) bool {
			visit(child, false, calleeHops(callable, route), calleeRoute(callable, authorityRoute))
			return false
		})
	}
	follow = func(callee *ast.Node, bindings nativeFlowBindings, route []string, authorityRoute []string) {
		if len(bindings) == 0 {
			return
		}
		key := nativeFlowBindingsKey(callee, bindings)
		if walk.followed[key] {
			return
		}
		walk.followed[key] = true
		var step func(*ast.Node) bool
		step = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node) {
				return false
			}
			if ast.IsCallExpression(node) || ast.IsNewExpression(node) || ast.IsTaggedTemplateExpression(node) {
				ordinary := []*ast.Node(nil)
				if selected := walk.analysis.nativeSelectedCallable(node); selected != nil {
					ordinary = []*ast.Node{selected}
				}
				flowed := walk.analysis.nativeInvokedCallables(node, bindings)
				for _, target := range flowed {
					if !nativeSameCallable(ordinary, target) {
						enter(target, route, authorityRoute)
					}
				}
				targets := flowed
				if len(targets) == 0 {
					targets = ordinary
				}
				for _, target := range targets {
					nextRoute := route
					nextAuthority := authorityRoute
					if !nativeInvokedWhereDefined(target) {
						nextRoute = calleeHops(target, route)
						nextAuthority = calleeRoute(target, authorityRoute)
					}
					follow(target, walk.analysis.nativeArgumentBindings(target, node, bindings), nextRoute, nextAuthority)
				}
			}
			if ast.IsPropertyAccessExpression(node) {
				ordinary := walk.analysis.nativeInvokedAccessors(node, nil)
				for _, accessor := range walk.analysis.nativeInvokedAccessors(node, bindings) {
					if !nativeSameCallable(ordinary, accessor) {
						enter(accessor, route, authorityRoute)
					}
				}
			}
			node.ForEachChild(step)
			return false
		}
		callee.ForEachChild(step)
	}
	enterCall := func(node *ast.Node, route []string, authorityRoute []string) {
		for _, callee := range walk.analysis.nativeInvokedCallables(node, nil) {
			enter(callee, route, authorityRoute)
			nextRoute := route
			nextAuthority := authorityRoute
			if !nativeInvokedWhereDefined(callee) {
				nextRoute = calleeHops(callee, route)
				nextAuthority = calleeRoute(callee, authorityRoute)
			}
			follow(callee, walk.analysis.nativeArgumentBindings(callee, node, nil), nextRoute, nextAuthority)
		}
	}
	visit = func(node *ast.Node, deferred bool, route []string, authorityRoute []string) {
		if node == nil {
			return
		}
		if ast.IsFunctionLikeDeclaration(node) && walk.entered[node] {
			return
		}
		if !deferred {
			walk.recordAmbient(node, authorityRoute)
		}
		if found == nil && nativeValueReference(node) {
			symbol := walk.analysis.checker.GetSymbolAtLocation(node)
			found = walk.dependencyForSymbol(symbol, nativeMemberRead(node), route)
		}
		if !deferred {
			if ast.IsCallExpression(node) || ast.IsNewExpression(node) || ast.IsTaggedTemplateExpression(node) {
				enterCall(node, route, authorityRoute)
			}
			if ast.IsCallExpression(node) {
				if provision, ok := walk.analysis.provideEdge(node); ok && provision.callbacks != nil {
					for _, callback := range walk.analysis.nativeFlowCallables(
						provision.callbacks, nil, newNativeFlowSeen(),
					) {
						enter(callback, route, authorityRoute)
					}
				}
			}
			if ast.IsPropertyAccessExpression(node) {
				for _, accessor := range walk.analysis.nativeInvokedAccessors(node, nil) {
					enter(accessor, route, authorityRoute)
				}
			}
		}
		nested := deferred || ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node)
		node.ForEachChild(func(child *ast.Node) bool {
			visit(child, nested, route, authorityRoute)
			return false
		})
	}
	visit(expression, false, hops, ambientRoute)
	return found
}

// nativeDependenciesForReference follows a value through imports, re-exports,
// project bindings and module-level initializers. It returns the module edge
// first so the existing first-path retention keeps the long-standing direct
// module route when the same requirement is reachable more than one way.
func (a *rowAnalysis) nativeDependenciesForReference(node *ast.Node) ([]nativeDependency, bool) {
	if node == nil {
		return nil, false
	}
	walk := &nativeLaunderWalk{
		analysis:    a,
		symbols:     make(map[*ast.Symbol]bool),
		starLookups: make(map[string]bool),
		ambient:     make(map[string][]string),
		origin:      ast.GetSourceFileOfNode(node),
		loaded:      make(map[string][]string),
		entered:     make(map[*ast.Node]bool),
		followed:    make(map[string]bool),
	}
	var edge *nativeDependency
	if symbol := a.checker.GetSymbolAtLocation(node); symbol != nil {
		edge = walk.dependencyForSymbol(symbol, nativeMemberRead(node), nil)
	}
	edges := make([]nativeDependency, 0, len(walk.ambient)+1)
	if edge != nil {
		edges = append(edges, *edge)
	}
	requirements := make([]string, 0, len(walk.ambient))
	for requirement := range walk.ambient {
		requirements = append(requirements, requirement)
	}
	sort.Strings(requirements)
	for _, requirement := range requirements {
		edges = append(edges, nativeDependency{requirement: requirement, hops: walk.ambient[requirement]})
	}
	requirements = requirements[:0]
	for requirement := range walk.loaded {
		requirements = append(requirements, requirement)
	}
	sort.Strings(requirements)
	for _, requirement := range requirements {
		edges = append(edges, nativeDependency{requirement: requirement, hops: walk.loaded[requirement]})
	}
	return edges, walk.asset
}

func importDeclarationAncestor(node *ast.Node) *ast.Node {
	for current := node; current != nil; current = current.Parent {
		if ast.IsImportDeclaration(current) {
			return current
		}
	}
	return nil
}

// importedModuleAt returns the authored module edge for an imported binding.
// It asks the alias declaration rather than its local name, so renamed and
// namespace imports classify identically. Type-only aliases contribute no
// runtime requirement.
func (a *rowAnalysis) importedModuleAt(node *ast.Node) *ast.Node {
	if node == nil {
		return nil
	}
	symbol := a.checker.GetSymbolAtLocation(node)
	if symbol == nil {
		return nil
	}
	for _, declaration := range symbol.Declarations {
		if ast.IsTypeOnlyImportOrExportDeclaration(declaration) {
			continue
		}
		statement := importDeclarationAncestor(declaration)
		if statement == nil {
			continue
		}
		if _, asset := a.assetImports[statement]; asset {
			// Asset loading is complete before runtime and contributes no
			// TypeScript/platform requirement to the importing function.
			continue
		}
		importDeclaration := statement.AsImportDeclaration()
		if importIsTypeOnly(importDeclaration) {
			continue
		}
		return importDeclaration.ModuleSpecifier
	}
	return nil
}

// foreignOriginModule follows the same authored-value provenance shapes as
// foreignValueProvenance, retaining the import edge when one is available so a
// node: module can be distinguished from the general TypeScript requirement.
func (a *rowAnalysis) foreignOriginModule(expression *ast.Node, seen map[*ast.Symbol]bool) *ast.Node {
	if expression == nil {
		return nil
	}
	switch expression.Kind {
	case ast.KindParenthesizedExpression, ast.KindAsExpression,
		ast.KindNonNullExpression, ast.KindAwaitExpression, ast.KindTypeAssertionExpression:
		return a.foreignOriginModule(expression.Expression(), seen)
	case ast.KindPropertyAccessExpression:
		return a.foreignOriginModule(expression.AsPropertyAccessExpression().Expression, seen)
	case ast.KindElementAccessExpression:
		return a.foreignOriginModule(expression.AsElementAccessExpression().Expression, seen)
	case ast.KindCallExpression:
		return a.foreignOriginModule(expression.AsCallExpression().Expression, seen)
	case ast.KindNewExpression:
		return a.foreignOriginModule(expression.AsNewExpression().Expression, seen)
	case ast.KindIdentifier:
		if module := a.importedModuleAt(expression); module != nil {
			return module
		}
		symbol := a.symbolAt(expression)
		if symbol == nil || seen[symbol] {
			return nil
		}
		seen[symbol] = true
		declaration := variableLikeDeclaration(symbol)
		if declaration != nil && ast.IsVariableDeclaration(declaration) {
			return a.foreignOriginModule(declaration.AsVariableDeclaration().Initializer, seen)
		}
	}
	return nil
}

func nativeValueReference(node *ast.Node) bool {
	if node == nil || !ast.IsIdentifier(node) || ast.IsPartOfTypeNode(node) ||
		isDeclarationNameIdentifier(node) || isPropertyNameIdentifier(node) {
		return false
	}
	parent := node.Parent
	if parent == nil {
		return true
	}
	if ast.IsQualifiedName(parent) && parent.AsQualifiedName().Right == node {
		return false
	}
	if ast.IsLabeledStatement(parent) || ast.IsBreakStatement(parent) || ast.IsContinueStatement(parent) {
		return false
	}
	return true
}

func (a *rowAnalysis) collectDirectNativeRequirements(fn *functionRows, paths map[string][]string) {
	name := nativeFunctionPathName(fn)
	for _, row := range fn.directRequirements {
		addNativeRequirement(paths, row.name, []string{name})
	}
	// The module load graph is transitive: evaluating this file evaluates every
	// project module it side-effect imports, and their side-effect imports in
	// turn. A direct foreign edge retains the historical one-hop route.
	loads := make(map[string][]string)
	a.collectNativeLoadGraph(fn.file, nil, make(map[string]bool), loads)
	for requirement, hops := range loads {
		addNativeRequirement(paths, requirement, append([]string{name}, hops...))
	}

	var visit func(*ast.Node) bool
	visit = func(node *ast.Node) bool {
		if node == nil || node != fn.node && ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node) {
			return false
		}
		if node.Kind == ast.KindAnyKeyword {
			addNativeRequirement(paths, "TypeScript", []string{name})
		}
		if ast.IsCallExpression(node) {
			call := node.AsCallExpression()
			if call.Expression != nil && call.Expression.Kind == ast.KindImportKeyword {
				addNativeRequirement(paths, "TypeScript", []string{name})
			}
			if call.Expression != nil && ast.IsIdentifier(call.Expression) &&
				(call.Expression.Text() == "eval" || call.Expression.Text() == "Function") &&
				a.isAmbientGlobalReference(call.Expression) {
				addNativeRequirement(paths, "TypeScript", []string{name})
			}
		}
		if ast.IsNewExpression(node) {
			expression := node.AsNewExpression().Expression
			if expression != nil && ast.IsIdentifier(expression) && expression.Text() == "Function" &&
				a.isAmbientGlobalReference(expression) {
				addNativeRequirement(paths, "TypeScript", []string{name})
			}
		}
		if ast.IsIdentifier(node) && nativeValueReference(node) {
			if _, forbidden := forbiddenHostGlobals[node.Text()]; forbidden && a.isAmbientGlobalReference(node) {
				addNativeRequirement(paths, "Host<"+strconv.Quote(node.Text())+">", []string{name})
			}
			for _, use := range a.ambientAuthorityUses(node) {
				requirement := use.requirement
				if requirement == "Host" {
					requirement = "Host<" + strconv.Quote(node.Text()) + ">"
				}
				addNativeRequirement(paths, requirement, []string{name})
			}
			dependencies, asset := a.nativeDependenciesForReference(node)
			for _, dependency := range dependencies {
				addNativeRequirement(paths, dependency.requirement,
					append([]string{name}, dependency.hops...))
			}
			if len(dependencies) == 0 && !asset && a.foreignValueProvenance(node, make(map[*ast.Symbol]bool)) {
				module := a.foreignOriginModule(node, make(map[*ast.Symbol]bool))
				if module != nil {
					addNativeRequirement(paths, a.nativeRequirementForModule(module), []string{name})
				} else {
					addNativeRequirement(paths, "TypeScript", []string{name})
				}
			}
		}
		node.ForEachChild(visit)
		return false
	}
	visit(fn.node)
}

func (a *rowAnalysis) collectNativePins(files []*ast.SourceFile) map[*functionRows]*ast.Node {
	pins := make(map[*functionRows]*ast.Node)
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if a.nativePinCall(node) {
				call := node.AsCallExpression()
				count := 0
				if call.Arguments != nil {
					count = len(call.Arguments.Nodes)
				}
				if count != 1 {
					a.report(node, "SMITHERS3005", "native pin is not checkable: the pin was given "+strconv.Itoa(count)+" arguments; native(...) takes one reference to a function declared in this project")
				} else {
					argument := ast.SkipParentheses(call.Arguments.Nodes[0])
					target := a.resolveFunctionReference(argument)
					if target == nil || !ast.IsFunctionDeclaration(target.node) ||
						target.node.Parent == nil || !ast.IsSourceFile(target.node.Parent) {
						a.report(argument, "SMITHERS3005", "native pin is not checkable: its argument does not resolve to an analyzed function declaration; native(...) takes one reference to a function declared in this project")
					} else if _, alreadyPinned := pins[target]; !alreadyPinned {
						pins[target] = node
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
	return pins
}

func (a *rowAnalysis) checkRetiredNativeMarkers() {
	for _, fn := range a.functions {
		if !ast.IsFunctionDeclaration(fn.node) {
			continue
		}
		for _, jsdoc := range fn.node.JSDoc(fn.file) {
			tags := jsdoc.AsJSDoc().Tags
			if tags == nil {
				continue
			}
			for _, tag := range tags.Nodes {
				if tag.TagName() == nil || tag.TagName().Text() != "native" {
					continue
				}
				a.reportWarning(tag, "SMITHERS3006", "the /** @native */ marker no longer pins a function; import { native } from \"smithers:native\" and write native("+fn.name+")")
			}
		}
	}
}

// checkNativePins builds one retained dependency path per requirement, then
// checks each assertion against the complete fixed-point graph. Ordinary
// Context capabilities (including Clock and Random) remain in the path table
// but do not block; only TypeScript and concrete host dependence do.
func (a *rowAnalysis) checkNativePins(files []*ast.SourceFile) {
	a.checkRetiredNativeMarkers()
	pins := a.collectNativePins(files)
	paths := make(map[*functionRows]map[string][]string, len(a.functions))
	for _, fn := range a.functions {
		paths[fn] = make(map[string][]string)
		a.collectDirectNativeRequirements(fn, paths[fn])
	}
	flowEdges := make(map[*functionRows][]nativeFlowCallEdge, len(a.functions))
	for _, fn := range a.functions {
		flowEdges[fn] = a.collectNativeFlowEdges(fn)
	}
	changed := true
	for changed {
		changed = false
		for _, fn := range a.functions {
			owner := nativeFunctionPathName(fn)
			for _, edge := range fn.calls {
				for requirement, dependencyPath := range paths[edge.callee] {
					changed = addNativeRequirement(paths[fn], requirement,
						append([]string{owner}, dependencyPath...)) || changed
				}
			}
			for _, edge := range flowEdges[fn] {
				for requirement, dependencyPath := range paths[edge.callee] {
					composed := append([]string{owner}, edge.hops...)
					if len(dependencyPath) != 0 && dependencyPath[0] == nativeFunctionPathName(edge.callee) {
						dependencyPath = dependencyPath[1:]
						composed = nativeAppendHop(composed, nativeCallableHop(edge.callee.node))
					}
					composed = append(composed, dependencyPath...)
					changed = addNativeRequirement(paths[fn], requirement, composed) || changed
				}
			}
			for _, provide := range fn.provides {
				if provide.callback == nil {
					continue
				}
				provided := make(map[string]bool, len(provide.provided))
				for _, row := range provide.provided {
					provided[row.name] = true
				}
				for requirement, dependencyPath := range paths[provide.callback] {
					if provided[requirement] && !nativePinBlocked(requirement) {
						continue
					}
					composed := []string{owner}
					if len(dependencyPath) != 0 && dependencyPath[0] == nativeFunctionPathName(provide.callback) {
						dependencyPath = dependencyPath[1:]
						composed = nativeAppendHop(composed, nativeCallableHop(provide.callback.node))
					}
					changed = addNativeRequirement(paths[fn], requirement,
						append(composed, dependencyPath...)) || changed
				}
			}
		}
	}
	a.nativeRequirementPaths = paths
	a.nativePins = pins
	// Function discovery order is source order over a path-sorted file list;
	// iterating that slice keeps diagnostics reproducible (map iteration does
	// not).
	for _, fn := range a.functions {
		assertion := pins[fn]
		if assertion == nil {
			continue
		}
		requirements := make([]string, 0, len(paths[fn]))
		for requirement := range paths[fn] {
			if nativePinBlocked(requirement) {
				requirements = append(requirements, requirement)
			}
		}
		sort.Strings(requirements)
		for _, requirement := range requirements {
			a.report(assertion, "SMITHERS3001", "native pin failed: "+requirement+
				" is required through "+strings.Join(paths[fn][requirement], " -> "))
		}
	}
}

func (a *rowAnalysis) nativePinArgument(node *ast.Node) bool {
	if node == nil {
		return false
	}
	argument := node
	for argument.Parent != nil && ast.IsParenthesizedExpression(argument.Parent) {
		argument = argument.Parent
	}
	if argument.Parent == nil || !ast.IsCallExpression(argument.Parent) ||
		!a.nativePinCall(argument.Parent) {
		return false
	}
	arguments := argument.Parent.AsCallExpression().Arguments
	return arguments != nil && len(arguments.Nodes) == 1 && arguments.Nodes[0] == argument
}
