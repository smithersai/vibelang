package main

// The **Effect Manifest** for the pinned Go fork — the static, sets-and-tables
// artifact a Flow publishes beside its body.
//
// `docs/DECISIONS.md` §PR-1 (pending ratification): "the compiler derives and
// publishes a static Effect Manifest per Flow — reachable Action identities,
// capability requirement row, external-input contracts, failure row, and site
// table. Sets and tables only: no control-flow edges, no branch structure, no
// execution counts."
//
// # What this unit may see, and what it may not
//
// The Manifest is derived from the CHECKED SOURCE, never from the Plan. That is
// the whole point of the artifact: after the continuation pivot there is no
// Plan to project one out of, so a Manifest that could only be projected out of
// a Plan would validate nothing.
//
// The rule is kept by construction, not by convention:
//
//   - durableEffectManifest takes the authored function node and the flow
//     identity, and NOTHING else from the lowering. It never reads
//     durableFunctionLowerer.nodes, .actions, .occurrences, or the map that
//     plan() returns.
//   - it runs a checked-source closure traversal, including helper/callback
//     implementations and implicit invocations, with no value environment,
//     control-flow edges, execution counts, or ordering.
//   - the one thing it borrows is CONTRACT DERIVATION: actionFor and
//     durableTypeDescriptor, reached through a scratch durableFunctionLowerer
//     that exists only to carry the checker. Those functions define what an
//     Action's and a signal's contract identity IS. A second transcription of
//     them would not test reachability; it would test whether two copies of one
//     hash agree, and would let a real divergence hide behind a spurious one.
//
// Both arms of a conditional are visited, without predicting which one runs.
// Unknown callees fail closed; an arbitrary `context` spelling or a library
// callback must not become a hole in the closure. This unit supports the
// currently shipped inspection artifact. It does not decide the September 6
// Plan-vs-body direction or certify that a descriptor is executable.
//
// The descriptor always includes its Manifest. `vibelangEffectManifest` only
// opts into separate .manifest.json/.plan.json companions for cross-checking.

import (
	"sort"
	"strconv"
	"strings"

	"github.com/microsoft/TypeScript/tsc/internal/ast"
	"github.com/microsoft/TypeScript/tsc/internal/checker"
	"github.com/microsoft/TypeScript/tsc/internal/scanner"
)

// requestedEffectManifest is set from the `vibelangEffectManifest` request
// option. The bridge serves exactly one request per process, so a request
// option is the whole lifetime of this value — the same reasoning
// `requestedComptimeTarget` is written under.
var requestedEffectManifest = false

// durableManifestArtifacts accumulates the extra artifacts one request asked
// for. Empty unless requestedEffectManifest is set, and read exactly once by
// compile().
var durableManifestArtifacts []artifact

type durableManifestBuilder struct {
	// contracts is a scratch lowerer used ONLY to reach actionFor and the type
	// descriptor derivation. Its nodes/actions maps are never read, and nothing
	// it accumulates reaches the Manifest.
	contracts    *durableFunctionLowerer
	owner        *lowerer
	logicalFile  string
	flowID       string
	flowVersion  int
	function     string
	actions      map[string]map[string]any
	requirements map[string]bool
	contractRows map[string]map[string]any
	failures     map[string]bool
	sites        []map[string]any
	occurrences  map[string]int
	// siteIDs is the assigned-id set the Plan lowerer keeps under the same
	// truncation (durable.go.txt, d.nodeIDs). A duplicate here is a duplicate
	// journal key inside a signed artifact, so it fails the Manifest closed
	// rather than appending a second row under one id. Values are
	// `kind@anchor`, for the message.
	siteIDs map[string]string
	// failed records the first reason the Manifest could not be stated
	// soundly. A Manifest with a hole in it is worse than no Manifest: PR-1
	// says a call the compiler cannot resolve "MUST either be rejected inside a
	// Flow body or force the Manifest to include the full effect set of the
	// callee's module; it MUST NOT silently narrow the Manifest".
	failed          string
	visit           func(*ast.Node) bool
	classes         map[*ast.Node]bool
	assignedSymbols map[*ast.Symbol]bool
}

// durableManifestAnchor is the authored `line:character` of a node, zero-based,
// matching the reference frontend's site anchors.
func (b *durableManifestBuilder) anchor(node *ast.Node) string {
	if b.owner.bodyProfile != nil {
		anchor, err := b.owner.bodyProfile.anchor(node)
		if err != nil {
			b.owner.bodyProfile.invalidProvenance = true
			b.fail(err.Error())
		}
		return anchor
	}
	start := scanner.SkipTrivia(b.owner.file.Text(), node.Pos())
	line, character := scanner.GetECMALineAndUTF16CharacterOfPosition(b.owner.file, start)
	return strconv.Itoa(line) + ":" + strconv.Itoa(int(character))
}

// recordSite appends one row to the site table.
//
// The occurrence counter exists ONLY so two syntactically distinct sites that
// hash alike cannot collide, exactly as the Plan lowerer's does. It is not an
// execution count and it never reaches a set the cross-check compares.
//
// The counter is keyed on durableDigest(identity) — the id's OWN tuple, with
// nothing added. It used to be keyed on that tuple plus the request kind:
//
//	bucket := durableCopyMap(identity)
//	bucket["requestKind"] = kind          // partitioned BY kind
//	occurrenceKey := durableDigest(bucket)
//	... "id": "src-" + durableDigest(withOccurrence)[:24]   // blind to kind
//
// A counter partitioned more finely than the value it disambiguates does not
// disambiguate it, it defeats it: two sites sharing (file, functionName,
// anchor, key) and differing only in kind each read occurrence 0 out of their
// own bucket and then mint the SAME id — under PR-2, the same journal key, in
// the artifact whose digest is meant to be signed. The reference frontend
// spelled it the same way, so no cross-backend digest comparison could see it;
// only a direct assertion can. The Plan lowerer's twin never had it
// (durable.go.txt, `occurrenceKey := durableDigest(identity)`).
//
// The old bucket was a strict refinement of this one, so every program whose
// site ids were already distinct keeps them byte-for-byte, and with them its
// Manifest digest. The answer moves only where it was a duplicate.
//
// `kind: "perform"` stays a constant in identity: a constant discriminates
// nothing either way, and splicing the real kind in would move every site id
// in every durable compile to fix nothing. The id's real discriminator is
// `anchor`, and two classified request sites cannot share one — a classified
// call's anchor is the start of its callee's leftmost token, so sharing it
// requires the inner call to BE the outer's callee prefix, and in that shape
// the outer's receiver is a call expression, which actionFor does not
// recognize. That invariant lives in the parser, far from here, which is why
// the id set below holds the line rather than resting on it: 24 hex digits is
// 96 bits of a SHA-256, and VIBE4199 is what the Plan lowerer raises on
// exactly this condition (durable.go.txt, `stable durable node id collision`).
func (b *durableManifestBuilder) recordSite(node *ast.Node, kind string, key string) {
	anchor := b.anchor(node)
	identity := map[string]any{
		"anchor":       anchor,
		"file":         b.logicalFile,
		"functionName": b.function,
		"kind":         "perform",
	}
	if key != "" {
		identity["key"] = key
	}
	occurrenceKey := durableDigest(identity)
	occurrence := b.occurrences[occurrenceKey]
	b.occurrences[occurrenceKey] = occurrence + 1
	withOccurrence := durableCopyMap(identity)
	withOccurrence["occurrence"] = occurrence
	id := "src-" + durableDigest(withOccurrence)[:24]
	if prior, taken := b.siteIDs[id]; taken {
		b.fail("stable durable node id collision " + id + ": " + kind + "@" + anchor + " and " + prior +
			" mint one site id, so one journal key would name two request sites")
		return
	}
	b.siteIDs[id] = kind + "@" + anchor
	row := map[string]any{
		"anchor": anchor,
		"id":     id,
		"kind":   kind,
	}
	if key != "" {
		row["key"] = key
	}
	b.sites = append(b.sites, row)
}

func (b *durableManifestBuilder) addContract(kind string, identity string, contractDigest string) {
	b.contractRows[kind+"#"+identity+"#"+contractDigest] = map[string]any{
		"contractDigest": contractDigest,
		"identity":       identity,
		"kind":           kind,
	}
}

func (b *durableManifestBuilder) fail(reason string) {
	if b.failed == "" {
		b.failed = reason
	}
}

// durableManifestProjectRoot is where main.go stages every authored and
// support file (virtualFileName). A declaration outside it is a default library
// or an ambient declaration the compiled project did not supply.
const durableManifestProjectRoot = "/src/"

// declaredInTheDefaultLibrary reports whether every declaration of a callee
// lives outside the compiled project.
//
// The reference twin (`poc/src/durable/effect-manifest.ts`) asks TypeScript's
// own `hasNoDefaultLib` mark instead of a path prefix. This fork's parser
// IGNORES the `no-default-lib` pragma, so the mark does not exist here and the
// staging root is the only thing that separates the two populations. The two
// spellings coincide on every program either backend can be handed: the bridge
// stages authored and support files under `/src/` and nothing else, and a
// `lib.*.d.ts` is never staged.
func (b *durableManifestBuilder) declaredInTheDefaultLibrary(symbol *ast.Symbol) bool {
	if symbol == nil || len(symbol.Declarations) == 0 {
		return false
	}
	valueDeclaration := false
	for _, declaration := range symbol.Declarations {
		// A type-only augmentation (notably the language's Error.is interface)
		// does not replace a default-library constructor's runtime value. A
		// user function/variable/namespace declaration still disqualifies it.
		if ast.IsInterfaceDeclaration(declaration) || ast.IsTypeAliasDeclaration(declaration) {
			continue
		}
		valueDeclaration = true
		file := ast.GetSourceFileOfNode(declaration)
		if file == nil || strings.HasPrefix(file.FileName(), durableManifestProjectRoot) {
			return false
		}
	}
	return valueDeclaration
}

// The authored upstream type of postfix ! is not its language value type.
// Follow only checked propagation and immutable unannotated aliases when
// selecting a payload member. Its native symbol, never the text "push"/"map",
// may establish the default-library exemption. Generated checking still owns
// assignability and callback parameter types.
func (b *durableManifestBuilder) valueType(node *ast.Node, seen map[*ast.Node]bool) *checker.Type {
	if node == nil {
		return nil
	}
	if seen[node] || len(seen) >= 128 {
		return b.owner.checker.GetTypeAtLocation(node)
	}
	seen[node] = true
	if ast.IsParenthesizedExpression(node) || ast.IsSatisfiesExpression(node) {
		return b.valueType(node.Expression(), seen)
	}
	if b.owner.analysis != nil && ast.IsNonNullExpression(node) {
		if receiver := b.owner.analysis.resultPropagations[node]; receiver != nil {
			if success, _, ok := b.owner.analysis.resultArguments(b.valueType(receiver, seen)); ok {
				return success
			}
		}
	}
	if ast.IsIdentifier(node) {
		if symbol := b.owner.checker.GetSymbolAtLocation(node); symbol != nil {
			if declaration := symbol.ValueDeclaration; declaration != nil && ast.IsVariableDeclaration(declaration) &&
				declaration.Type() == nil && ast.IsVarConst(declaration) && declaration.Initializer() != nil {
				return b.valueType(declaration.Initializer(), seen)
			}
		}
	}
	if member := b.valueMember(node, seen); member != nil {
		return b.owner.checker.GetTypeOfSymbolAtLocation(member, node)
	}
	return b.owner.checker.GetTypeAtLocation(node)
}

func (b *durableManifestBuilder) valueMember(node *ast.Node, seen map[*ast.Node]bool) *ast.Symbol {
	if node == nil {
		return nil
	}
	selection, selected := b.owner.prelude.selectMember(ast.SkipParentheses(node))
	if !selected {
		return nil
	}
	receiver := b.valueType(selection.receiver, seen)
	if receiver == nil {
		return nil
	}
	return b.owner.checker.GetPropertyOfType(b.owner.checker.GetApparentType(receiver), selection.text)
}

// unaccounted FAILS CLOSED on a call this walk did not account for.
//
// Until `MIGRATION-PLAN.md` step 11 no such call could reach here: the Plan
// lowerer refused every call it could not name as 4112 before the Manifest was
// ever consulted. That wall is withdrawn, so the Manifest is now the only thing
// between a Flow body and a silent narrowing, and `docs/DECISIONS.md` (Locked)
// says which way it has to fall: "A call whose callee the compiler cannot
// resolve is either rejected inside a Flow body or forces the Manifest to
// include the full effect set of the callee's module; it never silently narrows
// the Manifest."
//
// Resolved project implementations are followed transitively. A method merely
// named context is ordinary code: exempting its spelling used to publish an
// empty Manifest even when the method performed an Action.
func (b *durableManifestBuilder) unaccounted(call *ast.Node, callee *ast.Node, symbol *ast.Symbol) {
	if b.declaredInTheDefaultLibrary(symbol) {
		return
	}
	if b.declaredInTheDefaultLibrary(b.valueMember(callee, map[*ast.Node]bool{})) {
		return
	}
	// The language prelude's ordinary adapters introduce no request sites of
	// their own. Context reads are classified before this point; callbacks are
	// still followed below. The resolved declaration owns this exemption, not
	// an authored member's spelling (including Layer/Result lookalikes).
	if signature := b.owner.checker.GetResolvedSignature(call); signature != nil {
		if declaration := signature.Declaration(); declaration != nil {
			if file := ast.GetSourceFileOfNode(declaration); file != nil && file.FileName() == preludeVirtualName {
				return
			}
		}
	}
	// A compiler-owned combinator's effects live in the callbacks it is handed,
	// and the descent has already visited those as CHILDREN of this call. The
	// list is the prelude's own exports rather than a hand-picked subset, so a
	// new intrinsic cannot arrive and be silently classified as project code.
	for _, intrinsic := range []string{"fanOut", "sequential", "loopWhile", "waitBroadcast", "dequeue", "sleep", "waitSignal"} {
		if b.owner.durableIntrinsic(callee, intrinsic) {
			return
		}
	}
	if b.owner.analysis != nil {
		if b.owner.analysis.namesMutableBinding(callee) {
			b.fail("this Flow body calls a mutable function binding whose effects the Effect Manifest cannot state")
			return
		}
		if ast.IsCallExpression(call) && callee.Kind == ast.KindSuperKeyword {
			// A synthesized base constructor has no functionRows entry. Its
			// work is nevertheless covered by the class construction closure.
			if class := enclosingClassLike(call); class != nil {
				b.visitClassConstruction(class)
				return
			}
		}
		if ast.IsNewExpression(call) {
			// The resolved signature names the constructor body, which does not
			// contain instance field initializers. An inherited signature even
			// names the BASE class, omitting the constructed subclass entirely.
			// Follow the actual class value, not its possibly narrowed type.
			if class := b.classValue(callee, 0); class != nil {
				b.visitClassConstruction(class)
			} else {
				b.fail("this Flow body constructs a value whose initialization effects the Effect Manifest cannot state")
			}
			return
		}
		if implementation := b.owner.analysis.resolveCallee(call); implementation != nil && implementation.node.Body() != nil &&
			!implementation.file.IsDeclarationFile && implementation.file == b.owner.file {
			b.visit(implementation.node)
			return
		}
	}
	b.fail("this Flow body calls a function whose effects the Effect Manifest cannot state; a Manifest that " +
		"omitted it would claim a Flow reaches no effect it does reach")
}

// classValue proves a constructor VALUE through immutable aliases and type-only
// wrappers. A selected construct signature is not value provenance: structural
// types can collapse two classes whose field initializers have different effects.
func (b *durableManifestBuilder) classValue(expression *ast.Node, depth int) *ast.Node {
	if expression == nil || depth > 128 {
		return nil
	}
	if operand := typeOnlyWrapperOperand(expression); operand != nil {
		return b.classValue(operand, depth+1)
	}
	if ast.IsClassExpression(expression) {
		return expression
	}
	if !ast.IsIdentifier(expression) {
		return nil
	}
	symbol := b.owner.durableCanonicalSymbolAt(expression)
	if symbol == nil || b.assignedSymbols[symbol] {
		return nil
	}
	if class := classDeclarationOf(symbol); class != nil {
		return class
	}
	return b.classValue(constantInitializerOfSymbol(symbol), depth+1)
}

// Instance initializers and constructor bodies are a single construction edge.
// Base classes contribute both even when there is no authored `super` call.
// Static initializers are module/class evaluation, not repeated by `new`.
func (b *durableManifestBuilder) visitClassConstruction(class *ast.Node) {
	if b.classes[class] || b.failed != "" {
		return
	}
	b.classes[class] = true
	if file := ast.GetSourceFileOfNode(class); file == nil || file != b.owner.file || file.IsDeclarationFile {
		b.fail("this Flow body constructs a class whose initialization effects the Effect Manifest cannot state")
		return
	}
	if heritage := class.ClassLikeData().HeritageClauses; heritage != nil {
		for _, clause := range heritage.Nodes {
			if clause.AsHeritageClause().Token != ast.KindExtendsKeyword || clause.AsHeritageClause().Types == nil {
				continue
			}
			for _, typeNode := range clause.AsHeritageClause().Types.Nodes {
				expression := typeNode.Expression()
				if base := b.classValue(expression, 0); base != nil {
					b.visitClassConstruction(base)
				} else if !b.declaredInTheDefaultLibrary(b.owner.durableCanonicalSymbolAt(expression)) {
					b.fail("this Flow body constructs a class whose base initialization effects the Effect Manifest cannot state")
				}
			}
		}
	}
	for _, member := range class.Members() {
		if ast.IsPropertyDeclaration(member) && !ast.HasStaticModifier(member) && member.Initializer() != nil {
			b.visit(member.Initializer())
		}
		if ast.IsConstructorDeclaration(member) && member.Body() != nil {
			b.visit(member)
		}
	}
}

// classify records what one call expression contributes to the Manifest.
func (b *durableManifestBuilder) classify(node *ast.Node) {
	if b.classifyBoundChild(node) {
		return
	}
	call := node.AsCallExpression()
	if b.owner.analysis != nil {
		if capability, recognized := b.owner.analysis.contextRequirement(node); recognized {
			key := b.capabilityIdentity(capability.symbol)
			if capability.kind != contextReceiverCapability || key == "" {
				b.fail("durable capability read has no statically bound nominal identity")
				return
			}
			b.recordSite(node, "get", key)
			return
		}
	}
	if ast.IsPropertyAccessExpression(call.Expression) && b.owner.durableIntrinsic(call.Expression, "run") {
		access := call.Expression.AsPropertyAccessExpression()
		action, recognized := b.contracts.actionFor(access.Expression)
		if !recognized {
			b.unaccounted(node, call.Expression, b.owner.durableCanonicalSymbolAt(access.Expression))
			return
		}
		// FAIL CLOSED on an Action whose durable contract cannot be derived —
		// an identity collision in its failure channel, or a signature outside
		// the derivable subset. There is no descriptor to state, and dropping
		// the site would publish "this Flow reaches no such Action" about a
		// Flow that performs it. The reference frontend refuses the same
		// programs for the same reason.
		if action.invalid != "" || action.collision != nil {
			b.fail("durable Action has no derivable durable contract, so its effect cannot be stated soundly")
			return
		}
		// The three schemas ride the row as of `MIGRATION-PLAN.md` step 12, so
		// the corpus observations of `successSchema`/`errorSchema` can move off
		// `Flow.plan.actions` and onto `Flow.manifest.actions`. They cost the
		// Manifest digest no discriminating power: `contractDigest` is already
		// `digest({id, version, inputSchema, successSchema, errorSchema})`
		// (`durableActionContractDescriptor`), so two Flows whose Actions differ
		// in any schema already differed in this row before the schemas were in
		// it. Read straight off `action.descriptor` rather than re-derived: a
		// second derivation would be a second answer to compare, not the
		// contract the digest was taken over.
		if prior := b.actions[action.id]; prior != nil && prior["contractDigest"] != action.descriptor["contractDigest"] {
			b.fail("Action id " + action.id + " resolves to incompatible durable contracts")
			return
		}
		b.actions[action.id] = map[string]any{
			"contractDigest": action.descriptor["contractDigest"],
			"errorSchema":    action.descriptor["errorSchema"],
			"id":             action.id,
			"inputSchema":    action.descriptor["inputSchema"],
			"successSchema":  action.descriptor["successSchema"],
			"version":        action.descriptor["version"],
		}
		b.requirements[action.id] = true
		for _, identity := range durableManifestFailureRow(b.contracts, action.failureType) {
			b.failures[identity] = true
		}
		for _, identity := range action.failureIDs {
			b.failures[identity] = true
		}
		b.recordSite(node, "perform", action.id)
		return
	}
	if b.owner.durableIntrinsic(call.Expression, "sleep") {
		b.recordSite(node, "sleep", "")
		return
	}
	inputKind, what := "", ""
	for _, input := range []struct{ name, kind, what string }{
		{"waitSignal", "signal", "signal"},
		{"waitBroadcast", "broadcast", "broadcast signal"},
		{"dequeue", "queue", "queue"},
	} {
		if b.owner.durableIntrinsic(call.Expression, input.name) {
			inputKind, what = input.kind, input.what
			break
		}
	}
	if inputKind != "" {
		if call.Arguments == nil || len(call.Arguments.Nodes) != 1 || call.TypeArguments == nil || len(call.TypeArguments.Nodes) != 1 {
			b.fail("durable " + what + " requires one explicit payload type and one static identity")
			return
		}
		identityExpression := ast.SkipParentheses(call.Arguments.Nodes[0])
		if !ast.IsStringLiteralLike(identityExpression) {
			b.fail("durable " + what + " identity must be a string literal, not a runtime value")
			return
		}
		signalID := identityExpression.Text()
		typeNode := call.TypeArguments.Nodes[0]
		descriptor, err := b.contracts.durableTypeDescriptor(
			b.owner.checker.GetTypeFromTypeNode(typeNode), typeNode, make(map[checker.TypeId]bool), false)
		if err != nil {
			b.fail("durable " + what + " payload is not structurally encodable: " + err.Error())
			return
		}
		payloadSchema := durableStructuralSchema("input", descriptor)
		contract := map[string]any{
			"payloadSchema": payloadSchema,
			"signalId":      signalID,
		}
		if inputKind == "broadcast" {
			contract["delivery"] = "broadcast"
		} else if inputKind == "queue" {
			contract = map[string]any{"queueId": signalID, "itemSchema": payloadSchema}
		}
		b.addContract(inputKind, signalID, durableDigest(contract))
		b.recordSite(node, inputKind, signalID)
		return
	}
	b.unaccounted(node, call.Expression, b.owner.durableCanonicalSymbolAt(call.Expression))
}

func (b *durableManifestBuilder) capabilityIdentity(symbol *ast.Symbol) string {
	declaration := classDeclarationOf(symbol)
	if declaration == nil || declaration.Name() == nil || b.owner.analysis == nil || b.owner.analysis.project == nil {
		return ""
	}
	file := ast.GetSourceFileOfNode(declaration)
	if file == nil {
		return ""
	}
	logical := b.owner.analysis.project.logicalByVirtual[file.FileName()]
	if file == b.owner.file {
		logical = b.logicalFile
	}
	if logical == "" {
		return ""
	}
	return logical + "#" + declaration.Name().Text()
}

// A selected overload names a signature, not its implementation. Resolve the
// unique checked body on the same symbol, as resolveCallee does for calls.
func (b *durableManifestBuilder) callbackImplementation(declaration *ast.Node) *functionRows {
	if declaration == nil || b.owner.analysis == nil {
		return nil
	}
	if fn := b.owner.analysis.byNode[declaration]; fn != nil {
		return fn
	}
	if (ast.IsFunctionDeclaration(declaration) || ast.IsMethodDeclaration(declaration)) && declaration.Name() != nil {
		return b.owner.analysis.bySymbol[b.owner.analysis.symbolAt(declaration.Name())]
	}
	return nil
}

// durableManifestFailureRow enumerates the durable failure identities one
// Action's declared failure channel carries.
//
// The same walk durableFailureIdentityCollides makes, reporting the identities
// instead of the collision. Derived from the Action's declared failure TYPE, not
// read back out of the contract's error schema, even though that schema now
// carries the same identities: PR-1 wants the Manifest to be an independent
// derivation from the checked source, and nothing in the cross-check compares
// the two artifacts' failure rows against each other.
func durableManifestFailureRow(d *durableFunctionLowerer, failure *checker.Type) []string {
	if failure == nil || d == nil {
		return nil
	}
	constituents := []*checker.Type{failure}
	if failure.IsUnion() {
		constituents = failure.Types()
	}
	identities := make([]string, 0, len(constituents))
	for _, constituent := range constituents {
		if d.durableTypeSymbol(constituent) == d.durableErrorSymbol() {
			identities = append(identities, "javascript:Error@1")
			continue
		}
		symbol := constituent.Symbol()
		if symbol == nil {
			continue
		}
		declaration := classDeclarationOf(d.owner.checker.GetMergedSymbol(symbol))
		if declaration == nil {
			continue
		}
		name := declaration.Name()
		if name == nil || !ast.IsIdentifier(name) {
			continue
		}
		identities = append(identities, durableFailureIdentity(
			d.schemaLogicalFile(ast.GetSourceFileOfNode(declaration)), name.Text()))
	}
	return identities
}

// durableEffectManifest derives one Flow's Effect Manifest from the authored
// function, or reports why it could not.
func durableEffectManifest(
	owner *lowerer,
	logicalFile string,
	flowID string,
	flowVersion int,
	function string,
	body *ast.Node,
) (map[string]any, string) {
	schemaLogicalName := ""
	if owner.bodyProfile != nil {
		schemaLogicalName = owner.bodyProfile.logicalFile
	}
	return durableEffectManifestWithIdentity(owner, logicalFile, flowID, flowVersion, function, body, schemaLogicalName)
}

// The standalone source API stages one input at a private address. Its
// schema identity remains the authored identity, without pretending that the
// source uses the executable-body emission profile or deriving from a Plan.
func durableEffectManifestWithIdentity(owner *lowerer, logicalFile, flowID string, flowVersion int, function string,
	body *ast.Node, schemaLogicalName string,
) (map[string]any, string) {
	builder := &durableManifestBuilder{
		contracts: &durableFunctionLowerer{
			owner:       owner,
			flowID:      flowID,
			flowVersion: flowVersion,
			function:    function,
			logicalFile: logicalFile,
			values:      make(map[*ast.Symbol]durableValue),
			actions:     make(map[string]durableAction),
			occurrences: make(map[string]int),
			nodeIDs:     make(map[string]bool),
			signalIDs:   make(map[string]bool),
		},
		owner:           owner,
		logicalFile:     logicalFile,
		flowID:          flowID,
		flowVersion:     flowVersion,
		function:        function,
		actions:         make(map[string]map[string]any),
		requirements:    make(map[string]bool),
		contractRows:    make(map[string]map[string]any),
		failures:        make(map[string]bool),
		occurrences:     make(map[string]int),
		siteIDs:         make(map[string]string),
		classes:         make(map[*ast.Node]bool),
		assignedSymbols: make(map[*ast.Symbol]bool),
	}
	builder.contracts.activeNodes = &builder.contracts.nodes
	builder.contracts.schemaLogicalName = schemaLogicalName
	var collectAssignments func(*ast.Node) bool
	collectAssignments = func(node *ast.Node) bool {
		if ast.IsIdentifier(node) && ast.IsAssignmentTarget(node) {
			if symbol := owner.durableCanonicalSymbolAt(node); symbol != nil {
				builder.assignedSymbols[symbol] = true
			}
		}
		return node.ForEachChild(collectAssignments)
	}
	owner.file.AsNode().ForEachChild(collectAssignments)
	// A lexical Layer discharges its capabilities, not the get-site itself.
	// Publish the fixed-point root requirement row and record all reachable
	// reads separately; adding every read to requirements would lose discharge.
	if owner.analysis != nil {
		if root := owner.analysis.byNode[body]; root != nil {
			for _, row := range root.requirements {
				if row.symbol != nil && owner.analysis.extendsContext(row.symbol) {
					key := builder.capabilityIdentity(row.symbol)
					if key == "" {
						builder.fail("durable capability requirement has no nominal identity")
					} else {
						builder.requirements[key] = true
					}
				}
			}
		}
	}

	visited := map[*ast.Node]bool{}
	var visit func(*ast.Node) bool
	visit = func(node *ast.Node) bool {
		if visited[node] || builder.failed != "" {
			return false
		}
		visited[node] = true
		if ast.IsCallExpression(node) {
			builder.classify(node)
		} else if isInvocationExpression(node) {
			var callee *ast.Node
			if ast.IsTaggedTemplateExpression(node) {
				callee = node.AsTaggedTemplateExpression().Tag
			} else {
				callee = node.AsNewExpression().Expression
			}
			builder.unaccounted(node, callee, owner.durableCanonicalSymbolAt(callee))
		}
		if owner.analysis != nil {
			for _, fn := range owner.analysis.accessorInvocations(node) {
				if fn.file == owner.file && fn.node.Body() != nil {
					visit(fn.node)
				} else {
					builder.fail("this Flow body invokes an accessor whose effects the Effect Manifest cannot state")
				}
			}
			for _, fn := range owner.analysis.implicitInvocations(node) {
				if fn.file == owner.file && fn.node.Body() != nil {
					visit(fn.node)
				} else {
					builder.fail("this Flow body implicitly invokes a function whose effects the Effect Manifest cannot state")
				}
			}
			// A callback carried by a checked function value is as reachable as
			// an inline callback. The visited set closes recursive helpers and
			// gives their request sites one row, regardless of call count.
			for _, argument := range invocationArguments(node) {
				for _, signature := range owner.checker.GetSignaturesOfType(owner.checker.GetTypeAtLocation(argument), checker.SignatureKindCall) {
					declaration := signature.Declaration()
					if fn := builder.callbackImplementation(declaration); fn != nil && !owner.analysis.namesMutableBinding(argument) && fn.file == owner.file && fn.node.Body() != nil {
						visit(fn.node)
					} else if declaration == nil || ast.GetSourceFileOfNode(declaration) == nil ||
						strings.HasPrefix(ast.GetSourceFileOfNode(declaration).FileName(), durableManifestProjectRoot) && ast.GetSourceFileOfNode(declaration).FileName() != preludeVirtualName {
						builder.fail("this Flow body passes a callback whose effects the Effect Manifest cannot state")
					}
				}
			}
		}
		node.ForEachChild(visit)
		return false
	}
	builder.visit = visit
	if bodyNode := body.Body(); bodyNode != nil {
		visit(bodyNode)
	}
	if builder.failed != "" {
		return nil, builder.failed
	}

	actionIDs := make([]string, 0, len(builder.actions))
	for id := range builder.actions {
		actionIDs = append(actionIDs, id)
	}
	sortUTF16(actionIDs)
	actions := make([]any, 0, len(actionIDs))
	requirements := make([]any, 0, len(builder.requirements))
	for _, id := range actionIDs {
		actions = append(actions, builder.actions[id])
	}
	requirementIDs := make([]string, 0, len(builder.requirements))
	for id := range builder.requirements {
		requirementIDs = append(requirementIDs, id)
	}
	sortUTF16(requirementIDs)
	for _, id := range requirementIDs {
		requirements = append(requirements, id)
	}

	contractKeys := make([]string, 0, len(builder.contractRows))
	for key := range builder.contractRows {
		contractKeys = append(contractKeys, key)
	}
	sortUTF16(contractKeys)
	contracts := make([]any, 0, len(contractKeys))
	for _, key := range contractKeys {
		contracts = append(contracts, builder.contractRows[key])
	}

	failureIDs := make([]string, 0, len(builder.failures))
	for identity := range builder.failures {
		failureIDs = append(failureIDs, identity)
	}
	sortUTF16(failureIDs)
	failures := make([]any, 0, len(failureIDs))
	for _, identity := range failureIDs {
		failures = append(failures, identity)
	}

	// Byte order is CORRECT here and is left alone deliberately, even though it
	// sits three lines under a `sortUTF16` and does reach the Manifest digest:
	// `recordSite` mints every site id as `src-` plus 24 hex digits, so these
	// are provably ASCII and the two orders agree. The authored strings in this
	// builder — Action ids, failure identities — are the ones sorted above.
	sort.SliceStable(builder.sites, func(left, right int) bool {
		return builder.sites[left]["id"].(string) < builder.sites[right]["id"].(string)
	})
	sites := make([]any, 0, len(builder.sites))
	for _, site := range builder.sites {
		sites = append(sites, site)
	}

	semantic := map[string]any{
		"actions":         actions,
		"contracts":       contracts,
		"failures":        failures,
		"flowId":          flowID,
		"flowVersion":     flowVersion,
		"manifestVersion": 1,
		"requirements":    requirements,
		"sites":           sites,
	}
	manifest := durableCopyMap(semantic)
	manifest["digest"] = durableDigest(semantic)
	return manifest, ""
}
