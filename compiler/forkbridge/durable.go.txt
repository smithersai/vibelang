//
// This unit intentionally has no evaluator. It never calls the source
// function, an Action implementation, or a compiler intrinsic. The only

// durableFailureCollision is the structured reason behind one specific
// underivable contract: two DIFFERENT Error class declarations in one Action's
// failure channel that mint a single durable failure identity.
//
// It is a port of the reference frontend's `DurableFailureIdentityCollision`
// (poc/src/durable/schema.ts), which travels out of a derivation failure for
// exactly the same purpose: the position a durable diagnostic can report at is
// the authored `run` call, far from where the collision was detected, so the
// two class names have to be carried there as data or the call site can only
// say something generic.
type durableFailureCollision struct {
	identity string
	// The two colliding class names, SORTED. Sorted rather than
	// claimant-then-collider so the sentence built from them does not depend on
	// the order a checker enumerated the union in; the reference sorts for the
	// same reason, and that is what lets both backends emit one text.
	names [2]string
}

// durableFailureIdentityCollisionMessage is the sentence SMITHERS4124 carries,
// byte-for-byte the reference's `durableFailureIdentityCollisionMessage`
// (poc/src/durable/source-compiler.ts). Kept identical on purpose: a conformance
// case may declare `messageContains`, and a case declares it once for both
// backends.
func durableFailureIdentityCollisionMessage(collision *durableFailureCollision) string {
	return "Error classes " + collision.names[0] + " and " + collision.names[1] +
		" in this Action's declared failure channel share one durable failure identity " + collision.identity +
		"; a decoder on the far side of a persistence boundary selects a handler by identity, so these two" +
		" classes cannot be told apart on the wire — rename one of them or declare it in its own module"
}

type durableAction struct {
	id         string
	descriptor map[string]any
	inputType  *checker.Type
	invalid    string
	// collision marks a declaration that IS a compiler-bound Action but whose
	// durable contract cannot be derived, because two Error classes in its
	// failure channel mint one durable identity. It carries the reason rather
	// than a bare flag so the authored `run` call can name both classes.
	//
	// It used to be `underivable bool`, and the call site then reported the
	// reference's generic unsupported-call sentence — "higher-order and dynamic
	// calls are unavailable in durable source lowering" — which is FALSE of the
	// program that reaches it: `X.run({ ... })` is an ordinary compiler-bound
	// Action call. Both backends now report SMITHERS4124 at the same position.
	// See lowerAction.
	collision *durableFailureCollision
	// failureType is the Action's declared failure channel, kept so the Effect
	// Manifest can state a failure row from the same type the contract's error
	// schema is derived from, rather than from the derived schema — the Manifest
	// is meant to be an independent derivation, and projecting it out of the
	// Plan would validate nothing about a world with no Plan in it.
	failureType *checker.Type
}

func durableCode(suffix string) string { return durableDiagnosticPrefix + suffix }

func (l *lowerer) durableCanonicalSymbolAt(node *ast.Node) *ast.Symbol {
	if node == nil {
		return nil
	}
	symbol := l.checker.GetSymbolAtLocation(node)
	if node.Parent != nil && ast.IsShorthandPropertyAssignment(node.Parent) {
		if value := l.checker.GetShorthandAssignmentValueSymbol(node.Parent); value != nil {
			symbol = value
		}
	}
	if symbol == nil {
		return nil
	}
	if symbol.Flags&ast.SymbolFlagsAlias != 0 {
		// A type-only alias is never a runtime intrinsic reference, even when
		// the checker can resolve its target declaration.
		for _, declaration := range symbol.Declarations {
			if ast.IsTypeOnlyImportOrExportDeclaration(declaration) {
				return nil
			}
		}
		if aliased := l.checker.GetAliasedSymbol(symbol); aliased != nil {
			symbol = aliased
		}
	}
	return l.checker.GetMergedSymbol(symbol)
}

// durableIntrinsic proves identity by the resolved declaration in the
// compiler-owned prelude. The use-site spelling is not inspected: renamed
// imports and namespace-property references resolve to the same symbol, while
// an authored same-named function resolves elsewhere and is left ordinary.
func (l *lowerer) durableIntrinsic(node *ast.Node, exported string) bool {
	symbol := l.durableCanonicalSymbolAt(ast.SkipParentheses(node))
	if symbol == nil {
		return false
	}
	for _, declaration := range symbol.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && file.FileName() == durableVirtualName && declarationName(declaration) == exported {
			return true
		}
	}
	return false
}

func durableResolvedModule(typeChecker *checker.Checker, specifier *ast.Node) bool {
	if typeChecker == nil || specifier == nil {
		return false
	}
	symbol := typeChecker.ResolveExternalModuleName(specifier)
	if symbol == nil {
		return false
	}
	symbol = typeChecker.GetMergedSymbol(symbol)
	for _, declaration := range symbol.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && file.FileName() == durableVirtualName {
			return true
		}
	}
	return false
}

func durablePreludeSpecifierFor(virtualName string) string {
	directory := path.Dir(virtualName)
	rest := strings.Trim(strings.TrimPrefix(directory, "/src"), "/")
	if rest == "" {
		return "./" + durableModuleBase
	}
	return strings.Repeat("../", strings.Count(rest, "/")+1) + durableModuleBase
}

func durableLogicalFile(file *ast.SourceFile) string {
	if file == nil {
		return "durable-source.sm"
	}
	return strings.TrimPrefix(path.Clean(file.FileName()), "/src/")
}

// durableMaxIdentityLength bounds a durable failure identity in UTF-16 code
// units, exactly as the reference frontend's MAX_IDENTITY_LENGTH does.
const durableMaxIdentityLength = 256

// durableFailureIdentity mints the identity a durable CONTRACT carries for one
// nominal Error class. It is a port of the reference frontend's `stableIdentity`
// (poc/src/durable/schema.ts): `smithers:<logical file>#<class name>@1` with
// every character outside [A-Za-z0-9._/@:+-] rewritten to `_`, falling back to
// an injective digest of (file, name) once the result is over the bound.
//
// It is deliberately NOT `stableErrorIdentity`, the RUNTIME spelling
// `smithers:<file>:<Name>` this bridge emits into `__smithersRegisterError`.
// The two spellings coexist on purpose and must not be unified here: `:`
// survives normalization and `#` does not, so the runtime spelling is injective
// over (file, name) while the contract spelling blurs the separator into the
// file and the name. That is precisely why a contract identity can collide
// while the runtime identity beside it does not, and why a guard on the runtime
// spelling would not see the collision at all.
//
// This IS emitted: `durableErrorDescriptor` puts it in every structural error
// variant, so it is hashed into the Action's `errorSchema` digest, its
// `contractDigest`, the Flow's `flowSchemas.error`, and the Plan digest. It was
// derived-and-discarded until 2026-08-28, when the error channel stopped being a
// stub, so the normalization and the bound below are now contract bytes rather
// than a refusal test — which is why they are defined on UTF-16 code units.
//
// The normalization and the bound are defined on UTF-16 code units because the
// reference is JavaScript string surgery: `String.prototype.replace` with a
// non-unicode character class rewrites each unit of a surrogate pair
// separately, and `String.prototype.length` counts units. A rune-wise port
// would disagree with the reference on exactly the inputs this exists to
// classify. The normalized units can never contain a surrogate — every
// non-ASCII unit is replaced by `_` — so decoding back is lossless.
func durableFailureIdentity(sourceName string, name string) string {
	units := utf16.Encode([]rune("smithers:" + sourceName + "#" + name + "@1"))
	for index, unit := range units {
		if !isStableIdentityPathUnit(unit) {
			units[index] = '_'
		}
	}
	if len(units) > durableMaxIdentityLength {
		return "smithers:error/" + durableDigest(map[string]any{"fileName": sourceName, "name": name})[:48] + "@1"
	}
	return string(utf16.Decode(units))
}

// durableFailureIdentityCollides answers the collision in one Action's checked
// failure channel — the identity and the two DIFFERENT Error class declarations
// that minted it — or nil when the channel is describable.
//
// It returns the pair rather than a bool because the diagnostic names both
// classes. Naming them is the whole reason SMITHERS4124 exists: with only a
// bool, the call site had nothing to say and reported the reference's generic
// unsupported-call sentence instead.
//
// The identity is a function of (logical source file, class name) alone, so it
// is not injective over class declarations. Two spellings of that are reachable
// from authored `.sm`: two same-named classes in one logical file, which this
// bridge already refuses earlier and module-wide as SMITHERS1150, and two
// DIFFERENT names that normalize together — `$Failed` and `_Failed` are the
// smallest such pair, and the only member of the class that reaches a running
// wire. A failure channel holding a collision is not describable on the far side
// of a persistence boundary: a decoder selects a handler by identity, so two
// Error classes arriving under one identity is a forgeable key, and the emitted
// worker bundle maps a returned failure by class name, so two variants sharing a
// name are unmappable as well.
//
// Keyed on the class DECLARATION, not on the identity string, so the rule is
// exactly "two different classes". That is not what keeps `Failed | Failed`, a
// type alias, or an import-equals alias compiling — the checker collapses those
// to one union constituent, so the loop only sees the class once — it is here so
// that a channel naming one class twice can never be refused.
//
// Constituents that do not resolve to a class declaration are skipped rather
// than refused: the built-in `Error` is an interface plus a var in lib.d.ts, and
// `Result<T, Error>` must keep compiling.
func (d *durableFunctionLowerer) durableFailureIdentityCollides(failure *checker.Type) *durableFailureCollision {
	if failure == nil {
		return nil
	}
	constituents := []*checker.Type{failure}
	if failure.IsUnion() {
		constituents = failure.Types()
	}
	type claim struct {
		declaration *ast.Node
		name        string
	}
	claimed := make(map[string]claim, len(constituents))
	for _, constituent := range constituents {
		symbol := constituent.Symbol()
		if symbol == nil {
			continue
		}
		declaration := classDeclarationOf(d.owner.checker.GetMergedSymbol(symbol))
		if declaration == nil {
			continue
		}
		name := declaration.Name()
		if name == nil || !ast.IsIdentifier(name) {
			continue
		}
		identity := durableFailureIdentity(durableLogicalFile(ast.GetSourceFileOfNode(declaration)), name.Text())
		claimant, claimedBefore := claimed[identity]
		if !claimedBefore {
			claimed[identity] = claim{declaration: declaration, name: name.Text()}
			continue
		}
		if claimant.declaration != declaration {
			// Authored class names. The reference sorts the same pair with
			// `compareText` (poc/src/durable/schema.ts:454) before building the
			// SMITHERS4124 sentence, so byte order here would name the two
			// classes in the other order for a program whose Error classes are
			// spelled with characters straddling the surrogate boundary — a
			// legal identifier spelling, since ID_Start admits astral letters.
			names := []string{claimant.name, name.Text()}
			sortUTF16(names)
			return &durableFailureCollision{identity: identity, names: [2]string{names[0], names[1]}}
		}
	}
	return nil
}

// durableMaxUnionVariants bounds an Error union, exactly as the reference
// frontend's MAX_UNION_VARIANTS does.
const durableMaxUnionVariants = 128

// durableMaxObjectFields bounds one Error payload's field count, exactly as the
// reference frontend's MAX_OBJECT_FIELDS does.
const durableMaxObjectFields = 1024

// durableErrorSymbol is the compiler-owned `Error` TYPE: the merged symbol of
// the prelude's own `Error` interface declaration, which the bridge already
// resolves once and every other Error rule here is keyed on.
//
// NOT `ResolveName("Error", <authored location>, ...)`. That spelling reads the
// name as an author's module sees it, so a module declaring its own `Error`
// would move the anchor of "does this class extend Error" and of "is this the
// weakened built-in channel" — the two questions that decide whether a failure
// channel is described structurally or left as the legacy stub, and therefore
// what the contract digest is. The reference is keyed on a compiler-owned
// declaration for the same reason: its durable source compiler takes
// `errorSymbol` from the `Result<A, E extends Error>` constraint in its own
// virtual `result.d.ts`, never from a name looked up at the authored site.
func (d *durableFunctionLowerer) durableErrorSymbol() *ast.Symbol {
	if d.owner.prelude == nil {
		return nil
	}
	return d.owner.prelude.error
}

// durableTypeSymbol is the reference's `typeReferenceSymbol(type, checker) ??
// canonicalSymbol(checker, type.symbol)`: the declaration a type names.
func (d *durableFunctionLowerer) durableTypeSymbol(t *checker.Type) *ast.Symbol {
	if t == nil {
		return nil
	}
	return d.owner.checker.GetMergedSymbol(t.Symbol())
}

// durableErrorSchemaFor derives the error schema one Action contract carries, or
// the legacy weakened schema for the one failure channel the reference also
// declines to describe.
//
// THE WEAKENING IS NOT A FALLBACK. `Result<Success, Error>` — the built-in
// `Error` and nothing else — carries no nominal payload either implementation
// can describe, and the reference's `weakenUnderivableErrors` reaches
// `derivedSchema("error")` for exactly that channel and no other. Every other
// underivable channel is refused on both sides. Widening this to "derivation
// threw, so emit the stub" would be the fail-open that produced this bridge's
// original defect: a stub schema is a DIFFERENT digest, so a silent fallback
// mints a contract identity the reference never mints and the two backends
// disagree again with no diagnostic anywhere.
func (d *durableFunctionLowerer) durableErrorSchemaFor(failure *checker.Type) (map[string]any, error) {
	errorSymbol := d.durableErrorSymbol()
	if errorSymbol == nil {
		return nil, fmt.Errorf("compiler could not resolve the built-in Error type")
	}
	if durableIsWhollyBuiltInErrorChannel(d, failure, errorSymbol) {
		return durableLegacySchema("error"), nil
	}
	descriptor, err := d.durableErrorDescriptor(failure, errorSymbol)
	if err != nil {
		return nil, err
	}
	return durableStructuralSchema("error", descriptor), nil
}

// durableIsWhollyBuiltInErrorChannel is the reference's
// `isWhollyBuiltInErrorChannel`: every constituent resolves to the built-in
// `Error` symbol itself, rather than to a class declaration extending it.
func durableIsWhollyBuiltInErrorChannel(d *durableFunctionLowerer, failure *checker.Type, errorSymbol *ast.Symbol) bool {
	if failure == nil {
		return false
	}
	constituents := []*checker.Type{failure}
	if failure.IsUnion() {
		constituents = failure.Types()
	}
	if len(constituents) == 0 {
		return false
	}
	for _, constituent := range constituents {
		if d.durableTypeSymbol(constituent) != errorSymbol {
			return false
		}
	}
	return true
}

// durableErrorDescriptor is the reference's `DescriptorBuilder.error`: one
// `{kind:"error", identity, name, payload}` variant per declared Error class,
// deduplicated and ordered canonically when the channel is a union.
//
// The union is joined with `durableCanonicalUnion`, which is the reference's own
// rule — dedupe by canonical JSON, order by UTF-16 code unit, collapse a
// single survivor to the bare variant. Flattening is a no-op here (a variant is
// always an `error` record, never a nested union) and is left to that one
// function rather than duplicated, so the ordering rule has one home.
func (d *durableFunctionLowerer) durableErrorDescriptor(
	failure *checker.Type,
	errorSymbol *ast.Symbol,
) (any, error) {
	if failure == nil {
		return nil, fmt.Errorf("checker returned no failure type")
	}
	if failure.IsUnion() {
		members := failure.Types()
		if len(members) > durableMaxUnionVariants {
			return nil, fmt.Errorf("failure channel exceeds the Error union variant limit")
		}
		variants := make([]any, 0, len(members))
		for _, member := range members {
			variant, err := d.durableErrorDescriptor(member, errorSymbol)
			if err != nil {
				return nil, err
			}
			variants = append(variants, variant)
		}
		if len(variants) == 0 {
			return nil, fmt.Errorf("failure channel has no Error variants")
		}
		return durableCanonicalUnion(variants), nil
	}
	if failure.Flags()&(checker.TypeFlagsAny|checker.TypeFlagsUnknown|checker.TypeFlagsTypeParameter) != 0 {
		return nil, fmt.Errorf("failure channel must name a concrete Error class or union")
	}
	declaration := classDeclarationOf(d.durableTypeSymbol(failure))
	if declaration == nil {
		return nil, fmt.Errorf("failure channel must be an ordinary named class extending Error")
	}
	name := declaration.Name()
	if name == nil || !ast.IsIdentifier(name) {
		return nil, fmt.Errorf("failure channel must be an ordinary named class extending Error")
	}
	if !d.durableExtendsError(declaration, errorSymbol, make(map[*ast.Node]bool)) {
		return nil, fmt.Errorf("failure channel class %s does not extend Error", name.Text())
	}
	payload, err := d.durableErrorPayload(declaration, errorSymbol, make(map[*ast.Node]bool))
	if err != nil {
		return nil, err
	}
	return map[string]any{
		"identity": durableFailureIdentity(durableLogicalFile(ast.GetSourceFileOfNode(declaration)), name.Text()),
		"kind":     "error",
		"name":     name.Text(),
		"payload":  payload,
	}, nil
}

// durableExtendsError answers the reference's `extendsError`: does this class
// reach the built-in `Error` through its extends chain.
func (d *durableFunctionLowerer) durableExtendsError(
	declaration *ast.Node,
	errorSymbol *ast.Symbol,
	visited map[*ast.Node]bool,
) bool {
	if declaration == nil || visited[declaration] {
		return false
	}
	visited[declaration] = true
	clauses := declaration.ClassLikeData().HeritageClauses
	if clauses == nil {
		return false
	}
	for _, clauseNode := range clauses.Nodes {
		clause := clauseNode.AsHeritageClause()
		if clause.Token != ast.KindExtendsKeyword || clause.Types == nil {
			continue
		}
		for _, base := range clause.Types.Nodes {
			symbol := d.durableTypeSymbol(d.owner.checker.GetTypeAtLocation(base))
			if symbol != nil && symbol == errorSymbol {
				return true
			}
			if baseClass := classDeclarationOf(symbol); baseClass != nil &&
				d.durableExtendsError(baseClass, errorSymbol, visited) {
				return true
			}
		}
	}
	return false
}

// durableErrorPayload is the reference's `errorPayload`: the object descriptor
// one Error class carries on the wire, assembled from its base classes' payload
// fields, its own property declarations, and its constructor's parameter
// properties — the three spellings that survive `structuredClone`.
//
// Base fields are merged FIRST so a derived class's own declaration of the same
// name wins, which is the reference's `Map#set` order. The final order is UTF-16
// on the field name, not insertion order, so the descriptor — and therefore the
// contract digest — cannot depend on how a class happened to be written.
func (d *durableFunctionLowerer) durableErrorPayload(
	declaration *ast.Node,
	errorSymbol *ast.Symbol,
	visited map[*ast.Node]bool,
) (any, error) {
	if visited[declaration] {
		return nil, fmt.Errorf("Error inheritance chain is recursive")
	}
	visited[declaration] = true
	defer delete(visited, declaration)

	fields := make(map[string]map[string]any)
	if clauses := declaration.ClassLikeData().HeritageClauses; clauses != nil {
		for _, clauseNode := range clauses.Nodes {
			clause := clauseNode.AsHeritageClause()
			if clause.Token != ast.KindExtendsKeyword || clause.Types == nil {
				continue
			}
			for _, base := range clause.Types.Nodes {
				symbol := d.durableTypeSymbol(d.owner.checker.GetTypeAtLocation(base))
				if symbol != nil && symbol == errorSymbol {
					continue
				}
				baseDeclaration := classDeclarationOf(symbol)
				if baseDeclaration == nil {
					return nil, fmt.Errorf("Error class has an unsupported base class")
				}
				inherited, err := d.durableErrorPayload(baseDeclaration, errorSymbol, visited)
				if err != nil {
					return nil, err
				}
				record, _ := inherited.(map[string]any)
				items, _ := record["fields"].([]any)
				for _, item := range items {
					field, _ := item.(map[string]any)
					if field == nil {
						continue
					}
					fieldName, _ := field["name"].(string)
					fields[fieldName] = field
				}
			}
		}
	}

	add := func(fieldName string, optional bool, typeNode *ast.Node, location *ast.Node) error {
		value, err := d.durableTypeDescriptor(
			d.owner.checker.GetTypeFromTypeNode(typeNode), location, make(map[checker.TypeId]bool), optional)
		if err != nil {
			return fmt.Errorf("Error payload field %s: %w", fieldName, err)
		}
		if value == nil {
			return fmt.Errorf("Error payload field %s has no encodable value type", fieldName)
		}
		fields[fieldName] = map[string]any{"name": fieldName, "optional": optional, "value": value}
		return nil
	}

	for _, member := range declaration.ClassLikeData().Members.Nodes {
		switch {
		case ast.IsPropertyDeclaration(member):
			if ast.HasSyntacticModifier(member, ast.ModifierFlagsStatic) {
				continue
			}
			memberName := member.Name()
			if memberName != nil && ast.IsPrivateIdentifier(memberName) {
				return nil, fmt.Errorf("Error class has a private payload field")
			}
			if ast.HasSyntacticModifier(member, ast.ModifierFlagsPrivate|ast.ModifierFlagsProtected) {
				return nil, fmt.Errorf("Error class has a private/protected payload field")
			}
			property := member.AsPropertyDeclaration()
			if property.Type == nil || memberName == nil ||
				!(ast.IsIdentifier(memberName) || ast.IsStringLiteral(memberName) || ast.IsNumericLiteral(memberName)) {
				return nil, fmt.Errorf("Error payload fields need static names and explicit types")
			}
			optional := property.PostfixToken != nil && property.PostfixToken.Kind == ast.KindQuestionToken
			if err := add(memberName.Text(), optional, property.Type, member); err != nil {
				return nil, err
			}
		case member.Kind == ast.KindConstructor:
			for _, parameter := range member.Parameters() {
				// A parameter property is exactly the reference's test: one of
				// `public`, `private`, `protected`, `readonly`. `override` is
				// deliberately not in the set — it does not declare a field.
				if !ast.HasSyntacticModifier(parameter, ast.ModifierFlagsPublic|ast.ModifierFlagsPrivate|
					ast.ModifierFlagsProtected|ast.ModifierFlagsReadonly) {
					continue
				}
				if ast.HasSyntacticModifier(parameter, ast.ModifierFlagsPrivate|ast.ModifierFlagsProtected) {
					return nil, fmt.Errorf("Error class has a private/protected constructor field")
				}
				data := parameter.AsParameterDeclaration()
				parameterName := parameter.Name()
				if parameterName == nil || !ast.IsIdentifier(parameterName) ||
					data.Type == nil || data.DotDotDotToken != nil {
					return nil, fmt.Errorf("Error constructor fields need identifier names and explicit non-rest types")
				}
				optional := data.QuestionToken != nil || data.Initializer != nil
				if err := add(parameterName.Text(), optional, data.Type, parameter); err != nil {
					return nil, err
				}
			}
		}
	}

	names := make([]string, 0, len(fields))
	for fieldName := range fields {
		names = append(names, fieldName)
	}
	// Author-controlled field names: UTF-16, not Go's byte order. See
	// durableTypeDescriptor's note and compiler/fork_utf16_order_test.go.
	sortUTF16(names)
	if len(names) > durableMaxObjectFields {
		return nil, fmt.Errorf("Error payload exceeds the field limit")
	}
	ordered := make([]any, 0, len(names))
	for _, fieldName := range names {
		ordered = append(ordered, fields[fieldName])
	}
	return map[string]any{"fields": ordered, "kind": "object"}, nil
}

func durableDeclarationName(call *ast.Node, function *ast.Node) string {
	if call != nil && call.Parent != nil && ast.IsVariableDeclaration(call.Parent) {
		if name := call.Parent.Name(); name != nil && ast.IsIdentifier(name) {
			return name.Text()
		}
	}
	if function != nil {
		if name := function.Name(); name != nil && ast.IsIdentifier(name) {
			return name.Text()
		}
	}
	return "Flow"
}

func (l *lowerer) resolveDurableFunction(argument *ast.Node) *ast.Node {
	argument = ast.SkipParentheses(argument)
	if ast.IsArrowFunction(argument) || ast.IsFunctionExpression(argument) {
		return argument
	}
	symbol := l.durableCanonicalSymbolAt(argument)
	if symbol == nil {
		return nil
	}
	assigned := false
	var inspectAssignment func(*ast.Node) bool
	inspectAssignment = func(node *ast.Node) bool {
		if ast.IsIdentifier(node) && ast.IsAssignmentTarget(node) && l.durableCanonicalSymbolAt(node) == symbol {
			assigned = true
			return true
		}
		node.ForEachChild(inspectAssignment)
		return assigned
	}
	l.file.AsNode().ForEachChild(inspectAssignment)
	if assigned {
		return nil
	}
	var found *ast.Node
	for _, declaration := range symbol.Declarations {
		var candidate *ast.Node
		switch {
		case ast.IsFunctionDeclaration(declaration) && declaration.Body() != nil:
			candidate = declaration
		case ast.IsVariableDeclaration(declaration):
			if !ast.IsVarConst(declaration) {
				continue
			}
			initializer := declaration.AsVariableDeclaration().Initializer
			if initializer != nil {
				initializer = ast.SkipParentheses(initializer)
				if ast.IsArrowFunction(initializer) || ast.IsFunctionExpression(initializer) {
					candidate = initializer
				}
			}
		}
		if candidate == nil || ast.GetSourceFileOfNode(candidate) != l.file {
			continue
		}
		if found != nil && found != candidate {
			return nil
		}
		found = candidate
	}
	return found
}

// lowerDurableCall is the single integration hook used by visitExpression.
// matched is true only for the resolved compiler-owned durable declaration.
func (l *lowerer) lowerDurableCall(node *ast.Node) (lowered *ast.Node, matched bool) {
	if !ast.IsCallExpression(node) {
		return nil, false
	}
	call := node.AsCallExpression()
	if !l.durableIntrinsic(call.Expression, "durable") {
		return nil, false
	}
	matched = true
	if call.QuestionDotToken != nil || call.TypeArguments != nil || call.Arguments == nil || len(call.Arguments.Nodes) != 1 {
		l.report(node, durableCode("4103"), "durable(...) requires exactly one statically resolvable function argument")
		return node, true
	}
	function := l.resolveDurableFunction(call.Arguments.Nodes[0])
	if function == nil {
		l.report(call.Arguments.Nodes[0], durableCode("4103"), "durable(...) argument is not an inline or statically resolvable function")
		return node, true
	}
	name := durableDeclarationName(node, function)
	functionName := name
	if declared := function.Name(); declared != nil && ast.IsIdentifier(declared) {
		functionName = declared.Text()
	}
	logical := durableLogicalFile(l.file)
	d := &durableFunctionLowerer{
		owner:       l,
		flowID:      logical + "#" + name,
		flowVersion: 1,
		function:    functionName,
		logicalFile: logical,
		values:      make(map[*ast.Symbol]durableValue),
		actions:     make(map[string]durableAction),
		occurrences: make(map[string]int),
		nodeIDs:     make(map[string]bool),
		signalIDs:   make(map[string]bool),
	}
	d.activeNodes = &d.nodes
	output, ok := d.lowerFunction(function)
	if !ok || d.failed {
		return node, true
	}
	plan := d.plan(output)
	// The SECOND artifact, beside the Plan (`DECISIONS.md` §PR-1). Derived from
	// `function` — the Plan lowerer's own input — by its own traversal in
	// `effectmanifest.go`; `d`, `d.nodes`, `d.actions` and `plan` are not
	// passed to it and it cannot reach them.
	//
	// Opt-in, so that with the option absent — every existing caller — not one
	// emitted byte moves. See the header of `effectmanifest.go`.
	if requestedEffectManifest {
		manifest, manifestFailure := durableEffectManifest(l, logical, d.flowID, d.flowVersion, functionName, function)
		durableManifestArtifacts = append(durableManifestArtifacts, artifact{
			Path:    logical + ".plan.json",
			Content: []byte(durableCanonicalJSON(plan)),
		})
		if manifestFailure != "" {
			durableManifestArtifacts = append(durableManifestArtifacts, artifact{
				Path:    logical + ".effect-manifest.json",
				Content: []byte(durableCanonicalJSON(map[string]any{"error": manifestFailure})),
			})
		} else {
			durableManifestArtifacts = append(durableManifestArtifacts, artifact{
				Path:    logical + ".effect-manifest.json",
				Content: []byte(durableCanonicalJSON(manifest)),
			})
		}
	}
	descriptor := map[string]any{
		"artifactSource": "static-plan-artifact",
		"id":             d.flowID,
		"plan":           plan,
		"version":        d.flowVersion,
	}
	return l.durableLiteralExpression(descriptor), true
}

func (d *durableFunctionLowerer) fail(node *ast.Node, suffix string, message string) (durableValue, bool) {
	if !d.failed {
		d.owner.report(node, durableCode(suffix), message)
		d.failed = true
	}
	return durableValue{}, false
}

func (d *durableFunctionLowerer) lowerFunction(function *ast.Node) (durableValue, bool) {
	body := function.Body()
	if body == nil || !ast.IsBlock(body) {
		return d.fail(function, "4104", "durable source functions require a block body with an explicit return")
	}
	if function.BodyData().AsteriskToken != nil || ast.HasSyntacticModifier(function, ast.ModifierFlagsAsync) {
		return d.fail(function, "4104", "async and generator durable source functions are outside the bounded lowering subset")
	}
	parameters := function.Parameters()
	if len(parameters) != 1 {
		return d.fail(function, "4104", "durable source functions require exactly one input parameter")
	}
	parameter := parameters[0]
	parameterData := parameter.AsParameterDeclaration()
	if parameter.Name() == nil || !ast.IsIdentifier(parameter.Name()) || parameterData.Initializer != nil || parameterData.DotDotDotToken != nil {
		return d.fail(parameter, "4104", "durable input must be one plain identifier without an initializer")
	}
	inputSymbol := d.owner.durableCanonicalSymbolAt(parameter.Name())
	if inputSymbol == nil {
		return d.fail(parameter, "4199", "compiler could not resolve the durable input binding")
	}
	inputDescriptor, err := d.durableTypeDescriptor(d.owner.checker.GetTypeAtLocation(parameter.Name()), parameter, make(map[checker.TypeId]bool), false)
	if err != nil {
		return d.fail(parameter, "4110", "durable Flow input is not structurally encodable: "+err.Error())
	}
	d.inputSchema = durableStructuralSchema("input", inputDescriptor)
	d.values[inputSymbol] = durableValue{ir: map[string]any{"kind": "input", "path": []any{}}}

	var output durableValue
	returned := false
	for _, statement := range body.AsBlock().Statements.Nodes {
		if returned {
			return d.fail(statement, "4109", "statements after the durable return are not supported")
		}
		if statement.Kind == ast.KindEmptyStatement {
			continue
		}
		if ast.IsVariableStatement(statement) {
			list := statement.AsVariableStatement().DeclarationList.AsVariableDeclarationList()
			if list.AsNode().Flags&ast.NodeFlagsConst == 0 {
				return d.fail(statement, "4105", "durable straight-line bindings must use const")
			}
			for _, declaration := range list.Declarations.Nodes {
				valueDeclaration := declaration.AsVariableDeclaration()
				if declaration.Name() == nil || !ast.IsIdentifier(declaration.Name()) || valueDeclaration.Initializer == nil {
					return d.fail(declaration, "4105", "durable const bindings require one identifier and an initializer")
				}
				symbol := d.owner.durableCanonicalSymbolAt(declaration.Name())
				if symbol == nil {
					return d.fail(declaration.Name(), "4199", "compiler could not resolve const binding")
				}
				value, ok := d.lowerExpression(valueDeclaration.Initializer, "const:"+declaration.Name().Text(), false)
				if !ok {
					return durableValue{}, false
				}
				d.values[symbol] = value
			}
			continue
		}
		if statement.Kind == ast.KindReturnStatement {
			expression := statement.AsReturnStatement().Expression
			if expression == nil {
				return d.fail(statement, "4109", "durable return requires a value")
			}
			var ok bool
			output, ok = d.lowerExpression(expression, "return", true)
			if !ok {
				return durableValue{}, false
			}
			returned = true
			continue
		}
		if statement.Kind == ast.KindExpressionStatement {
			expression := ast.SkipParentheses(statement.AsExpressionStatement().Expression)
			if ast.IsCallExpression(expression) {
				if _, recognized, ok := d.lowerTimer(expression, "sleep"); recognized {
					if !ok {
						return durableValue{}, false
					}
					continue
				}
				if _, recognized, ok := d.lowerSequential(expression, "sequential"); recognized {
					if !ok {
						return durableValue{}, false
					}
					continue
				}
			}
			return d.fail(statement, "4108", "only compiler-owned sleep(...) and sequential(...) are supported as durable expression statements")
		}
		if statement.Kind == ast.KindIfStatement || statement.Kind == ast.KindSwitchStatement {
			return d.fail(statement, "4106", "runtime branches require explicit Plan branch lowering, which this subset does not guess")
		}
		if ast.IsIterationStatement(statement, false) {
			return d.fail(statement, "4107", "runtime loops require parameterized Plan templates and are not unrolled by this subset")
		}
		return d.fail(statement, "4108", "unsupported durable statement "+statement.Kind.String())
	}
	if !returned {
		return d.fail(body, "4109", "durable source function must return a value")
	}
	// The output expression is complete here, and so are `d.nodes`, `d.actions`
	// and `d.inputSchema` — everything the boundary walk reads. It covers the
	// output AND every value a Plan node consumes; an Action input is the
	// position that compiled and ran with an unsatisfiable projection until this
	// covered it.
	//
	// The walk's OUTPUT descriptor is the Flow's success contract, and this is
	// where `d.successSchema` is now set from. It used to be derived here only
	// for an action-free Flow, and from the return expression's TypeScript TYPE
	// rather than from the Plan — two different answers to two different
	// questions. The TypeScript type is WIDENED (`return { ok: true }` typed
	// `{ ok: boolean }`), while the reference derives `{kind:"literal",
	// value:true}` from the Plan value expression, so even the action-free case
	// the fork did derive could disagree; a Flow with an Action got the legacy
	// stub and disagreed always.
	if !d.checkFlowBoundary(function, output) {
		return durableValue{}, false
	}
	return output, true
}

func (d *durableFunctionLowerer) lowerExpression(node *ast.Node, anchor string, allowFinalAction bool) (durableValue, bool) {
	expression := ast.SkipParentheses(node)
	if ast.IsIdentifier(expression) {
		symbol := d.owner.durableCanonicalSymbolAt(expression)
		if value, ok := d.values[symbol]; ok {
			return value, true
		}
		return d.fail(expression, "4110", "unsupported runtime capture "+expression.Text()+"; only input and prior const bindings are available")
	}
	switch expression.Kind {
	case ast.KindNullKeyword:
		return durableValue{ir: map[string]any{"kind": "literal", "value": nil}}, true
	case ast.KindTrueKeyword:
		return durableValue{ir: map[string]any{"kind": "literal", "value": true}}, true
	case ast.KindFalseKeyword:
		return durableValue{ir: map[string]any{"kind": "literal", "value": false}}, true
	case ast.KindStringLiteral, ast.KindNoSubstitutionTemplateLiteral:
		return durableValue{ir: map[string]any{"kind": "literal", "value": expression.Text()}}, true
	case ast.KindNumericLiteral:
		value, err := strconv.ParseFloat(expression.Text(), 64)
		if err != nil || math.IsInf(value, 0) || math.IsNaN(value) {
			return d.fail(expression, "4111", "non-finite numeric literal is not durable")
		}
		return durableValue{ir: map[string]any{"kind": "literal", "value": value}}, true
	case ast.KindPrefixUnaryExpression:
		prefix := expression.AsPrefixUnaryExpression()
		if prefix.Operator == ast.KindMinusToken && ast.IsNumericLiteral(prefix.Operand) {
			value, err := strconv.ParseFloat(prefix.Operand.Text(), 64)
			value = -value
			if err != nil || math.IsInf(value, 0) || math.IsNaN(value) || math.Signbit(value) && value == 0 {
				return d.fail(expression, "4111", "non-canonical numeric literal is not durable")
			}
			return durableValue{ir: map[string]any{"kind": "literal", "value": value}}, true
		}
	}
	if ast.IsObjectLiteralExpression(expression) {
		fields := make(map[string]any)
		for _, property := range expression.AsObjectLiteralExpression().Properties.Nodes {
			var name string
			var initializer *ast.Node
			switch {
			case ast.IsPropertyAssignment(property):
				propertyName := property.Name()
				if propertyName == nil || !(ast.IsIdentifier(propertyName) || ast.IsStringOrNumericLiteralLike(propertyName)) {
					return d.fail(property, "4111", "durable object field names must be static")
				}
				name = propertyName.Text()
				initializer = property.AsPropertyAssignment().Initializer
			case ast.IsShorthandPropertyAssignment(property):
				name = property.Name().Text()
				initializer = property.Name()
			default:
				return d.fail(property, "4111", "durable objects do not support spreads, methods, or accessors in this subset")
			}
			if _, duplicate := fields[name]; duplicate {
				return d.fail(property, "4111", "duplicate durable object field "+name)
			}
			value, ok := d.lowerExpression(initializer, anchor+"."+name, false)
			if !ok {
				return durableValue{}, false
			}
			fields[name] = value.ir
		}
		return durableValue{ir: map[string]any{"fields": fields, "kind": "object"}}, true
	}
	if ast.IsArrayLiteralExpression(expression) {
		items := make([]any, 0, len(expression.AsArrayLiteralExpression().Elements.Nodes))
		for index, element := range expression.AsArrayLiteralExpression().Elements.Nodes {
			if element.Kind == ast.KindOmittedExpression || element.Kind == ast.KindSpreadElement {
				return d.fail(element, "4111", "durable arrays cannot contain holes or spreads")
			}
			value, ok := d.lowerExpression(element, fmt.Sprintf("%s[%d]", anchor, index), false)
			if !ok {
				return durableValue{}, false
			}
			items = append(items, value.ir)
		}
		return durableValue{ir: map[string]any{"items": items, "kind": "array"}}, true
	}
	if ast.IsPropertyAccessExpression(expression) {
		access := expression.AsPropertyAccessExpression()
		if access.QuestionDotToken != nil {
			return d.fail(expression, "4106", "optional projections require explicit Plan branch lowering")
		}
		base, ok := d.lowerExpression(access.Expression, anchor, false)
		if !ok {
			return durableValue{}, false
		}
		return d.project(base, access.Name().Text(), expression)
	}
	if ast.IsElementAccessExpression(expression) {
		access := expression.AsElementAccessExpression()
		if access.QuestionDotToken != nil {
			return d.fail(expression, "4106", "optional projections require explicit Plan branch lowering")
		}
		argument := ast.SkipParentheses(access.ArgumentExpression)
		if argument == nil || !ast.IsStringOrNumericLiteralLike(argument) {
			return d.fail(expression, "4111", "durable projection keys must be static string or numeric literals")
		}
		base, ok := d.lowerExpression(access.Expression, anchor, false)
		if !ok {
			return durableValue{}, false
		}
		return d.project(base, argument.Text(), expression)
	}
	if ast.IsConditionalExpression(expression) {
		return d.lowerConditional(expression, anchor, allowFinalAction)
	}
	if ast.IsNonNullExpression(expression) {
		candidate := ast.SkipParentheses(expression.AsNonNullExpression().Expression)
		if ast.IsCallExpression(candidate) {
			value, recognized, ok := d.lowerAction(candidate, anchor)
			if recognized {
				if ok {
					d.sequence = value.ir.(map[string]any)["nodeId"].(string)
				}
				return value, ok
			}
		}
		return d.fail(expression, "4112", "postfix ! is supported only directly on a compiler-bound Action.run(...) Result")
	}
	if ast.IsCallExpression(expression) {
		if value, recognized, ok := d.lowerSignal(expression, anchor); recognized {
			return value, ok
		}
		if value, recognized, ok := d.lowerTimer(expression, anchor); recognized {
			return value, ok
		}
		if value, recognized, ok := d.lowerSequential(expression, anchor); recognized {
			return value, ok
		}
		call := expression.AsCallExpression()
		for _, unsupported := range []struct {
			name    string
			suffix  string
			message string
		}{
			{name: "fanOut", suffix: "4117", message: "compiler-owned fanOut requires a bounded stable-key template not implemented by the Go durable subset"},
			{name: "loopWhile", suffix: "4121", message: "compiler-owned loopWhile requires a bounded loop template not implemented by the Go durable subset"},
			{name: "waitBroadcast", suffix: "4122", message: "compiler-owned waitBroadcast is not implemented by the Go durable subset"},
			{name: "dequeue", suffix: "4123", message: "compiler-owned dequeue is not implemented by the Go durable subset"},
		} {
			if d.owner.durableIntrinsic(call.Expression, unsupported.name) {
				return d.fail(expression, unsupported.suffix, unsupported.message)
			}
		}
		if value, recognized, ok := d.lowerAction(expression, anchor); recognized {
			if ok && !allowFinalAction {
				return d.fail(expression, "4115", "an intermediate Action.run(...) must use postfix !; only the final returned Result may remain wrapped")
			}
			return value, ok
		}
		if d.isChildFlowRun(expression) {
			return d.fail(expression, "4120", "child Flow embedding is not implemented by the Go durable subset")
		}
		return d.fail(expression, "4112", "higher-order and dynamic calls are unavailable in durable source lowering")
	}
	return d.fail(expression, "4111", "unsupported durable expression "+expression.Kind.String())
}

func (d *durableFunctionLowerer) project(value durableValue, key string, node *ast.Node) (durableValue, bool) {
	record, ok := value.ir.(map[string]any)
	if !ok {
		return d.fail(node, "4111", fmt.Sprintf("cannot project %q from this durable expression", key))
	}
	kind, _ := record["kind"].(string)
	switch kind {
	case "input", "node":
		pathValue, _ := record["path"].([]any)
		copyRecord := make(map[string]any, len(record))
		for name, item := range record {
			copyRecord[name] = item
		}
		copyRecord["path"] = append(append([]any(nil), pathValue...), key)
		return durableValue{ir: copyRecord}, true
	case "object":
		if fields, ok := record["fields"].(map[string]any); ok {
			if projected, exists := fields[key]; exists {
				return durableValue{ir: projected}, true
			}
		}
	case "array":
		index, err := strconv.Atoi(key)
		if items, ok := record["items"].([]any); ok && err == nil && index >= 0 && index < len(items) {
			return durableValue{ir: items[index]}, true
		}
	}
	return d.fail(node, "4111", fmt.Sprintf("cannot project %q from this durable expression", key))
}

func durableBooleanType(t *checker.Type) bool {
	if t == nil {
		return false
	}
	if t.IsUnion() {
		if len(t.Types()) == 0 {
			return false
		}
		for _, item := range t.Types() {
			if !durableBooleanType(item) {
				return false
			}
		}
		return true
	}
	return t.Flags()&(checker.TypeFlagsBoolean|checker.TypeFlagsBooleanLiteral) != 0
}

func (d *durableFunctionLowerer) lowerConditional(node *ast.Node, anchor string, allowFinalAction bool) (durableValue, bool) {
	conditional := node.AsConditionalExpression()
	if !durableBooleanType(d.owner.checker.GetTypeAtLocation(conditional.Condition)) {
		return d.fail(conditional.Condition, "4106", "durable conditional expressions require a statically boolean condition")
	}
	condition, ok := d.lowerExpression(conditional.Condition, anchor+":condition", false)
	if !ok {
		return durableValue{}, false
	}
	parentNodes := d.activeNodes
	parentSequence := d.sequence
	trueNodes := []any{}
	d.activeNodes = &trueNodes
	d.sequence = ""
	whenTrue, ok := d.lowerExpression(conditional.WhenTrue, anchor+":true", allowFinalAction)
	if !ok {
		d.activeNodes = parentNodes
		d.sequence = parentSequence
		return durableValue{}, false
	}
	falseNodes := []any{}
	d.activeNodes = &falseNodes
	d.sequence = ""
	whenFalse, ok := d.lowerExpression(conditional.WhenFalse, anchor+":false", allowFinalAction)
	d.activeNodes = parentNodes
	d.sequence = parentSequence
	if !ok {
		return durableValue{}, false
	}
	identity := map[string]any{
		"anchor":       anchor,
		"condition":    condition.ir,
		"file":         d.logicalFile,
		"flowId":       d.flowID,
		"flowVersion":  d.flowVersion,
		"functionName": d.function,
		"kind":         "branch",
	}
	id, ok := d.stableNodeID(node, identity)
	if !ok {
		return durableValue{}, false
	}
	branch := map[string]any{
		"condition":           condition.ir,
		"controlDependencies": durableControlDependencies(parentSequence),
		// The `?` TOKEN, not the conditional expression. Every other node kind
		// here anchors its callSite on the construct's own start, and a branch
		// looks like it should too — but a conditional expression starts at the
		// same character its CONDITION starts at, so that anchor names a position
		// that is already the condition's and says nothing about the branch. The
		// reference picks `expression.questionToken.getStart(...)` for that
		// reason (poc/src/durable/source-compiler.ts, lowerConditionalExpression).
		//
		// This is not cosmetic: `debug` is inside the Plan node, so it is hashed
		// into `plan.digest`. Anchoring on the expression was the LAST structural
		// difference between the two backends' Plans once the three stub schemas
		// below were derived — it is why static-plan-shape-is-digest-pinned still
		// minted two digests when every schema in it already agreed.
		"debug": d.debug(conditional.QuestionToken, "conditional"),
		"dependencies":        durableExpressionDependencies(condition.ir),
		"id":                  id,
		"kind":                "branch",
		"whenFalse":           map[string]any{"nodes": falseNodes, "output": whenFalse.ir},
		"whenTrue":            map[string]any{"nodes": trueNodes, "output": whenTrue.ir},
	}
	*d.activeNodes = append(*d.activeNodes, branch)
	d.sequence = id
	return durableValue{ir: map[string]any{"kind": "node", "nodeId": id, "path": []any{}}}, true
}

func (d *durableFunctionLowerer) lowerAction(node *ast.Node, anchor string) (durableValue, bool, bool) {
	if !ast.IsCallExpression(node) {
		return durableValue{}, false, false
	}
	call := node.AsCallExpression()
	if !ast.IsPropertyAccessExpression(call.Expression) {
		return durableValue{}, false, false
	}
	access := call.Expression.AsPropertyAccessExpression()
	if !d.owner.durableIntrinsic(access.Name(), "run") {
		return durableValue{}, false, false
	}
	action, ok := d.actionFor(access.Expression)
	if !ok {
		return durableValue{}, false, false
	}
	if action.invalid != "" {
		value, valid := d.fail(access.Expression, "4113", action.invalid)
		return value, true, valid
	}
	// A declaration whose durable contract cannot be derived is SKIPPED by the
	// reference rather than reported: `deriveSameFileActions` wraps derivation
	// in a `try`/`catch` whose comment says "A declaration whose signature is
	// outside the derivable subset is skipped, not fatal: its `run` calls then
	// find no descriptor and the lowerer reports the ordinary unsupported-call
	// diagnostic against the authored call site." That call site is still where
	// this is reported, and it is still where the reference reports it.
	//
	// The CODE and the SENTENCE are no longer the generic ones. Both backends
	// used to answer SMITHERS4112, "higher-order and dynamic calls are
	// unavailable in durable source lowering", which is false of the program
	// that reaches here — `X.run({ ... })` is an ordinary compiler-bound Action
	// call, and an author was sent looking for a higher-order call that does not
	// exist. The reference now keeps the collision's reason out of its `catch`
	// and re-reports it at the same node as SMITHERS4124; this does the same,
	// from its own detector, with the same text.
	//
	// Still reported here rather than by falling through to the shared
	// unsupported-call branch below, because falling through would first meet
	// `isChildFlowRun`, which recognises any compiler-owned `.run` whose
	// receiver is not `Action` itself and would answer SMITHERS4120 instead.
	if action.collision != nil {
		value, valid := d.fail(node, "4124", durableFailureIdentityCollisionMessage(action.collision))
		return value, true, valid
	}
	if call.QuestionDotToken != nil || access.QuestionDotToken != nil || call.TypeArguments != nil || call.Arguments == nil || len(call.Arguments.Nodes) != 1 {
		value, valid := d.fail(node, "4113", action.id+".run requires exactly one input argument in durable source")
		return value, true, valid
	}
	argumentType := d.owner.checker.GetTypeAtLocation(call.Arguments.Nodes[0])
	if action.inputType == nil || argumentType == nil || !d.owner.checker.IsTypeAssignableTo(argumentType, action.inputType) {
		value, valid := d.fail(call.Arguments.Nodes[0], "4100", action.id+".run input is not assignable to the Action's checked input contract")
		return value, true, valid
	}
	input, valid := d.lowerExpression(call.Arguments.Nodes[0], anchor+":input", false)
	if !valid {
		return durableValue{}, true, false
	}
	d.actions[action.id] = action
	identity := map[string]any{
		"actionContractDigest": action.descriptor["contractDigest"],
		"actionId":             action.id,
		"actionVersion":        1,
		"anchor":               anchor,
		"file":                 d.logicalFile,
		"flowId":               d.flowID,
		"flowVersion":          d.flowVersion,
		"functionName":         d.function,
		"input":                input.ir,
	}
	id, valid := d.stableNodeID(node, identity)
	if !valid {
		return durableValue{}, true, false
	}
	actionNode := map[string]any{
		"actionContractDigest": action.descriptor["contractDigest"],
		"actionId":             action.id,
		"actionVersion":        1,
		"controlDependencies":  durableControlDependencies(d.sequence),
		"debug":                d.debug(node, action.id),
		"dependencies":         durableExpressionDependencies(input.ir),
		"id":                   id,
		"input":                input.ir,
		"kind":                 "action",
	}
	*d.activeNodes = append(*d.activeNodes, actionNode)
	return durableValue{ir: map[string]any{"kind": "node", "nodeId": id, "path": []any{}}}, true, true
}

func (d *durableFunctionLowerer) actionFor(reference *ast.Node) (durableAction, bool) {
	symbol := d.owner.durableCanonicalSymbolAt(ast.SkipParentheses(reference))
	declaration := classDeclarationOf(symbol)
	if symbol == nil || declaration == nil || declaration.ClassLikeData().HeritageClauses == nil {
		return durableAction{}, false
	}
	bound := false
	var signatureType *checker.Type
	for _, clauseNode := range declaration.ClassLikeData().HeritageClauses.Nodes {
		clause := clauseNode.AsHeritageClause()
		if clause.Token != ast.KindExtendsKeyword || clause.Types == nil {
			continue
		}
		for _, item := range clause.Types.Nodes {
			if ast.IsExpressionWithTypeArguments(item) && d.owner.durableIntrinsic(item.AsExpressionWithTypeArguments().Expression, "Action") {
				bound = true
				typeArguments := item.AsExpressionWithTypeArguments().TypeArguments
				if typeArguments != nil && len(typeArguments.Nodes) == 1 {
					signatureType = d.owner.checker.GetTypeFromTypeNode(typeArguments.Nodes[0])
				}
			}
		}
	}
	if !bound {
		return durableAction{}, false
	}
	if signatureType == nil {
		return durableAction{invalid: "Action subclass requires exactly one checked function signature type"}, true
	}
	signatures := d.owner.checker.GetSignaturesOfType(signatureType, checker.SignatureKindCall)
	if len(signatures) != 1 || len(signatures[0].Parameters()) != 1 || signatures[0].HasRestParameter() {
		return durableAction{invalid: "Action signature must contain exactly one non-rest input parameter"}, true
	}
	inputType := d.owner.checker.GetTypeOfSymbolAtLocation(signatures[0].Parameters()[0], declaration)
	returned := d.owner.checker.GetReturnTypeOfSignature(signatures[0])
	if promised := d.owner.checker.GetPromisedTypeOfPromise(returned); promised != nil {
		returned = promised
	}
	if inputType == nil || d.owner.analysis == nil {
		return durableAction{invalid: "compiler could not derive the Action input contract"}, true
	}
	successType, failureType, result := d.owner.analysis.resultArguments(returned)
	if !result || successType == nil {
		return durableAction{invalid: "Action signature must return Result<Success, Error>"}, true
	}
	file := ast.GetSourceFileOfNode(declaration)
	logical := durableLogicalFile(file)
	id := logical + "#" + symbol.Name
	inputDescriptor, err := d.durableTypeDescriptor(inputType, declaration, make(map[checker.TypeId]bool), false)
	if err != nil {
		return durableAction{invalid: "Action input is not structurally encodable: " + err.Error()}, true
	}
	successDescriptor, err := d.durableTypeDescriptor(successType, declaration, make(map[checker.TypeId]bool), false)
	if err != nil {
		return durableAction{invalid: "Action success is not structurally encodable: " + err.Error()}, true
	}
	// The failure channel must be describable before it can be derived: the
	// reference refuses to derive a contract whose declared Error classes cannot
	// be told apart on the wire. Checked first, and separately, because it is
	// the one underivable failure channel whose reason a diagnostic names.
	if collision := d.durableFailureIdentityCollides(failureType); collision != nil {
		return durableAction{collision: collision}, true
	}
	inputSchema := durableStructuralSchema("input", inputDescriptor)
	successSchema := durableStructuralSchema("success", successDescriptor)
	errorSchema, err := d.durableErrorSchemaFor(failureType)
	if err != nil {
		return durableAction{invalid: "Action failure channel is not structurally encodable: " + err.Error()}, true
	}
	contract := map[string]any{
		"errorSchema":   errorSchema,
		"id":            id,
		"inputSchema":   inputSchema,
		"successSchema": successSchema,
		"version":       1,
	}
	descriptor := durableCopyMap(contract)
	descriptor["contractDigest"] = durableDigest(contract)
	return durableAction{id: id, descriptor: descriptor, inputType: inputType, failureType: failureType}, true
}

func (d *durableFunctionLowerer) lowerTimer(node *ast.Node, anchor string) (durableValue, bool, bool) {
	call := node.AsCallExpression()
	if !d.owner.durableIntrinsic(call.Expression, "sleep") {
		return durableValue{}, false, false
	}
	if call.QuestionDotToken != nil || call.TypeArguments != nil || call.Arguments == nil || len(call.Arguments.Nodes) != 1 {
		value, ok := d.fail(node, "4116", "compiler-owned sleep(...) requires one statically bound duration argument")
		return value, true, ok
	}
	duration, ok := d.lowerExpression(call.Arguments.Nodes[0], anchor+":duration", false)
	if !ok {
		return durableValue{}, true, false
	}
	if record, isRecord := duration.ir.(map[string]any); isRecord && record["kind"] == "literal" {
		number, isNumber := record["value"].(float64)
		if !isNumber || number < 0 || number > 9007199254740991 || math.Trunc(number) != number {
			value, valid := d.fail(call.Arguments.Nodes[0], "4116", "durable sleep duration must be a non-negative safe integer number of milliseconds")
			return value, true, valid
		}
	}
	identity := map[string]any{
		"anchor":       anchor,
		"durationMs":   duration.ir,
		"file":         d.logicalFile,
		"flowId":       d.flowID,
		"flowVersion":  d.flowVersion,
		"functionName": d.function,
		"kind":         "timer",
	}
	id, valid := d.stableNodeID(node, identity)
	if !valid {
		return durableValue{}, true, false
	}
	timer := map[string]any{
		"controlDependencies": durableControlDependencies(d.sequence),
		"debug":               d.debug(node, "sleep"),
		"dependencies":        durableExpressionDependencies(duration.ir),
		"durationMs":          duration.ir,
		"id":                  id,
		"kind":                "timer",
	}
	*d.activeNodes = append(*d.activeNodes, timer)
	d.sequence = id
	return durableValue{ir: map[string]any{"kind": "node", "nodeId": id, "path": []any{}}}, true, true
}

func (d *durableFunctionLowerer) lowerSequential(node *ast.Node, anchor string) (durableValue, bool, bool) {
	call := node.AsCallExpression()
	if !d.owner.durableIntrinsic(call.Expression, "sequential") {
		return durableValue{}, false, false
	}
	if call.QuestionDotToken != nil || call.TypeArguments != nil || call.Arguments == nil || len(call.Arguments.Nodes) != 2 {
		value, ok := d.fail(node, "4119", "compiler-owned sequential(first, second) requires exactly two direct imported Action.run(...) arguments")
		return value, true, ok
	}
	firstExpression := ast.SkipParentheses(call.Arguments.Nodes[0])
	first, recognized, ok := d.lowerAction(firstExpression, anchor+":first")
	if !recognized {
		value, valid := d.fail(call.Arguments.Nodes[0], "4119", "sequential(...) arguments must be direct imported Action.run(...) calls")
		return value, true, valid
	}
	if !ok {
		return durableValue{}, true, false
	}
	d.sequence = first.ir.(map[string]any)["nodeId"].(string)
	secondExpression := ast.SkipParentheses(call.Arguments.Nodes[1])
	second, recognized, ok := d.lowerAction(secondExpression, anchor+":second")
	if !recognized {
		value, valid := d.fail(call.Arguments.Nodes[1], "4119", "sequential(...) arguments must be direct imported Action.run(...) calls")
		return value, true, valid
	}
	if !ok {
		return durableValue{}, true, false
	}
	d.sequence = second.ir.(map[string]any)["nodeId"].(string)
	return durableValue{ir: map[string]any{"items": []any{first.ir, second.ir}, "kind": "array"}}, true, true
}

func (d *durableFunctionLowerer) lowerSignal(node *ast.Node, anchor string) (durableValue, bool, bool) {
	call := node.AsCallExpression()
	if !d.owner.durableIntrinsic(call.Expression, "waitSignal") {
		return durableValue{}, false, false
	}
	if call.QuestionDotToken != nil || call.Arguments == nil || len(call.Arguments.Nodes) != 1 || call.TypeArguments == nil || len(call.TypeArguments.Nodes) != 1 {
		value, ok := d.fail(node, "4118", "provisional compiler-owned waitSignal<Payload>(\"identity\") requires one explicit payload type and one static identity")
		return value, true, ok
	}
	identityExpression := ast.SkipParentheses(call.Arguments.Nodes[0])
	if !ast.IsStringLiteralLike(identityExpression) {
		value, ok := d.fail(call.Arguments.Nodes[0], "4118", "durable signal identity must be a string literal, not a runtime value")
		return value, true, ok
	}
	signalID := identityExpression.Text()
	if !durablePortableIdentity.MatchString(signalID) {
		value, ok := d.fail(identityExpression, "4118", "durable signal identity must be 1-128 portable characters")
		return value, true, ok
	}
	if d.signalIDs[signalID] {
		value, ok := d.fail(identityExpression, "4118", "durable signal identity "+signalID+" is duplicated in this Flow")
		return value, true, ok
	}
	typeNode := call.TypeArguments.Nodes[0]
	descriptor, err := d.durableTypeDescriptor(d.owner.checker.GetTypeFromTypeNode(typeNode), typeNode, make(map[checker.TypeId]bool), false)
	if err != nil {
		value, ok := d.fail(typeNode, "4118", "durable signal payload is not structurally encodable: "+err.Error())
		return value, true, ok
	}
	payloadSchema := durableStructuralSchema("input", descriptor)
	contractDigest := durableDigest(map[string]any{"payloadSchema": payloadSchema, "signalId": signalID})
	identity := map[string]any{
		"anchor":               anchor,
		"file":                 d.logicalFile,
		"flowId":               d.flowID,
		"flowVersion":          d.flowVersion,
		"functionName":         d.function,
		"kind":                 "signal",
		"signalContractDigest": contractDigest,
		"signalId":             signalID,
	}
	id, ok := d.stableNodeID(node, identity)
	if !ok {
		return durableValue{}, true, false
	}
	d.signalIDs[signalID] = true
	signal := map[string]any{
		"controlDependencies":  durableControlDependencies(d.sequence),
		"debug":                d.debug(node, "signal:"+signalID),
		"dependencies":         []any{},
		"id":                   id,
		"kind":                 "signal",
		"payloadSchema":        payloadSchema,
		"signalContractDigest": contractDigest,
		"signalId":             signalID,
	}
	*d.activeNodes = append(*d.activeNodes, signal)
	d.sequence = id
	return durableValue{ir: map[string]any{"kind": "node", "nodeId": id, "path": []any{}}}, true, true
}

func (d *durableFunctionLowerer) durableTypeDescriptor(t *checker.Type, location *ast.Node, seen map[checker.TypeId]bool, omitUndefined bool) (any, error) {
	if t == nil {
		return nil, fmt.Errorf("checker returned no type")
	}
	flags := t.Flags()
	if flags&checker.TypeFlagsUndefined != 0 {
		if omitUndefined {
			return nil, nil
		}
		return nil, fmt.Errorf("undefined is not a canonical JSON value")
	}
	if flags&checker.TypeFlagsNull != 0 {
		return map[string]any{"kind": "null"}, nil
	}
	if flags&checker.TypeFlagsString != 0 {
		return map[string]any{"kind": "string"}, nil
	}
	if flags&checker.TypeFlagsNumber != 0 {
		return map[string]any{"kind": "number"}, nil
	}
	if flags&checker.TypeFlagsBoolean != 0 {
		return map[string]any{"kind": "boolean"}, nil
	}
	if flags&checker.TypeFlagsLiteral != 0 {
		value := t.AsLiteralType().Value()
		switch value := value.(type) {
		case string, bool:
			return map[string]any{"kind": "literal", "value": value}, nil
		default:
			number, err := strconv.ParseFloat(fmt.Sprint(value), 64)
			if err != nil || math.IsInf(number, 0) || math.IsNaN(number) {
				return nil, fmt.Errorf("unsupported literal type")
			}
			return map[string]any{"kind": "literal", "value": number}, nil
		}
	}
	if t.IsUnion() {
		variants := make([]any, 0, len(t.Types()))
		for _, item := range t.Types() {
			variant, err := d.durableTypeDescriptor(item, location, seen, omitUndefined)
			if err != nil {
				return nil, err
			}
			if variant != nil {
				variants = append(variants, variant)
			}
		}
		if len(variants) == 0 {
			return nil, fmt.Errorf("union has no encodable variants")
		}
		return durableCanonicalUnion(variants), nil
	}
	if flags&(checker.TypeFlagsAny|checker.TypeFlagsUnknown|checker.TypeFlagsNever|checker.TypeFlagsBigIntLike|checker.TypeFlagsESSymbolLike|checker.TypeFlagsVoid) != 0 {
		return nil, fmt.Errorf("type is not a canonical JSON persistence type")
	}
	if flags&checker.TypeFlagsObject == 0 {
		return nil, fmt.Errorf("unsupported persistence type")
	}
	if seen[t.Id()] {
		return nil, fmt.Errorf("recursive persistence types are outside the bounded schema subset")
	}
	seen[t.Id()] = true
	defer delete(seen, t.Id())
	if checker.IsTupleType(t) {
		arguments := d.owner.checker.GetTypeArguments(t)
		items := make([]any, 0, len(arguments))
		for _, argument := range arguments {
			item, err := d.durableTypeDescriptor(argument, location, seen, false)
			if err != nil {
				return nil, err
			}
			items = append(items, item)
		}
		return map[string]any{"items": items, "kind": "tuple"}, nil
	}
	if d.owner.checker.IsArrayType(t) {
		arguments := d.owner.checker.GetTypeArguments(t)
		if len(arguments) != 1 {
			return nil, fmt.Errorf("array element type is unavailable")
		}
		element, err := d.durableTypeDescriptor(arguments[0], location, seen, false)
		if err != nil {
			return nil, err
		}
		return map[string]any{"element": element, "kind": "array"}, nil
	}
	if len(d.owner.checker.GetIndexInfosOfType(t)) != 0 || len(d.owner.checker.GetSignaturesOfType(t, checker.SignatureKindCall)) != 0 {
		return nil, fmt.Errorf("index signatures and callable objects are not durable structural records")
	}
	properties := append([]*ast.Symbol(nil), d.owner.checker.GetPropertiesOfType(t)...)
	// Field names are authored text, and this order IS the contract: it is the
	// order the descriptor records, so it is hashed into the successSchema
	// digest, the Action contractDigest, and the Plan digest. The reference
	// sorts the same fields with `compareText` (poc/src/durable/schema.ts:401),
	// which is UTF-16 code-unit order. Byte order was used here until
	// 2026-08-28 and silently minted a different contract for any Action whose
	// field names straddle the surrogate boundary; see comptime.go.txt's
	// "Ordering author-controlled strings" and
	// conformance/corpus/17-durable/action-success-field-order-is-utf16-not-utf8.
	sortUTF16By(properties, func(property *ast.Symbol) string { return property.Name })
	fields := make([]any, 0, len(properties))
	for _, property := range properties {
		optional := property.Flags&ast.SymbolFlagsOptional != 0
		propertyType := d.owner.checker.GetTypeOfSymbolAtLocation(property, location)
		value, err := d.durableTypeDescriptor(propertyType, location, seen, optional)
		if err != nil {
			return nil, fmt.Errorf("field %s: %w", property.Name, err)
		}
		if value == nil {
			return nil, fmt.Errorf("field %s has no encodable value type", property.Name)
		}
		fields = append(fields, map[string]any{"name": property.Name, "optional": optional, "value": value})
	}
	return map[string]any{"fields": fields, "kind": "object"}, nil
}

func durableCanonicalUnion(variants []any) any {
	flattened := make([]any, 0, len(variants))
	for _, variant := range variants {
		if record, ok := variant.(map[string]any); ok && record["kind"] == "union" {
			if nested, ok := record["variants"].([]any); ok {
				flattened = append(flattened, nested...)
				continue
			}
		}
		flattened = append(flattened, variant)
	}
	byCanonical := make(map[string]any, len(flattened))
	keys := make([]string, 0, len(flattened))
	for _, variant := range flattened {
		key := durableCanonicalJSON(variant)
		if _, exists := byCanonical[key]; !exists {
			byCanonical[key] = variant
			keys = append(keys, key)
		}
	}
	sortUTF16(keys)
	ordered := make([]any, 0, len(keys))
	for _, key := range keys {
		ordered = append(ordered, byCanonical[key])
	}
	if len(ordered) == 1 {
		return ordered[0]
	}
	return map[string]any{"kind": "union", "variants": ordered}
}

// ---------------------------------------------------------------------------
// Flow output projection check
// ---------------------------------------------------------------------------
//
// `project` above builds Plan IR: it appends a path component to an `input` or
// `node` expression whenever the authored source reads a member of one. It does
// NOT check the component against the durable descriptor the value actually
// carries, and nothing downstream did either — so
//
//	return { count: input.items.length }
//
// lowered to `{"kind":"input","path":["items","length"]}`, compiled clean, and
// ran. `.length` is the sharpest spelling of the hole: TypeScript accepts it on
// an array, so no stock diagnostic fires, the durable input descriptor has no
// such field, and the engine's `pathValue` refuses a non-numeric part on an
// array with a runtime ProjectionDefect. A compile-time-knowable error became a
// runtime fault, silently.
//
// This walk is the reference frontend's `flowSuccessDescriptor`
// (poc/src/durable/source-compiler.ts) ported to the Plan IR this bridge builds.
// It derives the descriptor the Flow output WOULD carry and reports every
// projection that descriptor cannot answer, at the same code (SMITHERS4110),
// the same position (the durable source function), and with the same sentence
// as the reference, so a conformance case can declare one text for both
// backends.
//
// It is ALSO the derivation. Until 2026-08-28 the descriptor this walk produced
// was discarded once it had answered the projection question, and `plan()` emitted
// `durableLegacySchema("success")` whenever the Flow used an Action and the return
// expression's widened TypeScript type otherwise — the sole structural difference
// between the two backends' Plans on every comparable program, and the reason
// `plan.digest` diverged even for all-ASCII source. `checkFlowBoundary` now records
// the output descriptor as the Flow's success schema, which is what the reference
// has always done with the same walk's result.
//
// ON THE CAUSE SPLIT THE REFERENCE NEEDED, AND WHY THIS BRIDGE DOES NOT.
// The reference has to tell two failures apart, because a legacy `Action.define`
// artifact reaches it through `smithers plan --bindings` carrying a success
// schema with no descriptor at all: a walk that reads one has learned nothing
// about the authored Flow, and refusing there would break every descriptor
// minted before schema derivation existed. So its `fail` takes a cause, a
// `non-structural` cause may be swallowed into the weaker contract, and a
// `defect` may not.
//
// This bridge has no such artifact. Every Action it lowers is a same-file class
// whose success descriptor `actionFor` derives structurally, and a declaration
// whose success type is NOT structurally encodable is refused outright there
// (SMITHERS4113) rather than admitted with a weaker contract. `fanOut`,
// `loopWhile`, `dequeue`, `waitBroadcast` and child Flows — the reference's
// other three non-structural sites — are refused wholesale by this subset. So
// every failure this walk can record is a defect, measured rather than assumed,
// and the split would be an enum with one member and an untestable branch.
//
// What DOES transfer is the reason the reference stopped aborting on the first
// failure: the walk keeps going, records every failure, and the diagnostic is
// the first one in a DETERMINISTIC traversal order. Aborting early is not what
// makes that wrong here — with one cause, first-failure and first-defect name
// the same failure. Unordered traversal is. Go randomizes `map` iteration, so a
// walk that visited an output object's fields in map order would report a
// different sentence on different runs of the same compiler over the same
// source. `flowOutputDescriptor` sorts field names with `sortUTF16`, which is
// the ordering the reference's `Object.keys(...).sort()` produces, so the two
// implementations also choose the SAME failure to report.
//
// A future lane that gives this bridge a binding surface for externally
// supplied Action descriptors must add the cause back before it adds the weaker
// contract: swallowing on "this Flow contains a legacy Action" is precisely the
// fail-open the reference closed.

// durableUnknownDescriptor stands in for a descriptor the walk could not derive.
//
// It is a distinct Go type rather than a sentinel value, so "is this the
// placeholder" is a type assertion and cannot be confused with any real
// descriptor. The reference's equivalent is a frozen `{ kind: "null" }` compared
// by object identity, and its own notes record that the guard keeping it out of
// a union is unverifiable there because it serializes exactly like a real `null`
// descriptor. Here it serializes like nothing at all: `durableWriteCanonical`
// panics on an unknown type, so `durableJoinDescriptors`' guard and the
// composite poisoning below are both load-bearing and mutation-visible rather
// than belt-and-braces.
//
// The invariant that makes that safe: this value NEVER nests inside a composite
// descriptor. `flowOutputDescriptor` collapses an object or tuple with any
// unknown part to the unknown descriptor itself, and `durableJoinDescriptors`
// and `durableProjectDescriptor` both short-circuit on it before any canonical
// JSON is written. It can never reach a Plan either, because a walk that records
// a failure refuses the Flow.
type durableUnknownDescriptor struct{}

func durableIsUnknownDescriptor(descriptor any) bool {
	_, unknown := descriptor.(durableUnknownDescriptor)
	return unknown
}

// durableOutputFail records one failure and yields the placeholder so the walk
// can continue instead of aborting at the first one.
type durableOutputFail func(message string) any

// durableDescriptorIndexKey is the reference's `^(0|[1-9][0-9]*)$`: the only
// projection spellings a tuple or array descriptor can answer.
func durableDescriptorIndexKey(key string) bool {
	if key == "0" {
		return true
	}
	if key == "" || key[0] < '1' || key[0] > '9' {
		return false
	}
	for index := 1; index < len(key); index++ {
		if key[index] < '0' || key[index] > '9' {
			return false
		}
	}
	return true
}

// durableJoinDescriptors is `durableCanonicalUnion` that keeps an unknown leg
// unknown instead of folding it into a variant. Losing this guard would send the
// placeholder into `durableCanonicalJSON`.
func durableJoinDescriptors(variants []any) any {
	for _, variant := range variants {
		if durableIsUnknownDescriptor(variant) {
			return durableUnknownDescriptor{}
		}
	}
	return durableCanonicalUnion(variants)
}

// durableProjectDescriptor answers one Plan IR path against a durable
// descriptor, recording a failure for every component the descriptor cannot
// answer. Ported from the reference's `projectDescriptor`, message for message.
func durableProjectDescriptor(descriptor any, path []any, fail durableOutputFail) any {
	if durableIsUnknownDescriptor(descriptor) {
		return descriptor
	}
	if len(path) == 0 {
		return descriptor
	}
	record, isRecord := descriptor.(map[string]any)
	if !isRecord {
		// Defensive: every descriptor this walk reads is built by
		// `durableTypeDescriptor` or by the walk itself, and both produce
		// records. A nil here would be a compiler defect, not an authoring one.
		return fail("projects through a value with no durable descriptor")
	}
	kind, _ := record["kind"].(string)
	if kind == "union" {
		variants, _ := record["variants"].([]any)
		projected := make([]any, 0, len(variants))
		for _, variant := range variants {
			projected = append(projected, durableProjectDescriptor(variant, path, fail))
		}
		return durableJoinDescriptors(projected)
	}
	head, _ := path[0].(string)
	tail := path[1:]
	switch kind {
	case "object":
		fields, _ := record["fields"].([]any)
		for _, item := range fields {
			field, _ := item.(map[string]any)
			if field == nil {
				continue
			}
			if name, _ := field["name"].(string); name == head {
				return durableProjectDescriptor(field["value"], tail, fail)
			}
		}
		return fail("projects missing durable field " + head)
	case "tuple":
		if durableDescriptorIndexKey(head) {
			items, _ := record["items"].([]any)
			index, err := strconv.Atoi(head)
			if err != nil || index >= len(items) {
				return fail("projects missing durable tuple index " + head)
			}
			return durableProjectDescriptor(items[index], tail, fail)
		}
	case "array":
		if durableDescriptorIndexKey(head) {
			return durableProjectDescriptor(record["element"], tail, fail)
		}
	}
	return fail("cannot project " + head + " from durable " + kind)
}

// durableFindNode resolves one Plan node id, descending into branch arms: an
// arm's own output references nodes that live inside that arm's node list, not
// in the Flow's top-level one.
func durableFindNode(nodes []any, id string) map[string]any {
	for _, item := range nodes {
		node, _ := item.(map[string]any)
		if node == nil {
			continue
		}
		if candidate, _ := node["id"].(string); candidate == id {
			return node
		}
		if kind, _ := node["kind"].(string); kind != "branch" {
			continue
		}
		for _, arm := range []string{"whenTrue", "whenFalse"} {
			fragment, _ := node[arm].(map[string]any)
			if fragment == nil {
				continue
			}
			nested, _ := fragment["nodes"].([]any)
			if found := durableFindNode(nested, id); found != nil {
				return found
			}
		}
	}
	return nil
}

// durableSchemaDescriptor reads the structural descriptor out of one durable
// schema, reporting absence rather than assuming it.
func durableSchemaDescriptor(schema map[string]any) (any, bool) {
	if schema == nil {
		return nil, false
	}
	if shape, _ := schema["shape"].(string); shape != "structural" {
		return nil, false
	}
	descriptor, present := schema["descriptor"]
	return descriptor, present
}

// flowNodeDescriptor derives the descriptor one Plan node's value carries, then
// answers the projection path recorded against it.
func (d *durableFunctionLowerer) flowNodeDescriptor(expression map[string]any, fail durableOutputFail) any {
	nodeID, _ := expression["nodeId"].(string)
	path, _ := expression["path"].([]any)
	node := durableFindNode(d.nodes, nodeID)
	if node == nil {
		return fail("references node " + nodeID + ", which this Plan does not declare")
	}
	kind, _ := node["kind"].(string)
	switch kind {
	case "action":
		actionID, _ := node["actionId"].(string)
		action, declared := d.actions[actionID]
		if !declared {
			return fail("references Action " + actionID + ", which this Plan does not declare")
		}
		successSchema, _ := action.descriptor["successSchema"].(map[string]any)
		descriptor, structural := durableSchemaDescriptor(successSchema)
		if !structural {
			// Unreachable today: `actionFor` mints every success schema with
			// `durableStructuralSchema` and refuses a success type it cannot
			// describe. Kept because it is the one site a binding surface would
			// make reachable, and it must fail closed until the cause split
			// above is put back with it.
			return fail("references an Action without a structural success descriptor")
		}
		return durableProjectDescriptor(descriptor, path, fail)
	case "branch":
		whenTrue, _ := node["whenTrue"].(map[string]any)
		whenFalse, _ := node["whenFalse"].(map[string]any)
		joined := durableJoinDescriptors([]any{
			d.flowOutputDescriptor(whenTrue["output"], fail),
			d.flowOutputDescriptor(whenFalse["output"], fail),
		})
		return durableProjectDescriptor(joined, path, fail)
	case "timer":
		// `sleep(...)` completes with no value; the reference derives the same
		// `null` descriptor for a timer node.
		return durableProjectDescriptor(map[string]any{"kind": "null"}, path, fail)
	case "signal":
		payloadSchema, _ := node["payloadSchema"].(map[string]any)
		descriptor, structural := durableSchemaDescriptor(payloadSchema)
		if !structural {
			return fail("references a signal without a structural payload descriptor")
		}
		return durableProjectDescriptor(descriptor, path, fail)
	}
	return fail("references a node without a supported success descriptor")
}

// flowOutputDescriptor derives the descriptor one Plan value expression carries.
func (d *durableFunctionLowerer) flowOutputDescriptor(value any, fail durableOutputFail) any {
	record, isRecord := value.(map[string]any)
	if !isRecord {
		return fail("holds a value expression with no durable descriptor")
	}
	kind, _ := record["kind"].(string)
	switch kind {
	case "literal":
		return map[string]any{"kind": "literal", "value": record["value"]}
	case "input":
		path, _ := record["path"].([]any)
		descriptor, structural := durableSchemaDescriptor(d.inputSchema)
		if !structural {
			return fail("reads a Flow input with no structural descriptor")
		}
		return durableProjectDescriptor(descriptor, path, fail)
	case "node":
		return d.flowNodeDescriptor(record, fail)
	case "object":
		fields, _ := record["fields"].(map[string]any)
		// Sorted, not map order: see the note above. Go randomizes `map`
		// iteration, so an unsorted walk would report a different one of several
		// defective fields on different runs of the same compiler.
		names := make([]string, 0, len(fields))
		for name := range fields {
			names = append(names, name)
		}
		sortUTF16(names)
		derived := make([]any, 0, len(names))
		unknown := false
		for _, name := range names {
			// Every field is walked even after one is unknown, so a defect in a
			// later field is still recorded rather than masked by an earlier one.
			field := d.flowOutputDescriptor(fields[name], fail)
			if durableIsUnknownDescriptor(field) {
				unknown = true
				continue
			}
			derived = append(derived, map[string]any{"name": name, "optional": false, "value": field})
		}
		if unknown {
			return durableUnknownDescriptor{}
		}
		return map[string]any{"fields": derived, "kind": "object"}
	case "array":
		items, _ := record["items"].([]any)
		derived := make([]any, 0, len(items))
		unknown := false
		for _, item := range items {
			element := d.flowOutputDescriptor(item, fail)
			if durableIsUnknownDescriptor(element) {
				unknown = true
				continue
			}
			derived = append(derived, element)
		}
		if unknown {
			return durableUnknownDescriptor{}
		}
		return map[string]any{"items": derived, "kind": "tuple"}
	}
	return fail("holds an unsupported durable value expression")
}

// durablePlanValue is one value a Plan NODE consumes, paired with the subject a
// diagnostic about it names.
type durablePlanValue struct {
	subject string
	value   any
}

// durablePlanNodeValues collects every value a Plan node consumes, in a
// deterministic order: node order (which is source order, because nodes are
// appended as they lower), and `whenTrue` before `whenFalse` inside a branch.
//
// The Flow OUTPUT is deliberately absent: it is walked separately because its
// descriptor is the Flow's success contract, while these are derived only to be
// discarded.
//
// This exists because the projection rule was half a rule. The output walk
// refuses `return { count: input.items.length }`, but `Step.run({ key:
// input.items.length })` reached no descriptor at all: the Plan carried
// `{"kind":"input","path":["items","length"]}` into the Action's input and the
// reference runtime's `pathValue` faulted on it with a `ProjectionDefect`. Same
// expression language, same descriptors, same question — so the same walk
// answers it, rather than a second walk that learns a shape this one does not.
func durablePlanNodeValues(nodes []any) []durablePlanValue {
	values := make([]durablePlanValue, 0, len(nodes))
	var visit func(fragmentNodes []any)
	visit = func(fragmentNodes []any) {
		for _, item := range fragmentNodes {
			node, _ := item.(map[string]any)
			if node == nil {
				continue
			}
			kind, _ := node["kind"].(string)
			switch kind {
			case "action":
				actionID, _ := node["actionId"].(string)
				values = append(values, durablePlanValue{subject: "Action " + actionID + " input", value: node["input"]})
			case "timer":
				values = append(values, durablePlanValue{subject: "sleep duration", value: node["durationMs"]})
			case "branch":
				values = append(values, durablePlanValue{subject: "branch condition", value: node["condition"]})
				for _, arm := range []string{"whenTrue", "whenFalse"} {
					fragment, _ := node[arm].(map[string]any)
					if fragment == nil {
						continue
					}
					nested, _ := fragment["nodes"].([]any)
					visit(nested)
				}
			case "signal":
				// A suspension consumes no Plan value; its payload schema is the
				// contract and is already pinned into the node's identity.
			default:
				// Fail closed. This subset mints exactly `action`, `branch`,
				// `timer` and `signal`; a fifth kind added without extending
				// this switch would otherwise reopen the hole silently, which is
				// how the Action-input case survived the output check.
				values = append(values, durablePlanValue{
					subject: "Plan node of kind " + kind,
					value:   nil,
				})
			}
		}
	}
	visit(nodes)
	return values
}

// checkFlowBoundary refuses a Flow that reads something its own durable
// descriptors cannot answer — in its output, and in every value one of its Plan
// nodes consumes. Reported at the durable source function, which is where the
// reference reports it, so both backends name one position.
//
// Output failures are collected first and named first, deliberately: every
// program that refuses today refuses on an output failure and must keep its
// exact sentence. The node-value walk is strictly narrowing — only a Flow whose
// output is clean can newly refuse — so no already-refused program changes its
// answer.
//
// On accepting, it records the output descriptor as `d.successSchema`. The
// reference does exactly this: `flowSchemas` runs the same walk and, when the
// output produced no failure, emits `structuralSchema("success",
// successDescriptor)`. Its two other outcomes are unreachable here — the weaker
// `derivedSchema("success")` needs a legacy Action descriptor this bridge has no
// binding surface for, and its `throw` is this function's refusal — so
// "accepted" and "structural success schema" are the same condition, and the
// schema is set on exactly the path that proved the descriptor is complete.
func (d *durableFunctionLowerer) checkFlowBoundary(function *ast.Node, output durableValue) bool {
	failures := make([]string, 0, 1)
	walk := func(value any, subject string) any {
		return d.flowOutputDescriptor(value, func(reason string) any {
			failures = append(failures, subject+" "+reason)
			return durableUnknownDescriptor{}
		})
	}
	successDescriptor := walk(output.ir, "Flow output")
	for _, entry := range durablePlanNodeValues(d.nodes) {
		walk(entry.value, entry.subject)
	}
	if len(failures) == 0 {
		d.successSchema = durableStructuralSchema("success", successDescriptor)
		return true
	}
	d.fail(function, "4110", "durable Flow boundary is not structurally encodable: "+failures[0])
	return false
}

func (d *durableFunctionLowerer) isChildFlowRun(node *ast.Node) bool {
	if !ast.IsCallExpression(node) || !ast.IsPropertyAccessExpression(node.AsCallExpression().Expression) {
		return false
	}
	access := node.AsCallExpression().Expression.AsPropertyAccessExpression()
	return d.owner.durableIntrinsic(access.Name(), "run") && !d.owner.durableIntrinsic(access.Expression, "Action")
}

func (d *durableFunctionLowerer) stableNodeID(node *ast.Node, identity map[string]any) (string, bool) {
	occurrenceKey := durableDigest(identity)
	occurrence := d.occurrences[occurrenceKey]
	d.occurrences[occurrenceKey] = occurrence + 1
	withOccurrence := durableCopyMap(identity)
	withOccurrence["occurrence"] = occurrence
	id := "src-" + durableDigest(withOccurrence)[:24]
	if d.nodeIDs[id] {
		d.fail(node, "4199", "stable durable node id collision "+id)
		return "", false
	}
	d.nodeIDs[id] = true
	return id, true
}

func (d *durableFunctionLowerer) debug(node *ast.Node, label string) map[string]any {
	start := scanner.SkipTrivia(d.owner.file.Text(), node.Pos())
	line, character := scanner.GetECMALineAndUTF16CharacterOfPosition(d.owner.file, start)
	return map[string]any{
		"callSite": fmt.Sprintf("%s:%d:%d", d.logicalFile, line+1, int(character)+1),
		"label":    label,
	}
}

func durableControlDependencies(sequence string) []any {
	if sequence == "" {
		return []any{}
	}
	return []any{sequence}
}

func durableExpressionDependencies(value any) []any {
	set := make(map[string]bool)
	var visit func(any)
	visit = func(value any) {
		switch value := value.(type) {
		case map[string]any:
			if value["kind"] == "node" {
				if id, ok := value["nodeId"].(string); ok {
					set[id] = true
				}
				return
			}
			for _, nested := range value {
				visit(nested)
			}
		case []any:
			for _, nested := range value {
				visit(nested)
			}
		}
	}
	visit(value)
	keys := make([]string, 0, len(set))
	for key := range set {
		keys = append(keys, key)
	}
	// Byte order is CORRECT here and is left alone deliberately. These are Plan
	// node ids minted by `stableNodeID` as `src-` plus 24 hex digits, so they
	// are provably ASCII and the two orders cannot disagree. Converting a site
	// that does not need the rule would hide which sites do.
	sort.Strings(keys)
	result := make([]any, len(keys))
	for index, key := range keys {
		result[index] = key
	}
	return result
}

func durableLegacySchema(role string) map[string]any {
	semantic := map[string]any{
		"format":        "canonical-json",
		"role":          role,
		"schemaVersion": 1,
		"shape":         "json-value",
		"source":        "compiler-derived-poc-stub",
	}
	schema := durableCopyMap(semantic)
	schema["digest"] = durableDigest(semantic)
	return schema
}

func durableStructuralSchema(role string, descriptor any) map[string]any {
	semantic := map[string]any{
		"descriptor":    descriptor,
		"format":        "canonical-json",
		"role":          role,
		"schemaVersion": 1,
		"shape":         "structural",
		"source":        "compiler-derived",
	}
	schema := durableCopyMap(semantic)
	schema["digest"] = durableDigest(semantic)
	return schema
}

// flowErrorSchema is the reference's `flowErrorSchema`: the Flow's failure
// channel is the union of its Actions' failure channels.
//
// The weakening is CONTAGIOUS and that is the reference's rule, not a shortcut:
// one Action whose error schema states no descriptor leaves the whole Flow's
// error channel undescribed, because a union missing one arm would claim the
// Flow can only fail in the ways the other Actions declare. `Result<Success,
// Error>` is the channel that reaches it — see durableErrorSchemaFor.
func (d *durableFunctionLowerer) flowErrorSchema(actionIDs []string) map[string]any {
	descriptors := make([]any, 0, len(actionIDs))
	for _, id := range actionIDs {
		schema, _ := d.actions[id].descriptor["errorSchema"].(map[string]any)
		descriptor, structural := durableSchemaDescriptor(schema)
		if !structural {
			return durableLegacySchema("error")
		}
		descriptors = append(descriptors, descriptor)
	}
	if len(descriptors) == 0 {
		return durableLegacySchema("error")
	}
	return durableStructuralSchema("error", durableCanonicalUnion(descriptors))
}

func (d *durableFunctionLowerer) plan(output durableValue) map[string]any {
	actionIDs := make([]string, 0, len(d.actions))
	for id := range d.actions {
		actionIDs = append(actionIDs, id)
	}
	sortUTF16(actionIDs)
	actions := make([]any, 0, len(actionIDs))
	requirements := make([]any, 0, len(actionIDs))
	for _, id := range actionIDs {
		actions = append(actions, d.actions[id].descriptor)
		requirements = append(requirements, id)
	}
	flowSchemas := map[string]any{
		"input":   d.inputSchema,
		"success": d.successSchema,
	}
	// Defensive only, and provably unreached: `lowerFunction` sets `inputSchema`
	// before it lowers a single statement and `checkFlowBoundary` sets
	// `successSchema` on the only path that returns an accepted Flow, so a nil
	// here would be a compiler defect. Kept as the weakened schema rather than a
	// panic because the reference's envelope admits it, and removed only when
	// something makes the two states distinguishable.
	if d.inputSchema == nil {
		flowSchemas["input"] = durableLegacySchema("input")
	}
	if d.successSchema == nil {
		flowSchemas["success"] = durableLegacySchema("success")
	}
	if len(actionIDs) != 0 {
		flowSchemas["error"] = d.flowErrorSchema(actionIDs)
	}
	semantic := map[string]any{
		"actions":       actions,
		"flowId":        d.flowID,
		"flowSchemas":   flowSchemas,
		"flowVersion":   d.flowVersion,
		"formatVersion": 1,
		"nodes":         d.nodes,
		"output":        output.ir,
		"requirements":  requirements,
	}
	plan := durableCopyMap(semantic)
	plan["digest"] = durableDigest(semantic)
	return plan
}

func durableCopyMap(source map[string]any) map[string]any {
	copy := make(map[string]any, len(source)+1)
	for key, value := range source {
		copy[key] = value
	}
	return copy
}

// durableCanonicalJSON is byte-compatible with the reference's recursive
// object-key ordering and JSON.stringify scalar rendering. In particular,
// JavaScript orders keys by UTF-16 code unit and uses Number::toString rather
// than Go's UTF-8 ordering and float formatter.
func durableCanonicalJSON(value any) string {
	var out strings.Builder
	durableWriteCanonical(&out, value)
	return out.String()
}

func durableWriteCanonical(out *strings.Builder, value any) {
	switch value := value.(type) {
	case nil:
		out.WriteString("null")
	case bool:
		if value {
			out.WriteString("true")
		} else {
			out.WriteString("false")
		}
	case string:
		out.WriteString(ctQuote(value))
	case int:
		out.WriteString(strconv.Itoa(value))
	case float64:
		out.WriteString(ctNumberText(value))
	case []string:
		out.WriteByte('[')
		for index, item := range value {
			if index != 0 {
				out.WriteByte(',')
			}
			durableWriteCanonical(out, item)
		}
		out.WriteByte(']')
	case []any:
		out.WriteByte('[')
		for index, item := range value {
			if index != 0 {
				out.WriteByte(',')
			}
			durableWriteCanonical(out, item)
		}
		out.WriteByte(']')
	case map[string]any:
		keys := make([]string, 0, len(value))
		for key := range value {
			keys = append(keys, key)
		}
		sortUTF16(keys)
		out.WriteByte('{')
		for index, key := range keys {
			if index != 0 {
				out.WriteByte(',')
			}
			out.WriteString(ctQuote(key))
			out.WriteByte(':')
			durableWriteCanonical(out, value[key])
		}
		out.WriteByte('}')
	default:
		panic(fmt.Sprintf("unsupported durable canonical JSON value %T", value))
	}
}

func durableDigest(value any) string {
	sum := sha256.Sum256([]byte(durableCanonicalJSON(value)))
	return hex.EncodeToString(sum[:])
}

// durableLiteralExpression recursively materializes Plan data with the fork's
// AST factory. It deliberately does not parse generated source or splice text.
func (l *lowerer) durableLiteralExpression(value any) *ast.Node {
	switch value := value.(type) {
	case nil:
		return l.factory.NewKeywordExpression(ast.KindNullKeyword)
	case bool:
		if value {
			return l.factory.NewKeywordExpression(ast.KindTrueKeyword)
		}
		return l.factory.NewKeywordExpression(ast.KindFalseKeyword)
	case string:
		return l.factory.NewStringLiteral(value, ast.TokenFlagsNone)
	case int:
		return l.factory.NewNumericLiteral(strconv.Itoa(value), ast.TokenFlagsNone)
	case float64:
		return l.factory.NewNumericLiteral(ctNumberText(value), ast.TokenFlagsNone)
	case []string:
		elements := make([]*ast.Node, len(value))
		for index, item := range value {
			elements[index] = l.durableLiteralExpression(item)
		}
		return l.factory.NewArrayLiteralExpression(l.factory.NewNodeList(elements), false)
	case []any:
		elements := make([]*ast.Node, len(value))
		for index, item := range value {
			elements[index] = l.durableLiteralExpression(item)
		}
		return l.factory.NewArrayLiteralExpression(l.factory.NewNodeList(elements), false)
	case map[string]any:
		keys := make([]string, 0, len(value))
		for key := range value {
			keys = append(keys, key)
		}
		sortUTF16(keys)
		properties := make([]*ast.Node, 0, len(keys))
		for _, key := range keys {
			propertyName := l.factory.NewStringLiteral(key, ast.TokenFlagsNone)
			if key == "__proto__" {
				// A quoted __proto__ in an object literal mutates the prototype
				// instead of defining an own data field. The computed spelling is
				// still factory-built and preserves the canonical JSON key.
				propertyName = l.factory.NewComputedPropertyName(propertyName)
			}
			properties = append(properties, l.factory.NewPropertyAssignment(
				nil,
				propertyName,
				nil,
				nil,
				l.durableLiteralExpression(value[key]),
			))
		}
		return l.factory.NewObjectLiteralExpression(l.factory.NewNodeList(properties), true)
	default:
		panic(fmt.Sprintf("unsupported durable literal %T", value))
	}
}
