package main

// Go-native Smithers lowering, executed inside the pinned TypeScript fork.
//
// This file is compiled into the fork's `cmd/tsc` package through the build
// overlay, so it can drive the fork's own parser, binder, checker, node
// factory, and printer directly. It replaces the identity transform for
// `LoweringInternal` requests: authored `.sm` statements are rewritten into
// Result variant constructions and nominal error dispatch on the real
// AST, and the fork's printer emits the lowered TypeScript together with an
// authored -> lowered source map.
//
// Nothing here inspects source text. Every decision is made from a resolved
// symbol or a checker type.

const (
	// preludeLogicalPath is the compiler-owned prelude injected into every
	// internally lowered project. It declares the runtime representation of
	// `Result`, publishes its type name globally, and
	// declares the nominal `Error.match` intrinsic, so authored `.sm` files
	// never import anything.
	preludeLogicalPath = "__smithers_prelude.ts"
	preludeVirtualName = "/src/" + preludeLogicalPath
	preludeModuleBase  = "__smithers_prelude.js"

	// tempPrefix names the temporaries introduced by postfix propagation, and
	// matchPrefix those introduced by nominal error matching.
	tempPrefix  = "__smithersUnwrapped"
	matchPrefix = "__smithersMatched"

// preludeImports is the fixed, ordered set of prelude exports the lowering may
// reference, together with the local aliases it imports them under. The local
// names are only ever produced by the lowering; authored code is never matched
// against them.
var preludeImports = []struct {
	exported string
	local    string
}{
	{exported: "SmithersOk", local: "__smithersOk"},
	{exported: "SmithersErr", local: "__smithersErr"},
	{exported: "Panic", local: "__smithersPanic"},
	{exported: "smithersMatchFailed", local: "__smithersMatchFailed"},
	{exported: "smithersTry", local: "__smithersTry"},
	{exported: "smithersTryPromise", local: "__smithersTryPromise"},
	{exported: "smithersResultAll", local: "__smithersResultAll"},
}

func preludeLocalName(exported string) string {
	for _, entry := range preludeImports {
		if entry.exported == exported {
			return entry.local
		}
	}
	panic("unknown prelude export " + exported)
}

// preludeText is the compiler-owned prelude. The classes are the runtime
// representation of a Result; the `declare global` block publishes the type
// name authored `.sm` files use without importing
// anything, and declares `Error.match` so nominal matching resolves to a
// compiler-owned declaration rather than to a spelling.
const preludeText = `export class Panic extends Error {
	constructor(readonly cause?: unknown) {
		super(cause instanceof Error ? cause.message : cause === undefined ? "panic" : String(cause));
		this.name = "Panic";
	}
}
export function panic(cause?: unknown): never {
	throw new Panic(cause);
}
export class SmithersExpectFailed extends Panic {
    constructor(message: string, readonly reason: unknown) {
        super(reason);
		this.message = message;
        this.name = "SmithersExpectFailed";
    }
}
const smithersContextFrames: Map<Function, unknown>[] = [];
export abstract class Context {
	static context<C extends abstract new (...args: never[]) => Context>(this: C): InstanceType<C> {
		for (let index = smithersContextFrames.length - 1; index >= 0; index--) {
			const value = smithersContextFrames[index].get(this);
			if (value !== undefined) return value as InstanceType<C>;
		}
		throw new Panic("unsatisfied Context requirement");
	}
}
export interface SmithersLayer<P> {
	readonly __smithersLayer: { readonly provides: P };
	readonly providers: Map<Function, unknown>;
}
export type Layer<P> = SmithersLayer<P>;
export const Layer = {
	succeed<C extends abstract new (...args: never[]) => Context>(capability: C, implementation: InstanceType<C>): SmithersLayer<C> {
		return { __smithersLayer: { provides: capability }, providers: new Map([[capability, implementation]]) };
	},
	merge<const L extends readonly SmithersLayer<unknown>[]>(...layers: L): SmithersLayer<L[number] extends SmithersLayer<infer P> ? P : never> {
		const providers = new Map<Function, unknown>();
		for (const layer of layers) for (const [capability, implementation] of layer.providers) providers.set(capability, implementation);
		return { __smithersLayer: { provides: undefined as never }, providers };
	},
	provide<L extends SmithersLayer<unknown>, A>(layer: L, body: () => A): A {
		smithersContextFrames.push(layer.providers);
		try { return body(); } finally { smithersContextFrames.pop(); }
	},
};
function smithersCall<A>(fallback: A | (() => A)): A {
    return typeof fallback === "function" ? (fallback as () => A)() : fallback;
}
export class SmithersOk<A> {
    readonly ok: true = true;
    constructor(readonly value: A) { }
    isOk(): boolean { return true; }
    isError(): boolean { return false; }
    unwrap(): A { return this.value; }
    unwrapOr(fallback: A | (() => A)): A { return this.value; }
    expect(message: string): A { return this.value; }
    match<R>(handlers: { ok: (value: A) => R; error: (error: never) => R }): R { return handlers.ok(this.value); }
    map<B>(mapper: (value: A) => B): SmithersOk<B> { return new SmithersOk(mapper(this.value)); }
    mapError<F>(mapper: (error: never) => F): SmithersOk<A> { return this; }
    andThen<B, F>(mapper: (value: A) => SmithersOk<B> | SmithersErr<F>): SmithersOk<B> | SmithersErr<F> { return mapper(this.value); }
	recover<B>(mapper: (error: never) => B): SmithersOk<A> { return this; }
	// tap observes the success variant and MUST NOT replace the Result it
	// observes: specification/failures.mdx, "Matching and Transformation",
	// lists tap/tapError among the required operations, and the observation is
	// defined to leave the observed Result unchanged. The receiver itself is
	// returned, not a copy, so identity is preserved as well as value.
	tap(observer: (value: A) => void): SmithersOk<A> { observer(this.value); return this; }
	tapError(observer: (error: never) => void): SmithersOk<A> { return this; }
}
export class SmithersErr<E> {
    readonly ok: false = false;
    constructor(readonly error: E) { }
    isOk(): boolean { return false; }
    isError(): boolean { return true; }
    unwrap(): never { throw this.error; }
    unwrapOr<A>(fallback: A | (() => A)): A { return smithersCall(fallback); }
    expect(message: string): never { throw new SmithersExpectFailed(message, this.error); }
    match<R>(handlers: { ok: (value: never) => R; error: (error: E) => R }): R { return handlers.error(this.error); }
    map<B>(mapper: (value: never) => B): SmithersErr<E> { return this; }
    mapError<F>(mapper: (error: E) => F): SmithersErr<F> { return new SmithersErr(mapper(this.error)); }
    andThen<B, F>(mapper: (value: never) => SmithersOk<B> | SmithersErr<F>): SmithersErr<E> { return this; }
	// recover handles a recoverable failure and MUST NOT intercept the
	// distinguished panic channel: specification/failures.mdx, "Panic", makes a
	// panic a defect only an explicit boundary may observe, and
	// poc/src/runtime/result.ts passes it through unchanged. A Panic therefore
	// survives the call as the error variant of the result.
	//
	// The mapper parameter stays E rather than Exclude<E, Panic>: .recover is
	// called on the Ok|Err union, whose combined signature intersects the two
	// parameter types, and a conditional type there collapses the mapper's
	// parameter to never at every call site. The runtime guard, not the
	// parameter type, is what keeps the mapper from seeing a Panic.
	recover<B>(mapper: (error: E) => B): SmithersOk<B> | SmithersErr<Extract<E, Panic>> {
		if (this.error instanceof Panic) return this as unknown as SmithersErr<Extract<E, Panic>>;
		return new SmithersOk(mapper(this.error));
	}
	// tap fires only on success and tapError only on failure; neither replaces
	// the Result. Unlike recover, tapError observes a Panic as well: observing
	// is not handling, so passing the distinguished channel to an observer does
	// not convert it into a success.
	tap(observer: (value: never) => void): SmithersErr<E> { return this; }
	tapError(observer: (error: E) => void): SmithersErr<E> { observer(this.error); return this; }
}
type SmithersAnyResult = SmithersOk<any> | SmithersErr<any>;
type SmithersOkValue<R> = R extends SmithersOk<infer A> ? A : never;
type SmithersErrError<R> = R extends SmithersErr<infer E> ? E : never;
// smithersResultAll collects every success and STOPS at the first error. The
// left-to-right scan is the contract, not an implementation detail: with two
// failing entries the collected Result must carry the first, so a right-to-left
// fold would be wrong.
export function smithersResultAll<const R extends readonly SmithersAnyResult[]>(
	results: R,
): SmithersOk<SmithersOkValue<R[number]>[]> | SmithersErr<SmithersErrError<R[number]>>;
export function smithersResultAll(results: readonly SmithersAnyResult[]): SmithersAnyResult {
	const values: unknown[] = [];
	for (const result of results) {
		if (!result.ok) return result;
		values.push(result.value);
	}
	return new SmithersOk(values);
}
type SmithersErrorConstructor<E extends Error = Error> = abstract new (...args: never[]) => E;
function smithersErrorIs<E extends Error>(error: unknown, type: SmithersErrorConstructor<E>): error is E {
	if (!(error instanceof Error) || typeof type !== "function") return false;
	try { return error instanceof type; } catch { return false; }
}
function smithersErrorMatches(error: unknown, ...types: readonly SmithersErrorConstructor[]): boolean {
	return types.some((type) => smithersErrorIs(error, type));
}
function smithersRootCause(error: Error): unknown {
	let current: unknown = error;
	const seen = new Set<object>();
	while (typeof current === "object" && current !== null && !seen.has(current)) {
		seen.add(current);
		const descriptor = Object.getOwnPropertyDescriptor(current, "cause");
		if (!descriptor || !("value" in descriptor) || descriptor.value === undefined) return current;
		current = descriptor.value;
	}
	return current;
}
for (const [name, method] of Object.entries({
	is(this: Error, type: SmithersErrorConstructor): boolean { return smithersErrorIs(this, type); },
	matches(this: Error, ...types: readonly SmithersErrorConstructor[]): boolean { return smithersErrorMatches(this, ...types); },
	rootCause(this: Error): unknown { return smithersRootCause(this); },
})) {
	if (Object.getOwnPropertyDescriptor(Error.prototype, name) === undefined) {
		Object.defineProperty(Error.prototype, name, {
			value: method,
			enumerable: false,
			configurable: false,
			writable: false,
		});
	}
}
export function smithersMatchFailed(error: never): never {
    throw new TypeError("non-exhaustive Error match");
}
export function smithersTry<A>(body: () => A): SmithersOk<A> | SmithersErr<Panic>;
export function smithersTry<A, E extends Error>(body: () => A, expected: abstract new (...args: never[]) => E): SmithersOk<A> | SmithersErr<E | Panic>;
export function smithersTry<A, E extends Error>(body: () => A, expected?: abstract new (...args: never[]) => E): SmithersOk<A> | SmithersErr<E | Panic> {
	try { return new SmithersOk(body()); }
	catch (cause) { return new SmithersErr(expected && cause instanceof expected ? cause : new Panic(cause)); }
}
export async function smithersTryPromise<A>(body: () => PromiseLike<A>): Promise<SmithersOk<A> | SmithersErr<Panic>>;
export async function smithersTryPromise<A, E extends Error>(body: () => PromiseLike<A>, expected: abstract new (...args: never[]) => E): Promise<SmithersOk<A> | SmithersErr<E | Panic>>;
export async function smithersTryPromise<A, E extends Error>(body: () => PromiseLike<A>, expected?: abstract new (...args: never[]) => E): Promise<SmithersOk<A> | SmithersErr<E | Panic>> {
	try { return new SmithersOk(await body()); }
	catch (cause) { return new SmithersErr(expected && cause instanceof expected ? cause : new Panic(cause)); }
}
declare global {
    type Result<A, E> = SmithersOk<A> | SmithersErr<E>;
    // The author-facing Result namespace carries only the documented
    // boundary adapters. Result.ok/err are deliberately absent: ordinary
    // return and throw lifting construct the variants.
    const Result: {
        try: typeof smithersTry;
        tryPromise: typeof smithersTryPromise;
        all: typeof smithersResultAll;
    };
    // Reflect.panic is the ambient spelling of the same distinguished channel
    // that panic(...) from smithers:exceptions charges. It is declared, never
    // emitted: every reachable use is rewritten into the error variant, and a
    // use in any other position is rejected, so the runtime Reflect object is
    // never asked for a member it does not have.
    namespace Reflect {
        function panic(cause?: unknown): never;
    }
    interface Error {
        is<E extends Error>(type: SmithersErrorConstructor<E>): this is E;
        matches<const Types extends readonly SmithersErrorConstructor[]>(...types: Types): boolean;
        match<R>(handlers: Record<string, (error: never) => R>): R;
        matchPartial<R, F>(handlers: Record<string, (error: never) => R>, fallback: (error: Error) => F): R | F;
        rootCause(): unknown;
    }
}
`

var compilerModuleSpecifiers = []string{
	"smthrs/context",
	"smthrs/provider",
	"smithers:exceptions",
	"smithers:comptime",
	"smithers:flows",
}

func configureInternalModules(options *core.CompilerOptions) {
	paths := collections.NewOrderedMapWithSizeHint[string, []string](len(compilerModuleSpecifiers))
	for _, specifier := range compilerModuleSpecifiers {
		if durableModuleSpecifiers[specifier] {
			// Durable declarations have a distinct module identity: resolving an
			// export from another compiler-owned module must not make it durable.
			paths.Set(specifier, []string{durableModuleAlias})
			continue
		}
		if specifier == comptimeModuleSpecifier {
			// The comptime intrinsic has its own compiler-owned module, so its
			// exported bindings cannot be reached through any other specifier.
			paths.Set(specifier, []string{comptimeModuleAlias})
			continue
		}
		paths.Set(specifier, []string{"./" + preludeLogicalPath})
	}
	options.Paths = paths
	options.PathsBasePath = "/src"
}

// preludeSymbols holds the declaration nodes of the compiler-owned prelude.
// Recognition is by node identity against these declarations, never by name
// spelling at the use site: a user-declared `Result` resolves to a different
// symbol and is left alone.
type preludeSymbols struct {
	checker *checker.Checker
	module  *ast.Symbol
	error   *ast.Symbol
	panic   *ast.Symbol
	context *ast.Symbol

	resultAlias  *ast.Node // the `Result` TypeAliasDeclaration in the prelude
	resultSymbol *ast.Symbol

	okClass  *ast.Node // the `SmithersOk` ClassDeclaration
	errClass *ast.Node // the `SmithersErr` ClassDeclaration

	resultUnwrap      map[*ast.Node]struct{} // retired `unwrap` members on the Result variants
	matchDecls        map[*ast.Node]struct{} // the global `Error.match` MethodSignature
	matchPartialDecls map[*ast.Node]struct{} // the global `Error.matchPartial` MethodSignature
	errorRuntimeDecls map[*ast.Node]struct{} // global Error helpers implemented by prelude side effects
	expectDecls       map[*ast.Node]struct{} // `expect` on the Result variants
	// resultConsumerDecls contains every receiver method that discharges a
	// Result must-consume obligation. The use-site member is recognized by its
	// prelude declaration identity, never by its spelling.
	resultConsumerDecls map[*ast.Node]struct{}
	// resultInspectionDecls contains the compiler-owned discriminant and payload
	// properties whose read counts as inspecting a Result. Authored reads are
	// resolved back to these declarations rather than matched by property name.
	resultInspectionDecls map[*ast.Node]struct{}
	contextDecls          map[*ast.Node]struct{} // the compiler-owned `Context.context`
	panicDecls            map[*ast.Node]struct{} // the compiler-owned `panic(...)` function
	// reflectPanicDecls is the ambient `Reflect.panic` declaration. It is a
	// separate set from panicDecls only so the prelude completeness check can
	// count each spelling once; both reach the same channel.
	reflectPanicDecls map[*ast.Node]struct{}
	layerDecls        map[string]map[*ast.Node]struct{}
	// resultNamespaceDecls are the members of the global `Result` value:
	// `try`/`tryPromise`, the two documented adapters that turn a throwing (or
	// rejecting) body into a Result whose failure channel retains the
	// distinguished Panic case, and `all`, the collecting combinator. None of
	// the three has a runtime existence under the `Result` spelling, so the
	// lowering replaces the callee with the prelude export that implements it.
	resultNamespaceDecls map[string]map[*ast.Node]struct{}
	// promiseCombinatorDecls are the ambient Promise constructor's own static
	// collection methods. Capturing their declaration identity here prevents a
	// same-spelled authored object from acquiring structured-concurrency trust.
	promiseCombinatorDecls map[string]map[*ast.Node]struct{}
}

// loadPreludeSymbols locates the prelude's declarations in the checked Program
// and resolves their symbols through the fork's checker.
func loadPreludeSymbols(program *compiler.Program, typeChecker *checker.Checker) (*preludeSymbols, error) {
	var preludeFile *ast.SourceFile
	for _, file := range program.SourceFiles() {
		if file.FileName() == preludeVirtualName {
			preludeFile = file
			break
		}
	}
	if preludeFile == nil {
		return nil, fmt.Errorf("the compiler-owned prelude %q is missing from the program", preludeLogicalPath)
	}

	symbols := &preludeSymbols{
		checker:               typeChecker,
		resultUnwrap:          make(map[*ast.Node]struct{}),
		matchDecls:            make(map[*ast.Node]struct{}),
		matchPartialDecls:     make(map[*ast.Node]struct{}),
		errorRuntimeDecls:     make(map[*ast.Node]struct{}),
		expectDecls:           make(map[*ast.Node]struct{}),
		resultConsumerDecls:   make(map[*ast.Node]struct{}),
		resultInspectionDecls: make(map[*ast.Node]struct{}),
		contextDecls:          make(map[*ast.Node]struct{}),
		panicDecls:            make(map[*ast.Node]struct{}),
		reflectPanicDecls:     make(map[*ast.Node]struct{}),
		layerDecls: map[string]map[*ast.Node]struct{}{
			"succeed": {},
			"merge":   {},
			"provide": {},
		},
		resultNamespaceDecls: map[string]map[*ast.Node]struct{}{
			"try":        {},
			"tryPromise": {},
			"all":        {},
		},
		promiseCombinatorDecls: map[string]map[*ast.Node]struct{}{
			"all": {}, "allSettled": {}, "race": {}, "any": {},
			"allKeyed": {}, "allSettledKeyed": {},
		},
	}
	var errorInterface *ast.Node
	var panicClass *ast.Node
	var contextClass *ast.Node
	var walk func(node *ast.Node) bool
	walk = func(node *ast.Node) bool {
		switch {
		case ast.IsTypeAliasDeclaration(node):
			if declarationName(node) == "Result" {
				symbols.resultAlias = node
			}
		case ast.IsClassDeclaration(node):
			switch declarationName(node) {
			case "SmithersOk":
				symbols.okClass = node
				collectResultInspectionMembers(node, symbols.resultInspectionDecls)
				for _, name := range resultConsumerMembers {
					collectMembers(node, ast.IsMethodDeclaration, name, symbols.resultConsumerDecls)
				}
				collectMembers(node, ast.IsMethodDeclaration, "unwrap", symbols.resultUnwrap)
				collectMembers(node, ast.IsMethodDeclaration, "expect", symbols.expectDecls)
			case "SmithersErr":
				symbols.errClass = node
				collectResultInspectionMembers(node, symbols.resultInspectionDecls)
				for _, name := range resultConsumerMembers {
					collectMembers(node, ast.IsMethodDeclaration, name, symbols.resultConsumerDecls)
				}
				collectMembers(node, ast.IsMethodDeclaration, "unwrap", symbols.resultUnwrap)
				collectMembers(node, ast.IsMethodDeclaration, "expect", symbols.expectDecls)
			case "Panic":
				panicClass = node
			case "Context":
				contextClass = node
				collectMembers(node, ast.IsMethodDeclaration, "context", symbols.contextDecls)
			}
			return false
		case ast.IsModuleDeclaration(node) && declarationName(node) == "Reflect":
			// The ambient `Reflect.panic` lives in its own namespace inside the
			// `declare global` block. Collecting it here rather than letting the
			// FunctionDeclaration case below claim it keeps the two authored
			// spellings — the imported `panic` and the ambient `Reflect.panic` —
			// countable one by one in the completeness check.
			collectMembers(moduleBodyOf(node), ast.IsFunctionDeclaration, "panic", symbols.reflectPanicDecls)
			return false
		case ast.IsFunctionDeclaration(node):
			if declarationName(node) == "panic" {
				symbols.panicDecls[node] = struct{}{}
			}
			return false
		case ast.IsInterfaceDeclaration(node):
			if declarationName(node) == "Error" {
				errorInterface = node
				collectMembers(node, isMethodSignature, "match", symbols.matchDecls)
				collectMembers(node, isMethodSignature, "matchPartial", symbols.matchPartialDecls)
				for _, name := range []string{"is", "matches", "rootCause"} {
					collectMembers(node, isMethodSignature, name, symbols.errorRuntimeDecls)
				}
			}
			return false
		case ast.IsVariableDeclaration(node):
			if declarationName(node) == "Result" && node.AsVariableDeclaration().Initializer == nil {
				typeNode := node.AsVariableDeclaration().Type
				if typeNode != nil && typeNode.Kind == ast.KindTypeLiteral {
					for _, member := range typeNode.AsTypeLiteralNode().Members.Nodes {
						if declarations, ok := symbols.resultNamespaceDecls[declarationName(member)]; ok {
							declarations[member] = struct{}{}
						}
					}
				}
				return false
			}
			if declarationName(node) == "Layer" {
				initializer := node.AsVariableDeclaration().Initializer
				if initializer != nil && ast.IsObjectLiteralExpression(initializer) {
					for _, member := range initializer.AsObjectLiteralExpression().Properties.Nodes {
						if declarations, ok := symbols.layerDecls[declarationName(member)]; ok {
							declarations[member] = struct{}{}
						}
					}
				}
				return false
			}
		}
		node.ForEachChild(walk)
		return false
	}
	preludeFile.AsNode().ForEachChild(walk)

	// Resolve Promise at a compiler-owned location where authored shadows are
	// out of scope, then retain the ambient constructor members by declaration
	// identity. The extended keyed names are optional library additions; the
	// standard `all` member must always be present in the embedded ES2022 libs.
	promiseSymbol := typeChecker.ResolveName("Promise", preludeFile.AsNode(), ast.SymbolFlagsValue, false)
	if promiseSymbol != nil {
		promiseType := typeChecker.GetTypeOfSymbolAtLocation(promiseSymbol, preludeFile.AsNode())
		for name, declarations := range symbols.promiseCombinatorDecls {
			property := typeChecker.GetPropertyOfType(promiseType, name)
			if property == nil {
				continue
			}
			for _, declaration := range property.Declarations {
				declarations[declaration] = struct{}{}
			}
		}
	}

	if symbols.resultAlias == nil || symbols.okClass == nil || symbols.errClass == nil ||
		len(symbols.resultUnwrap) != 2 ||
		len(symbols.matchDecls) != 1 || len(symbols.matchPartialDecls) != 1 || len(symbols.errorRuntimeDecls) != 3 ||
		len(symbols.expectDecls) != 2 || len(symbols.resultConsumerDecls) != 2*len(resultConsumerMembers) ||
		len(symbols.resultInspectionDecls) != 4 ||
		panicClass == nil || contextClass == nil || len(symbols.contextDecls) != 1 ||
		len(symbols.panicDecls) != 1 || len(symbols.reflectPanicDecls) != 1 ||
		errorInterface == nil || len(symbols.layerDecls["succeed"]) != 1 ||
		len(symbols.layerDecls["merge"]) != 1 || len(symbols.layerDecls["provide"]) != 1 ||
		len(symbols.resultNamespaceDecls["try"]) != 1 || len(symbols.resultNamespaceDecls["tryPromise"]) != 1 ||
		len(symbols.resultNamespaceDecls["all"]) != 1 || len(symbols.promiseCombinatorDecls["all"]) == 0 {
		return nil, fmt.Errorf("the compiler-owned prelude is incomplete")
	}
	symbols.module = symbols.mergedSymbolAt(preludeFile.AsNode())
	symbols.resultSymbol = symbols.mergedSymbolAt(symbols.resultAlias.Name())
	symbols.error = symbols.mergedSymbolAt(errorInterface.Name())
	symbols.panic = symbols.mergedSymbolAt(panicClass.Name())
	symbols.context = symbols.mergedSymbolAt(contextClass.Name())
	if symbols.module == nil || symbols.resultSymbol == nil ||
		symbols.error == nil || symbols.panic == nil || symbols.context == nil {
		return nil, fmt.Errorf("the compiler-owned prelude's Result declarations have no symbol")
	}
	return symbols, nil
}

func isMethodSignature(node *ast.Node) bool { return node.Kind == ast.KindMethodSignature }

// moduleBodyOf returns the block a namespace declaration encloses, so member
// collection can walk the statements directly rather than the declaration.
func moduleBodyOf(node *ast.Node) *ast.Node {
	if node == nil || !ast.IsModuleDeclaration(node) {
		return nil
	}
	return node.AsModuleDeclaration().Body
}

// collectMembers records the direct members of a declaration that match a kind
// predicate and a member name. The name is only ever read inside the prelude,
// which is a file the compiler itself wrote.
func collectMembers(declaration *ast.Node, matches func(*ast.Node) bool, name string, into map[*ast.Node]struct{}) {
	if declaration == nil {
		return
	}
	declaration.ForEachChild(func(child *ast.Node) bool {
		if matches(child) && declarationName(child) == name {
			into[child] = struct{}{}
		}
		return false
	})
}

// collectResultInspectionMembers indexes the Result variant's direct `ok`
// field and the readonly constructor parameter that becomes its `value` or
// `error` field. Names are consulted only inside the compiler-authored prelude;
// authored reads are matched to these nodes by the checker.
func collectResultInspectionMembers(declaration *ast.Node, into map[*ast.Node]struct{}) {
	if declaration == nil {
		return
	}
	declaration.ForEachChild(func(child *ast.Node) bool {
		if ast.IsPropertyDeclaration(child) && declarationName(child) == "ok" {
			into[child] = struct{}{}
			return false
		}
		if !ast.IsConstructorDeclaration(child) {
			return false
		}
		for _, parameter := range child.Parameters() {
			name := declarationName(parameter)
			if name == "value" || name == "error" {
				into[parameter] = struct{}{}
			}
		}
		return false
	})
}

func declarationName(node *ast.Node) string {
	name := node.Name()
	if name == nil || !ast.IsIdentifier(name) {
		return ""
	}
	return name.Text()
}

func (s *preludeSymbols) mergedSymbolAt(node *ast.Node) *ast.Symbol {
	if node == nil {
		return nil
	}
	symbol := s.checker.GetSymbolAtLocation(node)
	if symbol == nil {
		return nil
	}
	return s.checker.GetMergedSymbol(symbol)
}

// declaresAnyOf reports whether a resolved symbol was declared by one of the
// given prelude declaration nodes. Node identity is the strongest available
// form of "this is the compiler's own type", and it survives merging, aliasing
// and re-export.
func declaresAnyOf(symbol *ast.Symbol, declarations ...*ast.Node) bool {
	if symbol == nil {
		return false
	}
	for _, declaration := range symbol.Declarations {
		for _, candidate := range declarations {
			if candidate != nil && declaration == candidate {
				return true
			}
		}
	}
	return false
}

func (s *preludeSymbols) isAliasSymbol(symbol *ast.Symbol, canonical *ast.Symbol, declaration *ast.Node) bool {
	if symbol == nil {
		return false
	}
	if canonical != nil && s.checker.GetMergedSymbol(symbol) == canonical {
		return true
	}
	return declaresAnyOf(symbol, declaration)
}

// isResultSymbol reports whether a resolved symbol is the prelude's `Result`.
func (s *preludeSymbols) isResultSymbol(symbol *ast.Symbol) bool {
	return s.isAliasSymbol(symbol, s.resultSymbol, s.resultAlias)
}

// memberSelection gives dotted access and literal element access one semantic
// representation. In TypeScript, x.member and x["member"] select the same
// resolved property and execute the same call. Compiler-owned members must
// therefore never depend on which of those two authored spellings was used.
// A computed, non-literal key is deliberately not guessed here; ordinary
// checker diagnostics own keys that cannot be resolved to a declared member.
type memberSelection struct {
	receiver *ast.Node
	name     *ast.Node
}

func selectMember(node *ast.Node) (memberSelection, bool) {
	if node == nil {
		return memberSelection{}, false
	}
	if ast.IsPropertyAccessExpression(node) {
		access := node.AsPropertyAccessExpression()
		return memberSelection{receiver: access.Expression, name: access.Name()}, true
	}
	if ast.IsElementAccessExpression(node) {
		access := node.AsElementAccessExpression()
		name := ast.SkipParentheses(access.ArgumentExpression)
		if name != nil && ast.IsStringLiteralLike(name) {
			return memberSelection{receiver: access.Expression, name: name}, true
		}
	}
	return memberSelection{}, false
}

func (s *preludeSymbols) memberSymbol(selection memberSelection) *ast.Symbol {
	symbol := s.checker.GetSymbolAtLocation(selection.name)
	if symbol == nil && selection.receiver != nil && selection.name != nil {
		symbol = s.checker.GetPropertyOfType(s.checker.GetTypeAtLocation(selection.receiver), selection.name.Text())
	}
	return symbol
}

// authoredConstructor reports a forbidden author-facing Result
// variant hook. The owner must resolve to the compiler prelude; spelling is
// consulted only after that identity proof to select the reserved hook.
func (s *preludeSymbols) authoredConstructor(node *ast.Node) (string, bool) {
	selection, selected := selectMember(node)
	if !selected {
		return "", false
	}
	if !ast.IsIdentifier(selection.receiver) {
		return "", false
	}
	symbol := s.checker.GetSymbolAtLocation(selection.receiver)
	name := selection.name.Text()
	if s.isResultSymbol(symbol) && (name == "ok" || name == "err" || name == "success" || name == "error") {
		return "Result." + name, true
	}
	return "", false
}

// resultNamespaceCall recognizes a call to a member of the compiler-owned
// global `Result` value through the prelude's own declaration nodes, so an
// unrelated object with a `try` or `all` method is never mistaken for it. It
// returns the member name, which selects the prelude export the call lowers to.
func (s *preludeSymbols) resultNamespaceCall(node *ast.Node) (string, bool) {
	if node == nil || !ast.IsCallExpression(node) {
		return "", false
	}
	expression := node.AsCallExpression().Expression
	selection, selected := selectMember(expression)
	if !selected {
		return "", false
	}
	symbol := s.memberSymbol(selection)
	for _, member := range resultNamespaceMembers {
		if symbolDeclaredBy(symbol, s.resultNamespaceDecls[member]) {
			return member, true
		}
	}
	return "", false
}

// resultBoundaryCall narrows resultNamespaceCall to the members that ADOPT the
// caller's throw scope. `Result.all` is deliberately not one of them: its
// argument is an array of already-computed Results, not a body the adapter
// invokes inside its own try/catch, so a foreign call written there must still
// be charged and wrapped.
func (s *preludeSymbols) resultBoundaryCall(node *ast.Node) (string, bool) {
	member, matched := s.resultNamespaceCall(node)
	if !matched || !resultBoundaryMembers[member] {
		return "", false
	}
	return member, true
}

// isPanicCall recognizes the compiler-owned panic intrinsic under either of its
// two authored spellings — `panic(...)` imported from `smithers:exceptions` and
// the ambient `Reflect.panic(...)` — through the prelude's own declarations.
// docs/DECISIONS.md, "Typed failures" (Locked): "Reflect.panic and
// compiler/runtime invariant failures enter the same distinguished channel."
// A user function named `panic`, and a user object with a `panic` member, both
// resolve elsewhere and stay ordinary calls.
func (s *preludeSymbols) isPanicCall(node *ast.Node) bool {
	if node == nil || !ast.IsCallExpression(node) {
		return false
	}
	expression := node.AsCallExpression().Expression
	if selection, selected := selectMember(expression); selected {
		// The member name carries the declaration: the receiver `Reflect` is
		// the ambient lib namespace the prelude augments, so asking the receiver
		// would prove nothing about which `panic` was selected.
		return symbolDeclaredBy(s.resolvedSymbolAt(selection.name), s.reflectPanicDecls)
	}
	return symbolDeclaredBy(s.resolvedSymbolAt(expression), s.panicDecls)
}

// resolvedSymbolAt follows an import alias to the symbol it names. `panic` is
// reached by importing it, so the binding at the call site is an alias and only
// the aliased symbol carries the prelude's own declaration.
func (s *preludeSymbols) resolvedSymbolAt(node *ast.Node) *ast.Symbol {
	if node == nil {
		return nil
	}
	symbol := s.checker.GetSymbolAtLocation(node)
	if symbol == nil {
		return nil
	}
	if symbol.Flags&ast.SymbolFlagsAlias != 0 {
		if resolved := s.checker.GetAliasedSymbol(symbol); resolved != nil {
			symbol = resolved
		}
	}
	return s.checker.GetMergedSymbol(symbol)
}

// resultNamespaceMembers is iterated instead of the map so recognition order is
// fixed; a compiler whose answer depends on map iteration is not reproducible.
var resultNamespaceMembers = []string{"try", "tryPromise", "all"}

// resultConsumerMembers is the locked receiver surface that discharges a
// Result ownership obligation. The names are consulted only while indexing the
// compiler-owned prelude; authored calls are recognized from the declaration
// nodes collected there.
var resultConsumerMembers = []string{
	"isOk", "isError", "match", "map", "mapError", "andThen",
	"recover", "tap", "tapError", "unwrapOr", "expect",
}

func (s *preludeSymbols) resultConsumerCall(node *ast.Node) bool {
	if node == nil || !ast.IsCallExpression(node) {
		return false
	}
	expression := node.AsCallExpression().Expression
	selection, selected := selectMember(expression)
	if !selected {
		return false
	}
	symbol := s.memberSymbol(selection)
	return symbolDeclaredBy(symbol, s.resultConsumerDecls)
}

func (s *preludeSymbols) resultInspection(node *ast.Node) bool {
	selection, selected := selectMember(node)
	if !selected {
		return false
	}
	symbol := s.memberSymbol(selection)
	return symbolDeclaredBy(symbol, s.resultInspectionDecls)
}

func (s *preludeSymbols) promiseCombinatorCall(node *ast.Node) bool {
	if node == nil || !ast.IsCallExpression(node) {
		return false
	}
	expression := node.AsCallExpression().Expression
	selection, selected := selectMember(expression)
	if !selected {
		return false
	}
	symbol := s.memberSymbol(selection)
	for _, name := range []string{"all", "allSettled", "race", "any", "allKeyed", "allSettledKeyed"} {
		if symbolDeclaredBy(symbol, s.promiseCombinatorDecls[name]) {
			return s.checker.GetPromisedTypeOfPromise(s.checker.GetTypeAtLocation(node)) != nil
		}
	}
	return false
}

// resultBoundaryMembers are the subset of the namespace whose first argument is
// a body the adapter invokes inside its own try/catch.
var resultBoundaryMembers = map[string]bool{"try": true, "tryPromise": true}

// resultNamespaceExports maps each member to the prelude runtime export that
// implements it.
var resultNamespaceExports = map[string]string{
	"try":        "smithersTry",
	"tryPromise": "smithersTryPromise",
	"all":        "smithersResultAll",
}

// isErrorRuntimeMember identifies one of the global Error helpers whose
// authored method call executes against the prelude-installed prototype
// implementation. The resolved declaration, not the property spelling, is the
// authority, so unrelated user methods do not pull in or depend on the prelude.
func (s *preludeSymbols) isErrorRuntimeMember(node *ast.Node) bool {
	selection, selected := selectMember(node)
	if !selected {
		return false
	}
	return symbolDeclaredBy(s.memberSymbol(selection), s.errorRuntimeDecls)
}

// isCompilerModuleSpecifier recognizes a compiler-owned virtual module by the
// module symbol the checker resolved it to. The authored spelling is never
// used to decide whether an import is intrinsic.
func (s *preludeSymbols) isCompilerModuleSpecifier(specifier *ast.Node) bool {
	if specifier == nil {
		return false
	}
	if durableResolvedModule(s.checker, specifier) {
		return true
	}
	if comptimeResolvedModule(s.checker, specifier) {
		return true
	}
	if s.module == nil {
		return false
	}
	symbol := s.checker.ResolveExternalModuleName(specifier)
	return symbol != nil && s.checker.GetMergedSymbol(symbol) == s.module
}

// isVariantUnion reports whether every constituent of a checker type was
// declared by one of the prelude's variant classes. This recognizes a Result
// after narrowing or instantiation has dropped the alias.
func isVariantUnion(t *checker.Type, classes ...*ast.Node) bool {
	if t == nil {
		return false
	}
	constituents := []*checker.Type{t}
	if t.IsUnion() {
		constituents = t.Types()
	}
	if len(constituents) == 0 {
		return false
	}
	for _, constituent := range constituents {
		if !declaresAnyOf(constituent.Symbol(), classes...) {
			return false
		}
	}
	return true
}

// isResultType reports whether a checker type is a Result: either it still
// carries the prelude's alias, or (after narrowing, instantiation or explicit
// variant typing) every constituent is one of the prelude's variant classes.
func (s *preludeSymbols) isResultType(t *checker.Type) bool {
	if t == nil {
		return false
	}
	if s.isResultSymbol(t.Alias().Symbol()) {
		return true
	}
	return isVariantUnion(t, s.okClass, s.errClass)
}

// unwrapKind distinguishes a retired Result `.unwrap()` call from an ordinary
// same-spelled method. It no longer denotes propagation.
type unwrapKind int

const (
	unwrapNone unwrapKind = iota
	unwrapResult
)

// unwrapCall reports whether a call expression is the retired
// `<result>.unwrap()` spelling, using the resolved property declaration and
// Result type identity rather than authored member text.
func (s *preludeSymbols) unwrapCall(node *ast.Node) (*ast.Node, unwrapKind) {
	selection, ok := s.zeroArgumentMethodCall(node)
	if !ok {
		return nil, unwrapNone
	}
	symbol := s.memberSymbol(selection)
	if symbol == nil {
		return nil, unwrapNone
	}
	result := false
	for _, declaration := range symbol.Declarations {
		if _, ok := s.resultUnwrap[declaration]; ok {
			result = true
		}
	}
	if !result {
		return nil, unwrapNone
	}
	receiverType := s.checker.GetTypeAtLocation(selection.receiver)
	if result && s.isResultType(receiverType) {
		return selection.receiver, unwrapResult
	}
	return nil, unwrapNone
}

func (s *preludeSymbols) zeroArgumentMethodCall(node *ast.Node) (memberSelection, bool) {
	if !ast.IsCallExpression(node) {
		return memberSelection{}, false
	}
	call := node.AsCallExpression()
	if call.Arguments == nil || len(call.Arguments.Nodes) != 0 {
		return memberSelection{}, false
	}
	return selectMember(call.Expression)
}

type errorMatchKind int

const (
	errorMatchNone errorMatchKind = iota
	errorMatchExhaustive
	errorMatchPartial
)

// matchCall reports whether a call expression is a compiler-owned Error.match
// or Error.matchPartial operation, resolved through the prelude declarations.
// Arguments are returned unvalidated so lowering can issue the language's
// precise shape/fallback diagnostic before ordinary checker cascades escape.
func (s *preludeSymbols) matchCall(node *ast.Node) (receiver *ast.Node, handlers *ast.Node, fallback *ast.Node, kind errorMatchKind) {
	if !ast.IsCallExpression(node) {
		return nil, nil, nil, errorMatchNone
	}
	call := node.AsCallExpression()
	selection, selected := selectMember(call.Expression)
	if !selected {
		return nil, nil, nil, errorMatchNone
	}
	symbol := s.memberSymbol(selection)
	if symbol == nil {
		return nil, nil, nil, errorMatchNone
	}
	kind = errorMatchNone
	for _, declaration := range symbol.Declarations {
		if _, exists := s.matchDecls[declaration]; exists {
			kind = errorMatchExhaustive
			break
		}
		if _, exists := s.matchPartialDecls[declaration]; exists {
			kind = errorMatchPartial
			break
		}
	}
	if kind == errorMatchNone {
		return nil, nil, nil, errorMatchNone
	}
	if call.Arguments != nil && len(call.Arguments.Nodes) > 0 {
		handlers = call.Arguments.Nodes[0]
	}
	if call.Arguments != nil && len(call.Arguments.Nodes) > 1 {
		fallback = call.Arguments.Nodes[1]
	}
	return selection.receiver, handlers, fallback, kind
}

func (s *preludeSymbols) expectCall(node *ast.Node) (*ast.Node, bool) {
	if !ast.IsCallExpression(node) {
		return nil, false
	}
	call := node.AsCallExpression()
	selection, selected := selectMember(call.Expression)
	if call.Arguments == nil || len(call.Arguments.Nodes) != 1 || !selected {
		return nil, false
	}
	if !symbolDeclaredBy(s.memberSymbol(selection), s.expectDecls) ||
		!s.isResultType(s.checker.GetTypeAtLocation(selection.receiver)) {
		return nil, false
	}
	return selection.receiver, true
}

// liftShape is the checked return contract of the function being lowered. It
// is derived from the declared return type annotation by resolved symbol
// identity, and it drives every lift decision inside the body.
type liftShape struct {
	async  bool // the declared return type is `Promise<...>` on an async function
	result bool // the (awaited) return type is the prelude's Result
}

func (s liftShape) lifts() bool { return s.result }

// loweringReport is a Smithers diagnostic raised by the lowering itself, at an
// authored position. The lowering never approximates: a construct it cannot
// lower soundly is left byte-for-byte intact and reported here.
type loweringReport struct {
	code     string
	category string
	message  string
	start    int
	length   int
}

func loweringReportCategory(report loweringReport) string {
	if report.category != "" {
		return report.category
	}
	return "error"
}

// errorRow is one member of the statically known error union at a match site.
// Rows live in this side table keyed by symbol; nothing about a row is ever
// encoded in a checker TypeFlags bit, which is a full uint32 the fork owns.
type errorRow struct {
	symbol *ast.Symbol
	name   string
}

// rowSet is a nominal row side table. Its keys are checker-resolved class or
// type-parameter symbols; TypeFlags remains entirely owned by the fork.
type rowSet map[*ast.Symbol]errorRow

func addRow(into rowSet, row errorRow) bool {
	if row.symbol == nil {
		return false
	}
	if _, exists := into[row.symbol]; exists {
		return false
	}
	into[row.symbol] = row
	return true
}

func addRows(into rowSet, from rowSet) bool {
	changed := false
	for _, row := range from {
		changed = addRow(into, row) || changed
	}
	return changed
}

func missingRowSet(values rowSet, covered rowSet) rowSet {
	missing := make(rowSet)
	for symbol, row := range values {
		if _, ok := covered[symbol]; !ok {
			missing[symbol] = row
		}
	}
	return missing
}

type foreignPolicy struct {

type functionRows struct {

func (fn *functionRows) inferredFallible() bool {
	return !fn.explicitReturn && len(fn.bodyFailures) != 0
}

// rowAnalysis is the whole-project function graph and its fixed-point rows.
// Every edge and member is checker-resolved; import spelling never enters the
// graph, so aliases, namespace access, relative modules, and cycles all share
// the same representation.
type rowAnalysis struct {
	checker *checker.Checker
	prelude *preludeSymbols
	project *project
	// comptime owns the nodes the comptime frontend replaces or erases. Whole
	// program row rules do not apply to source that never reaches the emitted
	// artifact, so reports against those nodes are dropped rather than shown.
	comptime      *comptimeAnalysis
	functions     []*functionRows
	byNode        map[*ast.Node]*functionRows
	bySymbol      map[*ast.Symbol]*functionRows
	layerBindings map[*ast.Symbol]*ast.Node
	rowNames      map[*ast.Symbol]string
	reports       map[*ast.SourceFile][]loweringReport
	topProvides   []rowProvideEdge
	foreignCalls  map[*ast.Node]foreignPolicy
	// resultPropagations maps each valid postfix `!` AST node to its Result
	// operand. The parser already represents postfix `!` as NonNullExpression;
	// the `.sm` semantic pass removes that old meaning and owns every such node.
	resultPropagations map[*ast.Node]*ast.Node
	// A policy can be queried by row inference, provenance, top-level checks,
	// and foreign-boundary checks. Invalid annotation diagnostics are emitted
	// once per authored call even when several of those passes query it.
	foreignAnnotationReports map[*ast.Node]bool
	foreignOrderReports      map[*ast.Node]bool
	// assetImports is keyed by the authored import/re-export declaration or
	// dynamic import call. Potential asset edges are recorded before validation
	// so neither a valid asset nor a fail-closed asset refusal can fall through
	// into foreign-module policy.
	assetImports          map[*ast.Node]assetImportEdge
	refusedForeignModules map[*ast.SourceFile]bool
}

func symbolDeclaredBy(symbol *ast.Symbol, declarations map[*ast.Node]struct{}) bool {
	if symbol == nil {
		return false
	}
	for _, declaration := range symbol.Declarations {
		if _, ok := declarations[declaration]; ok {
			return true
		}
	}
	return false
}

func (s *preludeSymbols) layerCall(node *ast.Node, method string) bool {
	if !ast.IsCallExpression(node) {
		return false
	}
	call := node.AsCallExpression()
	selection, selected := selectMember(call.Expression)
	if !selected {
		return false
	}
	declarations := s.layerDecls[method]
	if len(declarations) == 0 {
		return false
	}
	symbol := s.memberSymbol(selection)
	return symbolDeclaredBy(symbol, declarations)
}

func (s *preludeSymbols) contextCall(node *ast.Node) *ast.Node {
	if !ast.IsCallExpression(node) {
		return nil
	}
	call := node.AsCallExpression()
	selection, selected := selectMember(call.Expression)
	if call.Arguments == nil || len(call.Arguments.Nodes) != 0 || !selected {
		return nil
	}
	if !symbolDeclaredBy(s.memberSymbol(selection), s.contextDecls) {
		return nil
	}
	return selection.receiver
}

func (a *rowAnalysis) canonical(symbol *ast.Symbol) *ast.Symbol {
	if symbol == nil {
		return nil
	}
	if symbol.Flags&ast.SymbolFlagsAlias != 0 {
		if resolved := a.checker.GetAliasedSymbol(symbol); resolved != nil {
			symbol = resolved
		}
	}
	return a.checker.GetMergedSymbol(symbol)
}

func (a *rowAnalysis) symbolAt(node *ast.Node) *ast.Symbol {
	if node == nil {
		return nil
	}
	return a.canonical(a.checker.GetSymbolAtLocation(node))
}

func (a *rowAnalysis) rowName(symbol *ast.Symbol) string {
	if symbol == nil {
		return "Error"
	}
	if name := a.rowNames[symbol]; name != "" {
		return name
	}
	return symbol.Name
}

func (a *rowAnalysis) row(symbol *ast.Symbol) errorRow {
	symbol = a.canonical(symbol)
	return errorRow{symbol: symbol, name: a.rowName(symbol)}
}

func (a *rowAnalysis) typeHasBaseSymbol(t *checker.Type, target *ast.Symbol, seen map[*checker.Type]bool) bool {
	if t == nil || target == nil || seen[t] {
		return false
	}
	seen[t] = true
	if symbol := a.canonical(t.Symbol()); symbol == target {
		return true
	}
	if t.Flags()&checker.TypeFlagsTypeParameter != 0 {
		constraint := a.checker.GetBaseConstraintOfType(t)
		return constraint != nil && constraint != t && a.typeHasBaseSymbol(constraint, target, seen)
	}
	if t.IsUnion() || t.IsIntersection() {
		for _, part := range t.Types() {
			if a.typeHasBaseSymbol(part, target, seen) {
				return true
			}
		}
		return false
	}
	if t.Flags()&checker.TypeFlagsObject == 0 {
		return false
	}
	for _, base := range a.checker.GetBaseTypes(t) {
		if a.typeHasBaseSymbol(base, target, seen) {
			return true
		}
	}
	return false
}

func (a *rowAnalysis) isErrorType(t *checker.Type) bool {
	if t == nil {
		return false
	}
	if t.IsUnion() {
		for _, part := range t.Types() {
			if !a.isErrorType(part) {
				return false
			}
		}
		return true
	}
	return a.typeHasBaseSymbol(t, a.prelude.error, make(map[*checker.Type]bool))
}

func (a *rowAnalysis) extendsContext(symbol *ast.Symbol) bool {
	symbol = a.canonical(symbol)
	if symbol == nil {
		return false
	}
	return a.typeHasBaseSymbol(a.checker.GetDeclaredTypeOfSymbol(symbol), a.prelude.context, make(map[*checker.Type]bool))
}

func (a *rowAnalysis) rowsOfType(t *checker.Type) (rowSet, bool) {
	rows := make(rowSet)
	if t == nil {
		return rows, false
	}
	if t.Flags()&checker.TypeFlagsNever != 0 {
		return rows, true
	}
	if t.IsUnion() {
		for _, part := range t.Types() {
			values, ok := a.rowsOfType(part)
			if !ok {
				return rows, false
			}
			addRows(rows, values)
		}
		return rows, true
	}
	if !a.isErrorType(t) {
		return rows, false
	}
	symbol := a.canonical(t.Symbol())
	if symbol == nil && t.Alias() != nil {
		symbol = a.canonical(t.Alias().Symbol())
	}
	if symbol == nil {
		return rows, false
	}
	addRow(rows, a.row(symbol))
	return rows, true
}

func (a *rowAnalysis) resultArguments(t *checker.Type) (success *checker.Type, failure *checker.Type, ok bool) {
	if t == nil || !a.prelude.isResultType(t) {
		return nil, nil, false
	}
	if alias := t.Alias(); alias != nil && a.prelude.isResultSymbol(alias.Symbol()) {
		arguments := alias.TypeArguments()
		if len(arguments) == 2 {
			return arguments[0], arguments[1], true
		}
	}
	constituents := []*checker.Type{t}
	if t.IsUnion() {
		constituents = t.Types()
	}
	for _, part := range constituents {
		if !declaresAnyOf(part.Symbol(), a.prelude.errClass) {
			continue
		}
		arguments := a.checker.GetTypeArguments(part)
		if len(arguments) == 1 {
			return nil, arguments[0], true
		}
	}
	return nil, nil, false
}

func (a *rowAnalysis) functionType(node *ast.Node) (*checker.Type, bool) {
	if node == nil || node.Type() == nil {
		return nil, false
	}
	t := a.checker.GetTypeFromTypeNode(node.Type())
	async := ast.HasSyntacticModifier(node, ast.ModifierFlagsAsync)
	// Failure-row contracts live in Result even when a plain function forwards
	// Promise<Result<...>> without using the async keyword. This analysis shape
	// is deliberately broader than lowerer.shapeOf: only async functions are
	// rewritten through their awaited value, but both spellings carry the same
	// statically declared failure row.
	if promised := a.checker.GetPromisedTypeOfPromise(t); promised != nil {
		t = promised
	}
	return t, async
}

func (a *rowAnalysis) declaredFunctionShape(node *ast.Node) (liftShape, rowSet, *checker.Type) {
	rows := make(rowSet)
	t, async := a.functionType(node)
	if t == nil {
		return liftShape{}, rows, nil
	}
	shape := liftShape{async: async}
	if _, failure, ok := a.resultArguments(t); ok {
		shape.result = true
		if failures, known := a.rowsOfType(failure); known {
			addRows(rows, failures)
		}
		return shape, rows, failure
	}
	return liftShape{}, rows, nil
}

const deferredRowFlags = checker.TypeFlagsTypeParameter | checker.TypeFlagsConditional |
	checker.TypeFlagsIndexedAccess | checker.TypeFlagsSubstitution

func hasDeferredRowType(t *checker.Type, seen map[*checker.Type]bool) bool {
	if t == nil || seen[t] {
		return false
	}
	seen[t] = true
	if t.Flags()&deferredRowFlags != 0 {
		return true
	}
	if t.IsUnion() || t.IsIntersection() {
		for _, part := range t.Types() {
			if hasDeferredRowType(part, seen) {
				return true
			}
		}
	}
	if alias := t.Alias(); alias != nil {
		for _, argument := range alias.TypeArguments() {
			if hasDeferredRowType(argument, seen) {
				return true
			}
		}
	}
	return false
}

func functionBinding(node *ast.Node) *ast.Node {
	if node == nil {
		return nil
	}
	if name := node.Name(); name != nil {
		return name
	}
	if node.Parent != nil && ast.IsVariableDeclaration(node.Parent) {
		return node.Parent.Name()
	}
	return nil
}

func functionIsExported(node *ast.Node) bool {
	if ast.HasSyntacticModifier(node, ast.ModifierFlagsExport) {
		return true
	}
	if node.Parent == nil || !ast.IsVariableDeclaration(node.Parent) ||
		node.Parent.AsVariableDeclaration().Initializer != node {
		return false
	}
	declarationList := node.Parent.Parent
	if declarationList == nil || !ast.IsVariableDeclarationList(declarationList) || declarationList.Parent == nil ||
		!ast.IsVariableStatement(declarationList.Parent) {
		return false
	}
	return ast.HasSyntacticModifier(declarationList.Parent, ast.ModifierFlagsExport)
}

func newRowAnalysis(typeChecker *checker.Checker, prelude *preludeSymbols, files []*ast.SourceFile, comptime *comptimeAnalysis, project *project) *rowAnalysis {
	analysis := &rowAnalysis{
		checker:                  typeChecker,
		prelude:                  prelude,
		project:                  project,
		comptime:                 comptime,
		byNode:                   make(map[*ast.Node]*functionRows),
		bySymbol:                 make(map[*ast.Symbol]*functionRows),
		layerBindings:            make(map[*ast.Symbol]*ast.Node),
		rowNames:                 make(map[*ast.Symbol]string),
		reports:                  make(map[*ast.SourceFile][]loweringReport),
		foreignCalls:             make(map[*ast.Node]foreignPolicy),
		resultPropagations:       make(map[*ast.Node]*ast.Node),
		foreignAnnotationReports: make(map[*ast.Node]bool),
		foreignOrderReports:      make(map[*ast.Node]bool),
		assetImports:             make(map[*ast.Node]assetImportEdge),
		refusedForeignModules:    make(map[*ast.SourceFile]bool),
	}
	analysis.buildRowNames(files)
	analysis.checkAssetImports(files)
	analysis.checkDynamicModuleImports(files)
	analysis.checkForeignModuleTrust(files)
	analysis.collectFunctions(files)
	analysis.checkNestedResultContracts()
	analysis.collectLayerBindings(files)
	for _, fn := range analysis.functions {
		analysis.collectFacts(fn)
	}
	analysis.checkForeignProvenanceEscapes(files)
	analysis.collectTopProvides(files)
	analysis.inferRows()
	analysis.collectInferredResultPropagations(files)
	analysis.checkContracts()
	analysis.checkBoundMustConsume(files)
	analysis.checkResultParameters(files)
	analysis.checkUnboundMustConsume(files)
	analysis.checkLayers()
	analysis.checkTopLevelRequirements(files)
	analysis.checkTopLevelCheckedChannels(files)
	analysis.checkCrossModuleEscapes(files)
	// Fail-closed frontend rules (smithershostrules.go). They run last because
	// none of them contributes a row: each one only decides whether a program
	// the language forbids may reach emit at all.
	analysis.checkHostGlobals(files)
	analysis.checkStaticBlocks(files)
	analysis.checkDuplicateErrorNames(files)
	analysis.checkForeignBoundaries(files)
	analysis.checkCallbackOwnership(files)
	analysis.checkRetiredTypeGrammar(files)
	analysis.checkRelativeModuleResolution(files)
	analysis.checkProjectImportBindings(files)
	return analysis
}

// checkNestedResultContracts rejects a Result success channel that is itself
// the compiler-owned Result. This is a fact about the authored contract, not a
// failed assignment in its body, so the diagnostic belongs at the return type
// annotation where the implicit normalization policy would have to be chosen.
func (a *rowAnalysis) checkNestedResultContracts() {
	for _, fn := range a.functions {
		t, _ := a.functionType(fn.node)
		success, _, ok := a.resultArguments(t)
		if !ok || !a.prelude.isResultType(success) {
			continue
		}
		a.report(fn.node.Type(), "SMITHERS1203", "nested Result normalization is not specified; make the conversion explicit")
	}
}

func (a *rowAnalysis) buildRowNames(files []*ast.SourceFile) {
	byName := make(map[string]map[*ast.Symbol]*ast.SourceFile)
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsClassDeclaration(node) && node.Name() != nil {
				symbol := a.symbolAt(node.Name())
				t := a.checker.GetTypeAtLocation(node.Name())
				if symbol != nil && (a.isErrorType(t) || a.extendsContext(symbol)) {
					items := byName[symbol.Name]
					if items == nil {
						items = make(map[*ast.Symbol]*ast.SourceFile)
						byName[symbol.Name] = items
					}
					items[symbol] = file
				}
				return false
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
	for name, items := range byName {
		if len(items) < 2 {
			continue
		}
		for symbol, file := range items {
			qualifier := strings.TrimPrefix(file.FileName(), "/src/")
			qualifier = strings.TrimSuffix(qualifier, ".sm")
			var cleaned strings.Builder
			for _, character := range qualifier {
				if character >= 'A' && character <= 'Z' || character >= 'a' && character <= 'z' ||
					character >= '0' && character <= '9' || strings.ContainsRune("._/-", character) {
					cleaned.WriteRune(character)
				} else {
					cleaned.WriteByte('_')
				}
			}
			a.rowNames[symbol] = name + "@" + cleaned.String()
		}
	}
}

func (a *rowAnalysis) collectFunctions(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) && node.Body() != nil {
				shape, declared, failureType := a.declaredFunctionShape(node)
				binding := functionBinding(node)
				symbol := a.symbolAt(binding)
				name := "<anonymous>"
				if binding != nil && binding.Text() != "" {
					name = binding.Text()
				}
				fn := &functionRows{
					node:               node,
					file:               file,
					symbol:             symbol,
					name:               name,
					exported:           functionIsExported(node),
					resultBoundaryBody: a.isResultBoundaryBody(node),
					explicitReturn:     node.Type() != nil,
					declaredShape:      shape,
					declaredFailures:   declared,
					directFailures:     make(rowSet),
					bodyFailures:       make(rowSet),
					failures:           make(rowSet),
					directRequirements: make(rowSet),
					requirements:       make(rowSet),
					genericTemplate:    len(node.TypeParameters()) != 0 && hasDeferredRowType(failureType, make(map[*checker.Type]bool)),
					unconsumedParams:   make(map[int]bool),
				}
				if shape.result {
					addRows(fn.directFailures, declared)
					addRows(fn.failures, declared)
				}
				a.functions = append(a.functions, fn)
				a.byNode[node] = fn
				if symbol != nil {
					a.bySymbol[symbol] = fn
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) collectLayerBindings(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsVariableDeclaration(node) {
				declaration := node.AsVariableDeclaration()
				if declaration.Initializer != nil {
					if symbol := a.symbolAt(node.Name()); symbol != nil {
						a.layerBindings[symbol] = declaration.Initializer
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) resolveCallee(call *ast.Node) *functionRows {
	if call == nil || !ast.IsCallExpression(call) {
		return nil
	}
	if signature := a.checker.GetResolvedSignature(call); signature != nil {
		if declaration := signature.Declaration(); declaration != nil {
			if fn := a.byNode[declaration]; fn != nil {
				return fn
			}
		}
	}
	expression := call.AsCallExpression().Expression
	location := expression
	if ast.IsPropertyAccessExpression(expression) {
		location = expression.AsPropertyAccessExpression().Name()
	}
	return a.bySymbol[a.symbolAt(location)]
}

func (a *rowAnalysis) isForeignDeclaration(node *ast.Node) bool {
	file := ast.GetSourceFileOfNode(node)
	if file == nil || file.FileName() == preludeVirtualName || file.FileName() == durableVirtualName ||
		file.FileName() == comptimeVirtualName {
		return false
	}
	name := strings.ToLower(file.FileName())
	return strings.HasPrefix(name, "/src/") && !strings.HasSuffix(name, ".sm")
}

func throwsAnnotation(declaration *ast.Node) string {
	file := ast.GetSourceFileOfNode(declaration)
	if file == nil {
		return ""
	}
	text := file.Text()
	start := declaration.Pos()
	end := scanner.SkipTrivia(text, start)
	if start < 0 || end < start || end > len(text) {
		return ""
	}
	prefix := text[start:end]
	open := strings.LastIndex(prefix, "/**")
	if open < 0 {
		return ""
	}
	close := strings.Index(prefix[open+3:], "*/")
	if close < 0 {
		return ""
	}
	comment := prefix[open : open+3+close+2]
	if exactModuleMarker(comment) {
		// A file-leading `@module` tag disambiguates the module initializer
		// trust claim from a function-level `@throws` claim on the first
		// declaration. It must never certify that declaration infallible.
		return ""
	}
	return exactThrowsAnnotation(comment)
}

func exactThrowsAnnotation(comment string) string {
	const marker = "@throws"
	for offset := 0; offset < len(comment); {
		index := strings.Index(comment[offset:], marker)
		if index < 0 {
			return ""
		}
		index += offset
		position := index + len(marker)
		for position < len(comment) && isJSDocWhitespace(comment[position]) {
			position++
		}
		if position >= len(comment) || comment[position] != '{' {
			offset = index + len(marker)
			continue
		}
		position++
		close := strings.IndexByte(comment[position:], '}')
		if close < 0 {
			return ""
		}
		return strings.TrimSpace(comment[position : position+close])
	}
	return ""
}

func (a *rowAnalysis) foreignPolicyForCall(call *ast.Node) (foreignPolicy, bool) {
	if call == nil || !ast.IsCallExpression(call) {
		return foreignPolicy{}, false
	}
	if a.assetValueProvenance(call.AsCallExpression().Expression, make(map[*ast.Symbol]bool)) {
		// A loader result is a compiler-issued pure-data module. Calling a typed
		// value it exports is governed by that checked module contract, never by
		// the JavaScript foreign-call/panic fallback.
		return foreignPolicy{}, false
	}
	signature := a.checker.GetResolvedSignature(call)
	if signature == nil || signature.Declaration() == nil || !a.isForeignDeclaration(signature.Declaration()) {
		return foreignPolicy{}, false
	}
	if file := ast.GetSourceFileOfNode(signature.Declaration()); file != nil && a.refusedForeignModules[file] {
		// The module edge already fails closed before emit. Charging a wrapped
		// call or a recoverable failure behind that refused initializer only
		// adds cascade diagnostics and diverges from the reference observation.
		return foreignPolicy{}, false
	}
	returned := a.checker.GetReturnTypeOfSignature(signature)
	async := a.checker.GetPromisedTypeOfPromise(returned) != nil
	annotation := throwsAnnotation(signature.Declaration())
	callee := call.AsCallExpression().Expression
	uncheckedCallee := a.foreignCalleeHasUncheckedResult(callee)
	// When the callee is itself the unchecked foreign call, that inner call is
	// the authored boundary whose result is being misused and reports 1507. Do
	// not add the same-position diagnostic again on the immediately invoked
	// outer call.
	orderUnsafe := !a.stableForeignCallee(callee) || uncheckedCallee ||
		annotation != "never" && a.foreignResultIsUsedAsValue(call)
	if uncheckedCallee && callee.Kind == ast.KindCallExpression {
		orderUnsafe = false
	}
	if orderUnsafe && !a.foreignOrderReports[call] {
		a.foreignOrderReports[call] = true
		a.report(call, "SMITHERS1507", "this foreign call/result use is not expression-order-safe in the POC; assign the checked result, propagate it with postfix !, and continue through an explicitly typed local adapter")
	}
	if annotation == "never" {
		return foreignPolicy{kind: "never", async: async}, true
	}
	if annotation != "" {
		symbol := a.canonical(a.checker.ResolveName(annotation, call, ast.SymbolFlagsType, false))
		if symbol != nil && a.isErrorType(a.checker.GetDeclaredTypeOfSymbol(symbol)) {
			return foreignPolicy{kind: "declared", errorSymbol: symbol, errorName: annotation, async: async}, true
		}
		if !a.foreignAnnotationReports[call] {
			a.foreignAnnotationReports[call] = true
			a.report(call, "SMITHERS1502", "foreign @throws {"+annotation+"} has no checker-matching Error constructor value in scope; import that Error class (named, aliased, or through a namespace) or provide an adapter")
		}
	}
	return foreignPolicy{kind: "panic", async: async}, true
}

func (a *rowAnalysis) stableForeignCallee(expression *ast.Node) bool {
	if expression == nil {
		return false
	}
	switch expression.Kind {
	case ast.KindParenthesizedExpression, ast.KindAsExpression,
		ast.KindTypeAssertionExpression:
		return a.stableForeignCallee(expression.Expression())
	case ast.KindNonNullExpression:
		// In `.sm` this AST kind is the checked postfix propagation boundary,
		// never TypeScript's non-null assertion.
		return true
	case ast.KindIdentifier:
		return true
	case ast.KindPropertyAccessExpression:
		return a.stableForeignCallee(expression.AsPropertyAccessExpression().Expression)
	case ast.KindElementAccessExpression:
		return a.stableForeignCallee(expression.AsElementAccessExpression().Expression)
	}
	return false
}

func (a *rowAnalysis) foreignCalleeHasUncheckedResult(expression *ast.Node) bool {
	if expression == nil {
		return false
	}
	switch expression.Kind {
	case ast.KindParenthesizedExpression, ast.KindAsExpression,
		ast.KindTypeAssertionExpression:
		return a.foreignCalleeHasUncheckedResult(expression.Expression())
	case ast.KindNonNullExpression:
		return false
	case ast.KindPropertyAccessExpression:
		return a.foreignCalleeHasUncheckedResult(expression.AsPropertyAccessExpression().Expression)
	case ast.KindElementAccessExpression:
		return a.foreignCalleeHasUncheckedResult(expression.AsElementAccessExpression().Expression)
	case ast.KindCallExpression:
		policy, foreign := a.foreignPolicyForCall(expression)
		return foreign && policy.kind != "never"
	case ast.KindIdentifier:
		symbol := a.canonical(a.checker.GetSymbolAtLocation(expression))
		declaration := variableLikeDeclaration(symbol)
		if declaration != nil && ast.IsVariableDeclaration(declaration) {
			initializer := declaration.AsVariableDeclaration().Initializer
			if initializer != nil && ast.IsCallExpression(initializer) {
				policy, foreign := a.foreignPolicyForCall(initializer)
				return foreign && policy.kind != "never"
			}
		}
	}
	return false
}

func (a *rowAnalysis) foreignResultIsUsedAsValue(call *ast.Node) bool {
	current := call
	parent := current.Parent
	for parent != nil && (ast.IsParenthesizedExpression(parent) || ast.IsAsExpression(parent) ||
		parent.Kind == ast.KindTypeAssertionExpression || ast.IsAwaitExpression(parent)) {
		current = parent
		parent = parent.Parent
	}
	if parent != nil && ast.IsNonNullExpression(parent) && parent.AsNonNullExpression().Expression == current {
		return false
	}
	if parent == nil {
		return false
	}
	if ast.IsCallExpression(parent) || ast.IsNewExpression(parent) {
		return true
	}
	if ast.IsPropertyAccessExpression(parent) && parent.AsPropertyAccessExpression().Expression == current {
		return true
	}
	if ast.IsElementAccessExpression(parent) && parent.AsElementAccessExpression().Expression == current {
		return true
	}
	if ast.IsBinaryExpression(parent) {
		binary := parent.AsBinaryExpression()
		if binary.OperatorToken.Kind == ast.KindEqualsToken && binary.Right == current {
			return true
		}
	}
	if (ast.IsForOfStatement(parent) || ast.IsForInStatement(parent)) && parent.AsForInOrOfStatement().Expression == current {
		return true
	}
	if ast.IsIfStatement(parent) && parent.AsIfStatement().Expression == current ||
		ast.IsWhileStatement(parent) && parent.AsWhileStatement().Expression == current ||
		ast.IsDoStatement(parent) && parent.AsDoStatement().Expression == current ||
		ast.IsSwitchStatement(parent) && parent.AsSwitchStatement().Expression == current {
		return true
	}
	if ast.IsThrowStatement(parent) || ast.IsSpreadElement(parent) || ast.IsSpreadAssignment(parent) {
		return true
	}
	if ast.IsConditionalExpression(parent) {
		return parent.AsConditionalExpression().Condition == current
	}
	if ast.IsArrayLiteralExpression(parent) || ast.IsVariableDeclaration(parent) || ast.IsPropertyAssignment(parent) ||
		ast.IsReturnStatement(parent) || ast.IsExpressionStatement(parent) ||
		ast.IsArrowFunction(parent) && parent.Body() == current {
		return false
	}
	return ast.IsExpression(parent)
}

func (a *rowAnalysis) addForeignFailures(into rowSet, policy foreignPolicy) {
	if policy.kind == "never" {
		return
	}
	addRow(into, a.row(a.prelude.panic))
	if policy.kind == "declared" && policy.errorSymbol != nil {
		addRow(into, a.row(policy.errorSymbol))
	}
}

func callPropagatesFailure(call *ast.Node, _ *preludeSymbols) bool {
	current := call
	for current.Parent != nil {
		parent := current.Parent
		switch {
		case ast.IsParenthesizedExpression(parent), ast.IsAsExpression(parent),
			ast.IsAwaitExpression(parent):
			current = parent
			continue
		case ast.IsNonNullExpression(parent) && parent.AsNonNullExpression().Expression == current:
			return true
		case ast.IsReturnStatement(parent):
			return true
		default:
			return false
		}
	}
	return false
}

func (a *rowAnalysis) instantiatedFailures(call *ast.Node, callee *functionRows) rowSet {
	if callee == nil || !callee.genericTemplate {
		return nil
	}
	signature := a.checker.GetResolvedSignature(call)
	if signature == nil {
		a.report(call, "SMITHERS1803", "the failure row of generic call '"+callee.name+"' cannot be resolved at this call site")
		return make(rowSet)
	}
	returned := a.checker.GetReturnTypeOfSignature(signature)
	if promised := a.checker.GetPromisedTypeOfPromise(returned); promised != nil {
		returned = promised
	}
	_, failure, ok := a.resultArguments(returned)
	if !ok || failure == nil || hasDeferredRowType(failure, make(map[*checker.Type]bool)) {
		a.report(call, "SMITHERS1803", "the failure row of generic call '"+callee.name+"' is a template over its type parameters and is still unresolved at this call site; supply concrete type arguments or wrap the call in an explicitly checked non-generic function")
		return make(rowSet)
	}
	rows, known := a.rowsOfType(failure)
	if !known {
		a.report(call, "SMITHERS1803", "the failure row of generic call '"+callee.name+"' cannot be determined soundly at this call site")
		return make(rowSet)
	}
	a.checkCallbackCoverage(call, callee, rows)
	return rows
}

func (a *rowAnalysis) nominalAncestry(symbol *ast.Symbol, seen map[*ast.Symbol]bool) rowSet {
	rows := make(rowSet)
	symbol = a.canonical(symbol)
	if symbol == nil || seen[symbol] {
		return rows
	}
	seen[symbol] = true
	addRow(rows, a.row(symbol))
	t := a.checker.GetDeclaredTypeOfSymbol(symbol)
	if t != nil && t.Flags()&checker.TypeFlagsObject != 0 {
		for _, base := range a.checker.GetBaseTypes(t) {
			if baseSymbol := a.canonical(base.Symbol()); baseSymbol != nil {
				addRows(rows, a.nominalAncestry(baseSymbol, seen))
			}
		}
	}
	return rows
}

func (a *rowAnalysis) checkCallbackCoverage(call *ast.Node, callee *functionRows, instantiated rowSet) {
	arguments := call.AsCallExpression().Arguments
	if arguments == nil {
		return
	}
	for _, argument := range arguments.Nodes {
		callback := a.byNode[argument]
		if callback == nil {
			location := argument
			if ast.IsPropertyAccessExpression(location) {
				location = location.AsPropertyAccessExpression().Name()
			}
			callback = a.bySymbol[a.symbolAt(location)]
		}
		if callback == nil || !callback.explicitReturn || !callback.declaredShape.result {
			continue
		}
		uncovered := make([]string, 0)
		for symbol, row := range callback.declaredFailures {
			ancestry := a.nominalAncestry(symbol, make(map[*ast.Symbol]bool))
			covered := false
			for expected := range instantiated {
				if _, ok := ancestry[expected]; ok {
					covered = true
					break
				}
			}
			if !covered {
				uncovered = append(uncovered, row.name)
			}
		}
		if len(uncovered) != 0 {
			sort.Strings(uncovered)
			a.report(call, "SMITHERS1806", "instantiating '"+callee.name+"' here publishes the failure row "+formatRowSet(instantiated)+", which a callback argument's declared "+strings.Join(uncovered, " | ")+" cannot produce; correct the type arguments or widen the row")
		}
	}
}

func (a *rowAnalysis) classReferenceSymbol(expression *ast.Node) *ast.Symbol {
	if expression == nil {
		return nil
	}
	// A local value alias (`const Chronometer = Clock`) has its own binding
	// symbol, but its checker type is still the nominal class constructor. The
	// type symbol is therefore the authoritative identity for capability rows;
	// this also handles typed aliases, alias chains, and property-held aliases.
	if valueType := a.checker.GetTypeAtLocation(expression); valueType != nil {
		if symbol := a.canonical(valueType.Symbol()); symbol != nil && a.extendsContext(symbol) {
			return symbol
		}
	}
	location := expression
	if selection, selected := selectMember(expression); selected {
		location = selection.name
	}
	return a.symbolAt(location)
}

func (a *rowAnalysis) resolveFunctionReference(expression *ast.Node) *functionRows {
	if expression == nil {
		return nil
	}
	if fn := a.byNode[expression]; fn != nil {
		return fn
	}
	location := expression
	if ast.IsPropertyAccessExpression(expression) {
		location = expression.AsPropertyAccessExpression().Name()
	}
	return a.bySymbol[a.symbolAt(location)]
}

func (a *rowAnalysis) resolveLayer(expression *ast.Node, seen map[*ast.Symbol]bool) (rowSet, bool) {
	provided := make(rowSet)
	if expression == nil {
		return provided, false
	}
	if ast.IsParenthesizedExpression(expression) {
		return a.resolveLayer(expression.AsParenthesizedExpression().Expression, seen)
	}
	if ast.IsIdentifier(expression) {
		symbol := a.symbolAt(expression)
		if symbol == nil || seen[symbol] {
			return provided, false
		}
		initializer := a.layerBindings[symbol]
		if initializer == nil {
			return provided, false
		}
		seen[symbol] = true
		return a.resolveLayer(initializer, seen)
	}
	if !ast.IsCallExpression(expression) {
		return provided, false
	}
	call := expression.AsCallExpression()
	if a.prelude.layerCall(expression, "succeed") {
		if call.Arguments == nil || len(call.Arguments.Nodes) == 0 {
			return provided, false
		}
		symbol := a.classReferenceSymbol(call.Arguments.Nodes[0])
		if symbol == nil || !a.extendsContext(symbol) {
			return provided, false
		}
		addRow(provided, a.row(symbol))
		return provided, true
	}
	if a.prelude.layerCall(expression, "merge") {
		complete := true
		if call.Arguments == nil {
			return provided, true
		}
		for _, argument := range call.Arguments.Nodes {
			rows, known := a.resolveLayer(argument, cloneSymbolSet(seen))
			addRows(provided, rows)
			complete = complete && known
		}
		return provided, complete
	}
	return provided, false
}

func cloneSymbolSet(values map[*ast.Symbol]bool) map[*ast.Symbol]bool {
	cloned := make(map[*ast.Symbol]bool, len(values))
	for symbol, value := range values {
		cloned[symbol] = value
	}
	return cloned
}

func (a *rowAnalysis) provideEdge(call *ast.Node) (rowProvideEdge, bool) {
	if !a.prelude.layerCall(call, "provide") {
		return rowProvideEdge{}, false
	}
	arguments := call.AsCallExpression().Arguments
	if arguments == nil || len(arguments.Nodes) < 2 {
		return rowProvideEdge{node: call}, true
	}
	layerArgument := arguments.Nodes[0]
	callbackArgument := arguments.Nodes[1]
	provided, complete := a.resolveLayer(layerArgument, make(map[*ast.Symbol]bool))
	return rowProvideEdge{
		node:      call,
		callback:  a.resolveFunctionReference(callbackArgument),
		provided:  provided,
		complete:  complete,
		argument:  layerArgument,
		callbacks: callbackArgument,
	}, true
}

func (a *rowAnalysis) collectFacts(fn *functionRows) {
	body := fn.node.Body()
	if body == nil {
		return
	}
	var visit func(*ast.Node, bool)
	visit = func(node *ast.Node, caught bool) {
		if node == nil || node != body && ast.IsFunctionLikeDeclaration(node) && !nativeInvokedWhereDefined(node) {
			return
		}
		if ast.IsTryStatement(node) {
			statement := node.AsTryStatement()
			visit(statement.TryBlock, caught || statement.CatchClause != nil)
			if statement.CatchClause != nil {
				visit(statement.CatchClause, caught)
			}
			if statement.FinallyBlock != nil {
				visit(statement.FinallyBlock, caught)
			}
			return
		}
		if ast.IsThrowStatement(node) && !caught {
			expression := node.AsThrowStatement().Expression
			if expression != nil {
				t := a.checker.GetTypeAtLocation(expression)
				rows, known := a.rowsOfType(t)
				if !known {
					a.report(node, "SMITHERS1103", "recoverable throw values must extend Error; use panic(...) for an unknown defect")
				} else {
					addRows(fn.directFailures, rows)
				}
			}
		}
		if ast.IsCallExpression(node) {
			if _, ok := a.prelude.expectCall(node); ok {
				addRow(fn.directFailures, a.row(a.prelude.panic))
			}
			if a.prelude.isPanicCall(node) {
				// specification/failures.mdx, "Foreign Exceptions": panic is
				// imported from smithers:exceptions and charges the
				// distinguished checked panic case of the enclosing function.
				addRow(fn.directFailures, a.row(a.prelude.panic))
			}
			if receiver := a.prelude.contextCall(node); receiver != nil {
				symbol := a.classReferenceSymbol(receiver)
				if symbol != nil && a.extendsContext(symbol) {
					addRow(fn.directRequirements, a.row(symbol))
				}
			}
			callee := a.resolveCallee(node)
			if callee != nil {
				fn.calls = append(fn.calls, rowCallEdge{
					node:                 node,
					callee:               callee,
					propagatesFailure:    callPropagatesFailure(node, a.prelude),
					instantiatedFailures: a.instantiatedFailures(node, callee),
				})
			}
			if policy, foreign := a.foreignPolicyForCall(node); foreign && !fn.resultBoundaryBody {
				// Inside an authored `Result.try`/`Result.tryPromise` body the
				// adapter is the catch scope: the foreign call keeps its throw,
				// the boundary converts it, and neither wrapping the call nor
				// charging its failure here would be correct.
				a.foreignCalls[node] = policy
				if policy.kind != "never" {
					if callPropagatesFailure(node, a.prelude) {
						a.addForeignFailures(fn.directFailures, policy)
					} else {
						a.report(node, "SMITHERS1301", "produced Result must be returned, matched, transformed, inspected, or propagated with postfix !")
					}
				}
			}
			if edge, ok := a.provideEdge(node); ok {
				fn.provides = append(fn.provides, edge)
			}
		}
		if ast.IsNonNullExpression(node) {
			operand := node.AsNonNullExpression().Expression
			if _, failure, ok := a.resultArguments(a.checker.GetTypeAtLocation(operand)); ok {
				a.resultPropagations[node] = operand
				if rows, known := a.rowsOfType(failure); known {
					addRows(fn.directFailures, rows)
				}
			}
		}
		if ast.IsPropertyAccessExpression(node) {
			access := node.AsPropertyAccessExpression()
			isCallTarget := node.Parent != nil && ast.IsCallExpression(node.Parent) && node.Parent.AsCallExpression().Expression == node
			// The RECEIVER's value provenance is the rule, not the property's
			// declaring file. A `.sm` local whose type is an interface declared
			// in a foreign module holds an authored value, and reading its
			// members executes no foreign accessor —
			// specification/compatibility.mdx, "Runtime TypeScript Dependency":
			// "A type-only import MUST NOT add that requirement."
			if !isCallTarget && a.foreignValueProvenance(access.Expression, make(map[*ast.Symbol]bool)) {
				symbol := a.checker.GetSymbolAtLocation(access.Name())
				for _, declaration := range symbolDeclarations(symbol) {
					if a.isForeignDeclaration(declaration) {
						a.report(node, "SMITHERS1506", "foreign property/accessor reads need an annotated adapter before they can enter checked code")
						break
					}
				}
			}
		}
		if ast.IsReturnStatement(node) {
			expression := node.AsReturnStatement().Expression
			if expression != nil {
				t := a.checker.GetTypeAtLocation(expression)
				if _, failure, ok := a.resultArguments(t); ok {
					if rows, known := a.rowsOfType(failure); known {
						addRows(fn.directFailures, rows)
					}
				}
			}
		}
		node.ForEachChild(func(child *ast.Node) bool {
			visit(child, caught)
			return false
		})
	}
	visit(body, false)
}

// collectInferredResultPropagations completes the postfix table after row
// inference. Explicit Result operands were recorded during collectFacts so
// their nominal errors could seed inference. This second pass covers a direct
// call whose authored TypeScript type is still its success type but whose
// checked Smithers callee or foreign boundary is known to produce a Result.
func (a *rowAnalysis) collectInferredResultPropagations(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsNonNullExpression(node) {
				operand := node.AsNonNullExpression().Expression
				if a.resultPropagationOperand(operand) {
					a.resultPropagations[node] = operand
				}
			}
			node.ForEachChild(visit)
			return false
		}
		visit(file.AsNode())
	}
}

func (a *rowAnalysis) resultPropagationOperand(expression *ast.Node) bool {
	if expression == nil {
		return false
	}
	if a.prelude.isResultType(a.checker.GetTypeAtLocation(expression)) {
		return true
	}
	switch expression.Kind {
	case ast.KindParenthesizedExpression, ast.KindAsExpression,
		ast.KindTypeAssertionExpression, ast.KindAwaitExpression:
		return a.resultPropagationOperand(expression.Expression())
	case ast.KindCallExpression:
		if callee := a.resolveCallee(expression); callee != nil {
			// An async call produces a Promise. The Result becomes available only
			// after an authored await, so postfix ! on the call itself must fail
			// closed as a non-Result operand.
			return !ast.HasSyntacticModifier(callee.node, ast.ModifierFlagsAsync) &&
				(callee.declaredShape.result || len(callee.failures) != 0)
		}
		if policy, foreign := a.foreignCalls[expression]; foreign && policy.kind != "never" {
			return true
		}
	}
	return false
}

func symbolDeclarations(symbol *ast.Symbol) []*ast.Node {
	if symbol == nil {
		return nil
	}
	return symbol.Declarations
}

func (a *rowAnalysis) collectTopProvides(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) {
				return false
			}
			if ast.IsCallExpression(node) {
				if edge, ok := a.provideEdge(node); ok {
					a.topProvides = append(a.topProvides, edge)
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) inferRows() {
	for _, fn := range a.functions {
		addRows(fn.bodyFailures, fn.directFailures)
		addRows(fn.requirements, fn.directRequirements)
	}
	changed := true
	for changed {
		changed = false
		for _, fn := range a.functions {
			for _, edge := range fn.calls {
				if edge.callee == nil {
					continue
				}
				if edge.propagatesFailure {
					failures := edge.callee.failures
					if edge.instantiatedFailures != nil {
						failures = edge.instantiatedFailures
					}
					changed = addRows(fn.bodyFailures, failures) || changed
				}
				changed = addRows(fn.requirements, edge.callee.requirements) || changed
			}
			for _, provide := range fn.provides {
				if provide.callback == nil {
					continue
				}
				changed = addRows(fn.requirements, missingRowSet(provide.callback.requirements, provide.provided)) || changed
			}
			if fn.explicitReturn && fn.declaredShape.result {
				changed = addRows(fn.failures, fn.declaredFailures) || changed
			} else {
				changed = addRows(fn.failures, fn.bodyFailures) || changed
			}
		}
	}
}

func formatRowSet(rows rowSet) string {
	names := make([]string, 0, len(rows))
	for _, row := range rows {
		names = append(names, row.name)
	}
	sort.Strings(names)
	return "{" + strings.Join(names, ", ") + "}"
}

func (a *rowAnalysis) reportWithCategory(node *ast.Node, code string, category string, message string) {
	file := ast.GetSourceFileOfNode(node)
	if file == nil {
		return
	}
	if a.comptime != nil && a.comptime.allowed[node] {
		// The node is inside a construct the comptime frontend owns. It has no
		// runtime existence, so a runtime rule cannot be broken by it.
		return
	}
	start := scanner.SkipTrivia(file.Text(), node.Pos())
	length := node.End() - start
	if length < 0 {
		length = 0
	}
	a.reports[file] = append(a.reports[file], loweringReport{code: code, category: category, message: message, start: start, length: length})
}

func (a *rowAnalysis) report(node *ast.Node, code string, message string) {
	a.reportWithCategory(node, code, "error", message)
}

func (a *rowAnalysis) reportWarning(node *ast.Node, code string, message string) {
	a.reportWithCategory(node, code, "warning", message)
}

func (a *rowAnalysis) checkContracts() {
	for _, fn := range a.functions {
		if fn.explicitReturn && !fn.declaredShape.result && len(fn.bodyFailures) != 0 {
			fn.hasContractError = true
			a.report(fn.node, "SMITHERS1101", "explicit return type cannot represent recoverable failures "+formatRowSet(fn.bodyFailures)+"; use Result<A, E>")
		}
		if fn.explicitReturn && fn.declaredShape.result {
			extra := missingRowSet(fn.bodyFailures, fn.declaredFailures)
			if len(extra) != 0 {
				fn.hasContractError = true
				a.report(fn.node, "SMITHERS1104", "Result contract omits reachable failures "+formatRowSet(extra))
			}
		}
		if fn.exported && !fn.explicitReturn && len(fn.bodyFailures) != 0 {
			fn.hasContractError = true
			a.report(fn.node, "SMITHERS1102", "exported fallible functions must spell Result<A, E> (or Promise<Result<A, E>>) in their public contract")
		}
	}
}

func (a *rowAnalysis) resultParameterReferenceConsumed(node *ast.Node) bool {
	current := node
	for current.Parent != nil {
		parent := current.Parent
		switch {
		case ast.IsNonNullExpression(parent) && parent.AsNonNullExpression().Expression == current:
			return a.resultPropagations[parent] != nil
		case ast.IsParenthesizedExpression(parent), ast.IsAsExpression(parent), parent.Kind == ast.KindSatisfiesExpression:
			current = parent
			continue
		case ast.IsReturnStatement(parent):
			return true
		case func() bool {
			selection, selected := selectMember(parent)
			return selected && selection.receiver == current
		}():
			// Inspecting a discriminant/property or invoking a Result operation
			// consumes the value at this boundary.
			return true
		default:
			return false
		}
	}
	return false
}

func (a *rowAnalysis) checkResultParameters(files []*ast.SourceFile) {
	for _, fn := range a.functions {
		if fn.symbol == nil {
			// Anonymous callback ownership is checked at its enclosing boundary;
			// it is not a separately callable parameter contract.
			continue
		}
		for index, parameter := range fn.node.Parameters() {
			name := parameter.Name()
			if name == nil || !ast.IsIdentifier(name) || !a.prelude.isResultType(a.checker.GetTypeAtLocation(name)) {
				continue
			}
			symbol := a.symbolAt(name)
			consumed := false
			var visit func(*ast.Node) bool
			visit = func(node *ast.Node) bool {
				if node != fn.node.Body() && ast.IsFunctionLikeDeclaration(node) {
					return false
				}
				if ast.IsIdentifier(node) && node != name && a.symbolAt(node) == symbol && a.resultParameterReferenceConsumed(node) {
					consumed = true
				}
				node.ForEachChild(visit)
				return false
			}
			fn.node.Body().ForEachChild(visit)
			if !consumed {
				fn.unconsumedParams[index] = true
				a.report(name, "SMITHERS1302", "Result parameter must be consumed by returning, matching, transforming, inspecting, or propagating it with postfix !")
			}
		}
	}

	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsCallExpression(node) {
				callee := a.resolveCallee(node)
				arguments := node.AsCallExpression().Arguments
				if callee != nil && arguments != nil {
					for index := range callee.unconsumedParams {
						if index < len(arguments.Nodes) && a.prelude.isResultType(a.checker.GetTypeAtLocation(arguments.Nodes[index])) {
							a.report(arguments.Nodes[index], "SMITHERS1301", "produced Result must be returned, matched, transformed, inspected, or propagated with postfix !")
						}
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) checkOneLayer(edge rowProvideEdge) {
	if edge.callback == nil {
		a.report(edge.node, "SMITHERS2103", "Layer.provide callback must resolve to a checked local function")
		return
	}
	if !edge.complete {
		target := edge.argument
		if target == nil {
			target = edge.node
		}
		a.report(target, "SMITHERS2104", "Layer expression is opaque; the compiler cannot prove its provided capability closure")
		return
	}
	if edge.callback.inferredFallible() && !edge.callback.declaredShape.result {
		target := edge.callbacks
		if target == nil {
			target = edge.node
		}
		a.report(target, "SMITHERS2105", "fallible Layer.provide callbacks need an explicit Result (or Promise<Result>) contract so the provided computation keeps its failure channel")
	}
	if enclosingFunction(edge.node) == nil {
		missing := missingRowSet(edge.callback.requirements, edge.provided)
		if len(missing) != 0 {
			a.report(edge.node, "SMITHERS2101", "Layer.provide is missing "+formatRowSet(missing))
		}
	}
}

func (a *rowAnalysis) checkLayers() {
	for _, fn := range a.functions {
		for _, edge := range fn.provides {
			a.checkOneLayer(edge)
		}
	}
	for _, edge := range a.topProvides {
		a.checkOneLayer(edge)
	}
}

func (a *rowAnalysis) checkTopLevelRequirements(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) {
				return false
			}
			if ast.IsCallExpression(node) {
				if callee := a.resolveCallee(node); callee != nil && len(callee.requirements) != 0 {
					a.report(node, "SMITHERS2102", "top-level call has unsatisfied requirements "+formatRowSet(callee.requirements))
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

// checkTopLevelCheckedChannels rejects every operation that would charge a
// failure row at module evaluation, where no enclosing function contract can
// carry it. Recognition follows the same checker identities used to build
// function rows: the prelude's Result.expect and panic declarations, analyzed
// local callees, and the resolved declaration of a foreign call. A method or
// function that merely shares one of those names remains ordinary.
//
// Top-level postfix propagation is owned by lowerPropagation (SMITHERS1202), and a
// top-level throw is owned by visitStatement (SMITHERS1511). Keeping this pass
// to the Panic-producing call family avoids duplicate or competing diagnoses.
func (a *rowAnalysis) checkTopLevelCheckedChannels(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsFunctionLikeDeclaration(node) || node.Kind == ast.KindClassStaticBlockDeclaration {
				return false
			}
			if ast.IsCallExpression(node) {
				if _, ok := a.prelude.expectCall(node); ok {
					a.report(node, "SMITHERS1505", "top-level Result.expect() panics on the error variant and cannot expose that checked panic channel; move it into a checked Result-returning function and consume that Result")
				} else if a.resolveCallee(node) != nil {
					// An authored function call exposes its Result normally. Its
					// returned value is governed by must-consume/unwrap/expect, not
					// by the foreign Panic boundary below.
					node.ForEachChild(visit)
					return false
				} else if a.prelude.isPanicCall(node) {
					a.report(node, "SMITHERS1505", "top-level panic cannot be represented as a checked Result; move it into a checked Result-returning function and consume that Result")
				} else if policy, foreign := a.foreignPolicyForCall(node); foreign && policy.kind != "never" {
					a.report(node, "SMITHERS1505", "an untrusted foreign call at top level cannot expose its checked panic channel; move it into a checked Result-returning function and consume that Result")
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func directCallReference(node *ast.Node) bool {
	if node == nil || node.Parent == nil {
		return false
	}
	if ast.IsCallExpression(node.Parent) && node.Parent.AsCallExpression().Expression == node {
		return true
	}
	if ast.IsPropertyAccessExpression(node.Parent) && node.Parent.AsPropertyAccessExpression().Name() == node &&
		node.Parent.Parent != nil && ast.IsCallExpression(node.Parent.Parent) &&
		node.Parent.Parent.AsCallExpression().Expression == node.Parent {
		return true
	}
	return false
}

func (a *rowAnalysis) checkCrossModuleEscapes(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsIdentifier(node) && !ast.IsPartOfTypeNode(node) &&
				!ast.IsDeclarationNameOrImportPropertyName(node) {
				target := a.bySymbol[a.symbolAt(node)]
				if target != nil && target.file != file && !directCallReference(node) {
					a.report(node, "SMITHERS1802", "cross-module function '"+target.name+"' escapes direct static call analysis; wrap the higher-order use in an explicitly checked local function")
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func trustedForeignModule(file *ast.SourceFile) bool {
	if file == nil || len(file.Statements.Nodes) == 0 {
		return false
	}
	text := file.Text()
	first := file.Statements.Nodes[0]
	end := scanner.SkipTrivia(text, first.Pos())
	if end < 0 || end > len(text) {
		return false
	}
	leading := text[:end]
	for offset := 0; offset < len(leading); {
		open := strings.Index(leading[offset:], "/**")
		if open < 0 {
			break
		}
		open += offset
		close := strings.Index(leading[open+3:], "*/")
		if close < 0 {
			break
		}
		close += open + 3
		comment := leading[open : close+2]
		if exactModuleMarker(comment) && exactThrowsNeverMarker(comment) {
			return true
		}
		offset = close + 2
	}
	return false
}

func exactModuleMarker(comment string) bool {
	lower := strings.ToLower(comment)
	const marker = "@module"
	for offset := 0; offset < len(lower); {
		index := strings.Index(lower[offset:], marker)
		if index < 0 {
			return false
		}
		index += offset
		next := index + len(marker)
		if next == len(lower) || lower[next] == '*' || isJSDocWhitespace(lower[next]) {
			return true
		}
		offset = next
	}
	return false
}

func exactThrowsNeverMarker(comment string) bool {
	lower := strings.ToLower(comment)
	const marker = "@throws"
	for offset := 0; offset < len(lower); {
		index := strings.Index(lower[offset:], marker)
		if index < 0 {
			return false
		}
		index += offset
		position := index + len(marker)
		for position < len(lower) && isJSDocWhitespace(lower[position]) {
			position++
		}
		if position >= len(lower) || lower[position] != '{' {
			offset = index + len(marker)
			continue
		}
		position++
		for position < len(lower) && isJSDocWhitespace(lower[position]) {
			position++
		}
		if !strings.HasPrefix(lower[position:], "never") {
			offset = index + len(marker)
			continue
		}
		position += len("never")
		for position < len(lower) && isJSDocWhitespace(lower[position]) {
			position++
		}
		if position < len(lower) && lower[position] == '}' {
			return true
		}
		offset = index + len(marker)
	}
	return false
}

func isJSDocWhitespace(value byte) bool {
	return value == ' ' || value == '\t' || value == '\r' || value == '\n'
}

// staticRuntimeModuleEdge returns the module specifier whose initializer runs
// for an authored import or re-export. Type-only, compiler-owned, and admitted
// compile-time asset edges are erased and therefore have no trust boundary.
func (a *rowAnalysis) staticRuntimeModuleEdge(statement *ast.Node) *ast.Node {
	if statement == nil {
		return nil
	}
	if _, asset := a.assetImports[statement]; asset {
		return nil
	}
	var specifier *ast.Node
	switch {
	case ast.IsImportDeclaration(statement):
		declaration := statement.AsImportDeclaration()
		if importIsTypeOnly(declaration) {
			return nil
		}
		specifier = declaration.ModuleSpecifier
	case ast.IsExportDeclaration(statement):
		declaration := statement.AsExportDeclaration()
		if declaration.ModuleSpecifier == nil || nativeExportIsTypeOnly(declaration) {
			return nil
		}
		specifier = declaration.ModuleSpecifier
	default:
		return nil
	}
	if a.prelude.isCompilerModuleSpecifier(specifier) {
		return nil
	}
	return specifier
}

func (a *rowAnalysis) foreignModuleSource(specifier *ast.Node) *ast.SourceFile {
	if specifier == nil {
		return nil
	}
	module := a.checker.ResolveExternalModuleName(specifier)
	if module == nil {
		return nil
	}
	for _, declaration := range module.Declarations {
		if a.isForeignDeclaration(declaration) {
			return ast.GetSourceFileOfNode(declaration)
		}
	}
	return nil
}

// checkDynamicModuleImports deliberately establishes the conservative base
// shared with the reference frontend: dynamic code-module imports are deferred
// until both backends can enforce one exact rewrite/trust policy. Compiler-owned
// dynamic asset imports are excluded because checkAssetImports already owns and
// erases those edges before runtime.
func (a *rowAnalysis) checkDynamicModuleImports(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsImportCall(node) {
				if _, asset := a.assetImports[node]; asset {
					return false
				}
				target := node
				call := node.AsCallExpression()
				if call.Arguments != nil && len(call.Arguments.Nodes) != 0 {
					target = call.Arguments.Nodes[0]
					if moduleFile := a.foreignModuleSource(target); moduleFile != nil {
						a.refusedForeignModules[moduleFile] = true
					}
				}
				a.report(target, "SMITHERS1510", "dynamic module import is deferred until the compiler can preserve one checked module-initialization trust boundary; use a static trusted import or a checked async foreign adapter")
				return false
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) checkForeignModuleTrust(files []*ast.SourceFile) {
	for _, file := range files {
		for _, statement := range file.Statements.Nodes {
			specifier := a.staticRuntimeModuleEdge(statement)
			if specifier == nil {
				continue
			}
			module := a.checker.ResolveExternalModuleName(specifier)
			if module == nil {
				// An unresolved runtime import cannot carry the leading module
				// initialization trust declaration. Compiler modules were already
				// excluded by resolved module identity above. A missing relative
				// Smithers module is instead owned by SMITHERS1801, because it is
				// a missing checked-closure edge rather than a foreign package.
				if ast.IsStringLiteral(specifier) {
					text := specifier.Text()
					if strings.HasPrefix(text, ".") && strings.HasSuffix(text, ".sm") {
						continue
					}
				}
				a.report(specifier, "SMITHERS1510", "foreign module initialization can panic before a checked call boundary; the module could not be resolved to a declaration carrying a leading @module and @throws {never} trust claim")
				continue
			}
			target := a.foreignModuleSource(specifier)
			if target != nil && !trustedForeignModule(target) {
				a.refusedForeignModules[target] = true
				a.report(specifier, "SMITHERS1510", "statically loaded TypeScript/JavaScript module does not declare a leading @module and @throws {never} initialization trust claim")
			}
		}
	}
}

// lowerer rewrites one authored `.sm` source file.
type lowerer struct {
	emitContext *printer.EmitContext
	factory     *printer.NodeFactory
	visitor     *ast.NodeVisitor
	checker     *checker.Checker
	prelude     *preludeSymbols
	analysis    *rowAnalysis
	comptime    *comptimeAnalysis
	file        *ast.SourceFile

	shape               liftShape
	canHoist            bool
	impure              bool
	guardedByCatch      bool
	pending             []*ast.Node
	nextTemp            int
	used                map[string]bool
	reports             []loweringReport
	needsPreludeRuntime bool
}

func newLowerer(typeChecker *checker.Checker, prelude *preludeSymbols, analysis *rowAnalysis, comptime *comptimeAnalysis) *lowerer {
	low := &lowerer{
		emitContext: printer.NewEmitContext(),
		checker:     typeChecker,
		prelude:     prelude,
		analysis:    analysis,
		comptime:    comptime,
		used:        make(map[string]bool),
	}
	low.factory = low.emitContext.Factory
	low.visitor = low.emitContext.NewNodeVisitor(low.visit)
	return low
}

// hoistableStatements are the statement kinds where a statement introduced
// before the statement runs exactly once, in the same order, as the expression
// it was extracted from. Propagation is only lowered inside these; anywhere
// else the authored call is left intact and reported.
var hoistableStatements = map[ast.Kind]struct{}{
	ast.KindVariableStatement:   {},
	ast.KindReturnStatement:     {},
	ast.KindExpressionStatement: {},
	ast.KindThrowStatement:      {},
	ast.KindIfStatement:         {},
}

func (l *lowerer) transform(file *ast.SourceFile) *ast.SourceFile {
	l.file = file
	lowered := l.visitor.VisitSourceFile(file)
	if len(l.used) == 0 && !l.needsPreludeRuntime {
		return lowered
	}
	importDeclaration := l.newPreludeImport(file.FileName())
	statements := append([]*ast.Node{importDeclaration}, lowered.Statements.Nodes...)
	list := l.factory.NewNodeList(statements)
	list.Loc = lowered.Statements.Loc
	return l.factory.UpdateSourceFile(lowered, list, lowered.EndOfFileToken).AsSourceFile()
}

// newPreludeImport builds `import { SmithersOk as __smithersOk, … } from
// "<relative>/__smithers_prelude.js"` over exactly the exports this file needed.
// The node is synthesized, so it carries no authored position and stays
// unmapped in the emitted source map.
func (l *lowerer) newPreludeImport(authoredName string) *ast.Node {
	specifiers := make([]*ast.Node, 0, len(preludeImports))
	for _, entry := range preludeImports {
		if !l.used[entry.exported] {
			continue
		}
		specifiers = append(specifiers, l.factory.NewImportSpecifier(
			false,
			l.factory.NewIdentifier(entry.exported),
			l.factory.NewIdentifier(entry.local),
		))
	}
	specifier := l.factory.NewStringLiteral(preludeSpecifierFor(authoredName), ast.TokenFlagsNone)
	if len(specifiers) == 0 {
		return l.factory.NewImportDeclaration(nil, nil, specifier, nil)
	}
	namedImports := l.factory.NewNamedImports(l.factory.NewNodeList(specifiers))
	clause := l.factory.NewImportClause(ast.KindUnknown, nil, namedImports)
	return l.factory.NewImportDeclaration(nil, clause, specifier, nil)
}

// preludeSpecifierFor computes the relative module specifier from one virtual
// project file to the injected prelude.
func preludeSpecifierFor(virtualName string) string {
	directory := path.Dir(virtualName)
	rest := strings.Trim(strings.TrimPrefix(directory, "/src"), "/")
	if rest == "" {
		return "./" + preludeModuleBase
	}
	return strings.Repeat("../", strings.Count(rest, "/")+1) + preludeModuleBase
}

func (l *lowerer) report(node *ast.Node, code string, message string) {
	start := node.Pos()
	if l.file != nil {
		start = scanner.SkipTrivia(l.file.Text(), start)
	}
	length := node.End() - start
	if length < 0 {
		length = 0
	}
	l.reports = append(l.reports, loweringReport{code: code, message: message, start: start, length: length})
}

type lowererScope struct {
	shape          liftShape
	canHoist       bool
	impure         bool
	guardedByCatch bool
	pending        []*ast.Node
}

func (l *lowerer) saveScope() lowererScope {
	return lowererScope{
		shape:          l.shape,
		canHoist:       l.canHoist,
		impure:         l.impure,
		guardedByCatch: l.guardedByCatch,
		pending:        l.pending,
	}
}

func (l *lowerer) restoreScope(saved lowererScope) {
	l.shape = saved.shape
	l.canHoist = saved.canHoist
	l.impure = saved.impure
	l.guardedByCatch = saved.guardedByCatch
	l.pending = saved.pending
}

func (l *lowerer) visit(node *ast.Node) *ast.Node {
	if node == nil {
		return nil
	}
	// The comptime frontend decided everything about this node before the
	// visitor ran. Substituting here — and returning without descending —
	// is what makes an unselected `comptime.target` branch, an erased
	// compile-time function body, and the compiler-owned import all disappear
	// from the emitted artifact rather than being visited as runtime code.
	if replacement, owned := l.comptimeSubstitution(node); owned {
		return replacement
	}
	if ast.IsFunctionLike(node) {
		saved := l.saveScope()
		l.shape = l.shapeOf(node)
		l.canHoist = false
		l.impure = false
		l.guardedByCatch = false
		l.pending = nil
		var result *ast.Node
		if l.shape.lifts() && ast.IsArrowFunction(node) && node.Body() != nil && node.Body().Kind != ast.KindBlock {
			result = l.lowerConciseBody(node)
		} else {
			result = l.visitor.VisitEachChild(node)
		}
		l.restoreScope(saved)
		return result
	}
	if node.Kind == ast.KindBlock {
		return l.visitBlock(node)
	}
	if ast.IsStatement(node) {
		return l.visitStatement(node)
	}
	return l.visitExpression(node)
}

func (l *lowerer) visitBlock(node *ast.Node) *ast.Node {
	block := node.AsBlock()
	statements := l.visitStatementSequence(block.Statements.Nodes)
	list := l.factory.NewNodeList(statements)
	list.Loc = block.Statements.Loc
	return l.factory.UpdateBlock(block, list, block.MultiLine)
}

// visitStatementSequence lowers each statement in source order.
func (l *lowerer) visitStatementSequence(statements []*ast.Node) []*ast.Node {
	result := make([]*ast.Node, 0, len(statements))
	for _, statement := range statements {
		result = appendVisitedStatement(result, l.visit(statement))
	}
	return result
}

func appendVisitedStatement(result []*ast.Node, visited *ast.Node) []*ast.Node {
	if visited == nil {
		return result
	}
	if visited.Kind == ast.KindSyntaxList {
		return append(result, visited.AsSyntaxList().Children...)
	}
	return append(result, visited)
}

// shapeOf derives the checked return contract from the declared return type
// annotation. `Promise` is only looked through for an `async` function, so a
// plain function that hands back a promise value is never lifted.
func (l *lowerer) shapeOf(node *ast.Node) liftShape {
	if l.analysis != nil {
		if fn := l.analysis.byNode[node]; fn != nil && fn.inferredFallible() {
			return liftShape{async: ast.HasSyntacticModifier(node, ast.ModifierFlagsAsync), result: true}
		}
	}
	typeNode := node.Type()
	if typeNode == nil {
		return liftShape{}
	}
	shape := liftShape{}
	if ast.HasSyntacticModifier(node, ast.ModifierFlagsAsync) {
		inner := l.promiseArgument(typeNode)
		if inner == nil {
			return liftShape{}
		}
		typeNode = inner
		shape.async = true
	}
	shape.result = l.classifyTypeNode(typeNode)
	if !shape.lifts() {
		return liftShape{}
	}
	return shape
}

// promiseArgument returns the awaited type node of a `Promise<T>` annotation.
// Promise identity comes from the checker's own definition of a promise, not
// from the spelling of the type name.
func (l *lowerer) promiseArgument(typeNode *ast.Node) *ast.Node {
	if !ast.IsTypeReferenceNode(typeNode) {
		return nil
	}
	reference := typeNode.AsTypeReferenceNode()
	if reference.TypeArguments == nil || len(reference.TypeArguments.Nodes) != 1 {
		return nil
	}
	if l.checker.GetPromisedTypeOfPromise(l.checker.GetTypeFromTypeNode(typeNode)) == nil {
		return nil
	}
	return reference.TypeArguments.Nodes[0]
}

// classifyTypeNode resolves an annotation to the compiler-owned Result by
// symbol identity.
func (l *lowerer) classifyTypeNode(typeNode *ast.Node) bool {
	if typeNode == nil || !ast.IsTypeReferenceNode(typeNode) {
		return false
	}
	reference := typeNode.AsTypeReferenceNode()
	symbol := l.checker.GetSymbolAtLocation(reference.TypeName)
	return l.prelude.isResultSymbol(symbol)
}

// lowerConciseBody wraps the expression body of a Result-returning
// arrow function, so `(): Result<A, E> => value` becomes `=> new __smithersOk(value)`.
// Propagation is not lowered out of a concise body: there is no statement list
// to hoist the guard into, and the prelude's runtime `unwrap` still applies.
func (l *lowerer) lowerConciseBody(node *ast.Node) *ast.Node {
	arrow := node.AsArrowFunction()
	original := arrow.Body
	body := l.visitor.VisitNode(original)
	lifted := l.liftValue(original, body)
	if lifted != body {
		l.emitContext.AssignSourceMapRange(lifted, original)
	}
	return l.factory.UpdateArrowFunction(
		arrow,
		l.visitor.VisitModifiers(arrow.Modifiers()),
		arrow.TypeParameters,
		l.visitor.VisitNodes(arrow.Parameters),
		arrow.Type,
		arrow.FullSignature,
		arrow.EqualsGreaterThanToken,
		lifted,
	)
}

// visitStatement gives every statement its own hoist scope so that temporaries
// introduced by propagation land immediately before the statement that needed
// them, and so evaluation-order accounting restarts at every statement.
func (l *lowerer) visitStatement(node *ast.Node) *ast.Node {
	saved := l.saveScope()
	_, hoistable := hoistableStatements[node.Kind]
	l.canHoist = hoistable && !declaresSeveralNames(node)
	l.impure = false
	l.pending = nil

	var lowered *ast.Node
	switch {
	case ast.IsImportDeclaration(node) && l.assetImportModule(node) != nil:
		lowered = l.lowerAssetImport(node, l.assetImportModule(node))
	case ast.IsExportDeclaration(node) && l.assetImportModule(node) != nil:
		lowered = l.lowerAssetExport(node, l.assetImportModule(node))
	case ast.IsImportDeclaration(node) && durableResolvedModule(l.checker, node.AsImportDeclaration().ModuleSpecifier):
		declaration := node.AsImportDeclaration()
		specifier := l.factory.NewStringLiteral(durablePreludeSpecifierFor(l.file.FileName()), ast.TokenFlagsNone)
		l.emitContext.AssignSourceMapRange(specifier, declaration.ModuleSpecifier)
		lowered = l.factory.UpdateImportDeclaration(
			declaration,
			l.visitor.VisitModifiers(declaration.Modifiers()),
			l.visitor.VisitNode(declaration.ImportClause),
			specifier,
			l.visitor.VisitNode(declaration.Attributes),
		)
	case ast.IsExportDeclaration(node) && durableResolvedModule(l.checker, node.AsExportDeclaration().ModuleSpecifier):
		declaration := node.AsExportDeclaration()
		specifier := l.factory.NewStringLiteral(durablePreludeSpecifierFor(l.file.FileName()), ast.TokenFlagsNone)
		l.emitContext.AssignSourceMapRange(specifier, declaration.ModuleSpecifier)
		lowered = l.factory.UpdateExportDeclaration(
			declaration,
			l.visitor.VisitModifiers(declaration.Modifiers()),
			declaration.IsTypeOnly,
			l.visitor.VisitNode(declaration.ExportClause),
			specifier,
			l.visitor.VisitNode(declaration.Attributes),
		)
	case ast.IsImportDeclaration(node) && l.prelude.isCompilerModuleSpecifier(node.AsImportDeclaration().ModuleSpecifier):
		declaration := node.AsImportDeclaration()
		specifier := l.factory.NewStringLiteral(preludeSpecifierFor(l.file.FileName()), ast.TokenFlagsNone)
		l.emitContext.AssignSourceMapRange(specifier, declaration.ModuleSpecifier)
		lowered = l.factory.UpdateImportDeclaration(
			declaration,
			l.visitor.VisitModifiers(declaration.Modifiers()),
			l.visitor.VisitNode(declaration.ImportClause),
			specifier,
			l.visitor.VisitNode(declaration.Attributes),
		)
	case ast.IsExportDeclaration(node) && l.prelude.isCompilerModuleSpecifier(node.AsExportDeclaration().ModuleSpecifier):
		declaration := node.AsExportDeclaration()
		specifier := l.factory.NewStringLiteral(preludeSpecifierFor(l.file.FileName()), ast.TokenFlagsNone)
		l.emitContext.AssignSourceMapRange(specifier, declaration.ModuleSpecifier)
		lowered = l.factory.UpdateExportDeclaration(
			declaration,
			l.visitor.VisitModifiers(declaration.Modifiers()),
			declaration.IsTypeOnly,
			l.visitor.VisitNode(declaration.ExportClause),
			specifier,
			l.visitor.VisitNode(declaration.Attributes),
		)
	case node.Kind == ast.KindTryStatement:
		lowered = l.visitTry(node)
	case node.Kind == ast.KindIfStatement && node.AsIfStatement().Initializer != nil:
		lowered = l.lowerConditionalDeclaration(node)
	case node.Kind == ast.KindThrowStatement && !l.guardedByCatch && enclosingFunction(node) == nil:
		// Module initialization has no checked channel to carry a failure, so a
		// top-level throw cannot be represented as a Result at all.
		l.report(node, "SMITHERS1511", "a top-level throw cannot be represented as a checked Result; move it into a checked Result-returning function and consume that Result")
		lowered = l.visitor.VisitEachChild(node)
	case l.shape.result && l.panicExit(node) != nil:
		lowered = l.lowerPanicExit(node, l.panicExit(node))
	case l.shape.result && node.Kind == ast.KindThrowStatement && !l.guardedByCatch:
		lowered = l.lowerThrow(node)
	case l.shape.lifts() && node.Kind == ast.KindReturnStatement:
		lowered = l.lowerReturn(node)
	default:
		lowered = l.visitor.VisitEachChild(node)
	}
	if len(l.pending) != 0 {
		lowered = l.factory.NewSyntaxList(append(l.pending, lowered))
	}

	l.restoreScope(saved)
	return lowered
}

// declaresSeveralNames reports whether a statement binds more than one name.
// Hoisting out of such a statement would move code above bindings the statement
// has already introduced, so propagation stays out of them entirely.
func declaresSeveralNames(node *ast.Node) bool {
	if node.Kind != ast.KindVariableStatement {
		return false
	}
	list := node.AsVariableStatement().DeclarationList
	return len(list.AsVariableDeclarationList().Declarations.Nodes) != 1
}

// visitTry marks the `try` block of a try statement that has a `catch` clause.
// Inside it, a recoverable exit must keep JavaScript throw behavior: rewriting
// it into an early `return` would make the authored `catch` silently dead. The
// `catch` and `finally` blocks are not themselves guarded by that clause.
func (l *lowerer) visitTry(node *ast.Node) *ast.Node {
	try := node.AsTryStatement()
	saved := l.guardedByCatch
	if try.CatchClause != nil {
		l.guardedByCatch = true
	}
	tryBlock := l.visitor.VisitNode(try.TryBlock)
	l.guardedByCatch = saved
	catchClause := l.visitor.VisitNode(try.CatchClause)
	finallyBlock := l.visitor.VisitNode(try.FinallyBlock)
	return l.factory.UpdateTryStatement(try, tryBlock, catchClause, finallyBlock)
}

// promiseChainCall reports whether a call is `<promise>.then/catch/finally(…)`.
// The receiver must be a promise by the checker's own definition, and the
// property must be that promise type's own member, resolved through the
// checker rather than matched on spelling.
func (l *lowerer) promiseChainCall(node *ast.Node) bool {
	if !ast.IsCallExpression(node) {
		return false
	}
	call := node.AsCallExpression()
	selection, selected := selectMember(call.Expression)
	if !selected {
		return false
	}
	receiver := l.checker.GetTypeAtLocation(selection.receiver)
	if receiver == nil || l.checker.GetPromisedTypeOfPromise(receiver) == nil {
		return false
	}
	symbol := l.prelude.memberSymbol(selection)
	if symbol == nil {
		return false
	}
	symbol = l.checker.GetMergedSymbol(symbol)
	for _, member := range []string{"then", "catch", "finally"} {
		property := l.checker.GetPropertyOfType(receiver, member)
		if property != nil && l.checker.GetMergedSymbol(property) == symbol {
			return true
		}
	}
	return false
}

// visitExpression decides the lowering for an expression before descending
// into it, then tracks whether anything with an observable effect has already
// been evaluated in the current statement. That flag is what makes hoisting
// order-preserving: a propagation point may only be hoisted while everything
// evaluated ahead of it in the same statement is effect-free.
func (l *lowerer) visitExpression(node *ast.Node) *ast.Node {
	if ast.IsNonNullExpression(node) {
		if receiver := l.analysis.resultPropagations[node]; receiver != nil {
			return l.lowerPropagation(node, receiver)
		}
		if producer := directPromiseProducer(node.AsNonNullExpression().Expression); producer != nil &&
			l.checker.GetPromisedTypeOfPromise(l.checker.GetTypeAtLocation(producer)) != nil {
			// The unbound producer analysis reports the missing await at the
			// Promise call. Adding SMITHERS1207 here would be a consequence-only
			// cascade for the same authored mistake.
			l.impure = true
			return l.visitor.VisitEachChild(node)
		}
		l.report(node, "SMITHERS1207", "postfix ! requires a Result operand; TypeScript non-null assertions are unavailable in .sm")
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}
	if ast.IsImportCall(node) && l.assetImportModule(node) != nil {
		return l.lowerDynamicAssetImport(node, l.assetImportModule(node))
	}
	if lowered, matched := l.lowerDurableCall(node); matched {
		return lowered
	}
	if hook, ok := l.prelude.authoredConstructor(node); ok {
		l.report(node, "SMITHERS1201", hook+" is a compiler hook, not an author-facing constructor; use ordinary return/throw lifting")
		return l.visitor.VisitEachChild(node)
	}
	if lowered, matched := l.lowerResultNamespaceCall(node); matched {
		return lowered
	}
	if receiver, ok := l.prelude.expectCall(node); ok {
		return l.lowerExpect(node, receiver)
	}
	if _, kind := l.prelude.unwrapCall(node); kind == unwrapResult {
		l.report(node, "SMITHERS1206", "Result.unwrap() is no longer the propagation spelling; use postfix !")
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}
	if receiver, handlers, fallback, kind := l.prelude.matchCall(node); kind != errorMatchNone {
		return l.lowerMatch(node, receiver, handlers, fallback, kind)
	}
	if l.prelude.isPanicCall(node) {
		if enclosingFunction(node) == nil {
			// The row analysis owns this situation as SMITHERS1505. Reporting
			// the expression-position SMITHERS1503 as well would obscure the
			// actionable fact that module evaluation has no checked channel.
			return l.visitor.VisitEachChild(node)
		}
		if owner := enclosingFunction(node); l.analysis != nil && owner != nil {
			if fn := l.analysis.byNode[owner]; fn != nil && fn.hasContractError {
				// The declaration-level contract diagnostic is the actionable root
				// cause; no emit occurs, so a lowering-placement cascade adds no
				// information and diverges from the reference.
				return l.visitor.VisitEachChild(node)
			}
		}
		// A panic that reached here is not one of the two exit positions
		// lowerPanicExit defines. Leaving it would keep the prelude's runtime
		// throw inside a value position whose Result meaning is undefined.
		l.report(node, "SMITHERS1503", "panic(...) lowering is supported only as an expression statement or a direct return")
		return l.visitor.VisitEachChild(node)
	}
	if l.prelude.isErrorRuntimeMember(node) {
		l.needsPreludeRuntime = true
	}
	if l.analysis != nil {
		if policy, ok := l.analysis.foreignCalls[node]; ok && policy.kind != "never" {
			return l.lowerForeignCall(node, policy)
		}
	}
	if l.promiseChainCall(node) {
		l.report(node, "SMITHERS1401", "Promise instance .then(), .catch(), and .finally() are unavailable in authored .sm; consume the Promise with await")
		selection, _ := selectMember(node.AsCallExpression().Expression)
		if producer := directPromiseProducer(selection.receiver); producer != nil {
			l.report(producer, "SMITHERS1402", "started Promise is not consumed with await, return, or an awaited recognized combinator")
		}
	}
	lowered := l.visitConditionalRegions(node)
	if isEffectful(node) {
		l.impure = true
	}
	return lowered
}

// lowerLabeledExpression consumes the fork's real LabeledExpression wrapper and
// its checker-computed join type. The construct runs at its authored evaluation
// point as a statement sequence hoisted immediately before the containing
// statement; every value break assigns the join temporary and exits the compiler
// wrapper, while a loop's normal/plain-break exit assigns ElseValue instead.

// lowerConditionalDeclaration consumes the fork binder's proved locals scope:
//
//	if (const x = f(); condition) { ... } else { ... }
//
// becomes a synthesized block containing the authored declaration followed by
// the same if with its Initializer cleared. The declaration and rewritten if
// retain authored source ranges; the block delimiters are compiler glue and
// therefore remain unmapped.
func (l *lowerer) lowerConditionalDeclaration(node *ast.Node) *ast.Node {
	statement := node.AsIfStatement()
	initializer := statement.Initializer
	if initializer == nil {
		return l.visitor.VisitEachChild(node)
	}
	if initializer.Kind != ast.KindVariableDeclarationList || statement.Locals == nil {
		l.report(node, "SMITHERS1717", "a conditional declaration must carry the fork binder's own locals table and one variable declaration list")
		return l.visitor.VisitEachChild(node)
	}
	declarations := initializer.AsVariableDeclarationList()
	if declarations.Flags&ast.NodeFlagsBlockScoped == 0 {
		l.report(node, "SMITHERS1717", "`var` in a conditional declaration hoists out of the conditional construct; use `const` or `let`")
		return l.visitor.VisitEachChild(node)
	}
	if !conditionalDeclarationBranchesAreBraced(node) {
		l.report(node, "SMITHERS1717", "a conditional declaration requires braced branches throughout its if/else-if/else chain")
		return l.visitor.VisitEachChild(node)
	}

	visitedInitializer := l.visitor.VisitNode(initializer)
	visitedCondition := l.visitor.VisitNode(statement.Expression)
	visitedThen := l.visitor.VisitEmbeddedStatement(statement.ThenStatement)
	visitedElse := l.visitor.VisitEmbeddedStatement(statement.ElseStatement)
	declaration := l.factory.NewVariableStatement(nil, visitedInitializer)
	l.emitContext.AssignSourceMapRange(declaration, initializer)
	l.emitContext.SetOriginal(declaration, initializer)
	conditional := l.factory.UpdateIfStatement(statement, nil, visitedCondition, visitedThen, visitedElse)
	return l.factory.NewBlock(l.factory.NewNodeList([]*ast.Node{declaration, conditional}), true)
}

func conditionalDeclarationBranchesAreBraced(node *ast.Node) bool {
	if node == nil || node.Kind != ast.KindIfStatement {
		return false
	}
	statement := node.AsIfStatement()
	if statement.ThenStatement == nil || statement.ThenStatement.Kind != ast.KindBlock {
		return false
	}
	if statement.ElseStatement == nil {
		return true
	}
	if statement.ElseStatement.Kind == ast.KindBlock {
		return true
	}
	return conditionalDeclarationBranchesAreBraced(statement.ElseStatement)
}

func (l *lowerer) lowerForeignCall(node *ast.Node, policy foreignPolicy) *ast.Node {
	visited := l.visitor.VisitEachChild(node)
	body := l.factory.NewArrowFunction(
		nil,
		nil,
		l.factory.NewNodeList([]*ast.Node{}),
		nil,
		nil,
		l.factory.NewToken(ast.KindEqualsGreaterThanToken),
		visited,
	)
	arguments := []*ast.Node{body}
	if policy.kind == "declared" && policy.errorName != "" {
		arguments = append(arguments, l.factory.NewIdentifier(policy.errorName))
	}
	helper := "smithersTry"
	if policy.async {
		helper = "smithersTryPromise"
	}
	lowered := l.newPreludeCall(helper, arguments...)
	l.emitContext.AssignSourceMapRange(lowered, node)
	l.emitContext.SetOriginal(lowered, node)
	l.impure = true
	return lowered
}

// lowerResultNamespaceCall rewrites an authored `Result.try(body)` /
// `Result.tryPromise(body)` / `Result.all(results)` call onto the prelude
// runtime that implements it. `Result` itself has no runtime existence — it is
// a `declare global` namespace — so the callee, not the whole call, is what is
// replaced.
//
// For the two adapters the body is left alone. Inside it a foreign call keeps
// its authored throw: the boundary is the catch scope, which is exactly why the
// row analysis does not charge or wrap foreign calls there.
func (l *lowerer) lowerResultNamespaceCall(node *ast.Node) (*ast.Node, bool) {
	member, matched := l.prelude.resultNamespaceCall(node)
	if !matched {
		return nil, false
	}
	exported := resultNamespaceExports[member]
	if exported == "" {
		return nil, false
	}
	call := node.AsCallExpression()
	l.used[exported] = true
	callee := l.factory.NewIdentifier(preludeLocalName(exported))
	l.emitContext.AssignSourceMapRange(callee, call.Expression)
	l.emitContext.SetOriginal(callee, call.Expression)
	lowered := l.factory.UpdateCallExpression(
		call,
		callee,
		call.QuestionDotToken,
		call.TypeArguments,
		l.visitor.VisitNodes(call.Arguments),
		call.Flags,
	)
	l.impure = true
	return lowered, true
}

func (a *rowAnalysis) foreignBoundaryExpression(node *ast.Node) (foreignPolicy, bool) {
	if node == nil {
		return foreignPolicy{}, false
	}
	if policy, ok := a.foreignCalls[node]; ok && policy.kind != "never" {
		return policy, true
	}
	switch node.Kind {
	case ast.KindParenthesizedExpression:
		return a.foreignBoundaryExpression(node.AsParenthesizedExpression().Expression)
	case ast.KindAsExpression:
		return a.foreignBoundaryExpression(node.AsAsExpression().Expression)
	case ast.KindNonNullExpression:
		return a.foreignBoundaryExpression(node.AsNonNullExpression().Expression)
	case ast.KindAwaitExpression:
		return a.foreignBoundaryExpression(node.AsAwaitExpression().Expression)
	}
	return foreignPolicy{}, false
}

// lowerExpect consumes a Result and converts its error variant into the
// compiler-owned Panic channel. Both the receiver and message are evaluated
// before the guard, in authored order.
func (l *lowerer) lowerExpect(node *ast.Node, receiver *ast.Node) *ast.Node {
	leave := func() *ast.Node {
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}
	statement, safe := safeUnwrapPlacement(node)
	if safe {
		_, safe = hoistableStatements[statement.Kind]
	}
	if !l.shape.result || l.guardedByCatch || !safe {
		return leave()
	}
	call := node.AsCallExpression()
	visitedReceiver := l.visitor.VisitNode(receiver)
	visitedMessage := l.visitor.VisitNode(call.Arguments.Nodes[0])
	name := fmt.Sprintf("%s%d", tempPrefix, l.nextTemp)
	l.nextTemp++
	messageName := name + "Message"

	receiverDeclaration := l.factory.NewVariableDeclaration(l.factory.NewIdentifier(name), nil, nil, visitedReceiver)
	receiverList := l.factory.NewVariableDeclarationList(l.factory.NewNodeList([]*ast.Node{receiverDeclaration}), ast.NodeFlagsConst)
	receiverBinding := l.factory.NewVariableStatement(nil, receiverList)
	l.emitContext.AssignSourceMapRange(receiverBinding, receiver)
	messageDeclaration := l.factory.NewVariableDeclaration(l.factory.NewIdentifier(messageName), nil, nil, visitedMessage)
	messageList := l.factory.NewVariableDeclarationList(l.factory.NewNodeList([]*ast.Node{messageDeclaration}), ast.NodeFlagsConst)
	messageBinding := l.factory.NewVariableStatement(nil, messageList)
	l.emitContext.AssignSourceMapRange(messageBinding, call.Arguments.Nodes[0])

	condition := l.factory.NewPrefixUnaryExpression(
		ast.KindExclamationToken,
		l.factory.NewPropertyAccessExpression(l.factory.NewIdentifier(name), nil, l.factory.NewIdentifier("ok"), ast.NodeFlagsNone),
	)
	panicValue := l.newVariant("Panic", l.factory.NewIdentifier(messageName))
	guard := l.factory.NewIfStatement(nil, condition, l.factory.NewReturnStatement(l.newVariant("SmithersErr", panicValue)), nil)
	l.emitContext.AssignSourceMapRange(guard, node)
	l.pending = append(l.pending, receiverBinding, messageBinding, guard)
	l.impure = false

	value := l.factory.NewPropertyAccessExpression(l.factory.NewIdentifier(name), nil, l.factory.NewIdentifier("value"), ast.NodeFlagsNone)
	l.emitContext.AssignSourceMapRange(value, node)
	l.emitContext.SetOriginal(value, node)
	return value
}

// visitConditionalRegions disables hoisting inside subexpressions that may not
// be evaluated at all. Hoisting out of a short-circuit right side or a
// conditional branch would make the propagation unconditional.
func (l *lowerer) visitConditionalRegions(node *ast.Node) *ast.Node {
	switch node.Kind {
	case ast.KindBinaryExpression:
		binary := node.AsBinaryExpression()
		if ast.IsLogicalOrCoalescingBinaryOperator(binary.OperatorToken.Kind) {
			left := l.visitor.VisitNode(binary.Left)
			right := l.withoutHoisting(binary.Right)
			return l.factory.UpdateBinaryExpression(binary, binary.Modifiers(), left, binary.Type, binary.OperatorToken, right)
		}
	case ast.KindConditionalExpression:
		conditional := node.AsConditionalExpression()
		condition := l.visitor.VisitNode(conditional.Condition)
		whenTrue := l.withoutHoisting(conditional.WhenTrue)
		whenFalse := l.withoutHoisting(conditional.WhenFalse)
		return l.factory.UpdateConditionalExpression(conditional, condition, conditional.QuestionToken, whenTrue, conditional.ColonToken, whenFalse)
	}
	return l.visitor.VisitEachChild(node)
}

func (l *lowerer) withoutHoisting(node *ast.Node) *ast.Node {
	savedCanHoist := l.canHoist
	l.canHoist = false
	visited := l.visitor.VisitNode(node)
	l.canHoist = savedCanHoist
	return visited
}

// isEffectful reports whether evaluating a node can be observed. Property and
// element reads count: an accessor can run arbitrary code, and moving a
// propagation point ahead of one would change the program.
func isEffectful(node *ast.Node) bool {
	switch node.Kind {
	case ast.KindCallExpression, ast.KindNewExpression, ast.KindTaggedTemplateExpression,
		ast.KindAwaitExpression, ast.KindYieldExpression, ast.KindDeleteExpression,
		ast.KindPropertyAccessExpression, ast.KindElementAccessExpression:
		return true
	case ast.KindPrefixUnaryExpression:
		operator := node.AsPrefixUnaryExpression().Operator
		return operator == ast.KindPlusPlusToken || operator == ast.KindMinusMinusToken
	case ast.KindPostfixUnaryExpression:
		return true
	case ast.KindBinaryExpression:
		return isAssignmentOperator(node.AsBinaryExpression().OperatorToken.Kind)
	}
	return false
}

func isAssignmentOperator(kind ast.Kind) bool {
	return kind >= ast.KindFirstAssignment && kind <= ast.KindLastAssignment
}

// lowerThrow rewrites `throw e` into `return new __smithersErr(e)` inside a
// Result-returning function.
// panicExit answers whether a statement's whole effect is one compiler-owned
// `panic(...)`. Those are the two positions whose Result lowering is defined —
// an expression statement and a direct return — and they are the two the
// reference frontend supports as well.
func (l *lowerer) panicExit(node *ast.Node) *ast.Node {
	switch node.Kind {
	case ast.KindExpressionStatement:
		expression := node.AsExpressionStatement().Expression
		if l.prelude.isPanicCall(expression) {
			return expression
		}
	case ast.KindReturnStatement:
		expression := node.AsReturnStatement().Expression
		if expression != nil && l.prelude.isPanicCall(expression) {
			return expression
		}
	}
	return nil
}

// lowerPanicExit rewrites `panic(cause)` into the error variant carrying the
// compiler-owned Panic. Left alone, `panic` would keep the prelude's runtime
// throw and abort a program whose declared contract says the caller's `match`
// error branch must see it — the exact silent wrongness the checked channel
// exists to prevent.
func (l *lowerer) lowerPanicExit(node *ast.Node, call *ast.Node) *ast.Node {
	arguments := call.AsCallExpression().Arguments
	visited := make([]*ast.Node, 0, 1)
	if arguments != nil {
		for _, argument := range arguments.Nodes {
			visited = append(visited, l.visitor.VisitNode(argument))
		}
	}
	constructed := l.newVariant("SmithersErr", l.newVariant("Panic", visited...))
	lowered := l.factory.NewReturnStatement(constructed)
	l.emitContext.AssignCommentAndSourceMapRanges(lowered, node)
	l.emitContext.SetOriginal(lowered, node)
	return lowered
}

func (l *lowerer) lowerThrow(node *ast.Node) *ast.Node {
	thrown := node.AsThrowStatement().Expression
	if thrown == nil {
		return l.visitor.VisitEachChild(node)
	}
	visited := l.visitor.VisitNode(thrown)
	constructed := l.newVariant("SmithersErr", visited)
	lowered := l.factory.NewReturnStatement(constructed)
	l.emitContext.AssignCommentAndSourceMapRanges(lowered, node)
	l.emitContext.SetOriginal(lowered, node)
	return lowered
}

// lowerReturn lifts the returned value into the function's declared Result
// channel.
func (l *lowerer) lowerReturn(node *ast.Node) *ast.Node {
	returned := node.AsReturnStatement().Expression
	if returned == nil {
		return l.visitor.VisitEachChild(node)
	}
	visited := l.visitor.VisitNode(returned)
	lifted := l.liftValue(returned, visited)
	if lifted == visited && visited == returned {
		return node
	}
	lowered := l.factory.NewReturnStatement(lifted)
	l.emitContext.AssignCommentAndSourceMapRanges(lowered, node)
	l.emitContext.SetOriginal(lowered, node)
	return lowered
}

// liftValue applies the declared channels to a returned expression. `original`
// is the authored node the checker is asked about; `visited` is its lowered
// form.
func (l *lowerer) liftValue(original *ast.Node, visited *ast.Node) *ast.Node {
	valueType := l.checker.GetTypeAtLocation(original)
	if l.shape.result {
		if l.isPropagationSuccessValue(original) {
			return l.newVariant("SmithersOk", visited)
		}
		if l.analysis != nil {
			if _, boundary := l.analysis.foreignBoundaryExpression(original); boundary {
				return visited
			}
		}
		if l.isResultValue(valueType) {
			return visited
		}
		return l.newVariant("SmithersOk", visited)
	}
	return visited
}

// isPropagationSuccessValue reports that lowering changes this authored
// Result-shaped expression into its success value. The stock checker still
// sees NonNullExpression as the Result itself, so liftValue must consult the
// Smithers side table rather than that stale TypeScript meaning.
func (l *lowerer) isPropagationSuccessValue(node *ast.Node) bool {
	if node == nil || l.analysis == nil {
		return false
	}
	if l.analysis.resultPropagations[node] != nil {
		return true
	}
	switch node.Kind {
	case ast.KindParenthesizedExpression, ast.KindAsExpression,
		ast.KindTypeAssertionExpression:
		return l.isPropagationSuccessValue(node.Expression())
	}
	return false
}

// isResultValue reports whether an expression's type already satisfies the
// Result channel. In an async function a promise of a Result satisfies it too,
// because the awaited value is what the caller observes.
func (l *lowerer) isResultValue(t *checker.Type) bool {
	if l.prelude.isResultType(t) {
		return true
	}
	return l.shape.async && l.prelude.isResultType(l.awaited(t))
}

func (l *lowerer) awaited(t *checker.Type) *checker.Type {
	if t == nil {
		return nil
	}
	return l.checker.GetPromisedTypeOfPromise(t)
}

func (l *lowerer) newVariant(exported string, arguments ...*ast.Node) *ast.Node {
	l.used[exported] = true
	return l.factory.NewNewExpression(
		l.factory.NewIdentifier(preludeLocalName(exported)),
		nil,
		l.factory.NewNodeList(arguments),
	)
}

func (l *lowerer) newPreludeCall(exported string, arguments ...*ast.Node) *ast.Node {
	l.used[exported] = true
	return l.factory.NewCallExpression(
		l.factory.NewIdentifier(preludeLocalName(exported)),
		nil,
		nil,
		l.factory.NewNodeList(arguments),
		ast.NodeFlagsNone,
	)
}

// safeUnwrapPlacement reports whether a propagation point sits where a checked
// early return preserves the authored evaluation order, and returns the
// statement its guard would be hoisted in front of. Only wrappers that neither
// evaluate anything ahead of the call nor make it conditional are walked
// through; anything else — an operand, an argument, a branch, a short-circuit
// right side, a further declarator — ends the walk unsafely.
func safeUnwrapPlacement(call *ast.Node) (*ast.Node, bool) {
	current := call
	for current.Parent != nil && !ast.IsStatement(current.Parent) && !ast.IsArrowFunction(current.Parent) {
		parent := current.Parent
		switch {
		case ast.IsParenthesizedExpression(parent), ast.IsAwaitExpression(parent),
			ast.IsAsExpression(parent):
		case ast.IsPropertyAccessExpression(parent) && parent.AsPropertyAccessExpression().Expression == current:
		case ast.IsBinaryExpression(parent) && parent.AsBinaryExpression().OperatorToken.Kind == ast.KindQuestionQuestionToken &&
			parent.AsBinaryExpression().Left == current:
			// The normative `result!?.member ?? fallback` shape is safe because
			// the propagation point is in the always-evaluated left operand. No
			// other compound form is admitted while precedence remains open.
		case ast.IsVariableDeclaration(parent) && parent.AsVariableDeclaration().Initializer == current:
		case ast.IsVariableDeclarationList(parent) && len(parent.AsVariableDeclarationList().Declarations.Nodes) == 1:
		default:
			return nil, false
		}
		current = parent
	}
	if current.Parent == nil || ast.IsArrowFunction(current.Parent) {
		// A concise arrow body has no statement list to hoist a guard into.
		return nil, false
	}
	return current.Parent, true
}

// isInRepeatedLoopHeader reports whether a node sits in a part of a loop that
// re-evaluates. A `for` initializer runs once and can safely carry an outer
// prologue; the condition and incrementor cannot.
func isInRepeatedLoopHeader(node *ast.Node) bool {
	current := node
	for current.Parent != nil {
		parent := current.Parent
		switch parent.Kind {
		case ast.KindWhileStatement, ast.KindDoStatement,
			ast.KindForInStatement, ast.KindForOfStatement:
			return current != parent.Statement()
		case ast.KindForStatement:
			data := parent.AsForStatement()
			return current != data.Statement && current != data.Initializer
		}
		current = parent
	}
	return false
}

// enclosingFunction returns the function-like declaration a node sits in, or
// nil at module top level.
func enclosingFunction(node *ast.Node) *ast.Node {
	for current := node.Parent; current != nil; current = current.Parent {
		if ast.IsFunctionLike(current) {
			return current
		}
	}
	return nil
}

// propagatedFailureNames names the failures a propagation point would charge to
// its owner, for the contract diagnostic. The names come from the resolved
// class symbols of the receiver's error type, never from authored text.
func (l *lowerer) propagatedFailureNames(receiverNode *ast.Node) string {
	if receiverNode == nil {
		return "the propagated failure"
	}
	names := make([]string, 0, 2)
	receiver := l.checker.GetTypeAtLocation(receiverNode)
	constituents := []*checker.Type{receiver}
	if receiver != nil && receiver.IsUnion() {
		constituents = receiver.Types()
	}
	for _, constituent := range constituents {
		if !declaresAnyOf(constituent.Symbol(), l.prelude.errClass) {
			continue
		}
		arguments := l.checker.GetTypeArguments(constituent)
		if len(arguments) == 0 {
			continue
		}
		errors := []*checker.Type{arguments[0]}
		if arguments[0].IsUnion() {
			errors = arguments[0].Types()
		}
		for _, item := range errors {
			if symbol := item.Symbol(); symbol != nil {
				names = append(names, symbol.Name)
			}
		}
	}
	if len(names) == 0 {
		return "the propagated failure"
	}
	sort.Strings(names)
	return strings.Join(names, ", ")
}

// lowerPropagation rewrites `<receiver>!` into a checked early return:
//
//	const __smithersUnwrappedN = <receiver>;
//	if (!__smithersUnwrappedN.ok)
//	    return __smithersUnwrappedN;
//	… __smithersUnwrappedN.value …
//
// The hoisted statements are emitted before the statement that contained the
// call, and the narrowing on the discriminant is what makes both the propagated
// variant and the extracted value check in the lowered program. Anything that
// cannot be hoisted while preserving evaluation order is left intact and
// reported.
func (l *lowerer) lowerPropagation(node *ast.Node, receiver *ast.Node) *ast.Node {
	discriminant := "ok"
	decline := func(target *ast.Node, code string, message string) *ast.Node {
		l.report(target, code, message)
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}
	statement, safe := safeUnwrapPlacement(node)
	if safe {
		if _, hoistable := hoistableStatements[statement.Kind]; !hoistable {
			safe = false
		}
	}
	switch {
	case !l.shape.result:
		// A function that *has* an explicit contract which cannot carry the
		// propagated failure is a contract error at the declaration, which is
		// where the author has to fix it. With no contract to name at all,
		// the propagation point itself is the only honest place to report.
		if owner := enclosingFunction(node); owner != nil && owner.Type() != nil {
			if l.analysis != nil {
				if fn := l.analysis.byNode[owner]; fn != nil && fn.hasContractError {
					l.impure = true
					return l.visitor.VisitEachChild(node)
				}
			}
			failures := l.propagatedFailureNames(receiver)
			return decline(owner, "SMITHERS1101", "explicit return type cannot represent recoverable failures "+failures+"; use Result<A, E>")
		}
		return decline(node, "SMITHERS1202", "postfix ! propagation requires an enclosing Result-returning function")
	case l.guardedByCatch:
		return decline(node, "SMITHERS1205", "postfix ! propagation inside a JavaScript try statement with a catch clause is not lowered because its early return would silently bypass the catch handler; move the propagation point outside the try or consume the value explicitly")
	case isInRepeatedLoopHeader(node):
		// A loop condition, incrementor, or iteration expression is evaluated
		// once per iteration, so the early-return guard this propagation lowers
		// to cannot be hoisted in front of the loop without changing when it
		// runs. It is a distinct refusal from the general placement rule.
		return decline(node, "SMITHERS1703", "postfix ! propagation in a loop condition, incrementor, or iteration expression needs per-iteration control-flow lowering; assign before the loop or propagate inside its body")
	case !safe:
		return decline(node, "SMITHERS1204", "postfix ! in this expression would require control-flow-aware evaluation-order rewriting; assign the Result to a local and propagate it in a simple statement")
	}

	visitedReceiver := l.visitor.VisitNode(receiver)
	name := fmt.Sprintf("%s%d", tempPrefix, l.nextTemp)
	l.nextTemp++

	declaration := l.factory.NewVariableDeclaration(l.factory.NewIdentifier(name), nil, nil, visitedReceiver)
	declarationList := l.factory.NewVariableDeclarationList(l.factory.NewNodeList([]*ast.Node{declaration}), ast.NodeFlagsConst)
	binding := l.factory.NewVariableStatement(nil, declarationList)
	l.emitContext.AssignSourceMapRange(binding, node)

	condition := l.factory.NewPrefixUnaryExpression(
		ast.KindExclamationToken,
		l.factory.NewPropertyAccessExpression(l.factory.NewIdentifier(name), nil, l.factory.NewIdentifier(discriminant), ast.NodeFlagsNone),
	)
	propagated := l.factory.NewIdentifier(name)
	guard := l.factory.NewIfStatement(nil, condition, l.factory.NewReturnStatement(propagated), nil)
	l.emitContext.AssignSourceMapRange(guard, node)

	l.pending = append(l.pending, binding, guard)
	// Everything up to and including the hoisted receiver is now emitted in its
	// authored order, so a later propagation point in the same statement is
	// still order-preserving.
	l.impure = false

	value := l.factory.NewPropertyAccessExpression(l.factory.NewIdentifier(name), nil, l.factory.NewIdentifier("value"), ast.NodeFlagsNone)
	l.emitContext.AssignSourceMapRange(value, node)
	l.emitContext.SetOriginal(value, node)
	return value
}

// matchCase is one checked nominal case of an Error.match operation.
type matchCase struct {
	label   *ast.Node // the authored case label identifier
	row     errorRow  // the resolved nominal identity it selects
	handler *ast.Node // the authored handler function
	arity   int
}

// lowerMatch rewrites `error.match({ NotFound: …, Timeout: … })` into
// constructor-keyed dispatch:
//
//	error instanceof NotFound ? ((e: NotFound) => …)(error)
//	    : error instanceof Timeout ? ((e: Timeout) => …)(error)
//	        : __smithersMatchFailed(error)
//
// The case keys are resolved to the Error class bindings they name, so import
// aliases and same-named classes in different modules stay exact, and the
// dispatch keys on the constructor itself rather than on any string identity.
// Exhaustiveness is checked against the statically known error union. The
// partial form uses its explicit fallback as the final branch and deliberately
// does not require complete nominal coverage.
func (l *lowerer) lowerMatch(node *ast.Node, receiver *ast.Node, handlers *ast.Node, fallback *ast.Node, kind errorMatchKind) *ast.Node {
	decline := func(target *ast.Node, code string, message string) *ast.Node {
		l.report(target, code, message)
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}
	operation := "Error.match"
	if kind == errorMatchPartial {
		operation = "Error.matchPartial"
	}
	if handlers == nil || !ast.IsObjectLiteralExpression(handlers) {
		return decline(node, "SMITHERS1251", operation+" requires an object literal so nominal cases can be checked and lowered")
	}
	argumentCount := 0
	if arguments := node.AsCallExpression().Arguments; arguments != nil {
		argumentCount = len(arguments.Nodes)
	}
	if kind == errorMatchExhaustive && argumentCount != 1 {
		return decline(node, "SMITHERS1251", "Error.match requires exactly one object literal argument so nominal cases can be checked and lowered")
	}
	if kind == errorMatchPartial && (argumentCount != 2 || fallback == nil) {
		return decline(node, "SMITHERS1255", "Error.matchPartial requires an explicit fallback(error) callback as its second argument")
	}
	// A case whose key is not a static Error class name is refused, but the
	// remaining cases are still collected: the union member the unusable key was
	// supposed to cover is now uncovered, and saying so is the more actionable
	// half of the answer. Reporting only the first refusal would hide it.
	cases := make([]matchCase, 0, len(handlers.AsObjectLiteralExpression().Properties.Nodes))
	casesAreStatic := true
	for _, member := range handlers.AsObjectLiteralExpression().Properties.Nodes {
		if !ast.IsPropertyAssignment(member) {
			casesAreStatic = false
			continue
		}
		name := member.Name()
		initializer := member.Initializer()
		if name == nil || !ast.IsIdentifier(name) || initializer == nil ||
			(!ast.IsArrowFunction(initializer) && !ast.IsFunctionExpression(initializer)) {
			casesAreStatic = false
			continue
		}
		row, ok := l.errorRowForLabel(name, handlers)
		if !ok {
			casesAreStatic = false
			continue
		}
		cases = append(cases, matchCase{label: name, row: row, handler: initializer, arity: len(initializer.Parameters())})
	}
	if !casesAreStatic || len(cases) == 0 {
		l.report(handlers, "SMITHERS1252", "Error match cases must use static Error class names and function handlers")
		casesAreStatic = false
	}

	if kind == errorMatchExhaustive {
		expected, ok := l.errorRowsOfType(l.checker.GetTypeAtLocation(receiver))
		if !ok {
			return decline(handlers, "SMITHERS1253", "Error.match cannot be proven exhaustive: the matched value's type is not a closed union of Error classes")
		}
		if missing := missingRows(expected, cases); len(missing) != 0 {
			return decline(handlers, "SMITHERS1253", "Error.match is not exhaustive; missing "+strings.Join(missing, ", "))
		}
		if extra := extraRows(expected, cases); len(extra) != 0 {
			return decline(handlers, "SMITHERS1254", "Error.match has cases outside the checked union: "+strings.Join(extra, ", "))
		}
	}
	if !casesAreStatic {
		l.impure = true
		return l.visitor.VisitEachChild(node)
	}

	scrutinee, ok := l.matchScrutinee(receiver)
	if !ok {
		return decline(node, "SMITHERS1204", "Error.match on a computed receiver in this expression would require control-flow-aware evaluation-order rewriting; assign the error to a local first")
	}

	var lowered *ast.Node
	if kind == errorMatchPartial {
		visitedFallback := l.visitor.VisitNode(fallback)
		lowered = l.factory.NewCallExpression(
			l.factory.NewParenthesizedExpression(visitedFallback),
			nil,
			nil,
			l.factory.NewNodeList([]*ast.Node{scrutinee()}),
			ast.NodeFlagsNone,
		)
		l.emitContext.AssignSourceMapRange(lowered, fallback)
	} else {
		lowered = l.newPreludeCall("smithersMatchFailed", scrutinee())
	}
	for index := len(cases) - 1; index >= 0; index-- {
		current := cases[index]
		label := l.factory.NewIdentifier(current.label.Text())
		l.emitContext.AssignSourceMapRange(label, current.label)
		condition := l.factory.NewBinaryExpression(
			nil,
			scrutinee(),
			nil,
			l.factory.NewToken(ast.KindInstanceOfKeyword),
			label,
		)
		l.emitContext.AssignSourceMapRange(condition, current.label)

		handler := l.annotateHandlerParameter(l.visitor.VisitNode(current.handler), current.label)
		arguments := []*ast.Node{}
		if current.arity > 0 {
			arguments = append(arguments, scrutinee())
		}
		invocation := l.factory.NewCallExpression(
			l.factory.NewParenthesizedExpression(handler),
			nil,
			nil,
			l.factory.NewNodeList(arguments),
			ast.NodeFlagsNone,
		)
		l.emitContext.AssignSourceMapRange(invocation, current.handler)

		lowered = l.factory.NewConditionalExpression(
			condition,
			l.factory.NewToken(ast.KindQuestionToken),
			invocation,
			l.factory.NewToken(ast.KindColonToken),
			lowered,
		)
	}
	l.emitContext.AssignSourceMapRange(lowered, node)
	l.emitContext.SetOriginal(lowered, node)
	l.impure = true
	return lowered
}

// matchScrutinee returns a factory for fresh references to the matched value.
// An identifier receiver is referenced directly, which keeps the checker's
// `instanceof` narrowing working and needs no statement to hoist into; anything
// else is bound to a temporary, which requires a hoistable placement.
func (l *lowerer) matchScrutinee(receiver *ast.Node) (func() *ast.Node, bool) {
	if ast.IsIdentifier(receiver) {
		text := receiver.Text()
		return func() *ast.Node {
			reference := l.factory.NewIdentifier(text)
			l.emitContext.AssignSourceMapRange(reference, receiver)
			return reference
		}, true
	}
	if !l.canHoist || l.impure {
		return nil, false
	}
	visited := l.visitor.VisitNode(receiver)
	name := fmt.Sprintf("%s%d", matchPrefix, l.nextTemp)
	l.nextTemp++
	declaration := l.factory.NewVariableDeclaration(l.factory.NewIdentifier(name), nil, nil, visited)
	declarationList := l.factory.NewVariableDeclarationList(l.factory.NewNodeList([]*ast.Node{declaration}), ast.NodeFlagsConst)
	statement := l.factory.NewVariableStatement(nil, declarationList)
	l.emitContext.AssignSourceMapRange(statement, receiver)
	l.pending = append(l.pending, statement)
	l.impure = false
	return func() *ast.Node {
		reference := l.factory.NewIdentifier(name)
		l.emitContext.AssignSourceMapRange(reference, receiver)
		return reference
	}, true
}

// annotateHandlerParameter gives an unannotated single-parameter handler the
// case's nominal type, so the lowered program checks the handler body against
// the error class the case selects instead of against an implicit `any`.
func (l *lowerer) annotateHandlerParameter(handler *ast.Node, label *ast.Node) *ast.Node {
	parameters := handler.Parameters()
	if len(parameters) != 1 {
		return handler
	}
	parameter := parameters[0].AsParameterDeclaration()
	if parameter.Type != nil {
		return handler
	}
	annotation := l.factory.NewTypeReferenceNode(l.factory.NewIdentifier(label.Text()), nil)
	updated := l.factory.UpdateParameterDeclaration(
		parameter,
		parameter.Modifiers(),
		parameter.DotDotDotToken,
		parameter.Name(),
		parameter.QuestionToken,
		annotation,
		parameter.Initializer,
	)
	list := l.factory.NewNodeList([]*ast.Node{updated})
	list.Loc = handler.ParameterList().Loc
	switch handler.Kind {
	case ast.KindArrowFunction:
		arrow := handler.AsArrowFunction()
		return l.factory.UpdateArrowFunction(arrow, arrow.Modifiers(), arrow.TypeParameters, list, arrow.Type, arrow.FullSignature, arrow.EqualsGreaterThanToken, arrow.Body)
	case ast.KindFunctionExpression:
		function := handler.AsFunctionExpression()
		return l.factory.UpdateFunctionExpression(function, function.Modifiers(), function.AsteriskToken, function.Name(), function.TypeParameters, list, function.Type, function.FullSignature, function.Body)
	}
	return handler
}

// errorRowForLabel resolves a case label to the Error class binding it names.
// The label is an ordinary in-scope value binding, so it is resolved at the
// handler object's location and then unaliased; the row identity is the
// resolved class symbol, never the authored spelling.
func (l *lowerer) errorRowForLabel(label *ast.Node, location *ast.Node) (errorRow, bool) {
	symbol := l.checker.ResolveName(label.Text(), location, ast.SymbolFlagsValue, false)
	if symbol == nil {
		return errorRow{}, false
	}
	if symbol.Flags&ast.SymbolFlagsAlias != 0 {
		if aliased := l.checker.GetAliasedSymbol(symbol); aliased != nil {
			symbol = aliased
		}
	}
	symbol = l.checker.GetMergedSymbol(symbol)
	declaration := classDeclarationOf(symbol)
	if declaration == nil {
		return errorRow{}, false
	}
	return errorRow{symbol: symbol, name: symbol.Name}, true
}

func classDeclarationOf(symbol *ast.Symbol) *ast.Node {
	if symbol == nil {
		return nil
	}
	for _, declaration := range symbol.Declarations {
		if ast.IsClassDeclaration(declaration) || ast.IsClassExpression(declaration) {
			return declaration
		}
	}
	return nil
}

// errorRowsOfType is the side table of nominal error identities a matched value
// can hold. Rows are keyed by resolved class symbol; nothing is written into
// any checker type flag.
func (l *lowerer) errorRowsOfType(t *checker.Type) ([]errorRow, bool) {
	if t == nil {
		return nil, false
	}
	constituents := []*checker.Type{t}
	if t.IsUnion() {
		constituents = t.Types()
	}
	if len(constituents) == 0 {
		return nil, false
	}
	rows := make([]errorRow, 0, len(constituents))
	for _, constituent := range constituents {
		symbol := constituent.Symbol()
		if symbol == nil {
			return nil, false
		}
		symbol = l.checker.GetMergedSymbol(symbol)
		if classDeclarationOf(symbol) == nil {
			return nil, false
		}
		rows = append(rows, errorRow{symbol: symbol, name: symbol.Name})
	}
	return rows, true
}

func missingRows(expected []errorRow, cases []matchCase) []string {
	covered := make(map[*ast.Symbol]struct{}, len(cases))
	for _, item := range cases {
		covered[item.row.symbol] = struct{}{}
	}
	missing := make([]string, 0)
	for _, row := range expected {
		if _, ok := covered[row.symbol]; !ok {
			missing = append(missing, row.name)
		}
	}
	sort.Strings(missing)
	return missing
}

func extraRows(expected []errorRow, cases []matchCase) []string {
	known := make(map[*ast.Symbol]struct{}, len(expected))
	for _, row := range expected {
		known[row.symbol] = struct{}{}
	}
	extra := make([]string, 0)
	for _, item := range cases {
		if _, ok := known[item.row.symbol]; !ok {
			extra = append(extra, item.row.name)
		}
	}
	sort.Strings(extra)
	return extra
}

// printLowered runs the fork's own printer over the lowered AST, producing the
// lowered TypeScript and the authored -> lowered source map.
func printLowered(emitContext *printer.EmitContext, file *ast.SourceFile, options *core.CompilerOptions) (string, *sourcemap.RawSourceMap) {
	printerOptions := printer.PrinterOptions{
		NewLine: core.NewLineKindLF,
		Target:  options.Target,
	}
	emitPrinter := printer.NewPrinter(printerOptions, printer.PrintHandlers{}, emitContext)
	generator := sourcemap.NewGenerator(
		path.Base(file.FileName())+".ts",
		"",
		path.Dir(file.FileName()),
		tspath.ComparePathsOptions{UseCaseSensitiveFileNames: true},
	)
	writer := printer.NewTextWriter("\n", 0)
	emitPrinter.Write(file.AsNode(), file, writer, generator)
	return writer.String(), generator.RawSourceMap()
}

// adoptGeneratedMappings records the printer's authored -> lowered mappings on
// the smithers input, using the same position semantics the externally lowered path
// validates. Positions the printer left unmapped (synthesized nodes such as the
// injected prelude import) stay unmapped.
func adoptGeneratedMappings(smithers *smithersInput, raw *sourcemap.RawSourceMap) error {
	if raw == nil {
		return fmt.Errorf("the printer produced no source map")
	}
	points := make([][]mappingPoint, smithers.loweredLines.lineCount())
	mappings := sourcemap.DecodeMappings(raw.Mappings)
	for mapping := range mappings.Values() {
		if mapping.GeneratedLine >= smithers.loweredLines.lineCount() {
			return fmt.Errorf("printer mapping targets lowered line %d beyond the lowered text", mapping.GeneratedLine)
		}
		point := mappingPoint{generatedCharacter: int(mapping.GeneratedCharacter)}
		if mapping.IsSourceMapping() {
			if int(mapping.SourceIndex) >= len(raw.Sources) {
				return fmt.Errorf("printer mapping source index %d is out of range", mapping.SourceIndex)
			}
			if mapping.SourceLine >= smithers.authoredLines.lineCount() {
				return fmt.Errorf("printer mapping targets authored line %d beyond the authored text", mapping.SourceLine)
			}
			point.hasSource = true
			point.sourceLine = mapping.SourceLine
			point.sourceCharacter = int(mapping.SourceCharacter)
			if limit := smithers.authoredLines.utf16Length(point.sourceLine); point.sourceCharacter > limit {
				point.sourceCharacter = limit
			}
		}
		points[mapping.GeneratedLine] = append(points[mapping.GeneratedLine], point)
	}
	if err := mappings.Error(); err != nil {
		return fmt.Errorf("printer mappings: %w", err)
	}
	for _, row := range points {
		sort.SliceStable(row, func(i, j int) bool { return row[i].generatedCharacter < row[j].generatedCharacter })
	}
	smithers.points = points
	return nil
}

// compileInternal is the Go-native lowering pipeline. The authored `.sm` files
// are parsed and bound by the fork, the fork's checker resolves the prelude's
// Result identity and the types the lowering depends on, the fork's
// factory builds the replacement nodes, and the fork's printer emits the
// lowered TypeScript plus its authored source map. The lowered program is then
// checked and emitted through the same path externally lowered input uses, so
// every diagnostic and every emitted map lands on authored `.sm` positions.

func syntaxSpan(node *ast.Node, text string) (int, int) {
	if node == nil {
		return 0, 0
	}
	start := scanner.SkipTrivia(text, node.Pos())
	length := node.End() - start
	if length < 0 {
		length = 0
	}
	return start, length
}

// recognizeSmithersSyntaxDiagnostics gives retired source forms their migration
// diagnostic, then relabels every other parser failure in an authored .sm file
// as the general Smithers grammar mismatch. Plain TypeScript files keep native
// TypeScript diagnostics.
func recognizeSmithersSyntaxDiagnostics(project *project, program *compiler.Program, items []*ast.Diagnostic) ([]*ast.Diagnostic, []diagnostic) {
	recognized := make([]diagnostic, 0)
	suppressAt := make(map[string][]int)
	for _, file := range program.SourceFiles() {
		logical, ok := project.logicalByVirtual[file.OriginalFileName()]
		if !ok {
			logical, ok = project.logicalByVirtual[file.FileName()]
		}
		if !ok {
			continue
		}
		if _, isSmithers := project.smithersByLogical[logical]; !isSmithers {
			continue
		}
		for _, report := range legacyAssetAssertionDiagnostics(project, file, logical) {
			recognized = append(recognized, report)
			if report.Span != nil {
				suppressAt[logical] = append(suppressAt[logical], report.Span.Start)
			}
		}
		for _, hit := range sweepRetiredSyntax(file.Text()) {
			recognized = append(recognized, diagnostic{
				Code: hit.code, Category: "error", Message: hit.message,
				File: logical, Span: &span{Start: hit.start, Length: 0}, Phase: "parse",
			})
			suppressAt[logical] = append(suppressAt[logical], hit.start)
		}
	}

	remaining := make([]*ast.Diagnostic, 0, len(items))
	for _, item := range items {
		logical, ok := authoredLogicalName(project, item.File())
		if !ok {
			remaining = append(remaining, item)
			continue
		}
		if withinSuppressionWindow(suppressAt[logical], item.Pos()) {
			continue
		}
		converted := project.convertDiagnostic(item, "parse")
		converted.Code = "SMITHERS1000"
		converted.Message = grammarMismatchMessage(converted.Message)
		recognized = append(recognized, converted)
	}
	return remaining, recognized
}

// authoredLogicalName resolves the logical path of an authored `.sm` input a
// diagnostic was reported against, and reports false for every other file.
func authoredLogicalName(project *project, file *ast.SourceFile) (string, bool) {
	if file == nil {
		return "", false
	}
	logical, ok := project.logicalByVirtual[file.OriginalFileName()]
	if !ok {
		logical, ok = project.logicalByVirtual[file.FileName()]
	}
	if !ok {
		return "", false
	}
	if _, isAuthored := project.smithersByLogical[logical]; !isAuthored {
		return "", false
	}
	return logical, true
}

// withinSuppressionWindow reports whether a raw parser failure is close enough
// to a construct already reported on to be its cascade rather than an
// independent fault.
func withinSuppressionWindow(anchors []int, position int) bool {
	for _, anchor := range anchors {
		distance := position - anchor
		if distance < 0 {
			distance = -distance
		}
		if distance < suppressionWindow {
			return true
		}
	}
	return false
}

func compileInternal(ctx context.Context, options *core.CompilerOptions, project *project) (result, error) {
	configureInternalModules(options)
	// Parse the authored program once before installing compiler-issued asset
	// declarations. Asset discovery needs the fork's real recovered AST, while
	// the second Program needs those declarations present from its first module-
	// resolution pass so imported values receive their generated module types.
	discoveryProgram := newCheckProgram(options, project)
	collector := newDiagnosticCollector(project)
	rawSyntax := discoveryProgram.GetSyntacticDiagnostics(ctx, nil)
	remainingSyntax, smithersSyntax := recognizeSmithersSyntaxDiagnostics(project, discoveryProgram, rawSyntax)
	collector.add("parse", remainingSyntax)
	for _, item := range smithersSyntax {
		collector.addConverted(item)
	}
	if hasError(collector.converted) {
		return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true}, nil
	}
	assetDiagnostics, err := prepareAssetModules(project, discoveryProgram)
	if err != nil {
		return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true}, err
	}
	for _, item := range assetDiagnostics {
		collector.addConverted(item)
	}
	if hasError(collector.converted) {
		return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true}, nil
	}
	program := newCheckProgram(options, project)

	typeChecker, done := program.GetTypeChecker(ctx)
	defer done()
	prelude, err := loadPreludeSymbols(program, typeChecker)
	if err != nil {
		return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true}, err
	}

	byName := make(map[string]*ast.SourceFile, len(program.SourceFiles()))
	for _, file := range program.SourceFiles() {
		byName[file.FileName()] = file
	}
	paths := make([]string, 0, len(project.smithersByLogical))
	for logicalPath := range project.smithersByLogical {
		paths = append(paths, logicalPath)
	}
	sort.Strings(paths)
	authoredFiles := make([]*ast.SourceFile, 0, len(paths))
	for _, logicalPath := range paths {
		smithers := project.smithersByLogical[logicalPath]
		file, ok := byName[smithers.authoredName]
		if !ok {
			return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true},
				fmt.Errorf("compiler input %q is absent from the checked program", logicalPath)
		}
		authoredFiles = append(authoredFiles, file)
	}
	// The comptime frontend runs before the row analysis and the Smithers
	// lowering, over the same checked program: it must see the authored tree,
	// both later passes must know which nodes it owns, and the lowering must
	// see the values it produced. When it refused anything, no comptime value
	// is substituted at all, so a refusal can never be half applied.
	comptime := newComptimeAnalysis(program, typeChecker, authoredFiles, requestedComptimeTarget)
	analysis := newRowAnalysis(typeChecker, prelude, authoredFiles, comptime, project)
	loweringDiagnostics := make([]diagnostic, 0)
	for _, logicalPath := range paths {
		smithers := project.smithersByLogical[logicalPath]
		file, ok := byName[smithers.authoredName]
		if !ok {
			return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true},
				fmt.Errorf("compiler input %q is absent from the checked program", logicalPath)
		}
		low := newLowerer(typeChecker, prelude, analysis, comptime)
		lowered := low.transform(file)
		text, raw := printLowered(low.emitContext, lowered, options)
		smithers.loweredText = text
		smithers.loweredLines = newLineIndex(text)
		if err := adoptGeneratedMappings(smithers, raw); err != nil {
			return result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true},
				fmt.Errorf("lowering %q: %w", logicalPath, err)
		}
		project.emitFiles[smithers.loweredName] = text
		for _, item := range analysis.reports[file] {
			loweringDiagnostics = append(loweringDiagnostics, diagnostic{
				Code:     item.code,
				Category: loweringReportCategory(item),
				Message:  item.message,
				File:     smithers.logicalPath,
				Span:     &span{Start: item.start, Length: item.length},
				Phase:    "lower",
			})
		}
		for _, item := range comptime.reports[file] {
			loweringDiagnostics = append(loweringDiagnostics, diagnostic{
				Code:     item.code,
				Category: loweringReportCategory(item),
				Message:  item.message,
				File:     smithers.logicalPath,
				Span:     &span{Start: item.start, Length: item.length},
				Phase:    "lower",
			})
		}
		for _, item := range low.reports {
			loweringDiagnostics = append(loweringDiagnostics, diagnostic{
				Code:     item.code,
				Category: loweringReportCategory(item),
				Message:  item.message,
				File:     smithers.logicalPath,
				Span:     &span{Start: item.start, Length: item.length},
				Phase:    "lower",
			})
		}
	}
	// A comptime refusal in a plain TypeScript module of the same project has no
	// `.sm` lowering pass to carry it, so it is emitted here against the file it
	// was reported on. Dropping it would leave a compiler-owned import to fail
	// at load time instead of at compile time.
	foreignComptimeDiagnostics := make([]diagnostic, 0)
	for file, items := range comptime.reports {
		logical, known := project.logicalByVirtual[file.FileName()]
		if !known || project.smithersByLogical[logical] != nil {
			continue
		}
		for _, item := range items {
			foreignComptimeDiagnostics = append(foreignComptimeDiagnostics, diagnostic{
				Code:     item.code,
				Category: loweringReportCategory(item),
				Message:  item.message,
				File:     logical,
				Span:     &span{Start: item.start, Length: item.length},
				Phase:    "lower",
			})
		}
	}
	// Map iteration is unordered, so these are sorted before they join the rest.
	// A compiler whose diagnostic order changes between identical runs is not
	// reproducible, however correct each individual diagnostic is.
	sort.SliceStable(foreignComptimeDiagnostics, func(left int, right int) bool {
		if foreignComptimeDiagnostics[left].File != foreignComptimeDiagnostics[right].File {
			return foreignComptimeDiagnostics[left].File < foreignComptimeDiagnostics[right].File
		}
		return foreignComptimeDiagnostics[left].Span.Start < foreignComptimeDiagnostics[right].Span.Start
	})
	loweringDiagnostics = append(loweringDiagnostics, foreignComptimeDiagnostics...)
	if len(loweringDiagnostics) != 0 {
		collector.converted = append(collector.converted, loweringDiagnostics...)
		if options.NoEmitOnError.IsTrue() && hasError(loweringDiagnostics) {
			// A construct the lowering refused is still present in the lowered
			// text. Emitting it would ship code whose runtime meaning the
			// language does not define, so the request stops here.
			return dedupeDiagnostics(result{Diagnostics: collector.converted, Artifacts: []artifact{}, EmitSkipped: true}), nil
		}
	}

	lowered, err := compileExternal(ctx, options, project)
	if err != nil {
		return lowered, err
	}
	lowered.Diagnostics = append(collector.converted, lowered.Diagnostics...)
	lowered = dedupeDiagnostics(lowered)
	for index := range lowered.Diagnostics {
		lowered.Diagnostics[index].Message = smithersMessage(lowered.Diagnostics[index].Message)
	}
	return lowered, nil
}

func dedupeDiagnostics(value result) result {
	seen := make(map[string]struct{}, len(value.Diagnostics))
	unique := make([]diagnostic, 0, len(value.Diagnostics))
	for _, item := range value.Diagnostics {
		key := item.Code + "\x00" + item.File + "\x00" + item.Message
		if item.Span != nil {
			key += fmt.Sprintf("\x00%d\x00%d", item.Span.Start, item.Span.Length)
		}
		if _, exists := seen[key]; exists {
			continue
		}
		seen[key] = struct{}{}
		unique = append(unique, item)
	}
	value.Diagnostics = unique
	return value
}

// smithersMessage restates the most common lowered-vocabulary assignability
// failures in Smithers's own vocabulary. The lowered variant classes are an
// implementation detail of the lowering, so a message that names them is
// accurate but not the language's. Only exactly recognized shapes are
// rewritten; anything else is reported verbatim rather than approximated.
func smithersMessage(message string) string {
	const prefix = "Type '"
	const middle = "' is not assignable to type '"
	const suffix = "'."
	if !strings.HasPrefix(message, prefix) || !strings.HasSuffix(message, suffix) {
		return message
	}
	body := strings.TrimSuffix(strings.TrimPrefix(message, prefix), suffix)
	index := strings.Index(body, middle)
	if index < 0 {
		return message
	}
	source := body[:index]
	target := body[index+len(middle):]

	type rule struct {
		variant   string
		container string
		position  int
		channel   string
	}
	for _, item := range []rule{
		{variant: "SmithersOk", container: "Result", position: 0, channel: "success"},
		{variant: "SmithersErr", container: "Result", position: 1, channel: "error"},
	} {
		inner, ok := typeArgument(source, item.variant, 0)
		if !ok {
			continue
		}
		expected, ok := typeArgument(target, item.container, item.position)
		if !ok {
			continue
		}
		return fmt.Sprintf("Type '%s' is not assignable to the %s type '%s' of '%s'.", inner, item.channel, expected, target)
	}
	return message
}

// typeArgument extracts one top-level type argument from a printed type name of
// the form `Name<A, B>`, respecting nesting.
func typeArgument(text string, name string, position int) (string, bool) {
	if !strings.HasPrefix(text, name+"<") || !strings.HasSuffix(text, ">") {
		return "", false
	}
	inner := text[len(name)+1 : len(text)-1]
	depth := 0
	start := 0
	arguments := make([]string, 0, 2)
	for index := 0; index < len(inner); index++ {
		switch inner[index] {
		case '<', '(', '[', '{':
			depth++
		case '>', ')', ']', '}':
			depth--
		case ',':
			if depth == 0 {
				arguments = append(arguments, strings.TrimSpace(inner[start:index]))
				start = index + 1
			}
		}
	}
	arguments = append(arguments, strings.TrimSpace(inner[start:]))
	if position >= len(arguments) {
		return "", false
	}
	return arguments[position], true
}
