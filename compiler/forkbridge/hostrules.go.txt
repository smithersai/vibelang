package main

// Fail-closed frontend rules over authored `.sm` modules.
//
// Nothing in this file rewrites a node. Each rule decides whether a program the
// language forbids reaches emit at all, which is why they live beside the
// lowering rather than inside it: a lowering that silently accepts what the
// language rejects is worse than one that refuses honestly.
//
//	SMITHERS1601/1602/1603  host globals and host-sensitive operations
//	SMITHERS1604            dynamic code evaluation (`eval`, `Function`)
//	SMITHERS1605            determinism-hostile globals (`WeakRef`,
//	                        `FinalizationRegistry`, `SharedArrayBuffer`, `Atomics`)
//	SMITHERS1107            class `static {}` initialization blocks
//	SMITHERS1150            two Error classes with one name in one module
//	SMITHERS1504            a foreign constructor without `@throws {never}`
//	SMITHERS1509            a callback escaping into UNTRUSTED foreign code
//	SMITHERS1404            an async callback with no proven owner
//
// Every decision resolves through the checker's symbols. A spelling is never
// the authority: a lexical `const Date = …` is an ordinary value wherever it
// appears, and an aliased import of a foreign export is foreign under any name.

import (
	"sort"
	"strings"

	"github.com/microsoft/TypeScript/tsc/internal/ast"
	"github.com/microsoft/TypeScript/tsc/internal/checker"
)

const (
	hostModulesLogicalPath = "__smithers_host_modules.d.ts"
	hostModulesVirtualName = "/src/" + hostModulesLogicalPath
)

// ---------------------------------------------------------------------------
// Host globals
// ---------------------------------------------------------------------------

// universalGlobals is the ECMAScript-262 global object: the names the LANGUAGE
// publishes, as opposed to the names a HOST publishes. Authored `.sm` may reach
// these and nothing else.
//
// This is an ALLOWLIST, and the inversion is the point. The rule used to be a
// list of eight forbidden spellings, which measured 22 of 38 sibling globals
// broken: `self`/`top`/`parent`/`frames` alias `globalThis` in every DOM and
// worker host and bypassed all eight at once; `XMLHttpRequest`/`WebSocket`/
// `EventSource`/`Worker` are the network and thread authority the specification
// names in the same breath as `process`; `navigator`/`location`/`localStorage`/
// `sessionStorage` are host identity and host-persistent state. A denylist over
// a namespace the host may extend at any time cannot be completed, so it was
// never a prohibition — only a spelling table.
//
// The criterion is specification/compatibility.mdx, "Host Globals" (Locked):
// "Platform-specific globals such as process, window, document, filesystem, and
// network MUST NOT be unconditional globals in authored .sm code", and
// "Facilities truly present in every supported JavaScript environment MAY be
// unconditional globals." docs/DECISIONS.md (Locked) restates the second.
//
// Read literally, "present in every supported environment" does not decide the
// edges: `console`, `fetch`, and `setTimeout` are present in all of them and are
// nevertheless refused, and so are `URL`, `TextEncoder`, and `AbortController`,
// which no clause mentions either way. ECMA-262 is the one line through that set
// that is both principled and COMPLETABLE — a global outside it is host-defined
// by construction, which is what the HTML, WinterCG, and Node global-scope
// specifications are — and the second clause is a MAY, so refusing a
// universally present host facility is permitted where admitting a
// platform-specific one is not.
//
// `globalThis` is in ECMA-262 and is deliberately absent here: it is the one
// language global whose purpose is to hand back the host's namespace.
//
// This set is byte-for-byte the reference implementation's UNIVERSAL_GLOBALS
// (poc/src/language/semantic.ts). The two backends must answer the same
// question the same way.
var universalGlobals = map[string]struct{}{
	// 19.1 Value Properties of the Global Object (`globalThis` excluded above).
	"Infinity": {}, "NaN": {}, "undefined": {},
	// 19.2 Function Properties of the Global Object. `eval` is deliberately
	// absent; see dynamicCodeGlobals.
	"isFinite": {}, "isNaN": {}, "parseFloat": {}, "parseInt": {},
	"decodeURI": {}, "decodeURIComponent": {}, "encodeURI": {}, "encodeURIComponent": {},
	// 19.3 Constructor Properties of the Global Object.
	"AggregateError": {}, "Array": {}, "ArrayBuffer": {}, "BigInt": {}, "BigInt64Array": {}, "BigUint64Array": {},
	"Boolean": {}, "DataView": {}, "Error": {}, "EvalError": {},
	// `FinalizationRegistry` is deliberately absent; see nondeterministicGlobals.
	// `Function` is deliberately absent; see dynamicCodeGlobals.
	"Float16Array": {}, "Float32Array": {}, "Float64Array": {},
	"Int8Array": {}, "Int16Array": {}, "Int32Array": {}, "Iterator": {}, "Map": {}, "Number": {}, "Object": {},
	"Promise": {}, "Proxy": {}, "RangeError": {}, "ReferenceError": {}, "RegExp": {}, "Set": {},
	// `SharedArrayBuffer` is deliberately absent; see nondeterministicGlobals.
	"String": {}, "Symbol": {}, "SyntaxError": {}, "TypeError": {},
	"Uint8Array": {}, "Uint8ClampedArray": {}, "Uint16Array": {}, "Uint32Array": {}, "URIError": {},
	// `WeakRef` is deliberately absent; see nondeterministicGlobals. `WeakMap`
	// and `WeakSet` stay: neither exposes collection, so neither can observe
	// garbage-collection timing.
	"WeakMap": {}, "WeakSet": {},
	// Explicit Resource Management, shipped in the same clause.
	"AsyncDisposableStack": {}, "DisposableStack": {}, "SuppressedError": {},
	// 19.4 Other Properties of the Global Object. `Atomics` is deliberately
	// absent; see nondeterministicGlobals.
	"JSON": {}, "Reflect": {},
	// Annex B, normative for every web-compatible host.
	"escape": {}, "unescape": {},
	// `Date`, `Math`, `Intl`, `performance`, and `crypto` are omitted on
	// purpose: hostSensitiveGlobals judges them per operation instead, which is
	// what keeps `Math.max` available while `Math.random` needs `Random`.
	// `eval` and `Function` are omitted for the same shape of reason, and
	// dynamicCodeUse judges them per operation.
}

// dynamicCodeGlobals are the two ECMA-262 names that turn a STRING into
// executable code in the host's own scope.
//
// The comment on universalGlobals above gives the reason `globalThis` is
// excluded — "the one language global whose purpose is to hand back the host's
// namespace" — and that sentence is true of these two verbatim. Measured on this
// backend before this rule existed, with a runtime oracle: 19 spellings compiled
// with `failures: [] requirements: []`, zero diagnostics, and RAN.
// `eval("process.platform")` printed `darwin` where the direct spelling is
// SMITHERS1601; `eval("Date.now()")` printed a wall clock where the direct
// spelling is SMITHERS1602; `eval("Math.random()")` bypassed SMITHERS1603; and
// `eval("globalThis.process.platform")` showed that refusing `globalThis` by
// name is defeated by the sibling spelling. `new Function`,
// `(function(){}).constructor`, `Function.prototype.constructor`,
// `Reflect.construct(Function, …)`, `Reflect.apply(Function, …)`, an alias, the
// indirect `(0, eval)` and an ES2015 shorthand `{ eval }` each reached the host
// the same way.
//
// These names are NOT in alwaysForbiddenHostGlobals, and that is the whole
// design: the rule refuses the OPERATION, per operation, and leaves the NAME
// resolvable. `crypto` is already exactly this shape — absent from
// universalGlobals, not refused by name, every USE refused because no capability
// can represent it — so `Function` stays a usable TYPE annotation and
// `x instanceof Function` stays a prototype test. Both are pinned as
// acceptances, because without them the rule can be widened to "any mention of
// `Function`" and nothing notices.
//
// The line is the READ, not the call, because every measured escape reaches the
// callee through a read: an alias, `(0, eval)`, `Reflect.apply(Function, …)` and
// a shorthand `{ eval }` were each measured executing host code. `Date` already
// draws the line in exactly this place — `typeof Date` is SMITHERS1602 today.
//
// This is byte-for-byte the reference implementation's DYNAMIC_CODE_GLOBALS
// (poc/src/language/semantic.ts). Which of the two compatibility.mdx sentences
// governs `eval("1 + 1")` — a call that reaches no host name at all — is a
// SPECIFICATION decision recorded in the corpus rather than settled here. What
// is not open is the direction: a call that confers ambient host authority may
// not report an empty row, so this fails closed at every spelling until that
// sentence is written.
var dynamicCodeGlobals = map[string]struct{}{
	"eval":     {},
	"Function": {},
}

// nondeterministicGlobals are the four ECMA-262 names whose results are a
// function of garbage-collection timing or another agent's schedule.
//
// specification/compatibility.mdx §Determinism-Sensitive Members states two
// rows verbatim: `WeakRef`/`FinalizationRegistry` and
// `SharedArrayBuffer`/`Atomics` "MUST NOT be unconditional globals". The reason
// is the one §Host Globals gives for the whole allowlist — a Flow body
// re-executes on every resumption — except that here there is no capability to
// reach for. `deref()` returning `undefined` is a pure function of collection
// timing; a shared-memory read observes another agent's schedule. Neither is a
// value a journal entry could record and replay.
//
// Measured 2026-08-28, before this set existed: all four compiled with zero
// diagnostics and an empty row on both backends, in the same file where the
// `Date.now()` control reported SMITHERS1602. Being LISTED in universalGlobals
// made the allowlist assert the opposite of the obligation rather than merely
// fail to enforce it.
//
// They get SMITHERS1605 rather than SMITHERS1601 because 1601's message ends
// "access it through a Context capability" and these rows say the opposite in as
// many words: "no capability can mediate it and no journal entry can describe
// it". SMITHERS1604 is the precedent — same "no capability could provide this"
// argument, own code, own reason.
//
// Unlike dynamicCodeGlobals the line here is the NAME, not the operation: every
// value use of `WeakRef` is construction and every member of `Atomics` is a
// shared-memory operation, so there is no safe read to preserve. The two
// directions a by-name rule can get wrong are pinned in
// poc/src/language/host-global-allowlist.test.ts — a type position and a lexical
// shadow stay legal, and `WeakMap`, `WeakSet`, `ArrayBuffer`, and the typed
// arrays stay available.
//
// This is byte-for-byte the reference implementation's NONDETERMINISTIC_GLOBALS
// (poc/src/language/semantic.ts). The two backends must answer the same question
// the same way.
var nondeterministicGlobals = map[string]struct{}{
	"WeakRef":              {},
	"FinalizationRegistry": {},
	"SharedArrayBuffer":    {},
	"Atomics":              {},
}

// alwaysForbiddenHostGlobals are refused by NAME, whether or not the ambient
// lib declares them.
//
// This is not the rule — universalGlobals is, and it decides every name the
// ambient environment actually publishes. This list exists for the names whose
// PRESENCE varies with the toolchain, and its job is to make the verdict not
// vary with it.
//
// That matters because the two backends do not share an ambient environment.
// The reference frontend runs against the installed `@types` packages, so
// `@types/node` declares `Buffer`, `require`, `module`, `exports`, `__dirname`,
// `__filename`, `global`, and `setImmediate`; this fork carries no ambient
// `@types/node` at all, so the same names resolve to nothing here. Measured
// before this list existed: `Buffer` was `ok: true` on the reference and
// `TS2591` on the fork, and `setImmediate` was `ok: true` there and `TS2304`
// here. Naming them makes both backends answer SMITHERS1601, which is also the
// honest answer: they are the Node host's globals, and the emitted module is
// ESM, where `__dirname`, `__filename`, `module`, and `exports` do not exist at
// all and would have been a guaranteed ReferenceError inside a function whose
// row read `failures: []`.
//
// A name that is NOT here and NOT declared is left to TypeScript, which refuses
// it as TS2304 — an ordinary typo must not be reported as a host global.
//
// specification/compatibility.mdx, "Host Globals" (Locked): "Platform-specific
// globals such as process, window, document, filesystem, and network MUST NOT
// be unconditional globals in authored .sm code. Host functionality MUST be
// represented by typed capabilities."

// hostSensitiveGlobals are the globals that MAY exist in every environment but
// whose host-sensitive operations still need a capability.
//
// specification/compatibility.mdx, "Host Globals" (Locked): "Facilities truly
// present in every supported JavaScript environment MAY be unconditional
// globals. Host-sensitive operations such as clock and random access MUST still
// use capabilities." The rule is therefore per-operation, not per-object:
// `Math.max` stays available from the same object whose `.random()` does not.
//
// `Intl` is here because `new Intl.DateTimeFormat("en").format()` — no
// argument — formats *now*: a wall-clock read reached through a global the old
// rule did not model at all, and it compiled clean on both backends. The rest
// of `Intl` needs no capability *here*, which is the same per-operation shape
// `Math` already has.
//
// That is the status quo, not a design statement, and it is a recorded gap.
// compatibility.mdx §Determinism-Sensitive Members row five says every
// ICU-backed operation MUST charge a `Locale` requirement — thirty members, of
// which this file charges none — and it is open as (SA-4). Its verb turns on
// whether `Locale` is given a source-language surface; row three's verb was
// decided on 2026-08-28 (charge, landing with migration step 7) and row five's
// was not. What IS by design is that the exemption is uniform across the whole
// class, which the corpus case
// 20-host-globals/intl-locale-formatting-is-not-a-clock-read pins across both
// backends: a rule that charges the `Intl` ROOT — the cheapest wrong
// implementation of row four or five — fails there rather than silently taking
// the standard library's ICU surface with it.
var hostSensitiveGlobals = map[string]struct{}{
	"Date":        {},
	"Math":        {},
	"performance": {},
	"crypto":      {},
	"Intl":        {},
}

// ambientAuthorityUse is one host-sensitive operation reached from an ambient
// global, together with the capability that must carry it instead.
type ambientAuthorityUse struct {
	requirement string // "Clock", "Random", or "Host"
	description string
	root        *ast.Node
}

func (a *rowAnalysis) checkHostGlobals(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			// `import.meta` is host authority by this allowlist's own criterion
			// — ECMA-262 hands its properties to the host through
			// HostGetImportMetaProperties — but it is a META-PROPERTY, not an
			// identifier, so the name-keyed rule below never saw it.
			// `import.meta.url` compiled with an empty row on both backends and
			// RAN, printing the host filesystem path, and `import.meta.dirname`
			// / `import.meta.filename` compiled on the reference while this fork
			// answered TS2339 — the exact backend divergence
			// alwaysForbiddenHostGlobals exists to close for their `__dirname`
			// and `__filename` siblings. `new.target` is the other
			// meta-property and is deliberately untouched: it is the language's
			// own and reads nothing from the host. An imported `.ts` module is
			// untouched too, because this pass walks authored `.sm` files only
			// and specification/compatibility.mdx §Source Relationship keeps a
			// `.ts` module's own complete syntax and behaviour.
			if ast.IsMetaProperty(node) && node.AsMetaProperty().KeywordToken == ast.KindImportKeyword &&
				!ast.IsPartOfTypeNode(node) {
				a.report(node, "SMITHERS1601",
					"ambient host global 'import.meta' is unavailable; access it through a Context capability")
			}
			if use, dynamic := a.dynamicCodeUse(node); dynamic {
				a.report(use.node, "SMITHERS1604",
					"ambient "+use.description+" is unavailable; it runs a string as code in the host's own scope, so the enclosing row cannot describe what it reads, throws, or requires")
			}
			// Row four: a `Date` INSTANCE member that reads the host time zone.
			// The same code and the same reason as `Date.now()`; see
			// dateZoneMemberUse.
			if zoneRead, zone := a.dateZoneMemberUse(node); zone {
				a.report(zoneRead, "SMITHERS1602",
					"ambient wall-clock access is unavailable; access it through Clock.context()")
			}
			if ast.IsIdentifier(node) {
				a.reportForbiddenHostGlobal(node)
				a.reportNondeterministicHostGlobal(node)
				for _, use := range a.ambientAuthorityUses(node) {
					switch use.requirement {
					case "Clock":
						a.report(use.root, "SMITHERS1602",
							"ambient "+use.description+" is unavailable; access it through Clock.context()")
					case "Random":
						a.report(use.root, "SMITHERS1603",
							"ambient "+use.description+" is unavailable; access it through Random.context()")
					default:
						a.report(use.root, "SMITHERS1601",
							"ambient host global '"+use.root.Text()+"' is unavailable; access it through a Context capability")
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) reportForbiddenHostGlobal(node *ast.Node) {
	if isDeclarationNameIdentifier(node) || isPropertyNameIdentifier(node) || ast.IsPartOfTypeNode(node) {
		return
	}
	if !a.isForbiddenAmbientGlobal(node) {
		return
	}
	a.report(node, "SMITHERS1601",
		"ambient host global '"+node.Text()+"' is unavailable; access it through a Context capability")
}

// reportNondeterministicHostGlobal refuses the determinism-hostile names with
// their own reason; see nondeterministicGlobals. Guarded by the same three
// position tests and the same lexical-shadow test as the allowlist rule above,
// so a type annotation and a local binding survive.
func (a *rowAnalysis) reportNondeterministicHostGlobal(node *ast.Node) {
	if isDeclarationNameIdentifier(node) || isPropertyNameIdentifier(node) || ast.IsPartOfTypeNode(node) {
		return
	}
	if _, hostile := nondeterministicGlobals[node.Text()]; !hostile {
		return
	}
	if a.ambientGlobalKind(node) == ambientGlobalLocal {
		return
	}
	a.report(node, "SMITHERS1605",
		"ambient host global '"+node.Text()+"' is unavailable; its result is a function of garbage-collection timing or another agent's schedule, which no capability can mediate and no journal entry can describe")
}

// isForbiddenAmbientGlobal is the host-global prohibition, as an allowlist.
//
// Every name the ambient environment publishes is refused unless ECMA-262
// publishes it too, or unless it is one of the host-sensitive objects that
// ambientAuthorityUses judges per operation. alwaysForbiddenHostGlobals is
// refused additionally by name, so that whether a `@types` package happens to
// be installed cannot decide whether `process` is legal.
func (a *rowAnalysis) isForbiddenAmbientGlobal(node *ast.Node) bool {
	kind := a.ambientGlobalKind(node)
	if kind == ambientGlobalLocal {
		return false
	}
	if _, forbidden := alwaysForbiddenHostGlobals[node.Text()]; forbidden {
		return true
	}
	if kind != ambientGlobalDeclared {
		return false
	}
	if _, universal := universalGlobals[node.Text()]; universal {
		return false
	}
	if _, sensitive := hostSensitiveGlobals[node.Text()]; sensitive {
		return false
	}
	// Refused by reportNondeterministicHostGlobal as SMITHERS1605 with a
	// truthful reason. Without this arm they would draw that refusal AND a
	// SMITHERS1601 whose remedy cannot be satisfied, at the same position.
	if _, hostile := nondeterministicGlobals[node.Text()]; hostile {
		return false
	}
	// The dynamic-code names are judged per operation by dynamicCodeUse, exactly
	// as the host-sensitive ones are by ambientAuthorityUses. Refusing them by
	// name here as well would erase the name — and with it `x instanceof
	// Function` and `cb: Function` — which is the half of compatibility.mdx:32
	// that survives a MUST NOT.
	_, dynamic := dynamicCodeGlobals[node.Text()]
	return !dynamic
}

// dynamicCodeEvaluation is one use of the dynamic-code-evaluation authority.
type dynamicCodeEvaluation struct {
	// node is where the refusal is reported: the node that NAMES the authority.
	node        *ast.Node
	description string
}

// dynamicCodeUse reports a use of the dynamic-code-evaluation authority; see
// dynamicCodeGlobals for what is refused and why the rule is per operation.
//
// Two arms, because the authority has two routes and only one of them has a name
// to key on:
//
//  1. A VALUE read of the ambient `eval` or `Function`. This covers every
//     spelling that mentions either name — the direct call, an alias
//     (`const e: any = eval`), the indirect `(0, eval)`, `new Function`,
//     `(Function as any)`, `Function.prototype.constructor`, a shorthand
//     `{ eval }`, and `Function` as an argument to `Reflect.construct` /
//     `Reflect.apply` — because all of them READ the binding, and reading it is
//     what hands the authority on.
//
//  2. A `constructor` selection off a CALLABLE receiver. `(function () {})
//     .constructor` IS the `Function` constructor, spelled without naming it,
//     and `new ((function(){}).constructor)("return process.platform")()`
//     measured `darwin` on this backend. The receiver test is what keeps this
//     narrow, and it is the honest test rather than a name test: `({}).constructor`
//     is `ObjectConstructor`, `[].constructor` is `ArrayConstructor`, a class
//     instance's is its own class, and a string's is `StringConstructor` — none
//     of which compile a string, and all four measured clean and running. The one
//     corpus case that reads `.constructor`
//     (09-foreign-calls/a-library-declared-member-of-a-foreign-value-still-needs-an-adapter)
//     reads it off an object literal and is untouched here; it stays refused by
//     the foreign-member rule that owns it.
//
// Two exemptions, both of which `Date` already has in
// ambientRequirementsForRootUse for the same reasons: a TYPE position reads
// nothing (`function f(cb: Function)` is an annotation), and the RIGHT operand
// of `instanceof` selects a prototype and evaluates nothing. Only the right
// operand — `Function instanceof Object` puts the object itself in a value
// position, where it can still be called.
//
// A receiver typed `any` is treated as CALLABLE, which closes what used to be
// recorded here as the standing `any` hole. `Object.getPrototypeOf(fn)` is
// declared `any` in lib.es5.d.ts, so `Object.getPrototypeOf(function* () {})
// .constructor` escaped, and so did `(JSON as any).constructor` — measured
// clean on both backends with an empty row. specification/compatibility.mdx
// §Dynamic Features: "`any` remains usable, but a receiver typed `any` in a
// position where the analysis must decide callability MUST be treated as
// callable — matching the fail-closed default already applied to a dynamically
// selected member." An `any` has neither call nor construct signatures, so the
// callability test below answered "not callable" for the one receiver type
// about which nothing is known.
//
// `unknown` is deliberately not here: it must be narrowed before use, so the
// selection does not compile and TypeScript refuses the program first.
func (a *rowAnalysis) dynamicCodeUse(node *ast.Node) (dynamicCodeEvaluation, bool) {
	if node == nil {
		return dynamicCodeEvaluation{}, false
	}
	if ast.IsIdentifier(node) {
		if _, dynamic := dynamicCodeGlobals[node.Text()]; !dynamic {
			return dynamicCodeEvaluation{}, false
		}
		if isDeclarationNameIdentifier(node) || isPropertyNameIdentifier(node) || ast.IsPartOfTypeNode(node) {
			return dynamicCodeEvaluation{}, false
		}
		if !a.isAmbientGlobalReference(node) {
			return dynamicCodeEvaluation{}, false
		}
		if parent := node.Parent; parent != nil && ast.IsBinaryExpression(parent) &&
			parent.AsBinaryExpression().OperatorToken.Kind == ast.KindInstanceOfKeyword &&
			parent.AsBinaryExpression().Right == node {
			return dynamicCodeEvaluation{}, false
		}
		return dynamicCodeEvaluation{
			node:        node,
			description: "dynamic code evaluation through '" + node.Text() + "'",
		}, true
	}
	if !ast.IsPropertyAccessExpression(node) && !ast.IsElementAccessExpression(node) {
		return dynamicCodeEvaluation{}, false
	}
	if ast.IsPartOfTypeNode(node) {
		return dynamicCodeEvaluation{}, false
	}
	selection, selected := a.prelude.selectMember(node)
	if !selected || selection.text != "constructor" {
		return dynamicCodeEvaluation{}, false
	}
	receiver := a.checker.GetTypeAtLocation(selection.receiver)
	if receiver == nil {
		return dynamicCodeEvaluation{}, false
	}
	if receiver.Flags()&checker.TypeFlagsAny == 0 &&
		len(a.checker.GetSignaturesOfType(receiver, checker.SignatureKindCall)) == 0 &&
		len(a.checker.GetSignaturesOfType(receiver, checker.SignatureKindConstruct)) == 0 {
		return dynamicCodeEvaluation{}, false
	}
	return dynamicCodeEvaluation{
		node:        selection.name,
		description: "dynamic code evaluation through a callable's 'constructor', which is the Function constructor",
	}, true
}

// ambientAuthorityUses reports the capabilities an identifier's use of a
// host-sensitive global would require. An empty result means the use needs no
// capability — which is the case for every ordinary member of a universal
// object, and for every binding that is not the ambient global at all.
func (a *rowAnalysis) ambientAuthorityUses(node *ast.Node) []ambientAuthorityUse {
	if _, sensitive := hostSensitiveGlobals[node.Text()]; !sensitive {
		return nil
	}
	if isDeclarationNameIdentifier(node) || isPropertyNameIdentifier(node) || ast.IsPartOfTypeNode(node) {
		return nil
	}
	if !a.isAmbientGlobalReference(node) {
		return nil
	}
	requirements := a.ambientRequirementsForRootUse(node)
	uses := make([]ambientAuthorityUse, 0, len(requirements))
	for _, requirement := range requirements {
		description := "randomness"
		if requirement == "Clock" {
			description = "wall-clock access"
			if node.Text() == "performance" {
				description = "monotonic-clock access"
			}
		}
		uses = append(uses, ambientAuthorityUse{requirement: requirement, description: description, root: node})
	}
	return uses
}

// ambientRequirementsForRootUse decides which members of the global the use
// reaches. A use that selects one member is judged on that member; a use whose
// selection cannot be seen statically escapes the whole object and is judged on
// the object.
func (a *rowAnalysis) ambientRequirementsForRootUse(root *ast.Node) []string {
	parent := root.Parent
	if parent == nil {
		return ambientRequirementsForWholeRoot(root.Text())
	}
	// `value instanceof Date` is a prototype test, not a host-sensitive
	// operation. specification/compatibility.mdx, "Host Globals" (Locked):
	// "Facilities truly present in every supported JavaScript environment MAY be
	// unconditional globals. Host-sensitive *operations* such as clock and
	// random access MUST still use capabilities." This function is already
	// per-operation rather than per-object — `Date.parse` and `Date.UTC` are
	// pure functions of their arguments, and `new Date(fixedInstant)` reads no
	// clock — and the right operand of `instanceof` is the same kind of use: it
	// selects a prototype and reads no host state. Only the RIGHT operand is
	// exempt; `Date instanceof Function` puts the object itself in an ordinary
	// value position where it can still escape, so that spelling keeps its
	// requirement.
	if ast.IsBinaryExpression(parent) &&
		parent.AsBinaryExpression().OperatorToken.Kind == ast.KindInstanceOfKeyword &&
		parent.AsBinaryExpression().Right == root {
		return nil
	}
	if (ast.IsPropertyAccessExpression(parent) || ast.IsElementAccessExpression(parent)) &&
		parent.Expression() == root {
		// The member is read through the SHARED selection helper (selectMember),
		// not a second literal test of this walk's own. The two answers had
		// already drifted, in the harmless-looking direction: an unresolved key
		// fell to the whole-root arm and charged everything, which is the right
		// verdict for `Date[NOW]` and an OVER-REFUSAL for `Date[PARSE]` and
		// `Math[MAX]`, which need no capability and were refused anyway — measured
		// on this backend, against a reference that runs both and prints
		// `1577836800000` and `2`. One helper makes the per-operation rule as
		// precise at an aliased, parenthesised or `satisfies`-wrapped key as it is
		// at the dotted one, in BOTH directions.
		selection, selected := a.prelude.selectMember(parent)
		if !selected {
			return ambientRequirementsForWholeRoot(root.Text())
		}
		return ambientRequirementsForMembers(root.Text(), []string{selection.text})
	}
	if root.Text() == "Date" {
		if ast.IsCallExpression(parent) && parent.AsCallExpression().Expression == root {
			return []string{"Clock"}
		}
		if ast.IsNewExpression(parent) && parent.AsNewExpression().Expression == root {
			// `new Date(fixedInstant)` reads no clock: the instant is authored.
			// The exemption is about the RUNTIME arity, though, and argument
			// NODES are not it: `new Date(...[])` is one syntactic argument and
			// zero actual ones, so it constructs the current time — measured,
			// clean, and executing, on both backends. A spread has no statically
			// known length, so it cannot prove an instant was supplied and does
			// not earn the exemption.
			arguments := parent.AsNewExpression().Arguments
			if arguments != nil && len(arguments.Nodes) > 0 && !containsSpreadArgument(arguments.Nodes) {
				return nil
			}
			return []string{"Clock"}
		}
	}
	if ast.IsVariableDeclaration(parent) && parent.AsVariableDeclaration().Initializer == root {
		name := parent.AsVariableDeclaration().Name()
		if name != nil && ast.IsObjectBindingPattern(name) {
			if members, known := a.bindingMemberNames(name); known {
				return ambientRequirementsForMembers(root.Text(), members)
			}
			return ambientRequirementsForWholeRoot(root.Text())
		}
	}
	return ambientRequirementsForWholeRoot(root.Text())
}

// containsSpreadArgument reports whether any argument node is a spread, whose
// runtime length the analysis cannot see.
func containsSpreadArgument(arguments []*ast.Node) bool {
	for _, argument := range arguments {
		if ast.IsSpreadElement(argument) {
			return true
		}
	}
	return false
}

// ambientRequirementsForWholeRoot judges a use where the whole object escaped,
// so any of its members could be reached.
func ambientRequirementsForWholeRoot(root string) []string {
	switch root {
	case "Date", "performance", "Intl":
		return []string{"Clock"}
	case "Math":
		return []string{"Random"}
	case "crypto":
		return []string{"Host"}
	}
	return nil
}

func ambientRequirementsForMembers(root string, members []string) []string {
	seen := make(map[string]bool, 2)
	for _, member := range members {
		switch {
		case root == "Date" && member == "now":
			seen["Clock"] = true
		case root == "Date" && member != "parse" && member != "UTC":
			// Every other `Date` member reaches an instance whose reads are
			// clock-shaped; `parse` and `UTC` are pure functions of arguments.
			seen["Clock"] = true
		case root == "Math" && member == "random":
			seen["Random"] = true
		case root == "performance":
			seen["Clock"] = true
		case root == "crypto" && (member == "randomUUID" || member == "getRandomValues"):
			seen["Random"] = true
		case root == "crypto":
			seen["Host"] = true
		case root == "Intl" && member == "DateTimeFormat":
			// Charged at CONSTRUCTION, not at the `format()` that reads the
			// clock. Which calls read it depends on the arity of a call on an
			// instance the analysis would have to track — `format()` formats
			// now, `format(instant)` does not — and `resolvedOptions().timeZone`
			// reads the host zone with no call at all. Charging the constructor
			// fails closed and costs only the formatter, not the rest of `Intl`.
			seen["Clock"] = true
		}
	}
	requirements := make([]string, 0, len(seen))
	for requirement := range seen {
		requirements = append(requirements, requirement)
	}
	// Byte order is CORRECT here and is left alone deliberately: `seen` is keyed
	// by this file's own closed capability vocabulary (`Clock`, `Host`, …), not
	// by anything the author spells, so it is provably ASCII.
	sort.Strings(requirements)
	return requirements
}

// bindingMemberNames lists the members an object binding pattern selects.
// `known` is false when a rest element or a computed name means the selection
// cannot be read statically, in which case the whole object escapes.
func (a *rowAnalysis) bindingMemberNames(pattern *ast.Node) (names []string, known bool) {
	for _, element := range pattern.AsBindingPattern().Elements.Nodes {
		binding := element.AsBindingElement()
		if binding.DotDotDotToken != nil {
			return nil, false
		}
		name := binding.PropertyName
		if name == nil {
			name = binding.Name()
		}
		if name == nil {
			return nil, false
		}
		switch {
		case ast.IsIdentifier(name), ast.IsStringLiteralLike(name), name.Kind == ast.KindNumericLiteral:
			names = append(names, name.Text())
		case ast.IsComputedPropertyName(name):
			// The same shared answer selectMember gives: `const { [NOW]: now } =
			// Date` for `const NOW = "now"` selects `now`, so it is charged as
			// `now` rather than falling to the whole-root arm — and
			// `const { [PARSE]: parse } = Date` is charged as `parse`, which needs
			// no capability and was over-refused before.
			key, static := a.prelude.staticKeyName(name.AsComputedPropertyName().Expression)
			if !static {
				return nil, false
			}
			names = append(names, key)
		default:
			return nil, false
		}
	}
	return names, true
}

// referencedValueSymbol answers the symbol an identifier READS, which is not
// always the symbol it resolves to.
//
// The name of an ES2015 shorthand property assignment declares a property and
// reads a value with one token, and GetSymbolAtLocation answers the *property*
// — a symbol whose only declaration is the ShorthandPropertyAssignment itself.
// Every provenance walk that follows a symbol to its declaration therefore
// dead-ends on `{ handler }` while following `{ handler: handler }` all the way
// to the foreign module it came from, and the dead end is FAIL-OPEN: the walk
// answers "not foreign" rather than "unknown". GetShorthandAssignmentValueSymbol
// exists for exactly this, and the host-global rule below has always used it;
// the value-provenance walks did not, which is what let a foreign callable
// through `Object.freeze({ handler })` on the reference.
func (a *rowAnalysis) referencedValueSymbol(node *ast.Node) *ast.Symbol {
	if node.Parent != nil && ast.IsShorthandPropertyAssignment(node.Parent) &&
		node.Parent.Name() == node {
		return a.checker.GetShorthandAssignmentValueSymbol(node.Parent)
	}
	return a.checker.GetSymbolAtLocation(node)
}

// ambientGlobalClass is how the ambient environment answers one identifier.
type ambientGlobalClass int

const (
	// ambientGlobalLocal: the program itself declares the name, or it is an
	// import alias. A lexical `const Date = …` is an ordinary value under any
	// spelling, and a re-exported binding that happens to be spelled `Math` is
	// not the host global either.
	ambientGlobalLocal ambientGlobalClass = iota
	// ambientGlobalDeclared: every declaration lives in a declaration file, so
	// the ambient lib publishes the name. THIS namespace is closed and
	// enumerable, which is what lets an allowlist over it be total where a
	// denylist over "whatever the host publishes" never can be.
	ambientGlobalDeclared
	// ambientGlobalUndeclared: the name resolves to nothing. TypeScript refuses
	// the program on its own here (TS2304/TS2591), so the language adds a
	// refusal only for alwaysForbiddenHostGlobals; treating every unresolved
	// identifier as a host global would answer an ordinary typo with "ambient
	// host global 'lenght' is unavailable" and preempt the honest TypeScript
	// diagnostic that names the real problem. A symbol with no declarations at
	// all (checker-synthesized bindings such as `arguments`) lands here for the
	// same reason: there is no declaration file to attribute it to.
	ambientGlobalUndeclared
)

// ambientGlobalKind decides identity, not spelling.
func (a *rowAnalysis) ambientGlobalKind(node *ast.Node) ambientGlobalClass {
	symbol := a.referencedValueSymbol(node)
	if symbol == nil {
		return ambientGlobalUndeclared
	}
	if symbol.Flags&ast.SymbolFlagsAlias != 0 {
		return ambientGlobalLocal
	}
	if len(symbol.Declarations) == 0 {
		return ambientGlobalUndeclared
	}
	for _, declaration := range symbol.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file == nil || !file.IsDeclarationFile {
			return ambientGlobalLocal
		}
	}
	return ambientGlobalDeclared
}

// ---------------------------------------------------------------------------
// Determinism-sensitive members: the rows that CHARGE, and the row that refuses
// ---------------------------------------------------------------------------

// dateZoneMembers are the `Date` INSTANCE members whose result is a function of
// the host time zone.
//
// specification/compatibility.mdx §Determinism-Sensitive Members row four:
// these "MUST charge `Clock` even when an explicit instant is supplied, because
// they read the host time zone. `getTime()` stays free."
//
// WIDER than the seven the row names, by the row's own criterion and by the
// precedent row five set when its four names measured twenty-six short. Every
// other LOCAL getter reads the zone for the identical reason. Absent
// deliberately: `getTime`, `valueOf`, `toISOString`, `toJSON`, `toUTCString`
// and every `getUTC*` member are absolute or UTC-anchored; `getMilliseconds` is
// zone-independent; and the SETTERS are writes rather than reads.
var dateZoneMembers = map[string]struct{}{
	// The seven the specification names.
	"getHours": {}, "getDay": {}, "getTimezoneOffset": {},
	"toLocaleString": {}, "toLocaleDateString": {}, "toLocaleTimeString": {}, "toString": {},
	// The seven that carry the identical hazard for the identical reason.
	"getFullYear": {}, "getMonth": {}, "getDate": {}, "getMinutes": {}, "getSeconds": {},
	"toDateString": {}, "toTimeString": {},
}

// dateZoneMemberUse reports a read of a `Date` INSTANCE member that reads the
// host time zone, and the node the refusal is positioned at.
//
// This is the type-directed analysis row four was blocked on. `Date` was
// analyzed at the ROOT IDENTIFIER only, so `new Date(instant)` returned an
// empty row — correctly, the instant is authored — and no instance member was
// ever inspected after it. Measured before this rule existed:
// `new Date(0).getTimezoneOffset()` compiled with zero diagnostics and an empty
// row on both backends, in the same file where the `Date.now()` control
// correctly reported SMITHERS1602.
//
// Reported as SMITHERS1602 and not as a new code, because it is the same
// refusal for the same reason as `Date.now()`.
//
// Keyed on the RECEIVER'S TYPE and not on the member name, which is the
// opposite of row five's key and deliberately so: `toString` and
// `toLocaleString` are members of nearly every value in the language, and a
// name-keyed rule here would take the standard library with it.
func (a *rowAnalysis) dateZoneMemberUse(node *ast.Node) (*ast.Node, bool) {
	if node == nil {
		return nil, false
	}
	if !ast.IsPropertyAccessExpression(node) && !ast.IsElementAccessExpression(node) {
		return nil, false
	}
	if ast.IsPartOfTypeNode(node) {
		return nil, false
	}
	selection, selected := a.prelude.selectMember(node)
	if !selected {
		return nil, false
	}
	if _, zone := dateZoneMembers[selection.text]; !zone {
		return nil, false
	}
	// The ROOT spelling `Date.toString` is the namespace object's own member and
	// is already judged by ambientRequirementsForMembers; this rule is about an
	// instance.
	if a.isAmbientNamespaceReceiver(selection.receiver) {
		return nil, false
	}
	if !a.isAmbientDateReceiver(selection.receiver) {
		return nil, false
	}
	return selection.name, true
}

// isAmbientNamespaceReceiver reports whether a receiver is one of the ambient
// namespace objects the per-operation rules already own, so the member rules
// above do not charge a row beside a refusal.
func (a *rowAnalysis) isAmbientNamespaceReceiver(receiver *ast.Node) bool {
	if receiver == nil || !ast.IsIdentifier(receiver) {
		return false
	}
	if _, sensitive := hostSensitiveGlobals[receiver.Text()]; !sensitive {
		return false
	}
	return a.isAmbientGlobalReference(receiver)
}

// isAmbientDateReceiver reports whether an expression's TYPE is the TypeScript
// library's own `Date`, resolved through the declaring file rather than through
// the name the author spelled.
func (a *rowAnalysis) isAmbientDateReceiver(receiver *ast.Node) bool {
	if receiver == nil {
		return false
	}
	t := a.checker.GetTypeAtLocation(receiver)
	if t == nil {
		return false
	}
	symbol := t.Symbol()
	if symbol == nil || symbol.Name != "Date" {
		return false
	}
	for _, declaration := range symbol.Declarations {
		file := ast.GetSourceFileOfNode(declaration)
		if file != nil && file.IsDeclarationFile {
			return true
		}
	}
	return false
}

// isAmbientGlobalReference reports whether the identifier reads the ambient
// environment rather than a binding the program declares.
func (a *rowAnalysis) isAmbientGlobalReference(node *ast.Node) bool {
	return a.ambientGlobalKind(node) != ambientGlobalLocal
}

// isDeclarationNameIdentifier answers whether this identifier only NAMES
// something, so it reads no value.
//
// The ES2015 shorthand property is the one form that does both with one token:
// `{ process }` is the Name of its ShorthandPropertyAssignment *and* a
// reference to `process`. Treating it as a declaration name skips the very
// check the reference needs, so `Object.freeze({ process })` slipped past
// SMITHERS1601 while `Object.freeze({ process: process })` was refused.
// ambientAuthorityUses already carved the shorthand back out for the
// Date/Math/performance/crypto rule; the carve-out belongs here, where both
// callers get it and neither can drift from the other.
func isDeclarationNameIdentifier(node *ast.Node) bool {
	if node.Parent == nil {
		return false
	}
	if ast.IsShorthandPropertyAssignment(node.Parent) && node.Parent.Name() == node {
		return false
	}
	return ast.IsDeclarationName(node) || ast.IsDeclarationNameOrImportPropertyName(node)
}

func isPropertyNameIdentifier(node *ast.Node) bool {
	parent := node.Parent
	if parent == nil {
		return false
	}
	if ast.IsPropertyAccessExpression(parent) && parent.AsPropertyAccessExpression().Name() == node {
		return true
	}
	switch parent.Kind {
	case ast.KindPropertyAssignment, ast.KindMethodDeclaration, ast.KindPropertyDeclaration:
		return parent.Name() == node
	}
	return false
}

// ---------------------------------------------------------------------------
// Class `static {}` initialization blocks
// ---------------------------------------------------------------------------

// checkStaticBlocks rejects a class static initialization block.
//
// specification/type-system.mdx, "Function Type": a Smithers function has a
// return type and an inferred requirement set; specification/failures.mdx,
// "Compiler Lifting", defines lifting only inside functions. A static block
// runs at module evaluation with no enclosing function channel to carry a
// Result or a requirement row — the same reason a top-level throw is rejected.
func (a *rowAnalysis) checkStaticBlocks(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if node.Kind == ast.KindClassStaticBlockDeclaration {
				a.report(node, "SMITHERS1107",
					"class `static {}` initialization blocks execute outside every checked function channel and are not analyzed or lowered; use static field initializers or an explicit checked function")
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

// ---------------------------------------------------------------------------
// Duplicate Error class names
// ---------------------------------------------------------------------------

// checkDuplicateErrorNames rejects two Error classes that share one name inside
// one module.
//
// specification/failures.mdx, "Error Classes", requires the compiler to provide
// "stable nominal identity, matching metadata, serialization evidence, and
// cross-realm transport metadata". Row identity is module-qualified, so the
// same name in two different modules is fine (and stays fine — the scan is
// per file); two in one module cannot both hold the same stable id.
func (a *rowAnalysis) checkDuplicateErrorNames(files []*ast.SourceFile) {
	for _, file := range files {
		seen := make(map[string]bool)
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if ast.IsClassDeclaration(node) && node.Name() != nil {
				if a.isNominalErrorClass(node) {
					name := node.Name().Text()
					if seen[name] {
						a.report(node.Name(), "SMITHERS1150",
							"duplicate Error class name '"+name+"' cannot receive a stable module-local identity")
					} else {
						seen[name] = true
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

// ---------------------------------------------------------------------------
// Foreign boundaries: constructors and escaping callbacks
// ---------------------------------------------------------------------------

// checkForeignProvenanceEscapes ports the reference's SMITHERS1508 boundary.
// A callable/object originating in TypeScript can carry hidden execution and
// panic behavior. Passing it through an unchecked higher-order call, retaining
// it in mutable storage, or returning it would lose that provenance entirely.
func (a *rowAnalysis) checkForeignProvenanceEscapes(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node, *functionRows) bool
		visit = func(node *ast.Node, owner *functionRows) bool {
			if fn := a.byNode[node]; fn != nil {
				owner = fn
			}
			if ast.IsCallExpression(node) {
				policy, foreign := a.foreignPolicyForCall(node)
				_, unwrapKind := a.prelude.unwrapCall(node)
				// An UNTRUSTED foreign call reports SMITHERS1509 for every
				// callable argument and owns the position. A TRUSTED one
				// (`@throws {never}` / `@throws {T}`) no longer does, so this
				// rule takes the position back for the case its own charter
				// covers: foreign callable PROVENANCE crossing the call. The
				// trust claim is about this callee, not about the panic
				// behaviour of a callable minted somewhere else.
				if (!foreign || policy.kind != "panic") && unwrapKind == unwrapNone && !a.prelude.isPanicCall(node) {
					arguments := node.AsCallExpression().Arguments
					if arguments != nil {
						for _, argument := range arguments.Nodes {
							if a.containsForeignExecutableValue(argument, make(map[*ast.Symbol]bool)) {
								a.reportForeignProvenanceEscape(owner, argument,
									"foreign callable provenance would escape through an unchecked higher-order call; wrap it in a local adapter that owns invocation and exposes an explicit Result/task contract")
							}
						}
					}
				}
			}
			a.checkImplicitForeignInvocation(owner, node)
			if ast.IsNewExpression(node) {
				// An AUTHORED class whose base chain reaches a foreign class.
				// `new X()` runs that foreign constructor through an implicit
				// `super(...)` that no call expression names, so
				// checkForeignConstruction — which asks about the constructed
				// expression's own provenance — never saw it.
				//
				// The rule is on the CONSTRUCTION and not on the `extends`
				// clause: a clause runs no constructor, it evaluates the base
				// expression and sets a prototype. Refusing the clause was
				// measurably wrong —
				// 17-durable/the-retired-vibelang-flows-specifier-is-not-compiler-owned
				// declares a subclass of an unresolvable foreign `Action`, never
				// constructs it, and declares the module edge as its whole
				// diagnostic set.
				//
				// It is reported from THIS walk rather than from
				// checkForeignBoundaries because this one carries the owner, so
				// the caller's panic row is charged the way the reference charges
				// it. checkForeignBoundaries has no owner, which is the standing
				// cause of the SMITHERS1509 row divergence recorded in
				// 09-foreign-calls/a-callback-handed-to-an-untrusted-host-is-still-rejected.
				constructed := node.AsNewExpression().Expression
				if !a.foreignValueProvenance(constructed, make(map[*ast.Symbol]bool)) &&
					a.foreignBaseClass(constructed, make(map[*ast.Symbol]bool)) {
					a.report(node, "SMITHERS1504",
						"constructing this class runs a FOREIGN base constructor through an implicit super(...) that no call expression names, and constructor Result lowering is deferred; wrap the base in an annotated factory or Smithers adapter and hold it as a field")
					if owner != nil {
						addRow(owner.directFailures, a.row(a.prelude.panic))
					}
				}
			}
			if ast.IsTaggedTemplateExpression(node) {
				a.checkForeignTemplateTag(owner, node)
			}
			if node.Kind == ast.KindDecorator {
				applied := node.AsDecorator().Expression
				if applied != nil && ast.IsCallExpression(applied) {
					applied = applied.AsCallExpression().Expression
				}
				if a.foreignValueProvenance(applied, make(map[*ast.Symbol]bool)) {
					a.report(node, "SMITHERS1504",
						"a foreign decorator is invoked when the declaration is evaluated, with no call expression to lower; apply it inside an annotated adapter instead")
					if owner != nil {
						addRow(owner.directFailures, a.row(a.prelude.panic))
					}
				}
			}
			if ast.IsVariableDeclaration(node) {
				declaration := node.AsVariableDeclaration()
				list := node.Parent
				if declaration.Initializer != nil && list != nil && ast.IsVariableDeclarationList(list) &&
					list.Flags&ast.NodeFlagsConst == 0 &&
					a.containsForeignExecutableValue(declaration.Initializer, make(map[*ast.Symbol]bool)) {
					a.reportForeignProvenanceEscape(owner, declaration.Initializer,
						"a mutable alias cannot retain foreign panic provenance in this POC; use a const local adapter with an explicit Result contract")
				}
			}
			if ast.IsReturnStatement(node) {
				expression := node.AsReturnStatement().Expression
				if expression != nil && !ast.IsNewExpression(expression) &&
					a.containsForeignExecutableValue(expression, make(map[*ast.Symbol]bool)) {
					a.reportForeignProvenanceEscape(owner, expression,
						"returning an executable foreign value would lose its panic provenance; return a Smithers-owned adapter with an explicit Result/task contract instead")
				}
			}
			if ast.IsBinaryExpression(node) {
				binary := node.AsBinaryExpression()
				if binary.OperatorToken.Kind == ast.KindEqualsToken &&
					a.containsForeignExecutableValue(binary.Right, make(map[*ast.Symbol]bool)) {
					a.reportForeignProvenanceEscape(owner, binary.Right,
						"storing a foreign callable through a mutable/opaque reference loses panic provenance; use an immutable local adapter with an explicit Result contract")
				}
			}
			node.ForEachChild(func(child *ast.Node) bool { return visit(child, owner) })
			return false
		}
		file.AsNode().ForEachChild(func(node *ast.Node) bool { return visit(node, nil) })
	}
}

// The three answers implicitInvocationProtocol can give.
//
// They are constants and not bare literals because BOTH halves of this rule
// consume them: checkImplicitForeignInvocation prints the answer in its
// diagnostic, and implicitInvocations (lowering.go) branches on it to pick the
// member an AUTHORED value runs at that position. A literal in one half and a
// literal in the other is exactly the drift this file's doc comment exists to
// prevent.
const (
	iterationProtocol   = "Symbol.iterator / Symbol.asyncIterator"
	enumerationProtocol = "its own enumerable getters"
	coercionProtocol    = "Symbol.toPrimitive / valueOf / toString"
)

// implicitInvocationProtocol answers ONE question about an expression's
// POSITION: does evaluating it here invoke an arbitrary method on the value?
//
// Every form below is a method call with no call expression to see. `for…of`,
// spread and `yield*` run `value[Symbol.iterator]()`; object spread and
// destructuring read every own enumerable property, running getters; template
// interpolation, arithmetic, loose comparison and a computed key run
// `Symbol.toPrimitive`/`valueOf`/`toString`. The panic-channel machinery on both
// backends is driven from call expressions and property reads, so all of them
// were unmodelled and a foreign value that throws produced a raw host Error out
// of a function whose row read `failures: []`.
//
// It is one predicate and not a list of call sites on purpose: each new sibling
// (`for await…of`, `yield*`, a spread into a call's arguments, a compound
// assignment, a computed property key) would otherwise be a separate edit at a
// separate site, and the ones nobody thought of stay fail-open silently. Asking
// the position a single question makes the answer total over the grammar.
//
// This mirrors implicitInvocationProtocol in poc/src/language/semantic.ts byte
// for byte in behaviour; the two must agree or the backends diverge.
//
// The answer is one of the three named constants below, never a bare literal:
// implicitInvocations in lowering.go branches on it to decide WHICH checked
// function the position runs, so the answer has to be comparable and not only
// printable.
func implicitInvocationProtocol(node *ast.Node) string {
	parent := node.Parent
	if parent == nil {
		return ""
	}
	switch parent.Kind {
	// --- iteration protocol ------------------------------------------------
	// `for (const x of E)` and `for await (const x of E)` are one node kind.
	case ast.KindForOfStatement:
		if parent.AsForInOrOfStatement().Expression == node {
			return iterationProtocol
		}
	// `[...E]`, `f(...E)`, `new C(...E)` — one node kind for all three.
	case ast.KindSpreadElement:
		if parent.AsSpreadElement().Expression == node {
			return iterationProtocol
		}
	case ast.KindYieldExpression:
		yielded := parent.AsYieldExpression()
		if yielded.AsteriskToken != nil && yielded.Expression == node {
			return iterationProtocol
		}
	// --- enumeration protocol ----------------------------------------------
	case ast.KindSpreadAssignment:
		if parent.AsSpreadAssignment().Expression == node {
			return enumerationProtocol
		}
	// --- coercion protocol -------------------------------------------------
	case ast.KindTemplateSpan:
		// A TAGGED template does not coerce its substitutions: they are handed
		// to the tag untouched. The tag itself is a call, refused separately.
		if template := parent.Parent; template != nil && template.Parent != nil &&
			ast.IsTaggedTemplateExpression(template.Parent) {
			return ""
		}
		return coercionProtocol
	case ast.KindPrefixUnaryExpression:
		if coercingPrefixOperators[parent.AsPrefixUnaryExpression().Operator] {
			return coercionProtocol
		}
	case ast.KindPostfixUnaryExpression:
		return coercionProtocol
	case ast.KindComputedPropertyName:
		// `{ [E]: v }` runs ToPropertyKey on E.
		return coercionProtocol
	case ast.KindElementAccessExpression:
		// `o[E]` runs ToPropertyKey on E; `E[k]` is the receiver and belongs to
		// the neighbouring property-read rule.
		if parent.AsElementAccessExpression().ArgumentExpression == node {
			return coercionProtocol
		}
	case ast.KindBinaryExpression:
		binary := parent.AsBinaryExpression()
		switch binary.OperatorToken.Kind {
		case ast.KindEqualsToken:
			// `[a] = E` / `({a} = E)` — destructuring ASSIGNMENT. The
			// declaration forms are reported at their binding pattern by the
			// neighbouring rule, so they are not claimed twice here.
			if binary.Right == node && ast.IsArrayLiteralExpression(binary.Left) {
				return iterationProtocol
			}
			if binary.Right == node && ast.IsObjectLiteralExpression(binary.Left) {
				return enumerationProtocol
			}
		case ast.KindInstanceOfKeyword:
			// `x instanceof E` runs `E[Symbol.hasInstance]`.
			if binary.Right == node {
				return coercionProtocol
			}
		case ast.KindInKeyword:
			// `E in o` runs ToPropertyKey on E.
			if binary.Left == node {
				return coercionProtocol
			}
		default:
			if coercingBinaryOperators[binary.OperatorToken.Kind] {
				return coercionProtocol
			}
		}
	}
	return ""
}

// coercingBinaryOperators omits `===`/`!==` because they never coerce, and
// `&&`/`||`/`??`/`,` because ToBoolean and sequencing run no user code.
var coercingBinaryOperators = map[ast.Kind]bool{
	ast.KindEqualsEqualsToken:                            true,
	ast.KindExclamationEqualsToken:                       true,
	ast.KindPlusToken:                                    true,
	ast.KindMinusToken:                                   true,
	ast.KindAsteriskToken:                                true,
	ast.KindAsteriskAsteriskToken:                        true,
	ast.KindSlashToken:                                   true,
	ast.KindPercentToken:                                 true,
	ast.KindLessThanToken:                                true,
	ast.KindGreaterThanToken:                             true,
	ast.KindLessThanEqualsToken:                          true,
	ast.KindGreaterThanEqualsToken:                       true,
	ast.KindAmpersandToken:                               true,
	ast.KindBarToken:                                     true,
	ast.KindCaretToken:                                   true,
	ast.KindLessThanLessThanToken:                        true,
	ast.KindGreaterThanGreaterThanToken:                  true,
	ast.KindGreaterThanGreaterThanGreaterThanToken:       true,
	ast.KindPlusEqualsToken:                              true,
	ast.KindMinusEqualsToken:                             true,
	ast.KindAsteriskEqualsToken:                          true,
	ast.KindAsteriskAsteriskEqualsToken:                  true,
	ast.KindSlashEqualsToken:                             true,
	ast.KindPercentEqualsToken:                           true,
	ast.KindAmpersandEqualsToken:                         true,
	ast.KindBarEqualsToken:                               true,
	ast.KindCaretEqualsToken:                             true,
	ast.KindLessThanLessThanEqualsToken:                  true,
	ast.KindGreaterThanGreaterThanEqualsToken:            true,
	ast.KindGreaterThanGreaterThanGreaterThanEqualsToken: true,
}

var coercingPrefixOperators = map[ast.Kind]bool{
	ast.KindPlusToken:       true,
	ast.KindMinusToken:      true,
	ast.KindTildeToken:      true,
	ast.KindPlusPlusToken:   true,
	ast.KindMinusMinusToken: true,
}

// checkImplicitForeignInvocation is the shared reporter for
// implicitInvocationProtocol. Two gates keep it from widening past the defect:
// the value must have FOREIGN provenance, and it must be able to CARRY a foreign
// member — foreignTypeCanExecute is false for a primitive, so a trusted
// binding's `string` may still be interpolated, iterated and spread, because
// String.prototype's protocol members are the language's, not the module's. An
// ESM namespace object is excluded for the same reason the property-read rule
// excludes it: its members are live-binding selection, not user accessors.
func (a *rowAnalysis) checkImplicitForeignInvocation(owner *functionRows, node *ast.Node) {
	protocol := implicitInvocationProtocol(node)
	if protocol == "" {
		return
	}
	if a.namespaceObjectReference(node) {
		return
	}
	if !a.foreignValueProvenance(node, make(map[*ast.Symbol]bool)) {
		return
	}
	// The can-execute gate reads the type of the value ITSELF, with parentheses
	// and type assertions stripped. `(foreign as unknown as number) == 1` still
	// runs `foreign.valueOf()` at runtime, so asking the cast's type would let an
	// author launder the gate with the same `as` that provenance already refuses
	// to follow. `!` is deliberately not stripped: in `.sm` it is the checked
	// propagation boundary and it really does change the value.
	if !a.foreignTypeCanExecute(a.checker.GetTypeAtLocation(withoutTypeAssertions(node))) {
		return
	}
	// `for (const n of foreign.inner)` is BOTH a foreign property read and an
	// implicit iteration of what that read returned. When the property rule has
	// already reported this node, the REPORT is suppressed so the same code does
	// not appear twice at one position — but the row is still charged, because
	// that rule reports without touching the row (the standing cause of the
	// SMITHERS1509 row divergence recorded in
	// 09-foreign-calls/a-callback-handed-to-an-untrusted-host-is-still-rejected)
	// and the reference charges Panic here.
	if !a.foreignEvaluationReports[node] {
		a.report(node, "SMITHERS1506",
			"this position invokes "+protocol+" on a foreign value with no call expression to lower; that step can throw, so bind the value through a checker-annotated adapter that returns an owned Smithers value first")
	}
	if owner != nil {
		addRow(owner.directFailures, a.row(a.prelude.panic))
	}
}

// withoutTypeAssertions strips parentheses and type assertions. The runtime
// value is unchanged by all of them.
//
// This is the transitive closure of `typeOnlyWrapperOperand` (THE ONE TABLE,
// lowering.go) and NOT a fourth copy of its contents: this function having
// learned `satisfies` while `foreignValueProvenance` three functions away had
// not is exactly the drift that table exists to end.
func withoutTypeAssertions(node *ast.Node) *ast.Node {
	for {
		operand := typeOnlyWrapperOperand(node)
		if operand == nil {
			return node
		}
		node = operand
	}
}

// namespaceObjectReference reports whether an identifier names an ESM namespace
// import. Reading through one is live-binding selection, not a user accessor.
func (a *rowAnalysis) namespaceObjectReference(node *ast.Node) bool {
	if !ast.IsIdentifier(node) {
		return false
	}
	symbol := a.referencedValueSymbol(node)
	if symbol == nil {
		return false
	}
	for _, declaration := range symbolDeclarations(symbol) {
		if declaration.Kind == ast.KindNamespaceImport {
			return true
		}
	}
	return false
}

// checkForeignTemplateTag refuses a tagged template whose TAG is foreign.
//
// A tagged template invokes its tag, so the rule that governs the ordinary
// call spelling governs it too — but the foreign-call pipeline is keyed on
// call expressions, so this form reached none of it and a throwing foreign
// tag escaped a `failures: []` row uncaught. Result lowering for the tagged form is deferred
// exactly as it is for `new`, so this follows the constructor precedent
// (SMITHERS1504): refused unless the RESOLVED signature carries
// `@throws {never}`, which keeps a genuinely trusted tag usable.
func (a *rowAnalysis) checkForeignTemplateTag(owner *functionRows, node *ast.Node) {
	tagged := node.AsTaggedTemplateExpression()
	if !a.foreignValueProvenance(tagged.Tag, make(map[*ast.Symbol]bool)) {
		return
	}
	signature := a.checker.GetResolvedSignature(node)
	var declaration *ast.Node
	if signature != nil {
		declaration = signature.Declaration()
	}
	returned := a.checker.GetTypeAtLocation(node)
	if signature != nil {
		returned = a.checker.GetReturnTypeOfSignature(signature)
	}
	annotation, refused := a.foreignThrowsAnnotation(declaration, a.carriesRejectionChannel(returned), node)
	if !refused && isNeverThrowsAnnotation(annotation) {
		// The substitutions are this call's ARGUMENTS. A trust claim about the
		// tag cannot speak for another module's panic provenance, so foreign
		// callable provenance handed through them stays refused exactly as it is
		// for the ordinary call spelling `tag(strings, value)`.
		if tagged.Template != nil && tagged.Template.Kind == ast.KindTemplateExpression {
			for _, span := range tagged.Template.AsTemplateExpression().TemplateSpans.Nodes {
				expression := span.AsTemplateSpan().Expression
				if a.containsForeignExecutableValue(expression, make(map[*ast.Symbol]bool)) {
					a.reportForeignProvenanceEscape(owner, expression,
						"foreign callable provenance would escape through an unchecked higher-order call; wrap it in a local adapter that owns invocation and exposes an explicit Result/task contract")
				}
			}
		}
		return
	}
	a.report(node, "SMITHERS1504",
		"a foreign tagged template invokes its tag and can execute JavaScript, but tagged-template Result lowering is deferred; call the tag as an ordinary function through an annotated adapter (only a checker-resolved @throws {never} tag is accepted)")
	if owner != nil {
		addRow(owner.directFailures, a.row(a.prelude.panic))
	}
}

func (a *rowAnalysis) reportForeignProvenanceEscape(owner *functionRows, node *ast.Node, message string) {
	a.report(node, "SMITHERS1508", message)
	if owner != nil {
		addRow(owner.directFailures, a.row(a.prelude.panic))
	}
}

func (a *rowAnalysis) containsForeignExecutableValue(expression *ast.Node, seen map[*ast.Symbol]bool) bool {
	if expression == nil {
		return false
	}
	if a.foreignValueProvenance(expression, make(map[*ast.Symbol]bool)) &&
		a.foreignTypeCanExecute(a.checker.GetTypeAtLocation(expression)) {
		return true
	}
	// Type-only wrappers from `typeOnlyWrapperOperand` (THE ONE TABLE,
	// lowering.go); `!` keeps its own branch because it is the checked
	// propagation boundary and not a type-level annotation.
	if operand := typeOnlyWrapperOperand(expression); operand != nil {
		return a.containsForeignExecutableValue(operand, seen)
	}
	switch expression.Kind {
	case ast.KindNonNullExpression:
		return a.containsForeignExecutableValue(expression.Expression(), seen)
	case ast.KindObjectLiteralExpression:
		for _, property := range expression.AsObjectLiteralExpression().Properties.Nodes {
			switch {
			case ast.IsPropertyAssignment(property):
				if a.containsForeignExecutableValue(property.AsPropertyAssignment().Initializer, copySymbolSet(seen)) {
					return true
				}
			case ast.IsShorthandPropertyAssignment(property):
				// The property's NAME node, walked exactly as the longhand
				// initializer is: resolving only the value symbol skipped the
				// foreign-declaration test above, so a directly IMPORTED
				// callable in `{ getHandler }` was accepted while
				// `{ getHandler: getHandler }` was refused.
				if a.containsForeignExecutableValue(property.Name(), copySymbolSet(seen)) {
					return true
				}
			case ast.IsSpreadAssignment(property):
				if a.containsForeignExecutableValue(property.AsSpreadAssignment().Expression, copySymbolSet(seen)) {
					return true
				}
			}
		}
	case ast.KindArrayLiteralExpression:
		for _, element := range expression.AsArrayLiteralExpression().Elements.Nodes {
			if a.containsForeignExecutableValue(element, copySymbolSet(seen)) {
				return true
			}
		}
	case ast.KindIdentifier:
		symbol := a.canonical(a.referencedValueSymbol(expression))
		return a.foreignSymbolValueCanExecute(symbol, seen)
	}
	return false
}

func (a *rowAnalysis) foreignSymbolValueCanExecute(symbol *ast.Symbol, seen map[*ast.Symbol]bool) bool {
	symbol = a.canonical(symbol)
	if symbol == nil || seen[symbol] {
		return false
	}
	seen[symbol] = true
	declaration := variableLikeDeclaration(symbol)
	return declaration != nil && ast.IsVariableDeclaration(declaration) &&
		a.containsForeignExecutableValue(declaration.AsVariableDeclaration().Initializer, seen)
}

func (a *rowAnalysis) foreignTypeCanExecute(t *checker.Type) bool {
	if t == nil {
		return false
	}
	if t.Flags()&(checker.TypeFlagsAny|checker.TypeFlagsUnknown) != 0 {
		return true
	}
	if t.IsUnion() {
		for _, part := range t.Types() {
			if a.foreignTypeCanExecute(part) {
				return true
			}
		}
		return false
	}
	return t.Flags()&checker.TypeFlagsObject != 0 ||
		len(a.checker.GetSignaturesOfType(t, checker.SignatureKindCall)) != 0 ||
		len(a.checker.GetSignaturesOfType(t, checker.SignatureKindConstruct)) != 0
}

func copySymbolSet(values map[*ast.Symbol]bool) map[*ast.Symbol]bool {
	copied := make(map[*ast.Symbol]bool, len(values))
	for value := range values {
		copied[value] = true
	}
	return copied
}

// checkForeignBoundaries rejects the two foreign-boundary shapes whose runtime
// meaning the language does not define.
//
// specification/failures.mdx, "Foreign Exceptions": "Calling any TypeScript,
// JavaScript, or other foreign runtime implementation MUST add the
// distinguished checked panic case by default … A caller MUST propagate panic,
// explicitly catch it, or use a trusted adapter."
//
//   - A `new Foreign(...)` executes foreign code, and constructor Result
//     lowering is deferred, so only a checker-resolved `@throws {never}`
//     constructor is accepted (SMITHERS1504).
//   - A callback handed to an UNTRUSTED foreign call has no checked invocation
//     count, no checked lifetime and no checked failure channel, which is also
//     what specification/requirements.mdx, "Scoping", requires of
//     caller-controlled background APIs (SMITHERS1509).
//
// The trust marker is honoured in the ARGUMENT position, mirroring the
// reference's checkForeignValueBoundaries. specification/compatibility.mdx,
// "Foreign Boundary": "Calling an unannotated foreign runtime value MUST add the
// checked `panic` case … Trusted `@throws {never}` metadata opts out;
// `@throws {T}` declares a more precise channel." The channel is a property of
// the CALL, so a claim made about the call covers the callback that call was
// handed; and specification/requirements.mdx, "Scoping", assigns the deferred
// half explicitly — "Imported JavaScript or TypeScript that starts hidden
// background work owns that work. Caller-controlled background APIs MUST expose
// explicit completion or disposal handles through their adapters" — putting that
// obligation on the adapter rather than on `.sm`. Requiring the callback itself
// to be panic-free is not available: specification/failures.mdx, "Panic Does Not
// Widen a Return Type", says a function "MUST therefore be able to abort with
// `panic(...)` while keeping a plain return type", so panic-freedom is not
// spellable and the rule would admit nothing.
//
// A trusted call therefore falls through to checkForeignProvenanceEscapes, so a
// FOREIGN callable handed on through it still reports SMITHERS1508: no claim
// about this callee speaks for another module's panic provenance.
func (a *rowAnalysis) checkForeignBoundaries(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			switch {
			case ast.IsCallExpression(node):
				if policy, foreign := a.foreignPolicyForCall(node); foreign && policy.kind == "panic" {
					arguments := node.AsCallExpression().Arguments
					if arguments != nil {
						for _, argument := range arguments.Nodes {
							if a.containsCallableValue(argument, make(map[*ast.Symbol]bool)) {
								a.report(argument, "SMITHERS1509",
									"a callback handed to an UNTRUSTED foreign call may escape beyond the checked call scope; register it through a trusted @throws {never} binding that owns the invocation, or expose an owned Smithers wrapper/adapter with an explicit Result or structured-task callback policy")
							}
						}
					}
				}
			case ast.IsNewExpression(node):
				a.checkForeignConstruction(node)
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

func (a *rowAnalysis) checkForeignConstruction(node *ast.Node) {
	expression := node.AsNewExpression().Expression
	if !a.foreignValueProvenance(expression, make(map[*ast.Symbol]bool)) {
		return
	}
	signature := a.checker.GetResolvedSignature(node)
	if signature != nil && signature.Declaration() != nil {
		annotation, refused := a.foreignThrowsAnnotation(signature.Declaration(), false, node)
		if !refused && isNeverThrowsAnnotation(annotation) {
			return
		}
	}
	a.report(node, "SMITHERS1504",
		"a foreign constructor can execute JavaScript but constructor Result lowering is deferred; expose an annotated factory function or Smithers adapter (only a checker-resolved @throws {never} constructor is accepted)")
}

// foreignBaseClass reports whether an authored class's `extends` chain reaches a
// foreign class. `new Derived()` runs every constructor up the chain, so a
// foreign base three levels up is still invoked by this construction, and still
// invoked with no call expression naming it.
func (a *rowAnalysis) foreignBaseClass(expression *ast.Node, seen map[*ast.Symbol]bool) bool {
	symbol := a.canonical(a.referencedValueSymbol(expression))
	if symbol == nil || seen[symbol] {
		return false
	}
	seen[symbol] = true
	for _, declaration := range symbolDeclarations(symbol) {
		if !ast.IsClassDeclaration(declaration) && !ast.IsClassExpression(declaration) {
			continue
		}
		heritage := declaration.ClassLikeData().HeritageClauses
		if heritage == nil {
			continue
		}
		for _, clause := range heritage.Nodes {
			if clause.AsHeritageClause().Token != ast.KindExtendsKeyword {
				continue
			}
			for _, base := range clause.AsHeritageClause().Types.Nodes {
				if a.foreignValueProvenance(base.Expression(), make(map[*ast.Symbol]bool)) {
					return true
				}
				if a.foreignBaseClass(base.Expression(), seen) {
					return true
				}
			}
		}
	}
	return false
}

func isNeverThrowsAnnotation(annotation string) bool {
	return annotation == "never"
}

// containsCallableValue reports whether an argument carries something foreign
// code could invoke. The type's call/construct signatures are the authority, so
// a callback reaches this whether it was written inline, passed by name, or
// nested inside an object or array literal.
func (a *rowAnalysis) containsCallableValue(expression *ast.Node, seen map[*ast.Symbol]bool) bool {
	if expression == nil {
		return false
	}
	if ast.IsFunctionExpressionOrArrowFunction(expression) {
		return true
	}
	// Type-only wrappers from `typeOnlyWrapperOperand` (THE ONE TABLE,
	// lowering.go); `!` keeps its own branch, as above.
	if operand := typeOnlyWrapperOperand(expression); operand != nil {
		return a.containsCallableValue(operand, seen)
	}
	switch expression.Kind {
	case ast.KindNonNullExpression:
		return a.containsCallableValue(expression.Expression(), seen)
	}
	if t := a.checker.GetTypeAtLocation(expression); t != nil {
		if len(a.checker.GetSignaturesOfType(t, checker.SignatureKindCall)) > 0 ||
			len(a.checker.GetSignaturesOfType(t, checker.SignatureKindConstruct)) > 0 {
			return true
		}
	}
	switch {
	case ast.IsObjectLiteralExpression(expression):
		for _, property := range expression.AsObjectLiteralExpression().Properties.Nodes {
			switch property.Kind {
			case ast.KindMethodDeclaration, ast.KindGetAccessor, ast.KindSetAccessor:
				return true
			case ast.KindPropertyAssignment:
				if a.containsCallableValue(property.AsPropertyAssignment().Initializer, seen) {
					return true
				}
			case ast.KindShorthandPropertyAssignment:
				if a.containsCallableValue(property.Name(), seen) {
					return true
				}
			case ast.KindSpreadAssignment:
				if a.containsCallableValue(property.AsSpreadAssignment().Expression, seen) {
					return true
				}
			}
		}
	case ast.IsArrayLiteralExpression(expression):
		for _, element := range expression.AsArrayLiteralExpression().Elements.Nodes {
			if a.containsCallableValue(element, seen) {
				return true
			}
		}
	case ast.IsIdentifier(expression):
		symbol := a.canonical(a.checker.GetSymbolAtLocation(expression))
		if symbol == nil || seen[symbol] {
			return false
		}
		seen[symbol] = true
		declaration := variableLikeDeclaration(symbol)
		if declaration != nil && ast.IsVariableDeclaration(declaration) {
			return a.containsCallableValue(declaration.AsVariableDeclaration().Initializer, seen)
		}
	}
	return false
}

// foreignValueProvenance reports whether the value an expression produces came
// out of a TypeScript/JavaScript module rather than out of authored `.sm`.
//
// This is provenance, not declaration-file membership. A `.sm` local whose TYPE
// is an interface declared in a foreign module is still an authored value —
// specification/compatibility.mdx, "Runtime TypeScript Dependency": "A
// type-only import MUST NOT add that requirement." Provenance is followed
// through resolved symbols, so an aliased import is foreign under any spelling.
//
// A SELECTING operator carries the provenance of whichever operand it yields,
// and which operand that will be is not decidable here — so this asks
// `valueBranches` (THE ONE TABLE, lowering.go) and folds every operand that can
// be the value, which is the fail-closed direction. That table is shared with
// `contextReceiverOf` rather than restated, because restating it is precisely
// how this walk came to know the property access and the call while knowing
// neither `??` nor the ternary.
//
// A TYPE-ONLY wrapper changes no value at all, so it is asked of
// `typeOnlyWrapperOperand` (the sibling ONE TABLE, lowering.go) and never
// restated here either — for the same reason, one round later. This walk knew
// the parenthesis, `as` and `<T>x` and not `satisfies`, while
// `withoutTypeAssertions` three functions below knew all four, so
// `(client satisfies T).dangerous` reported nothing while `client.dangerous`
// was refused and a raw host `Error` escaped a checked function through it.
//
// `await` and `!` keep their own branches below: each really does change the
// value, so neither belongs in a table shared with walks that must treat them
// differently.
func (a *rowAnalysis) foreignValueProvenance(expression *ast.Node, seen map[*ast.Symbol]bool) bool {
	if expression == nil {
		return false
	}
	if a.assetValueProvenance(expression, make(map[*ast.Symbol]bool)) {
		// Compiler-owned asset modules are compile-time pure-data edges. Their
		// values must never acquire JavaScript accessor/call provenance merely
		// because the checker representation lives outside an authored .sm file.
		return false
	}
	if operand := typeOnlyWrapperOperand(expression); operand != nil {
		return a.foreignValueProvenance(operand, seen)
	}
	switch expression.Kind {
	case ast.KindNonNullExpression, ast.KindAwaitExpression:
		return a.foreignValueProvenance(expression.Expression(), seen)
	case ast.KindPropertyAccessExpression:
		return a.foreignValueProvenance(expression.AsPropertyAccessExpression().Expression, seen)
	case ast.KindElementAccessExpression:
		return a.foreignValueProvenance(expression.AsElementAccessExpression().Expression, seen)
	case ast.KindCallExpression:
		if _, foreign := a.foreignPolicyForCall(expression); foreign {
			return true
		}
		return a.foreignValueProvenance(expression.AsCallExpression().Expression, seen)
	case ast.KindNewExpression:
		return a.foreignValueProvenance(expression.AsNewExpression().Expression, seen)
	case ast.KindIdentifier:
		symbol := a.canonical(a.referencedValueSymbol(expression))
		if symbol == nil || seen[symbol] {
			return false
		}
		seen[symbol] = true
		declaration := symbol.ValueDeclaration
		if declaration == nil {
			declaration = variableLikeDeclaration(symbol)
		}
		if declaration == nil {
			return false
		}
		if a.isForeignDeclaration(declaration) {
			return true
		}
		if ast.IsVariableDeclaration(declaration) {
			return a.foreignValueProvenance(declaration.AsVariableDeclaration().Initializer, seen)
		}
	}
	for _, branch := range valueBranches(expression) {
		// Each branch is walked with its OWN copy of the visited set: two
		// operands are two independent paths, and a symbol one of them has
		// already declined must not silence the other.
		if a.foreignValueProvenance(branch, copySymbolSet(seen)) {
			return true
		}
	}
	return false
}

func variableLikeDeclaration(symbol *ast.Symbol) *ast.Node {
	if symbol == nil {
		return nil
	}
	if symbol.ValueDeclaration != nil && ast.IsVariableDeclaration(symbol.ValueDeclaration) {
		return symbol.ValueDeclaration
	}
	for _, declaration := range symbol.Declarations {
		if ast.IsVariableDeclaration(declaration) {
			return declaration
		}
	}
	return symbol.ValueDeclaration
}

// Async callback ownership
// ---------------------------------------------------------------------------

// checkCallbackOwnership rejects an async callback whose Promise nobody owns.
//
// docs/DECISIONS.md, "Concurrency" (Locked): "Authored .sm code must
// transitively consume every started Promise with await: either directly or
// through a recognized combinator such as Promise.all whose result is awaited."
// specification/requirements.mdx, "Scoping": "Fire-and-forget and detached work
// are initially unavailable in authored .sm."
//
// An async function value handed to a general higher-order function starts a
// Promise at a call site the compiler cannot see, so no await can be proven.
// The two recognized owners are excluded: `Layer.provide`, whose computation
// the provide call itself carries, and `Result.tryPromise`, which invokes its
// body exactly once and awaits it.
func (a *rowAnalysis) checkCallbackOwnership(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			// A constructor is a consumer: `new Runner(cb)` hands `cb` across the
			// same opaque edge `run(cb)` does, so a NewExpression's argument list
			// is walked exactly like a call's. So is a TAGGED TEMPLATE's
			// substitution list: `` hostTag`x${cb}` `` is `hostTag(parts, cb)`,
			// and reading only call/`new` made SMITHERS1303 and SMITHERS1404
			// silent on the template spelling of the very same handover.
			if isInvocationExpression(node) {
				for index, argument := range invocationArguments(node) {
					a.checkFallibleCallbackArgument(node, index, argument)
					if ast.IsCallExpression(node) || ast.IsTaggedTemplateExpression(node) {
						// SMITHERS1404 (async invocation ownership) is a separate rule
						// about STARTED work rather than a lost failure channel, and it
						// is deliberately left on the argument itself. Widening its
						// reach the way the contract check is widened is not this
						// rule's question and would change which programs it accepts. A
						// tagged template IS an argument position, though — the
						// substitution is handed to the tag and started there — so the
						// two spellings of one handover answer alike. `new` stays out,
						// as it always has.
						a.checkAsyncCallbackArgument(node, index, argument)
					}
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

// collectCallbackValues follows every function VALUE an argument carries.
//
// A callback boundary is a value edge, not a syntactic argument position:
// `match({ error: cb })` hands `cb` to the callee exactly as `map(cb)` does, and
// the callee invokes it exactly the same way, so the contract that makes the
// handover safe has to be required in both spellings. Reading only the top-level
// argument node let an inferred-fallible function cross the boundary inside an
// object literal, an array literal, a parenthesis, or an `as` cast, where its
// lowered Result return silently became the callee's plain success value
// (specification/failures.mdx, "Compiler Lifting").
//
// Function BODIES are never entered: a callback nested inside another callback's
// body is a boundary of its own and is checked at its own call site. A spread
// ELEMENT's operand is followed, but an object spread is not — its members are
// not syntactically present, so nothing here could name them.
func collectCallbackValues(argument *ast.Node) []*ast.Node {
	carried := make([]*ast.Node, 0, 1)
	var walk func(*ast.Node)
	walk = func(node *ast.Node) {
		if node == nil {
			return
		}
		// Type-only wrappers from `typeOnlyWrapperOperand` (THE ONE TABLE,
		// lowering.go). This walk already knew all four, which is precisely
		// what made the drift invisible: it and `containsCallableValue` decide
		// the same SMITHERS1509 position and disagreed about `satisfies`.
		if operand := typeOnlyWrapperOperand(node); operand != nil {
			walk(operand)
			return
		}
		switch node.Kind {
		case ast.KindNonNullExpression:
			walk(node.Expression())
			return
		case ast.KindObjectLiteralExpression:
			for _, property := range node.AsObjectLiteralExpression().Properties.Nodes {
				switch property.Kind {
				case ast.KindPropertyAssignment:
					walk(property.AsPropertyAssignment().Initializer)
				case ast.KindShorthandPropertyAssignment:
					carried = append(carried, property.Name())
				case ast.KindMethodDeclaration, ast.KindGetAccessor, ast.KindSetAccessor:
					carried = append(carried, property)
				}
			}
			return
		case ast.KindArrayLiteralExpression:
			for _, element := range node.AsArrayLiteralExpression().Elements.Nodes {
				if element.Kind == ast.KindOmittedExpression {
					continue
				}
				if element.Kind == ast.KindSpreadElement {
					walk(element.Expression())
					continue
				}
				walk(element)
			}
			return
		}
		carried = append(carried, node)
	}
	walk(argument)
	return carried
}

// crossingCallbacks is every checked function value a call or `new` hands
// across its argument boundary.
//
// It walks EXACTLY the value edge checkFallibleCallbackArgument walks for
// SMITHERS1303 and checkAsyncCallbackArgument walks for SMITHERS1404 — the same
// collectCallbackValues plus resolveFunctionReference — so the three channels
// cannot drift apart on which handovers they see. The callback-boundary rules
// were asymmetric before this existed: a callback that could FAIL was refused
// and an ASYNC callback was refused, while a callback that REQUIRED a
// capability crossed the same boundary with the requirement deleted from every
// row. `xs.map(() => Db.context()...)` published `requirements: []`, read `Db`
// through the runtime frame stack, and the top-level unsatisfied-requirement
// check certified it.
//
// `Layer.provide`'s computation (argument 1) is the one argument excluded:
// checkOneLayer already reconciles that callback's row against the layer's
// provided closure, and charging it here would republish to the caller exactly
// the capabilities the provide site satisfies.
func (a *rowAnalysis) crossingCallbacks(node *ast.Node) []*functionRows {
	if !isInvocationExpression(node) {
		return nil
	}
	provideComputation := -1
	if ast.IsCallExpression(node) && a.prelude.layerCall(node, "provide") {
		provideComputation = 1
	}
	carriedFunctions := make([]*functionRows, 0)
	// A tagged template hands its SUBSTITUTIONS to the tag exactly as a call
	// hands its arguments; see invocationArguments.
	for index, argument := range invocationArguments(node) {
		if index == provideComputation {
			continue
		}
		for _, carried := range collectCallbackValues(argument) {
			if callback := a.resolveFunctionReference(carried); callback != nil {
				carriedFunctions = append(carriedFunctions, callback)
			}
		}
	}
	return carriedFunctions
}

// checkFallibleCallbackArgument rejects a function value that carries a failure
// row across a callback boundary without declaring one.
//
// specification/failures.mdx, "Inference": "Public, abstract, ambient, and
// declaration-only contracts MUST express fallibility directly with
// Result<A, E>." An argument position is such a boundary. The callee invokes the
// function somewhere the compiler cannot see, so the only description of the
// function's failures that survives the hand-off is the function's own declared
// type — and a plain `(key: string) => string` has nowhere to put them. Accepted,
// the typed failure would simply vanish at the edge, which is the one outcome
// the failure channel exists to prevent.
//
// The rule is about the CALLBACK's contract, not the callee's parameter type:
// annotating the function value with `Result<A, E>` clears it wherever it is
// passed, and no signature inspection of the receiving function is involved.
//
// Two argument positions are excluded because a different rule already owns
// them, and reporting here as well would double-report the same fact:
// `Layer.provide`'s computation (SMITHERS2105, in checkOneLayer) and — through
// the row analysis, which does not charge a foreign call inside an adapter body
// — `Result.try`/`Result.tryPromise`, where the adapter IS the boundary that
// converts the throw into the Result.
func (a *rowAnalysis) checkFallibleCallbackArgument(call *ast.Node, index int, argument *ast.Node) {
	if index == 1 && a.prelude.layerCall(call, "provide") {
		return
	}
	if a.isDurableIntrinsicCall(call) {
		// A durable flow body is not a callback at all. The durable frontend
		// compiles it into a static plan at compile time — it never becomes a
		// closure handed to code the compiler cannot see — and the plan carries
		// each step's declared `Result<A, E>` explicitly. The premise of this
		// rule, that the failure channel disappears at an opaque edge, is false
		// here, so it is a recognized owner exactly like Layer.provide.
		return
	}
	// The contract check follows the function VALUES the argument carries, not
	// the argument's syntactic shape; see collectCallbackValues.
	for _, carried := range collectCallbackValues(argument) {
		callback := a.resolveFunctionReference(carried)
		if callback == nil {
			// A parameter, an import, or anything else that is not a checked local
			// function has no row the compiler computed, so there is nothing here to
			// prove lost. This is what keeps `keys.map(transform)` — where
			// `transform` is a parameter — an ordinary call.
			continue
		}
		if len(callback.bodyFailures) == 0 || callback.declaredShape.result {
			continue
		}
		a.report(carried, "SMITHERS1303",
			"an inferred-fallible function value cannot cross a callback boundary; add an explicit Result contract or handle its failures before passing it")
	}
}

// isDurableIntrinsicCall reports whether a call's callee resolves to one of the
// compiler-owned durable declarations. Identity, not spelling: the intrinsic is
// recognized by the declaration file its symbol was declared in, so a renamed
// import (`durable as compileFlow`) is still the intrinsic and a user function
// named `durable` still is not.
func (a *rowAnalysis) isDurableIntrinsicCall(call *ast.Node) bool {
	if call == nil || !ast.IsCallExpression(call) {
		return false
	}
	symbol := a.prelude.resolvedSymbolAt(ast.SkipParentheses(call.AsCallExpression().Expression))
	if symbol == nil {
		return false
	}
	for _, declaration := range symbol.Declarations {
		if file := ast.GetSourceFileOfNode(declaration); file != nil && file.FileName() == durableVirtualName {
			return true
		}
	}
	return false
}

func (a *rowAnalysis) checkAsyncCallbackArgument(call *ast.Node, index int, argument *ast.Node) {
	function := argument
	if !ast.IsFunctionExpressionOrArrowFunction(function) {
		resolved := a.resolveFunctionReference(argument)
		if resolved == nil {
			return
		}
		function = resolved.node
	}
	if !ast.HasSyntacticModifier(function, ast.ModifierFlagsAsync) {
		return
	}
	// `Layer.provide(layer, async () => …)`: the provide call is the owner, so
	// long as the Promise it returns is itself consumed.
	if index == 1 && a.prelude.layerCall(call, "provide") && promiseValueIsConsumed(call) {
		return
	}
	// `Result.tryPromise(async () => …)`: the boundary awaits its own body.
	if index == 0 && a.isResultBoundaryBody(function) {
		return
	}
	a.report(argument, "SMITHERS1404",
		"async callback invocation ownership is not proven here; use an explicit structured-concurrency combinator or await/return a recognized Layer.provide computation")
}

// promiseValueIsConsumed reports whether the Promise a call produces is awaited
// or returned at the site itself. Anything less is not proof of consumption, so
// the answer is deliberately conservative.
func promiseValueIsConsumed(call *ast.Node) bool {
	current := call
	for current.Parent != nil {
		parent := current.Parent
		switch {
		case ast.IsParenthesizedExpression(parent), ast.IsAsExpression(parent), ast.IsNonNullExpression(parent):
			current = parent
			continue
		case ast.IsAwaitExpression(parent), ast.IsReturnStatement(parent):
			return true
		default:
			return false
		}
	}
	return false
}

// Type-only imports

// importIsTypeOnly reports whether an import contributes no runtime value.
//
// specification/type-system.mdx, "Foreign Boundaries": "Type-only imports add
// neither runtime requirement." Both spellings count: `import type { T }`, and
// a named-import list whose every element carries its own `type` marker.
func importIsTypeOnly(declaration *ast.ImportDeclaration) bool {
	clause := declaration.ImportClause
	if clause == nil {
		// A bare `import "./x.ts"` is exactly a module-initializer edge.
		return false
	}
	if clause.IsTypeOnly() {
		return true
	}
	data := clause.AsImportClause()
	if data.Name() != nil || data.NamedBindings == nil || !ast.IsNamedImports(data.NamedBindings) {
		return false
	}
	elements := data.NamedBindings.AsNamedImports().Elements
	if elements == nil || len(elements.Nodes) == 0 {
		return false
	}
	for _, element := range elements.Nodes {
		if !element.AsImportSpecifier().IsTypeOnly {
			return false
		}
	}
	return true
}

// isResultBoundaryBody reports whether a function IS the body argument of
// `Result.try` / `Result.tryPromise`. The adapter invokes it exactly once
// inside its own try/catch, so it owns both the throw scope and, for the async
// form, the callback's lifetime.
func (a *rowAnalysis) isResultBoundaryBody(node *ast.Node) bool {
	if !ast.IsFunctionExpressionOrArrowFunction(node) || node.Parent == nil {
		return false
	}
	parent := node.Parent
	if !ast.IsCallExpression(parent) {
		return false
	}
	arguments := parent.AsCallExpression().Arguments
	if arguments == nil || len(arguments.Nodes) == 0 || arguments.Nodes[0] != node {
		return false
	}
	_, matched := a.prelude.resultBoundaryCall(parent)
	return matched
}

// Retired type grammar

// checkRetiredTypeGrammar rejects the two retired type markers with the
// migration diagnostic instead of a raw TypeScript grammar error.
//
// specification/type-system.mdx, "Absence" (Locked): "The earlier ?T,
// payload-capture, orelse, and .? grammar MUST NOT be part of the language."
// specification/failures.mdx, "Inference", and
// specification/type-system.mdx, "Public Boundaries", both say Smithers MUST
// NOT add a `!T` failure-annotation grammar.
//
// Identity, not a token scan: the fork's parser already recovers `!T` as a
// JSDoc non-nullable type and `?T` as a JSDoc nullable type, so the retired
// grammar IS a node kind. A postfix expression `x!` is a different node and is
// owned by the propagation pass (or rejected there for a non-Result); the
// postfix `T!` type form is not this retired prefix marker either.
func (a *rowAnalysis) checkRetiredTypeGrammar(files []*ast.SourceFile) {
	for _, file := range files {
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			switch node.Kind {
			case ast.KindJSDocNonNullableType:
				if isPrefixJSDocTypeMarker(node) {
					a.report(node, "SMITHERS1001", "the `!T` return marker was removed; use Result<T, E>")
				}
			case ast.KindJSDocNullableType:
				if isPrefixJSDocTypeMarker(node) {
					a.report(node, "SMITHERS1001", "the `?T` type grammar was removed; use T | undefined")
				}
			}
			node.ForEachChild(visit)
			return false
		}
		file.AsNode().ForEachChild(visit)
	}
}

// isPrefixJSDocTypeMarker separates `!T` from `T!`. The parser uses one node for
// both and distinguishes them by whether the marker precedes the type, which is
// the same test the fork's own grammar check performs.
func isPrefixJSDocTypeMarker(node *ast.Node) bool {
	inner := node.Type()
	return inner != nil && node.Pos() != inner.Pos()
}

// Project closure
// ---------------------------------------------------------------------------

// checkRelativeModuleResolution rejects a relative `.sm` import the supplied
// project does not contain.
//
// specification/requirements.mdx, "Satisfaction": "When the compiler knows the
// complete closure, an unsatisfied capability MUST be a compile error." Failure
// and requirement rows are computed over the whole supplied module set, so a
// module the compiler cannot see means the closure is NOT known. The analysis
// must say so rather than proceed as though the absent module contributed an
// empty row — an assumption that would let a real failure or requirement
// disappear because the file carrying it was missing.
//
// The rule is deliberately narrow, and each narrowing is load-bearing:
//
//   - Only a RELATIVE specifier. A bare specifier names a package, whose absence
//     is a packaging question TypeScript already answers.
//   - Only a `.sm` specifier. A relative `.ts`, `.js`, or asset import is not a
//     Smithers module and does not carry rows.
//   - Type-only imports are included. The corpus pins this: the closure the row
//     analysis needs is the module set, and a type-only edge still names a module
//     whose Error and Context declarations the rows are built from. Excluding it
//     would let the one import spelling that produces no value cascade also
//     silently produce no closure check.
func (a *rowAnalysis) checkRelativeModuleResolution(files []*ast.SourceFile) {
	for _, file := range files {
		for _, statement := range file.Statements.Nodes {
			if !ast.IsImportDeclaration(statement) {
				continue
			}
			specifier := statement.AsImportDeclaration().ModuleSpecifier
			if specifier == nil || !ast.IsStringLiteral(specifier) {
				continue
			}
			text := specifier.Text()
			if !strings.HasPrefix(text, ".") || !strings.HasSuffix(text, ".sm") {
				continue
			}
			// Resolution is the checker's own, so this asks the real module
			// graph rather than re-deriving path lookup rules that could drift
			// from it. ResolveExternalModuleName ignores errors, so asking is
			// free of side effects.
			if a.checker.ResolveExternalModuleName(specifier) != nil {
				continue
			}
			a.report(specifier, "SMITHERS1801",
				"relative Smithers module '"+text+"' is not present in the supplied project source set")
		}
	}
}

// checkCompilerResultConstructors refuses a value binding in authored `.sm`
// that resolves to one of the compiler's own Result variant constructors.
//
// specification/failures.mdx, "Compiler Lifting" (Locked): "Authors MUST NOT
// need to write `Result.ok(...)` or `Result.err(...)`. Those constructors MUST
// NOT be part of the ordinary Smithers authoring API." SMITHERS1201 already
// carries that sentence for the `Result.ok(...)` spelling; this is the same
// sentence at the other spelling, where the constructor is reached by NAME
// through a compiler-owned module rather than through the `Result` namespace.
// A Result the compiler did not construct at a checked exit is a Result whose
// failure channel means nothing, so the rule is about the constructor being
// reachable at all — not about which module name reached it.
//
// The proof is checker-backed and never textual: a reference is refused only
// when its unaliased, merged symbol is declared by the prelude's own
// `SmithersOk` / `SmithersErr` class declaration. An author's own class of the
// same name resolves to its own declaration and is untouched, and a renamed,
// namespaced, or re-exported binding resolves to the same prelude declaration
// and is refused identically.
//
// Only value positions are refused. A type-only import initializes nothing and
// constructs nothing, so it is left to the module-trust rule, which is
// deliberately silent on type-only edges for every module.
//
// One report per constructor per file: the import that brought the name in is
// the actionable site, and repeating it at every use would bury it.
func (a *rowAnalysis) checkCompilerResultConstructors(files []*ast.SourceFile) {
	if a.prelude == nil || a.prelude.okClass == nil || a.prelude.errClass == nil {
		return
	}
	for _, file := range files {
		reported := make(map[*ast.Node]bool, 2)
		refuse := func(node *ast.Node) {
			if node == nil {
				return
			}
			declaration := classDeclarationOf(a.symbolAt(node))
			if declaration != a.prelude.okClass && declaration != a.prelude.errClass {
				return
			}
			if reported[declaration] {
				return
			}
			reported[declaration] = true
			a.report(node, "SMITHERS1201", "'"+declarationName(declaration)+"' is a compiler-owned Result constructor, not an author-facing one; ordinary return and throw lifting construct the variants")
		}
		var visit func(*ast.Node) bool
		visit = func(node *ast.Node) bool {
			if node == nil {
				return false
			}
			switch {
			case ast.IsImportDeclaration(node):
				declaration := node.AsImportDeclaration()
				if declaration.ImportClause == nil || declaration.ImportClause.IsTypeOnly() {
					return false
				}
				clause := declaration.ImportClause.AsImportClause()
				refuse(clause.Name())
				if clause.NamedBindings != nil && ast.IsNamedImports(clause.NamedBindings) {
					for _, element := range clause.NamedBindings.AsNamedImports().Elements.Nodes {
						if !element.AsImportSpecifier().IsTypeOnly {
							refuse(element.Name())
						}
					}
				}
				return false
			case ast.IsExportDeclaration(node):
				declaration := node.AsExportDeclaration()
				if declaration.IsTypeOnly || declaration.ExportClause == nil ||
					!ast.IsNamedExports(declaration.ExportClause) {
					return false
				}
				for _, element := range declaration.ExportClause.AsNamedExports().Elements.Nodes {
					if !element.AsExportSpecifier().IsTypeOnly {
						refuse(element.Name())
					}
				}
				return false
			case ast.IsPropertyAccessExpression(node):
				refuse(node.AsPropertyAccessExpression().Name())
			}
			return node.ForEachChild(visit)
		}
		file.AsNode().ForEachChild(visit)
	}
}

// checkProjectImportBindings rejects a value binding that does not resolve to
// an export of the supplied relative Smithers module. The proof is entirely
// checker-backed: the module is the symbol ResolveExternalModuleName selected,
// and a binding is valid only when its unaliased/merged symbol is one of that
// module's unaliased/merged exports. Thus `present as absent` is valid even
// though the local spelling matches a missing-name example, while an actually
// unresolved binding is refused with SMITHERS1804 before stock TS2305 can
// stand in for the language's row-closure rule.
func (a *rowAnalysis) checkProjectImportBindings(files []*ast.SourceFile) {
	for _, file := range files {
		for _, statement := range file.Statements.Nodes {
			if !ast.IsImportDeclaration(statement) {
				continue
			}
			declaration := statement.AsImportDeclaration()
			specifier := declaration.ModuleSpecifier
			if specifier == nil || !ast.IsStringLiteral(specifier) {
				continue
			}
			text := specifier.Text()
			if !strings.HasPrefix(text, ".") || !strings.HasSuffix(text, ".sm") || declaration.ImportClause == nil {
				continue
			}
			clause := declaration.ImportClause.AsImportClause()
			if declaration.ImportClause.IsTypeOnly() {
				continue
			}
			module := a.checker.ResolveExternalModuleName(specifier)
			if module == nil {
				// SMITHERS1801 owns the absent-module situation.
				continue
			}
			exported := make(map[*ast.Symbol]bool)
			for _, symbol := range a.checker.GetExportsOfModule(module) {
				if canonical := a.canonical(symbol); canonical != nil {
					exported[canonical] = true
				}
			}
			checkBinding := func(binding *ast.Node) {
				if binding == nil {
					return
				}
				resolved := a.symbolAt(binding)
				if resolved != nil && exported[resolved] {
					return
				}
				a.report(binding, "SMITHERS1804", "import '"+binding.Text()+"' does not resolve to an exported value in '"+text+"'")
			}
			checkBinding(clause.Name())
			if clause.NamedBindings == nil || !ast.IsNamedImports(clause.NamedBindings) {
				continue
			}
			for _, element := range clause.NamedBindings.AsNamedImports().Elements.Nodes {
				specifier := element.AsImportSpecifier()
				if !specifier.IsTypeOnly {
					checkBinding(specifier.Name())
				}
			}
		}
	}
}
